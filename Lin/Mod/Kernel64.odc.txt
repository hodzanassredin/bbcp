MODULE LinKernel64;
(** 64-bit Linux platform kernel. Provides mmap, munmap, signal handling for amd64. **)

	IMPORT SYSTEM, Kernel64;

	CONST
		clusterSize* = 262144; (* 256KB *)

	TYPE
		ADDRESS = LONGINT;

	VAR
		monoCluster-: BOOLEAN;

	PROCEDURE [ccall] mmap(addr: LONGINT; len: INTEGER; prot, flags, fd, off: INTEGER): LONGINT;

	PROCEDURE [ccall] munmap(addr: LONGINT; len: INTEGER): INTEGER;

	PROCEDURE AllocateCluster* (size: LONGINT; VAR r: Kernel64.Cluster);
		VAR adr: LONGINT;
	BEGIN
		IF monoCluster THEN
			adr := mmap(0, 1572864000, 3, 34, -1, 0) (* 1.5GB *)
		ELSE
			adr := mmap(0, SHORT(size), 3, 34, -1, 0)
		END;
		IF adr < 0 THEN r := NIL
		ELSE
			r := SYSTEM.VAL(Kernel64.Cluster, adr);
			r.size := SHORT(size)
		END
	END AllocateCluster;

	PROCEDURE Init*;
	BEGIN
		Kernel64.InitHeap;
	END Init;

END LinKernel64.
