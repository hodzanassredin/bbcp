	PROCEDURE IncompleteType (typ: HrCPT.Struct): BOOLEAN;
	BEGIN
		IF typ.form = Pointer THEN
			IF typ = HrCPT.sysptrtyp THEN RETURN FALSE END;
			typ := typ.BaseTyp
		END;
		RETURN (typ = HrCPT.undftyp) OR (typ.comp = Record) & (typ.BaseTyp = HrCPT.undftyp)
	END IncompleteType;
	
	PROCEDURE SetType (struct: HrCPT.Struct; obj: HrCPT.Object; typ: HrCPT.Struct; name: HrCPT.String);
		VAR u: Elem;
	BEGIN
		IF obj # NIL THEN obj.typ := typ ELSE struct.BaseTyp := typ END;
		IF name # NIL THEN
			NEW(u); u.struct := struct; u.obj := obj; u.pos := HrM.errpos; u.name := name;
			u.next := userList; userList := u
		END
	END SetType;

	PROCEDURE SetType2 (t: T.Type; obj: T.Object; type: T.Type; name: HrCPT.String);
		VAR f: Fixup;
	BEGIN ASSERT((t = NIL) OR (t IS T.Pointer) OR (t IS T.Structure) OR (t IS T.Forward), 20);
		IF obj # NIL THEN obj.type := type; 
			IF ~(obj IS T.TypeName) & ~(type IS T.Forward)  THEN INCL(type.flags, T.used) 
			END
		ELSE 
			WITH t: T.Record DO t.base := type(T.Record)
			| t: T.RecPtr DO WITH type: T.Record DO t.base := type ELSE HALT(22) END
			| t: T.Array DO t.elTyp := type
			| t: T.ArrayPtr DO t.base := type(T.Array)
			| t: T.Procedure DO t.res := type
			| t: T.Forward DO t.base := type(T.Forward)
			ELSE HALT(23)
			END
		END;
		IF type # NIL THEN WITH type: T.Forward DO NEW(f); f.t := t; f.obj := obj; f.link := fixups; fixups := f ELSE END;
		(*IF type.typeName # NIL THEN MarkTypeUsed(type) END*)
		END
	END SetType2;

	PROCEDURE CheckAlloc (VAR typ: HrCPT.Struct; dynAllowed: BOOLEAN; pos: INTEGER);
	BEGIN
		typ.pvused := TRUE;
		IF typ.comp = DynArr THEN
			IF ~dynAllowed THEN HrM.Mark(88, pos); typ := HrCPT.undftyp END
		ELSIF typ.comp = Record THEN
			IF (typ.attribute = absAttr) OR (typ.attribute = limAttr) & (typ.mno # 0) THEN
				HrM.Mark(193, pos); typ := HrCPT.undftyp
			END
		END
	END CheckAlloc;

	PROCEDURE CheckRecursiveType (outer, inner: HrCPT.Struct; pos: INTEGER);
		VAR fld: HrCPT.Object;
	BEGIN
		IF outer = inner THEN HrM.Mark(58, pos)
		ELSIF inner.comp IN {Array, DynArr} THEN CheckRecursiveType(outer, inner.BaseTyp, pos)
		ELSIF inner.comp = Record THEN
			fld := inner.link;
			WHILE (fld # NIL) & (fld.mode = Fld) DO
				CheckRecursiveType(outer, fld.typ, pos);
				fld := fld.link
			END;
			IF inner.BaseTyp # NIL THEN CheckRecursiveType(outer, inner.BaseTyp, pos) END
		END
	END CheckRecursiveType;
	
	PROCEDURE CheckRecursiveType2 (a, b: T.Type; pos: INTEGER);
		VAR i: INTEGER;
	BEGIN
		IF a = b THEN HrM.Mark(-58, pos)
		ELSE 
			WITH b: T.Array DO CheckRecursiveType2(a, b.elTyp, pos)
			| b: T.Record DO
				IF b.fields = NIL THEN i := -1 ELSE i := LEN(b.fields) - 1 END;
				WHILE i >= 0 DO CheckRecursiveType2(a, b.fields[i].type, pos); DEC(i) END;
				IF b.base # NIL THEN CheckRecursiveType2(a, b.base, pos) END
			ELSE
			END
		END
	END CheckRecursiveType2;
	
	PROCEDURE FixType (struct: HrCPT.Struct; obj: HrCPT.Object; typ: HrCPT.Struct; pos: INTEGER);
	(* fix forward reference *)
		VAR t: HrCPT.Struct; f, bf: HrCPT.Object; i: SHORTINT;
	BEGIN
		IF obj # NIL THEN
			IF obj.mode = Var THEN	(* variable type *)
				IF struct # NIL THEN	(* receiver type *)	(* a receiver can never be specified with a forward reference! *)
					IF (typ.form # Pointer) OR (typ.BaseTyp # struct) THEN HrM.Mark(180, pos) END;
				ELSE CheckAlloc(typ, obj.mnolev > level, pos)	(* TRUE for parameters *)
				END
			ELSIF obj.mode = VarPar THEN	(* varpar type *)
				IF struct # NIL THEN	(* varpar receiver type *) (* a receiver cannot be specified with a forward reference! *)
					IF typ # struct THEN HrM.Mark(180, pos) END
				END
			ELSIF obj.mode = Fld THEN	(* field type *)
				CheckAlloc(typ, FALSE, pos);
				CheckRecursiveType(struct, typ, pos)
			ELSIF obj.mode = TProc THEN	(* proc return type *)
				IF typ.form = Comp THEN typ := HrCPT.undftyp; HrM.Mark(54, pos) END
			ELSIF obj.mode = Typ THEN	(* alias type *)
				IF typ.form IN {Byte..Set, Char16, Int64} THEN	(* make alias structure *)
					t := HrCPT.NewStr(typ.form, Basic); i := t.ref;
					t^ := typ^; t.ref := i; t.strobj := obj; t.mno := 0;
					t.BaseTyp := typ; typ := t
				END;
				IF obj.vis # internal THEN
					IF typ.comp = Record THEN typ.exp := TRUE
					ELSIF typ.form = Pointer THEN typ.BaseTyp.exp := TRUE
					END
				END
			ELSE HALT(100)
			END;
			obj.typ := typ
		ELSE
			IF struct.form = Pointer THEN	(* pointer base type *)
				IF typ.comp = Record THEN HrM.PropagateRecPtrSysFlag(typ.sysflag, struct.sysflag)
				ELSIF typ.comp IN {Array, DynArr} THEN HrM.PropagateArrPtrSysFlag(typ.sysflag, struct.sysflag)
				ELSE typ := HrCPT.undftyp; HrM.Mark(57, pos)
				END;
				struct.untagged := struct.sysflag > 0;
				IF (struct.strobj # NIL) & (struct.strobj.vis # internal) THEN typ.exp := TRUE END;
			ELSIF struct.comp = Array THEN	(* array base type *)
				CheckAlloc(typ, FALSE, pos);
				CheckRecursiveType(struct, typ, pos)
			ELSIF struct.comp = DynArr THEN	(* array base type *)
				CheckAlloc(typ, TRUE, pos);
				CheckRecursiveType(struct, typ, pos)
			ELSIF struct.comp = Record THEN	(* record base type *)	
			(*i don't need to replicate this, if I forbid forward base type references *)
				IF typ.form = Pointer THEN typ := typ.BaseTyp END;
				typ.pvused := TRUE; struct.extlev := SHORT(SHORT(typ.extlev + 1));
				HrM.PropagateRecordSysFlag(typ.sysflag, struct.sysflag);
				IF (typ.attribute = 0) OR (typ.attribute = limAttr) & (typ.mno # 0) THEN HrM.Mark(181, pos)
				ELSIF (struct.attribute = absAttr) & (typ.attribute # absAttr) THEN HrM.Mark(191, pos)
				ELSIF (typ.attribute = limAttr) & (struct.attribute # limAttr) THEN HrM.Mark(197, pos)
				END;
				f := struct.link;
				WHILE f # NIL DO	(* check for field name conflicts *)
					HrCPT.FindField(f.name, typ, bf);
					IF bf # NIL THEN HrM.Mark(1, pos) END;
					f := f.link
				END;
				CheckRecursiveType(struct, typ, pos);
				struct.untagged := struct.sysflag > 0;
			ELSIF struct.form = ProcTyp THEN	(* proc type return type *)
				IF typ.form = Comp THEN typ := HrCPT.undftyp; HrM.Mark(54, pos) END;
			ELSE HALT(100)
			END;
			struct.BaseTyp := typ
		END
	END FixType;

	PROCEDURE CheckForwardTypes;
		VAR u, next: Elem; progress: BOOLEAN;
	BEGIN
		u := userList; userList := NIL;
		WHILE u # NIL DO
			next := u.next; HrS.name := u.name$; HrCPT.Find(HrS.name, u.base);
			IF u.base = NIL THEN HrM.Mark(0, u.pos)
			ELSIF u.base.mode # Typ THEN HrM.Mark(72, u.pos)	(*72?! procedure must be bound to ...*)
			ELSE u.next := userList; userList := u	(* reinsert *)
			END;
			u := next
		END;
		REPEAT	(* iteration for multy level alias *)
			u := userList; userList := NIL; progress := FALSE;
			WHILE u # NIL DO
				next := u.next;
				IF IncompleteType(u.base.typ) THEN
					u.next := userList; userList := u	(* reinsert *)
				ELSE
					progress := TRUE;
					FixType(u.struct, u.obj, u.base.typ, u.pos)
				END;
				u := next
			END
		UNTIL (userList = NIL) OR ~progress;
		u := userList;	(* remaining type relations are cyclic *)
		WHILE u # NIL DO
			IF (u.obj = NIL) OR (u.obj.mode = Typ) THEN HrM.Mark(58, u.pos) END;
			u := u.next
		END;
	END CheckForwardTypes;

	PROCEDURE BuildPtrType (actual: T.Type): T.Pointer;
		VAR rp: T.RecPtr; ap: T.ArrayPtr; p: T.Pointer; 
	BEGIN ASSERT(actual # NIL);
		WITH actual: T.Record DO rp := T.dir.RecPtr(); rp.base := actual; p := rp (* Propagate sysflags *)
		| actual: T.Array DO ap := T.dir.ArrayPtr(); ap.base := actual; p := ap; 
			p.flags := HrM.MPropagateArrPtrSysFlag(ap.flags) (* Propagate sysflags *)
		ELSE err(57)
		END;
		RETURN p
	END BuildPtrType;
	
	PROCEDURE Resolve (VAR fwd: T.Type);
		VAR a: T.Type;
	BEGIN
		WITH fwd: T.Forward DO
			WHILE (fwd.actual # NIL) & (fwd.actual IS T.Forward) DO 
				fwd := fwd.actual(T.Forward) 
			END;
			IF (fwd.actual = NIL) & (fwd.base = NIL) THEN 
				IF ~(31 IN fwd.flags) THEN Mark(-128, 0, fwd.pos); INCL(fwd.flags, 31) END
			ELSIF fwd.base # NIL THEN fwd.actual := BuildPtrType(fwd.base.actual) 
			END;
			IF fwd.actual # NIL THEN a := fwd.actual ELSE a := T.errtp END
		ELSE (*ASSERT(fwd = T.errtp); *) Mark(-128, 0, fwd.pos); a := T.errtp
		END;
		(*INCL(a.typeName.flags, T.used);*)
		ASSERT(~(a IS T.Forward), 40);
		fwd := a
	END Resolve;
	
	PROCEDURE CheckForwardTypes2;
		CONST report = TRUE;
		VAR f: Fixup; o: T.Object; a, s, fwd: T.Type; tn: T.TypeName;
	BEGIN
		f := fixups; 
		WHILE f # NIL DO o := f.obj; a := NIL; tn := NIL;
			IF o # NIL THEN Resolve(o.type); a := o.type;
				WITH o: T.TypeName DO tn := o;
					IF o.type.typeName # NIL THEN a := NIL END
					(* Alias types are duplicated in old structure...*)
				| o: T.Var DO (* CheckAlloc *)
				| o: T.Field DO (* CheckAlloc *) 
					ASSERT((f.t # NIL) & (f.t IS T.Record), 20);
					CheckRecursiveType2(f.t, o.type, o.textpos)
				ELSE 
				END
			ELSIF f.t # NIL THEN s := f.t;
				WITH s: T.Array DO Resolve(s.elTyp); a := s.elTyp;	(* CheckAlloc *)
					CheckRecursiveType2(s, s.elTyp, s.pos)
				(*| s: T.Record DO CpjsM.Mark(eForwardBaseType, s.txtpos); HALT(1);*)
				| s: T.Procedure DO Resolve(s.res); a := s.res;
					IF (s.res IS T.Structure) THEN s.res := T.errtp; err(54) END
				| s: T.Forward DO ASSERT(s.actual # NIL); s.actual(T.Pointer).pos := s.pos 
				ELSE HALT(60)
				END
			END;
			IF a # NIL THEN IF a.typeName = NIL THEN T.RegisterType(a, tn) ELSE ASSERT(tn = NIL) END END;
			f := f.link
		END
	END CheckForwardTypes2;
	
	PROCEDURE CheckForwardProcedures;	(* in current scope T.nowScope *)
		VAR o, obj: T.Object; first: BOOLEAN; i, j: INTEGER; rec, base: T.Record; t: T.Type; p: T.TProc;
	BEGIN 
		obj := T.nowScope.head; WHILE obj # NIL DO
			WITH obj: T.Proc DO IF fwdFlag IN obj.flags THEN HrM.Mark(-129, obj.textpos) END
			ELSE 
			END;
			obj := obj.next
		END;
		i := 0; WHILE i < topRecs DO rec := recs[i]; base := rec.base; first := TRUE;
			(* recs contains only non-abstract records that directly extend abstract records *)
			WHILE base # NIL DO 
				j := 0; WHILE j < base.nproc DO p := base.proc[j];
					IF T.abstract IN p.flags THEN
						o := T.FindInTree(rec.tree, p.name);
						IF (o = NIL) OR ~(o IS T.TProc) THEN 
							IF first THEN HrM.Mark(192, rec.pos); first := FALSE END;	
							HrM.LogWLn; HrM.LogWStr("  "); 
							IF rec.typeName # NIL THEN HrM.LogWPar("#Dev:NotImplementedIn", SHORT(p.name^), 
									SHORT(rec.typeName.name^))
							ELSE HrM.LogWPar("#Dev:NotImplemented", SHORT(p.name^), "")
							END
						END
					END;
					INC(j)
				END;
				base := base.base
			END;
			INC(i)
		END
	END CheckForwardProcedures;
	
	PROCEDURE CheckUnimpl (m: HrCPT.Object; typ: HrCPT.Struct; pos: INTEGER);
		VAR obj: HrCPT.Object;
	BEGIN
		IF m # NIL THEN
			IF (m.mode = TProc) & (absAttr IN m.conval.setval) THEN
				HrCPT.FindField(m.name^, typ, obj);
				IF (obj = NIL) OR (obj.mode # TProc) OR (absAttr IN obj.conval.setval) THEN
					HrM.Mark(192, pos);
					HrM.LogWLn; HrM.LogWStr("  "); 
					IF typ.strobj # NIL THEN
						HrM.LogWPar("#Dev:NotImplementedIn", m.name, typ.strobj.name)
					ELSE
						HrM.LogWPar("#Dev:NotImplemented", m.name, "")
					END
				END
			END;
			CheckUnimpl(m.left, typ, pos);
			CheckUnimpl(m.right, typ, pos)
		END
	END CheckUnimpl;
	
	PROCEDURE CheckRecords (rec: Elem);
		VAR b: HrCPT.Struct;
	BEGIN
		WHILE rec # NIL DO	(* check for unimplemented methods in base type *)
			b := rec.struct.BaseTyp;
			WHILE (b # NIL) & (b # HrCPT.undftyp) DO
				CheckUnimpl(b.link, rec.struct, rec.pos);
				b := b.BaseTyp
			END;
			rec := rec.next
		END
	END CheckRecords;


	PROCEDURE CheckSym(s: SHORTINT);
	BEGIN
		IF sym = s THEN HrS.Get(sym) ELSE HrM.err(s) END
	END CheckSym;

	PROCEDURE qualident (VAR id: HrCPT.Object; VAR ob: T.Object);
		VAR obj: HrCPT.Object; lev: BYTE;
			o: T.Object; nst: INTEGER;
	BEGIN (*sym = ident*)
		HrCPT.Find(HrS.name, obj); o := T.Find(HrS.name$); 
		IF (obj # NIL) & (obj.mode # Attr) THEN ASSERT(o # NIL, 60) ELSIF obj = NIL THEN ASSERT(o = NIL, 60) END;
		HrS.Get(sym);
		IF (sym = period) & (obj # NIL) & (obj.mode = Mod) THEN
			HrS.Get(sym);
			IF sym = ident THEN
				HrCPT.FindImport(HrS.name, obj, obj); 
				IF o IS T.Mod THEN o := T.FindImport(o(T.Mod), HrS.name$) ELSE err(eNotAModule) END; 
				ASSERT((obj = NIL) = (o = NIL), 60);
				HrS.Get(sym)
			ELSE err(ident); obj := NIL; o := T.errid
			END
		END ;
		IF obj = NIL THEN err(0); Err(0, 1);
			obj := HrCPT.NewObj(); obj.mode := Var; obj.typ := HrCPT.undftyp; obj.adr := 0;
			o := T.errid
		ELSE lev := obj.mnolev;
			IF (obj.mode IN {Var, VarPar}) & (lev # level) THEN
				obj.leaf := FALSE;
				IF lev > 0 THEN HrB.StaticLink(SHORT(SHORT(level-lev)), TRUE) END	(* !!! *)
			END;
			IF (o # NIL) & (o IS T.Var) THEN 
				WITH o: T.Var DO nst := o.owner.nesting; IF nst # level THEN INCL(o.flags, T.ref); 
						IF nst > 0 THEN B.MarkNesting(level - nst, TRUE) END
					END
				END
			END
		END ;
		id := obj; ob := o
		;ASSERT((ob # NIL) OR (obj.mode = Attr));
	END qualident;

	PROCEDURE ConstExpression(VAR x: HrCPT.Node; OUT e: T.Expression; OUT lit: T.Literal);
		(*VAR e: T.Expression; lit: T.Literal;*)
	BEGIN Expression(x, e);
		IF x.class # Nconst THEN
			err(50); x := HrB.NewIntConst(1) 
		END;
		(*WITH e: T.Literal DO lit := e | e: T.ConDes DO lit := e.const.val ELSE err(-50); lit := T.errcon END;*)
		IF T.const IN e.flags THEN 
			WITH e: T.Literal DO lit := e
			| e: T.ConDes DO lit := e.const.val
			ELSE
				IF e.type.what IN T.integers THEN lit := T.ThisInt(HrB.EvalInt(e))
				ELSIF e.type.what = T.fSet THEN lit := T.ThisSet(HrB.EvalSet(e))
				ELSE HALT(126)
				END
			END
		ELSE Err(-50, 0); lit := T.errcon
		END
	END ConstExpression;
	
	PROCEDURE LibEnt (o: T.Object; IN n1, n2: T.Name);
		VAR lib, ent: POINTER TO T.Name;
	BEGIN

		WITH o: T.TProc DO o.lib := lib; o.ent := ent
		| o: T.LProc DO  o.lib := lib; o.ent := ent
		| o: T.CProc DO  o.lib := lib; o.ent := ent
		| o: T.XProc DO  o.lib := lib; o.ent := ent
		| o: T.GlobVar DO  o.lib := lib; o.ent := ent
		| o: T.TypeName DO  o.lib := lib; o.ent := ent
		| o: T.Const DO  o.lib := lib; o.ent := ent

		END
	END LibEnt;

	PROCEDURE CheckMark(obj: HrCPT.Object; symbol: INTEGER; OUT flags: SET; 
		OUT lib, ent: POINTER TO T.Name; OUT modifiers: POINTER TO ARRAY OF POINTER TO T.String
	);	(* !!! *)
		VAR n: INTEGER; mod: ARRAY 256 OF HrCPT.String;
			(*n1, n2: T.Name;*)
	BEGIN HrS.Get(sym); flags := {}; 
		IF (sym = times) OR (sym = minus) THEN
			IF (level > 0) OR ~(obj.mode IN {Var, Fld, TProc}) & (sym = minus) THEN err(41) END ;
			IF (level > 0) OR (sym = minus) & ~(symbol IN {sGVar, sField, sTProc}) THEN err(41) END;
			IF sym = times THEN obj.vis := external ELSE obj.vis := externalR END ;
			IF sym = times THEN INCL(flags, T.exported) ELSE INCL(flags, T.readOnly) END;
			HrS.Get(sym)
		ELSE obj.vis := internal
		END;
		IF (obj.mode IN {TProc, LProc, XProc, CProc, Var, Typ, Con, Fld}) & (sym = lbrak) THEN
		(* where is else? what error is reported? *)
		IF (sym = lbrak) & (symbol IN {sXProc, sTProc, sGVar, sType, sConst, sField}) THEN
			(* this seems to be all-encompassing list: all possible symbol types are allowed *)
			HrS.Get(sym);
			IF (sym = ident) & (HrS.name = "js") THEN INCL(flags, T.js); mod[0] := HrM.JS; n := 1; HrS.Get(sym) 
			ELSE n := 0 
			END;
			IF (sym = semicolon) OR (n = 0) THEN 
				IF sym = semicolon THEN HrS.Get(sym) END;
				IF (sym = number) & (HrS.numtyp = char) THEN
					NEW(HrS.str, 2); HrS.str[0] := SHORT(CHR(HrS.intval)); HrS.str[1] := 0X; sym := string
				END;
				IF sym = string THEN (*n1 := ''; n2 := '';*)
					IF HrS.str^ # "" THEN obj.entry := HrS.str; NEW(ent); ent^ := HrS.str$ END;
					HrS.Get(sym); (*n := 0;*)
					IF (sym = comma) & (obj.mode IN {LProc, XProc, CProc, Var, Con}) THEN
					IF (sym = comma) & (symbol IN {sXProc, sCProc, sGVar, sConst}) THEN
						HrS.Get(sym);
						IF (sym = number) & (HrS.numtyp = char) THEN
							NEW(HrS.str, 2); HrS.str[0] := SHORT(CHR(HrS.intval)); HrS.str[1] := 0X; sym := string
						END;
						IF sym = string THEN
							obj.library := obj.entry; obj.entry := NIL;
							IF HrS.str^ # "" THEN obj.entry := HrS.str; lib := ent; NEW(ent); ent^ := HrS.str$
							END;
							HrS.Get(sym);
						ELSE err(string)
						END
					END;
					END;
					(*IF n2 # '' THEN IF n1 # '' THEN LibEnt(o, n1, n2) END
					ELSE IF n1 # '' THEN LibEnt(o, '', n1) END
					END;*)
					WHILE sym = comma DO
						HrS.Get(sym);
						IF (sym = number) & (HrS.numtyp = char) THEN
							NEW(HrS.str, 2); HrS.str[0] := SHORT(CHR(HrS.intval)); HrS.str[1] := 0X; sym := string
						END;
						IF sym = string THEN
							IF n < LEN(mod) THEN mod[n] := HrS.str; INC(n)
							ELSE err(235)
							END;
							HrS.Get(sym)
						ELSE err(string)
						END
					END;
				ELSE err(string)
				END;
			END;
			IF n > 0 THEN
				NEW(obj.modifiers, n);
				NEW(modifiers, n);
				WHILE n > 0 DO DEC(n); obj.modifiers[n] := mod[n] (*END;
				WHILE n > 0 DO DEC(n)*); NEW(modifiers[n], 1 + LEN(mod[n]$)); modifiers[n]^ := mod[n]$ END
			END;
			CheckSym(rbrak);
			IF HrM.options * {HrM.interface, HrM.java, HrM.javascript} = {} THEN err(225) END
		END
		END
	END CheckMark;

	PROCEDURE CheckSysFlag (VAR sysflag: SHORTINT;
		GetSF: PROCEDURE(IN id: ARRAY OF SHORTCHAR; num: SHORTINT; VAR flag: SHORTINT)
	);
		VAR x: HrCPT.Object; i: SHORTINT;
			o: T.Object;
	BEGIN
		sysflag := 0;
		IF sym = lbrak THEN
			HrS.Get(sym);
			WHILE (sym = number) OR (sym = ident) OR (sym = string) DO
				IF sym = number THEN
					IF HrS.numtyp = integer THEN
						i := SHORT(HrS.intval); GetSF("", i, sysflag)
					ELSE err(-1225)
					END
				ELSIF sym = ident THEN
					HrCPT.Find(HrS.name, x);
					o := T.Find(HrS.name$);
					IF (x # NIL) & (x.mode = Con) & (x.typ.form IN {Int8, Int16, Int32}) THEN 
					IF (o # NIL) & (o IS T.Const) & (o(T.Const).val IS T.Int) THEN 
						i := SHORT(SHORT(o(T.Const).val(T.Int).long)); GetSF("", i, sysflag)	(*potential trap*)
					END;
						i := SHORT(x.conval.intval); GetSF("", i, sysflag)
					ELSE
						GetSF(HrS.name, 0, sysflag)
					END
				ELSE
					GetSF(HrS.str, 0, sysflag)
				END;
				HrS.Get(sym);
				IF (sym = comma) OR (sym = plus) THEN HrS.Get(sym) END
			END;
			CheckSym(rbrak)
		END
	END CheckSysFlag;

	PROCEDURE Receiver (VAR mode, vis: BYTE; VAR name: HrCPT.Name; VAR typ, rec: HrCPT.Struct;
		OUT rcv: T.Par; OUT record: T.Record
	);
		VAR tname: HrCPT.String;
			t: T.Type;
	BEGIN typ := HrCPT.undftyp; rec := NIL; vis := 0;
		rcv := T.dir.Par();
		IF sym = var THEN HrS.Get(sym); mode := VarPar; rcv.flags := T.varK
		ELSIF sym = in THEN HrS.Get(sym); mode := VarPar; vis := inPar; rcv.flags := T.inK	(* ??? *)
		ELSIF sym = out THEN HrS.Get(sym); mode := VarPar; rcv.flags := T.varK; err(70)
		ELSE mode := Var; rcv.flags := T.valK
		END ;
		rcv.name := ThisName(HrS.name); rcv.textpos := HrM.startpos;
		name := HrS.name; CheckSym(ident); CheckSym(colon);
		IF sym # ident THEN err(ident) END;
		Type(typ, tname, t, NIL);
		IF tname = NIL THEN
			IF typ.form = Pointer THEN  rec := typ.BaseTyp ELSE rec := typ END;
			IF ~((mode = Var) & (typ.form = Pointer) & (rec.comp = Record) OR
				(mode = VarPar) & (typ.comp = Record)) THEN err(70); rec := NIL END;
			IF (rec # NIL) & (rec.mno # level) THEN err(72); rec := NIL END
		ELSE err(0); Err(0, 0)
		END;
		WITH t: T.Record DO record := t | t: T.RecPtr DO record := t.base 
		ELSE IF t IS T.Forward THEN err(298) ELSE err(70) END; t := T.errtp
		END;
		IF (record # NIL) & (record.typeName.owner.nesting # 0) THEN err(-299) END;
		rcv.type := t;
		CheckSym(rparen);
		IF rec = NIL THEN rec := HrCPT.NewStr(Comp, Record); rec.BaseTyp := NIL END
	END Receiver;
	
	PROCEDURE FormalParameters (
		VAR firstPar: HrCPT.Object; VAR resTyp: HrCPT.Struct; VAR name: HrCPT.String;
		rcv: T.Par; checkMatch: BOOLEAN; fp: POINTER TO ARRAY OF T.Par; fres: T.Type; checkNames: BOOLEAN; 
		OUT pars: POINTER TO ARRAY OF T.Par; OUT res: T.Type
	);
		VAR mode, vis: BYTE; sys: SHORTINT;
				par, first, last, newPar, iidPar: HrCPT.Object; typ: HrCPT.Struct;
				fplen, i, num, sameType: INTEGER; flags, kind: SET; p: T.Par; type: T.Type;
				pp: ARRAY 32 OF T.Par;
		PROCEDURE CheckKindName;
		BEGIN
			IF num > fplen THEN
			ELSIF num = fplen THEN HrM.Mark(294, p.textpos)
			ELSE IF checkNames & (fp[num].name^ # p.name^) THEN HrM.Mark(291, p.textpos) END;
				IF kind # fp[num].flags * T.kindMask THEN HrM.Mark(290, p.textpos) END
			END
		END CheckKindName;
		
	BEGIN IF fp = NIL THEN fplen := 0 ELSE fplen := LEN(fp) END;
		num := 0; sameType := 0;
		IF rcv # NIL THEN INCL(rcv.flags, T.receiver); pp[0] := rcv; 
			IF checkMatch THEN p := rcv; kind := rcv.flags * T.kindMask; CheckKindName END; 
			num := 1; sameType := 1
		ELSE num := 0; sameType := 0
		END;
		IF sym = lparen THEN HrS.Get(sym);
			first := NIL; last := firstPar;
			newPar := NIL; iidPar := NIL;
			IF (sym = ident) OR (sym = var) OR (sym = in) OR (sym = out) THEN
				LOOP 
					sys := 0; vis := 0; flags := {};
					IF sym = var THEN HrS.Get(sym); mode := VarPar; kind := T.varK
					ELSIF sym = in THEN HrS.Get(sym); mode := VarPar; vis := inPar; kind := T.inK
					ELSIF sym = out THEN HrS.Get(sym); mode := VarPar; vis := outPar; kind := T.outK
					ELSE mode := Var; kind := T.valK
					END ;
					IF mode = VarPar THEN CheckSysFlag(sys, HrM.GetVarParSysFlag) END;
					IF ODD(sys DIV nilBit) THEN flags := flags + T.nilPF END;
					IF ODD(sys DIV inBit) THEN vis := inPar; kind := T.inK
					ELSIF ODD(sys DIV outBit) THEN vis := outPar; kind := T.outK
					END;
					IF ODD(sys DIV newBit) & (vis # outPar) THEN err(225); HALT(126)	(* com-related *)
					ELSIF ODD(sys DIV iidBit) & (vis # inPar) THEN err(225); HALT(126)
					END;
					LOOP
						IF sym = ident THEN 
							HrCPT.Insert(HrS.name, par); 
							p := T.dir.Par(); NEW(p.name); p.name^ := HrS.name$; p.flags := flags + kind;
							i := 0; WHILE (i < num) & (pp[i].name^ # p.name^) DO INC(i) END;
							IF i < num THEN err(1) END;
							IF checkMatch THEN CheckKindName END;
							pp[num] := p; p.num := num; INC(num);
							(*IF sameType = none THEN sameType := num END;*)
							HrS.Get(sym); par.textpos := HrM.startpos;
							p.textpos := HrM.startpos;
							par.mode := mode; par.link := NIL; par.vis := vis; par.sysflag := SHORT(sys);
							IF first = NIL THEN first := par END ;
							IF firstPar = NIL THEN firstPar := par ELSE last.link := par END ;
							last := par;
						ELSE err(ident)
						END;
						IF sym = comma THEN HrS.Get(sym)
						ELSIF sym = ident THEN err(comma)
						ELSIF sym = var THEN err(comma); HrS.Get(sym)
						ELSE EXIT
						END
					END ;
					CheckSym(colon); Type(typ, name, type, NIL);
					IF mode # VarPar THEN CheckAlloc(typ, TRUE, HrM.errpos) END;	(*?*)
					(* IN only allowed for records and arrays *)
					IF (mode = VarPar) & (vis = inPar) & (typ.form # Undef) & (typ.form # Comp) & (typ.sysflag = 0) THEN 
						err(177)
					END;
					IF (kind = T.inK) & ~(type IS T.Record) & ~(type IS T.Array) & (type.flags = {}) THEN err(177) END;
					(* typ.pbused is set when parameter type name is parsed *)
					WHILE first # NIL DO
						SetType (NIL, first, typ, name);
						IF HrM.com IN HrM.options THEN	(*? no idea what this is all about...*)
							IF ODD(sys DIV newBit) THEN
								IF (newPar # NIL) OR (typ.form # Pointer) OR (typ.sysflag # interface) THEN err(168) END;
								newPar := first
							ELSIF ODD(sys DIV iidBit) THEN
								IF (iidPar # NIL) OR (typ # HrCPT.guidtyp) THEN err(168) END;
								iidPar := first
							END
						END;
						first := first.link
					END;
					WHILE sameType < num DO SetType2(NIL, pp[sameType], type, name); 
						IF checkMatch & (sameType < fplen) & ~T.EqualTypes(fp[sameType].type, type) THEN err(295) END;
						INC(sameType) 
					END;
					IF sym = semicolon THEN HrS.Get(sym)
					ELSIF sym = ident THEN err(semicolon)
					ELSE EXIT
					END
				END
			END;
			CheckSym(rparen);
			IF (newPar = NIL) # (iidPar = NIL) THEN err(168) END;
			name := NIL;
			IF sym = colon THEN
				HrS.Get(sym);
				Type(resTyp, name, res, NIL);
				IF resTyp.form = Comp THEN resTyp := HrCPT.undftyp; err(54) END;
				IF (res IS T.Record) OR (res IS T.Array) THEN res := T.errtp; err(-54) END
			ELSE resTyp := HrCPT.notyp
			END;
		END;
			IF checkMatch & ((res = NIL) # (fres = NIL) OR (res # NIL) & (fres # NIL) & 
				((~T.EqualTypes(res, fres) & ((rcv = NIL) OR ~T.Is(res, fres)))))
			THEN err(-117); IF (res = NIL) & (fres # NIL) THEN res := fres END
			END;
		IF num > 0 THEN NEW(pars, num); REPEAT DEC(num); pars[num] := pp[num] UNTIL num = 0 END
	END FormalParameters;
	
	PROCEDURE CheckOverwrite (proc, base: HrCPT.Object; rec: HrCPT.Struct; baseProc: T.TProc);
		VAR o, bo: HrCPT.Object;
	BEGIN
		IF base # NIL (* baseProc # NIL *) THEN
			IF base.conval.setval * {absAttr, empAttr, extAttr} = {} THEN err(182) END;
			IF baseProc.flags * {T.abstract, T.empty, T.extensible} = {} THEN err(182) END;
			IF (proc.link.mode # base.link.mode) OR (proc.link.vis # base.link.vis)
				OR ~HrCPT.Extends(proc.link.typ, base.link.typ) THEN err(115) END;	(*hm.. ~Extends is impossible *)
			o := proc.link; bo := base.link;
			WHILE (o # NIL) & (bo # NIL) DO
				IF (bo.sysflag # 0) & (o.sysflag = 0) THEN	(* propagate sysflags *)
					o.sysflag := bo.sysflag	(*?*)
				END;
				o := o.link; bo := bo.link
			END;
			HrB.CheckParameters(proc.link.link, base.link.link, FALSE);
			IF ~HrCPT.Extends(proc.typ, base.typ) THEN err(117) END;
			IF (base.vis # proc.vis) & ((proc.vis # internal) OR rec.exp) THEN err(183) END;
			INCL(proc.conval.setval, isRedef)
		END;
	END CheckOverwrite;

	PROCEDURE GetAttributes (proc, base: HrCPT.Object; owner: HrCPT.Struct;
		oproc, obase: T.TProc; rec: T.Record
	);	(* read method attributes *)
		VAR attr, battr: SET; o: HrCPT.Object;
			obj: T.Object; p: POINTER TO ARRAY OF T.Par; i, len: INTEGER; VAR oattr: SET;
	BEGIN
		attr := {}; oattr := {};
		IF sym = comma THEN	(* read attributes *)
			HrS.Get(sym);
			IF sym = ident THEN
				HrCPT.Find(HrS.name, o);
				IF HrS.name$ = "NEW" THEN INCL(oattr, T.new) ELSE err(1178) END;
				IF (o # NIL) & (o.mode = SProc) & (o.adr = newfn) THEN
					IF ~(HrM.oberon IN HrM.options) THEN INCL(attr, newAttr) ELSE err(178) END;
					HrS.Get(sym);
					IF sym = comma THEN
						HrS.Get(sym);
						IF sym = ident THEN HrCPT.Find(HrS.name, o);
							IF HrS.name$ = "ABSTRACT" THEN INCL(oattr, T.abstract)
							ELSIF HrS.name$ = "EXTENSIBLE" THEN INCL(oattr, T.extensible)
							ELSIF HrS.name$ = "EMPTY" THEN INCL(oattr, T.empty)
							END
						ELSE o := NIL; err(ident) 
						END
					ELSE o := NIL
					END
				END; 
				IF o # NIL THEN
					IF (o.mode # Attr) OR (o.adr = limAttr) OR (HrM.oberon IN HrM.options) THEN err(178)
					ELSE INCL(attr, o.adr)
					END;
					HrS.Get(sym)
				END
			ELSIF sym = new THEN INCL(oattr, T.new)
			ELSIF sym = abstract THEN INCL(oattr, T.abstract)
			ELSIF sym = extensible THEN INCL(oattr, T.extensible)
			ELSIF sym = empty THEN INCL(oattr, T.empty)
			ELSE err(ident)
			END;
			IF (oattr # {}) & (HrM.oberon IN HrM.options) THEN err(178) END
		END;
		IF (base = NIL) & ~(newAttr IN attr) THEN err(185); INCL(attr, newAttr)
		ELSIF (base # NIL) & (newAttr IN attr) THEN err(186)
		END;
		IF (obase = NIL) & ~(T.new IN oattr) THEN err(185); INCL(oattr, T.new)
		ELSIF (obase # NIL) & (T.new IN oattr) THEN err(186)
		END;
		IF absAttr IN attr THEN
			IF owner.attribute # absAttr THEN err(190) END;
			IF (proc.vis = internal) & owner.exp THEN err(179) END
		END;
		IF T.abstract IN oattr THEN
			IF ~(T.abstract IN rec.flags) THEN err(190) END;
			IF ~(T.exported IN oproc.flags) & (T.exported IN rec.flags) THEN err(179) END
		END;
		IF (owner.attribute = 0) OR (owner.attribute = limAttr) THEN
			IF (empAttr IN attr) & (newAttr IN attr) THEN err(187)
(*
			ELSIF extAttr IN attr THEN err(188)
*)
			END
		END;
		IF (rec # NIL) & ({} = rec.flags * { T.abstract, T.extensible }) THEN
			IF oattr * {T.new, T.empty} = {T.new, T.empty} THEN err(187) 
			ELSIF T.extensible IN oattr THEN err(188) 
			END
		END;
		IF base # NIL THEN
			battr := base.conval.setval;
			IF empAttr IN battr THEN
				IF absAttr IN attr THEN err(189) END
			ELSIF ~(absAttr IN battr) THEN
				IF (absAttr IN attr) OR (empAttr IN attr) THEN err(189) END
			END
		END;
		IF obase # NIL THEN
			IF (T.empty IN obase.flags) & (T.abstract IN oattr)
			OR ~(T.abstract IN obase.flags) & ({} # oattr * {T.abstract, T.empty}) THEN err(189)
			END
		END;
		IF empAttr IN attr THEN
			IF proc.typ # HrCPT.notyp THEN err(195)
			ELSE
				o := proc.link; WHILE (o # NIL) & (o.vis # outPar) DO o := o.link END;
				IF o # NIL THEN err(195) END
			END
		END;
		IF (T.empty IN oattr) THEN
			IF oproc.type # NIL THEN err(-195) END;
			p := oproc.par;
			IF p # NIL THEN len := LEN(p);
				i := 0; WHILE i < len DO 
					IF p[i].flags * T.kindMask = T.outK THEN HrM.Mark(-195, p[i].textpos) END; INC(i)
				END
			END
		END;
		IF (owner.sysflag = interface) & ~(absAttr IN attr) THEN err(162) END;
		IF (rec # NIL) & (T.interface IN rec.flags) & ~(T.abstract IN oattr) THEN err(162) END;
		proc.conval.setval := attr;
		oproc.flags := oproc.flags + oattr
	END GetAttributes;
	
	PROCEDURE RecordType(VAR typ: HrCPT.Struct; attr: HrCPT.Object; att: INTEGER; OUT rec: T.Record);
		VAR fld, first, last: HrCPT.Object; r: Elem; ftyp: HrCPT.Struct; name: HrCPT.String;
			base: T.Record; t: T.Type; rc: RecCheck;
			flds: ARRAY 64 OF T.Field; num, sameType: INTEGER; f: T.Field;
	BEGIN typ := HrCPT.NewStr(Comp, Record); typ.BaseTyp := NIL;
		rec := T.dir.Record(); rec.flags := {}; rec.pos := HrM.startpos;
		CheckSysFlag(typ.sysflag, HrM.GetRecordSysFlag);
		IF typ.sysflag > 0 THEN INCL(rec.flags, T.untagged); 
			CASE typ.sysflag OF 1: ELSE HALT(126) END
		END;
		IF attr # NIL THEN
			IF ~(HrM.oberon IN HrM.options) & (attr.adr # empAttr) THEN typ.attribute := SHORT(SHORT(attr.adr))
			ELSE err(178)
			END
		END;
		ASSERT(att IN {0, T.new, T.abstract, T.extensible, T.limited}, 20);
		IF (att # 0) & ~(HrM.oberon IN HrM.options) THEN INCL(rec.flags, att) 
		ELSIF att # 0 THEN err(178)
		END;
		IF typ.sysflag = interface THEN	(* ? *)
			IF (HrS.str # NIL) & (HrS.str[0] = "{") THEN typ.ext := HrS.str END;
			IF typ.attribute # absAttr THEN err(163) END;
			IF sym # lparen THEN err(160) END
		END;
		IF sym = lparen THEN
			HrS.Get(sym); (* record extension *)
			IF sym = ident THEN
				Type(ftyp, name, t, NIL);
				IF ftyp.form = Pointer THEN ftyp := ftyp.BaseTyp END;
				IF t # NIL THEN 
					WITH t: T.RecPtr DO base := t.base | t: T.Record DO base := t 
					| t: T.Forward DO err(eForwardBaseType)
					ELSE err(53)(*might be a forward declaration*)
					(* couldn't we forbid (yet) undeclared type names in base type specification? *)
					END
				END;
				IF base # NIL THEN SetType2(rec, NIL, base, name) END;
				SetType(typ, NIL, ftyp, name);
				IF (ftyp.comp = Record) & (ftyp # HrCPT.anytyp) THEN
					ftyp.pvused := TRUE; typ.extlev := SHORT(SHORT(ftyp.extlev + 1));
					HrM.PropagateRecordSysFlag(ftyp.sysflag, typ.sysflag);
					IF (ftyp.attribute = 0) OR (ftyp.attribute = limAttr) & (ftyp.mno # 0) THEN err(181)
					ELSIF (typ.attribute = absAttr) & (ftyp.attribute # absAttr) & ~(HrM.java IN HrM.options) THEN err(191)
					ELSIF (ftyp.attribute = limAttr) & (typ.attribute # limAttr) THEN err(197)
					END
				ELSIF ftyp # HrCPT.undftyp THEN err(53)
				END;
				IF base # NIL THEN INCL(base.flags, T.pvused); rec.level := base.level + 1;
					(* PropagateRecordSysflag *)
					IF (att = 0) & (ftyp.attribute = 0) OR (att = T.limited) & (base.typeName.imno # 0) THEN err(181)
					ELSIF (att = T.abstract) & (T.abstract IN base.flags) & ~(HrM.java IN HrM.options) THEN err(191)
					ELSIF (T.limited IN base.flags) & ~(T.limited IN rec.flags) THEN err(197)
					END
				END
			ELSE err(ident)
			END ;
			IF typ.attribute # absAttr THEN	(* save typ for unimplemented method check *)
				NEW(r); r.struct := typ; r.pos := HrM.errpos; r.next := recList; recList := r
			END;
			IF att # T.abstract THEN 
				NEW(rc); rc.rec := rec; rc.pos := HrM.errpos; rc.link := recChecks; recChecks := rc
			END;
			CheckSym(rparen)
		END;
(*
		CpjsCPT.OpenScope(0, NIL);
*)
		first := NIL; last := NIL;
		num := 0; 
		LOOP sameType := none;
			IF sym = ident THEN
				LOOP
					IF sym = ident THEN
						IF (typ.BaseTyp # NIL) & (typ.BaseTyp # HrCPT.undftyp) THEN
							HrCPT.FindBaseField(HrS.name, typ, fld);
							IF fld # NIL THEN err(1) END
						END ;
						IF (base # NIL) & (T.FindInRec(base, HrS.name$) # NIL) THEN err(1) END;
						HrCPT.InsertField(HrS.name, typ, fld);
						f := T.dir.Field(); NEW(f.name); f.name^ := HrS.name$; f.textpos := HrM.startpos;
						flds[num] := f; IF sameType = none THEN sameType := num END; INC(num); 
						fld.mode := Fld; fld.link := NIL; fld.typ := HrCPT.undftyp; fld.textpos := HrM.startpos;
						CheckMark(fld, sField, f.flags, mark.lib, mark.ent, f.modifiers);
						IF first = NIL THEN first := fld END ;
						IF last = NIL THEN typ.link := fld ELSE last.link := fld END ;
						last := fld
					ELSE err(ident)
					END ;
					IF sym = comma THEN HrS.Get(sym)
					ELSIF sym = ident THEN err(comma)
					ELSE EXIT
					END
				END ;
				CheckSym(colon); Type(ftyp, name, t, NIL); 
				CheckAlloc(ftyp, FALSE, HrM.errpos);	(*?*)
				WHILE first # NIL DO
					SetType(typ, first, ftyp, name); first := first.link
				END;
				WHILE sameType < num DO SetType2(rec, flds[sameType], t, name); INC(sameType) END;
				IF typ.sysflag = interface THEN err(161) END	(*?*)
			END;
			IF sym = semicolon THEN HrS.Get(sym)
			ELSIF sym = ident THEN err(semicolon)
			ELSE EXIT
			END
		END;
		IF num > 0 THEN NEW(rec.fields, num); 
			REPEAT DEC(num); rec.fields[num] := flds[num]; T.Insert(rec.tree, flds[num]) UNTIL num = 0 
		END;
(*
		IF typ.link # NIL THEN ASSERT(typ.link = CpjsCPT.topScope.right) END;
		typ.link := CpjsCPT.topScope.right; CpjsCPT.CloseScope;
*)
		typ.untagged := typ.sysflag > 0;
		HrB.Inittd(TDinit, lastTDinit, typ); (*?*)
		CheckSym(end);
		IF (rec # NIL) & ~(T.abstract IN rec.flags) & (base # NIL) & (T.abstract IN base.flags) THEN 
			IF topRecs = capRecs THEN err(234) ELSE recs[topRecs] := rec; INC(topRecs) END
		END
	END RecordType;

	PROCEDURE ArrayType(VAR typ: HrCPT.Struct; OUT arr: T.Array);
		VAR x: HrCPT.Node; n(*, eFlag*): INTEGER; sysflag: SHORTINT; name: HrCPT.String;
			el: T.Type; dim: T.Literal; arrEl: T.Array; _: T.Expression;
	BEGIN sysflag := 0; CheckSysFlag(sysflag, HrM.GetArraySysFlag); 
		arr := T.dir.Array(); (*arr.mno := 1;*) (*?arr.sysflag := sysflag*)
		IF sysflag > 0 THEN INCL(arr.flags, T.untagged); CASE sysflag OF 1: ELSE HALT(126) END END;
		IF sym = of THEN	(* open array *)
			typ := HrCPT.NewStr(Comp, DynArr); typ.mno := 0; typ.sysflag := sysflag;
			HrS.Get(sym); Type(typ.BaseTyp, name, el, NIL); SetType(typ, NIL, typ.BaseTyp, name);
			SetType2(arr, NIL, el, name);
			CheckAlloc(typ.BaseTyp, TRUE, HrM.errpos);	(*?*)
			IF typ.BaseTyp.comp = DynArr THEN typ.n := typ.BaseTyp.n + 1 ELSE typ.n := 0 END;
			INCL(arr.flags, T.open);
			IF (el IS T.Array) & (T.open IN el.flags) THEN arr.len := el(T.Array).len + 1 ELSE arr.len := 0 END
		ELSE
			typ := HrCPT.NewStr(Comp, Array); typ.sysflag := sysflag; ConstExpression(x, _, dim);
			IF x.typ.form IN {Int8, Int16, Int32} THEN n := x.conval.intval;
				IF (n <= 0) OR (n > HrM.MaxIndex) THEN err(63); n := 1 END
			ELSE err(42); n := 1
			END ;
			typ.n := n;
			WITH dim: T.Int DO n := SHORT(dim.long);
				IF (n <= 0) OR (n > HrM.MaxIndex) THEN err(-63); n := 1 END
			ELSE n := 1; err(42) 
			END;
			arr.len := n;
			IF sym = of THEN
				HrS.Get(sym); Type(typ.BaseTyp, name, el, NIL); SetType(typ, NIL, typ.BaseTyp, name);
				SetType2(arr, NIL, el, name);
				CheckAlloc(typ.BaseTyp, FALSE, HrM.errpos)	(*?*)
			ELSIF sym = comma THEN
				HrS.Get(sym);
				IF sym # of THEN ArrayType(typ.BaseTyp, arrEl); arr.elTyp := arrEl END
			ELSE err(35)
			END
		END;
		(*IF (CpjsM.js = sysflag) & ~(typ.BaseTyp.form IN charSet) THEN CpjsM.errpos := eFlag; err(CpjsM.eJString) END;*)
		typ.untagged := typ.sysflag > 0
	END ArrayType;

	PROCEDURE PointerType(VAR typ: HrCPT.Struct; OUT p: T.Type; register: T.TypeName);
		VAR name: HrCPT.String; base: T.Type; tn: T.TypeName;
	BEGIN typ := HrCPT.NewStr(Pointer, Basic); CheckSysFlag(typ.sysflag, HrM.GetPointerSysFlag);
		CheckSym(to);
		IF (register # NIL) THEN tn := T.dir.TypeName(); NEW(tn.name); tn.name^ := register.name + '^' END;
		Type(typ.BaseTyp, name, base, tn);
		SetType(typ, NIL, typ.BaseTyp, name);
		IF (typ.BaseTyp # HrCPT.undftyp) & (typ.BaseTyp.comp = Basic) THEN
			typ.BaseTyp := HrCPT.undftyp; err(57)
		END;
		ASSERT(base # NIL);
		WITH | base: T.Record DO p := T.dir.RecPtr(); ASSERT(typ.BaseTyp.sysflag = 0, 126)
		| base: T.Array DO p := T.dir.ArrayPtr(); p.flags := HrM.MPropagateArrPtrSysFlag(base.flags)
		| base: T.Forward DO p := T.dir.Forward()(*; p.pos := CpjsM.startpos*)
		ELSE err(57); base := T.errtp
		END;
		SetType2(p, NIL, base, name); 
		IF typ.sysflag > 0 THEN INCL(p.flags, T.untagged); CASE typ.sysflag OF 1: ELSE HALT(126) END END;
		IF typ.BaseTyp.comp = Record THEN HrM.PropagateRecPtrSysFlag(typ.BaseTyp.sysflag, typ.sysflag)
		ELSIF typ.BaseTyp.comp IN {Array, DynArr} THEN HrM.PropagateArrPtrSysFlag(typ.BaseTyp.sysflag, typ.sysflag)
		END;
		(* Propagate sysflags *)
		typ.untagged := typ.sysflag > 0
	END PointerType;
	
	PROCEDURE Type (VAR typ: HrCPT.Struct; VAR name: HrCPT.String; OUT type: T.Type; register: T.TypeName);
	(* name # NIL => forward reference *)
		VAR id: HrCPT.Object; tname: HrCPT.String;
			qd: T.Object; rec: T.Record; attr, pos: INTEGER; arr: T.Array; proc: T.Procedure; fwd: T.Forward; 
			tn: T.TypeName; base: T.Type;
	BEGIN pos := HrM.startpos;
		typ := HrCPT.undftyp; name := NIL;
		IF sym < lparen THEN err(12);
			REPEAT HrS.Get(sym) UNTIL sym >= lparen
		END ;
		IF sym = ident THEN
			HrCPT.Find(HrS.name, id);
			qd := T.Find(HrS.name$);
			IF (id = NIL) OR (id.mode = -1) OR (id.mode = Typ) & IncompleteType(id.typ) THEN	(* forward type decl *)
				IF (qd # NIL) & ~(qd IS T.Mod) (*forward type decl*) THEN type := qd.type END;
				name := HrCPT.NewName(HrS.name); HrS.Get(sym); 
				IF (id = NIL) (*qd = NIL*)& (sym = period) THEN	(* missing module *)
					err(0); type := T.errtp; HrS.Get(sym); name := NIL;
					IF sym = ident THEN HrS.Get(sym) END
				ELSIF (qd = NIL) & (sym # record) THEN 
					IF unknownTypesAllowed THEN
						fwd := T.dir.Forward(); fwd.pos := pos; type := fwd; 
						tn := T.dir.TypeName(); fwd.typeName := tn; tn.type := fwd;
						T.AppendThis(tn, ThisName(name$), pos)
					ELSE err(0); type := T.errtp
					END
				ELSIF sym = record THEN	(* wrong attribute *)
					err(178); HrS.Get(sym); name := NIL; RecordType(typ, NIL, 0, rec); type := rec
				END
			ELSE
				qualident(id, qd);
				IF (id # NIL) & (id.mode # Attr) THEN
					IF qd = NIL THEN err(0) 
					ELSIF qd IS T.TypeName THEN type := qd.type; ASSERT(type # NIL);
						IF ~(HrM.oberon IN HrM.options)
						& ((type = T.lreal64typ) OR (type = T.lint64typ) OR (type = T.lchar16typ)) THEN err(-198)
						END;
					ELSE err(-52) 
					END
				END;
				IF id.mode = Typ THEN
					IF ~(HrM.oberon IN HrM.options)
						& ((id.typ = HrCPT.lreal64typ) OR (id.typ = HrCPT.lint64typ) OR (id.typ = HrCPT.lchar16typ))
					THEN err(198)
					END;
					typ := id.typ
				ELSIF id.mode = Attr THEN
					IF sym = record THEN
						HrS.Get(sym); 
						CASE id.adr OF 0: attr := 0 
						| absAttr: attr := T.abstract | limAttr: attr := T.limited | extAttr: attr := T.extensible 
						ELSE err(-178)
						END;
						RecordType(typ, id, attr, rec); type := rec
					ELSE err(12)
					END
				ELSE err(52)
				END;
			END
		ELSE
			IF sym = array THEN
				HrS.Get(sym); ArrayType(typ, arr); type := arr; type.pos := pos
			ELSIF (*(sym = record) OR*) (sym = abstract) OR (sym = extensible) OR (sym = limited) THEN
				attr := 0; 
				CASE sym OF record: attr := 0 | abstract: attr := T.abstract | limited: attr := T.limited 
				| extensible: attr := T.extensible 
				END;
				IF sym # record THEN HrS.Get(sym); CheckSym(record) ELSE HrS.Get(sym) END;
				RecordType(typ, NIL, attr, rec); type := rec; type.pos := pos
			ELSIF sym = record THEN
				HrS.Get(sym); RecordType(typ, NIL, 0, rec); type := rec; type.pos := pos
			ELSIF sym = pointer THEN 
				HrS.Get(sym); PointerType(typ, type, register); type.pos := pos
			ELSIF sym = procedure THEN
				HrS.Get(sym); typ := HrCPT.NewStr(ProcTyp, Basic);
				CheckSysFlag(typ.sysflag, HrM.GetProcTypSysFlag);
				proc := T.dir.Procedure(); type := proc;
				CASE typ.sysflag OF 0: ELSE HALT(126) END;
				typ.untagged := typ.sysflag > 0;
				IF sym = lparen THEN
					(*CpjsS.Get(sym);*) HrCPT.OpenScope(level, NIL); 
					FormalParameters(typ.link, typ.BaseTyp, tname, NIL, FALSE, NIL, NIL, FALSE, proc.par, proc.res);
					SetType(typ, NIL, typ.BaseTyp, tname);
					SetType2(proc, NIL, proc.res, tname);
					HrCPT.CloseScope
				ELSE typ.BaseTyp := HrCPT.notyp; typ.link := NIL
				END;
				type.pos := pos
			ELSE err(12); type := T.errtp
			END;
			(*IF (type.typeName = NIL) & (type # T.errtp) & (register = NIL) THEN 
				register := T.dir.TypeName(); T.NameAnon(register.name); register.textpos := pos
			END*)
		END ;
		LOOP
			IF (sym >= semicolon) & (sym <= else) OR (sym = rparen) OR (sym = eof)
				OR (sym = number) OR (sym = comma) OR (sym = string) THEN EXIT END;
			err(15); IF sym = ident THEN EXIT END;
			HrS.Get(sym)
		END;
		ASSERT(type # NIL);
		IF type IS T.Forward THEN (*ASSERT(type.typeName # NIL); *)
			IF register # NIL THEN register.type := type; T.Append(register) END
			;IF register # NIL THEN ASSERT(register.name^ # "AADDR", 99) END
		ELSE 
			IF type.typeName = NIL (* not an alias *) THEN T.RegisterType(type, register) 
			ELSIF register # NIL THEN register.type := type; T.Append(register)
			END
		END
		(*;IF register # NIL THEN ASSERT(register.name^ # 'PRec') END*)
	END Type;
	
	PROCEDURE ActualParameters(VAR aparlist: HrCPT.Node; fpar: HrCPT.Object; VAR pre, lastp: HrCPT.Node; fp: POINTER TO ARRAY OF T.Par; i: INTEGER; ap: POINTER TO ARRAY OF T.Expression );
		VAR apar, last, newPar, iidPar, n: HrCPT.Node;
			e: T.Expression; len: INTEGER;
	BEGIN IF fp = NIL THEN len := 0 ELSE len := LEN(fp) END;
		aparlist := NIL; last := NIL;
		IF sym # rparen THEN 
			newPar := NIL; iidPar := NIL; 
			LOOP Expression(apar, e);
				IF (fpar # NIL) & (i < len) THEN
					IF (apar.typ.form = Pointer) & (fpar.typ.form = Comp) THEN HrB.DeRef(apar) END;
					HrB.Param(apar, fpar); 
					(* with some bad designators, CP2 still sends a fpar into here, while Cpjs does not; be careful *)
					IF i # none THEN ap[i] := B.CheckParam(e, fp[i]) END;
					(*IF (fpar.mode = Var) OR (fpar.vis = inPar) THEN HrB.CheckBuffering(apar, NIL, fpar, pre, lastp) END;*)
					HrB.Link(aparlist, last, apar);
					IF ODD(fpar.sysflag DIV newBit) THEN newPar := apar ;HALT(126)	(* COM-and java-related stuff *)
					ELSIF ODD(fpar.sysflag DIV iidBit) THEN iidPar := apar ;HALT(126)
					END;
					IF (newPar # NIL) & (iidPar # NIL) THEN HrB.CheckNewParamPair(newPar, iidPar) END;
					IF anchorVarPar & (fpar.mode = VarPar) & ~(HrM.java IN HrM.options)
						OR (HrM.allSysVal IN HrM.options)	(* source output: avoid double evaluation *)
							 & ((fpar.mode = VarPar) & (fpar.typ.comp = Record) & ~fpar.typ.untagged
								OR (fpar.typ.comp = DynArr) & ~fpar.typ.untagged) THEN
						n := apar;
						WHILE n.class IN {Nfield, Nindex, Nguard} DO n := n.left END;
						IF (n.class = Nderef) & (n.subcl = 0) THEN
							IF n.left.class = Nguard THEN n := n.left END;
							(*CpjsB.CheckVarParBuffering(n.left, pre, lastp)*)
						END
					END;
					fpar := fpar.link; INC(i)
				ELSE IF i # -1 THEN err(64); err(-64) END
				END;
				IF sym = comma THEN HrS.Get(sym)
				ELSIF (lparen <= sym) & (sym <= ident) THEN err(comma)
				ELSE EXIT
				END
			END
		END;
		IF (fpar # NIL) & (i < len) THEN err(65); err(-65) END
	END ActualParameters;
	
	PROCEDURE ThisRecord (t: T.Type): T.Record;
		VAR rec: T.Record;
	BEGIN IF t # NIL THEN WITH t: T.Record DO rec := t | t: T.RecPtr DO rec := t.base ELSE END END; RETURN rec
	END ThisRecord;
	
	PROCEDURE Call (VAR x: HrCPT.Node; VAR des: T.Designator; proper: T.Proc; isSuper: BOOLEAN; ap: POINTER TO ARRAY OF T.Expression; OUT stmt: T.Stat );
		VAR apar, pre, lastp: HrCPT.Node; fpar: HrCPT.Object;
			fp: POINTER TO ARRAY OF T.Par; num: INTEGER;
	BEGIN
		pre := NIL; lastp := NIL;
		HrB.PrepCall(x, fpar); 
		B.PrepCall(des, proper, isSuper, fp, num, ap, stmt);
		(*IF (x.obj # NIL) & (x.obj.mode = TProc) THEN CpjsB.CheckBuffering(x.left, NIL, x.obj.link, pre, lastp) END;*)
		IF sym = lparen THEN
			HrS.Get(sym); ActualParameters(apar, fpar, pre, lastp, fp, num, ap); CheckSym(rparen)
		ELSE apar := NIL;
			IF (fpar # NIL) & (fp # NIL) & (LEN(fp) > num) THEN err(65) END
		END ;
		HrB.Call(x, apar, fpar); (*B.Call(proper, des, ap, stmt);*)
		IF (x.typ # HrCPT.notyp) & (stmt # NIL) THEN err(55) END;
		(*IF pre # NIL THEN SetPos(x); CpjsB.Construct(Ncomp, pre, x); x := pre END;*)
		IF level > 0 THEN HrCPT.topScope.link.leaf := FALSE END;
		IF level > 0 THEN INCL(T.nowScope.flags, T.ref) END;
		IF proper # NIL THEN INCL(proper.flags, T.used) END
	END Call;

	PROCEDURE selector(VAR x: HrCPT.Node; VAR des: T.Designator; inFactor: BOOLEAN; 
		VAR pr: T.Proc; OUT isSuper: BOOLEAN; OUT ap: POINTER TO ARRAY OF T.Expression	(* could be global *)
	);
		VAR obj, proc, p, fpar: HrCPT.Object; y, apar, pre, lastp: HrCPT.Node; typ: HrCPT.Struct; name: HrCPT.Name;
			o: T.Object; yy: T.Expression; xx: T.Designator; rec: T.Record; proper, iiscall, isguard: BOOLEAN; 
			(*pr,*) base: T.TProc; ccall: T.CallDes;
			rcv: T.Designator; fp: POINTER TO ARRAY OF T.Par; (*ap: POINTER TO ARRAY OF T.Expression; *)
			bkmrk, num, periodpos: INTEGER; _: T.Stat; rcvt: T.Type; frcv: T.Par;
	BEGIN xx := des; isSuper := FALSE;	(* handle des = NIL after a proper procedure id *)
		LOOP
			IF (xx = B.incompdes) & ((sym = lbrak) OR (sym = period) OR (sym = arrow) OR (sym = dollar)) THEN 
				xx := T.errdes;
				IF sym = dollar THEN Err(-90, 4)	(*$118*)
				ELSIF sym = arrow THEN Err(-78, 1)
				ELSIF sym = period THEN Err(-83, 1)
				ELSIF sym = lbrak THEN Err(-79, 1)
				ELSE err(-535)	(*$-1*)
				END
			END;
			IF sym = lbrak THEN HrS.Get(sym);
				LOOP
					IF (x.typ # NIL) & (x.typ.form = Pointer) THEN HrB.DeRef(x) END ;
					(*IF xx.type IS T.ArrayPtr THEN xx := B.Deref(xx) END;*)
					Expression(y, yy); HrB.Index(x, y);
					xx := B.Index(xx, yy);
					IF sym = comma THEN HrS.Get(sym) ELSE EXIT END
				END ;
				CheckSym(rbrak); xx.pos := HrM.startpos	(* index's position is set after ] *)
			ELSIF sym = period THEN HrS.Get(sym); periodpos := HrM.startpos;
				IF sym = ident THEN name := HrS.name; HrS.Get(sym); 
					IF x.typ # NIL THEN
						IF x.typ.form = Pointer THEN HrB.DeRef(x) END ;
						rec := ThisRecord(xx.type);
						IF x.typ.comp = Record (*rec # NIL*) THEN
							typ := x.typ; HrCPT.FindField(name, typ, obj); HrB.Field(x, obj);
							IF xx # T.errdes THEN	(* this check could go higher up when black text is removed *)
								o := T.FindInRec(rec, name$);
								IF o = NIL THEN err(-83); xx := T.errdes	(*$114*)
								ELSE 
									WITH o: T.Field DO xx := B.Field(xx, o) 
									| o: T.TProc DO 
										pr := o; NEW(ap, LEN(pr.par)); 
										frcv := pr.par[0]; 
										IF (frcv.type IS T.RecPtr) & (xx.type IS T.Record) THEN err(-71); xx := T.errdes END;	(*$*)
										ap[0] := B.CheckParam(xx, pr.par[0]);IF xx # T.errdes THEN xx := B.incompdes END
									ELSE err(-592); xx := T.errdes
									END 
								END
							END;
							IF (obj # NIL) & (obj.mode = TProc) THEN
								IF sym = arrow THEN  (* super call *) HrS.Get(sym);
									IF (xx # T.errdes) & (pr # NIL) & (pr IS T.TProc) THEN 	(* this could be moved up *)
										ASSERT(ap # NIL); isSuper := TRUE;
										rcv := ap[0](T.Designator);
										IF rcv IS T.Deref THEN rcv := rcv(T.Deref).what END;
										WITH rcv: T.VarDes DO 
											IF T.receiver IN rcv.var.flags THEN 
												rec := ThisRecord(rcv.var.type).base;
												IF rec = NIL THEN err(-593); xx := T.errdes END	(*$113*)
											ELSE err(-594); xx := T.errdes	(*$113*)
											END
										ELSE err(-594); xx := T.errdes	(*$113*)
										END;
										(* rec is the base type of receiver or NIL *)
										IF rec # NIL THEN o := T.FindInRec(rec, pr.name);
											IF o # NIL THEN 
												WITH o: T.TProc DO pr := o(*; isSuper := TRUE*) ELSE err(-598) END 
											ELSE err(-74); xx := T.errdes 	(*$113*)
											END
										END
									END;
									y := x.left;
									IF y.class = Nderef THEN y := y.left END ;	(* y = record variable *)
									IF y.obj # NIL THEN
										proc := HrCPT.topScope;	(* find innermost scope which owner is a TProc *)
										WHILE (proc.link # NIL) & (proc.link.mode # TProc) DO proc := proc.left END ;
										IF (proc.link = NIL) OR (proc.link.link # y.obj) (* OR (proc.link.name^ # name) *) THEN 
											err(75)
										END ;
										typ := y.obj.typ;
										IF typ.form = Pointer THEN typ := typ.BaseTyp END ;
										HrCPT.FindBaseField(x.obj.name^, typ, p);
										IF p # NIL THEN
											x.subcl := super; x.typ := p.typ;	(* correct result type *)
											IF p.conval.setval * {absAttr, empAttr} # {} THEN err(194) END;
											IF (p.vis = externalR) & (p.mnolev < 0) & (proc.link.name^ # name) THEN err(196) END;
										ELSE err(74)
										END
									ELSE err(75)
									END
								ELSE
									proc := obj; 
									IF (x.left.readonly) & (proc.link.mode = VarPar) & (proc.link.vis = 0)  THEN err(76) END;
									IF (xx # T.errdes) & (pr # NIL) & (pr IS T.TProc) THEN 	(* this could be moved up *)
										isSuper := FALSE;
										IF (T.immutable IN ap[0].flags) & (pr.par[0].flags * T.kindMask = T.varK)
										THEN err(-529) 	(*$107*)
										END;
										base := pr(T.TProc);
										WHILE (base # NIL) & ~(T.new IN base.flags) DO
											rec := ThisRecord(base.par[0].type).base; 
											IF rec = NIL THEN base := NIL 
											ELSE o := T.FindInRec(rec, base.name); base := NIL;
												IF o # NIL THEN WITH o: T.TProc DO base := o ELSE err(-598) END 
												ELSE err(-597) 
												END
											END
										END;
										IF base = NIL THEN err(-599)
										ELSIF (T.readOnly IN base.flags) & (base.imno # 0) THEN err(-196) 	(*$*)
										END
									END;
									WHILE (proc.mnolev >= 0) & ~(newAttr IN proc.conval.setval) & (typ.BaseTyp # NIL) DO
										(* find base method *)
										typ := typ.BaseTyp; HrCPT.FindField(name, typ, proc);
									END;
									IF (proc.vis = externalR) & (proc.mnolev < 0) THEN err(196) END;
								END ;
								IF (obj.typ # HrCPT.notyp) & (sym # lparen) THEN err(lparen); Err(-lparen, 0) END
							END
						ELSE err(53); HrM.Mark(-53, periodpos); xx := T.errdes
						END
					ELSE err(52)	(* what can trigger this error here? x.typ # NIL... identifier does not denote a type *)
					END
				ELSE err(ident); Err(-ident, 1); xx := T.errdes
				END
			ELSIF sym = arrow THEN HrS.Get(sym); IF xx # T.errdes THEN HrB.DeRef(x); END; xx := B.Deref(xx)
			ELSIF sym = dollar THEN
				IF x.typ.form = Pointer THEN HrB.DeRef(x) END;
				HrS.Get(sym); HrB.StrDeref(x);
				xx := B.StrSel(xx); xx.pos := HrM.startpos
			ELSIF sym = lparen THEN bkmrk := HrM.startpos + 1;
				IF (x.obj # NIL) & (x.obj.mode IN {XProc, LProc, CProc, TProc}) THEN typ := x.obj.typ
				ELSIF x.typ.form = ProcTyp THEN typ := x.typ.BaseTyp
				ELSIF x.class = Nproc (* pr IS T.SProc *) THEN EXIT	(* standard procedure *)
				ELSE typ := NIL
				END;
				isguard := FALSE; 
				IF xx = B.incompdes THEN proper := pr.type = NIL
				ELSIF xx IS T.ProcDes THEN HALT(60)
				ELSIF (xx.type IS T.Procedure) (*& ~(xx IS T.TypeDes)*) THEN proper := xx.type(T.Procedure).res = NIL
				ELSIF xx.type = NIL THEN HALT(60) (* only a ProcDes for a proper procedure will  *)
				ELSIF (xx.type IS T.SysPtr) OR (xx.type IS T.RecPtr) OR (xx.type IS T.Record) THEN isguard := TRUE
				ELSE EXIT (* this "(" is neither a call nor a type guard: it is not part of this designator  *)
				END;
				
				IF (typ # HrCPT.notyp) & isguard THEN	(* bad structure: a call of a proper procedure  *)
					HrS.Get(sym);
					IF (typ = NIL) & isguard THEN	(* type guard *)
						IF sym = ident THEN 
							qualident(obj, o);
							IF (o # NIL) THEN xx := B.Guard(xx, o, bkmrk)
								(*WITH o: T.TypeName DO xx := B.Guard(xx, o) 
								ELSE IF o # T.errid THEN Err(-48, 2) END; xx := T.errdes END*)	(*$116*)
							ELSE err(-0); xx := T.errdes
							END;
							IF obj.mode = Typ THEN HrB.TypTest(x, obj, TRUE)
							ELSE err(52)
							END;
							HrS.Get(sym) 
						ELSE err(ident); Err(-48, 3); xx := T.errdes; HrS.Get(sym); CheckSym(rparen)	(*$116*)
						END;
					END
				ELSIF inFactor THEN (* function call *) 
					IF ~proper THEN Call(x, xx, pr, isSuper, ap, _) ELSE err(-528); xx := T.errdes; EXIT END
				ELSIF proper THEN EXIT (* proper procedure call with parameters will be done in Stat *)
				ELSE (* call statement // function or procedure call outside of a Factor *) 
					ASSERT(xx = B.incompdes); ASSERT(pr # NIL); 
					(* pr, super and ap should hold correct values *); xx := T.errdes;
					EXIT
				END
			ELSE
				IF xx = T.errdes THEN pr := NIL
				ELSIF xx = B.incompdes THEN ASSERT(pr # NIL);
					INCL(pr.flags, T.used);
					IF inFactor THEN xx := B.ProcDes(pr)	
					ELSIF pr.type = NIL THEN (* pr, super and ap should hold correct values *); xx := T.errdes
					ELSE err(-55); xx := T.errdes
					END
				END;
				EXIT
			END
		END;
		ASSERT(xx IS T.Designator, 60); des := xx(T.Designator)
	END selector;

	PROCEDURE StandProcCall(VAR x: HrCPT.Node; num: INTEGER; OUT ex: T.Expression; OUT stmt: T.Stat);
		VAR y: HrCPT.Node; m: BYTE; n: SHORTINT;
			actuals: ARRAY maxStdPar OF T.Expression; par: INTEGER; xx: T.Expression;
	BEGIN m := SHORT(SHORT(x.obj.adr)); n := 0; par := 2;
		IF sym = lparen THEN HrS.Get(sym);
			IF sym # rparen THEN
				LOOP
					IF n = 0 THEN Expression(x, actuals[0]); HrB.StPar0(x, m); n := 1
					ELSIF n = 1 THEN Expression(y, actuals[1]); HrB.StPar1(x, y, m); n := 2
					ELSE Expression(y, actuals[par]); HrB.StParN(x, y, m, n); INC(n); INC(par);
					END ;
					IF sym = comma THEN HrS.Get(sym)
					ELSIF (lparen <= sym) & (sym <= ident) THEN err(comma)
					ELSE EXIT
					END
				END ;
				CheckSym(rparen)
			ELSE HrS.Get(sym)
			END ;
			HrB.StFct(x, m, n);
			B.StdCall2(actuals[0], actuals[1], actuals, num, n, ex, stmt)
		ELSE err(lparen); Err(-lparen, 0); ex := T.errex
		END ;
		IF (level > 0) & ((m = newfn) OR (m = sysnewfn)) THEN HrCPT.topScope.link.leaf := FALSE END
	END StandProcCall;
	
	PROCEDURE Element(VAR x: HrCPT.Node; OUT xx: T.Expression);
		VAR y: HrCPT.Node; yy: T.Expression;
	BEGIN Expression(x, xx);
		IF sym = upto THEN
			HrS.Get(sym); Expression(y, yy); HrB.SetRange(x, y); xx := B.SetRange(xx, yy)
		ELSE HrB.SetElem(x); xx := B.SetElem(xx)
		END
		;ASSERT(xx # NIL)
	END Element;

	PROCEDURE OpToFn (op: INTEGER): INTEGER;	(** Eliminate! Temporary procedure that  *)
		VAR res: INTEGER;
	BEGIN
		CASE op OF
		| plus: res := T.plusfn | minus: res := T.minusfn | times: res := T.timesfn | slash: res := T.slashfn
		| mod: res := T.modfn | div: res := T.divfn | and: res := T.andfn | or: res := T.orfn
		| eql: res := T.eqlfn | neq: res := T.neqfn | lss: res := T.lssfn | leq: res := T.leqfn | gtr: res := T.gtrfn
		| geq: res := T.geqfn | in: res := T.infn | is: res := T.isfn
		ELSE res := -1
		END;
		RETURN res
	END OpToFn;
	
	PROCEDURE Sets(VAR x: HrCPT.Node; OUT xx: T.Expression);
		VAR y: HrCPT.Node; yy: T.Expression; bkmrk: INTEGER;
	BEGIN
		IF sym # rbrace THEN 
			Element(x, xx);
			LOOP
				IF sym = comma THEN HrS.Get(sym); bkmrk := HrM.startpos;
				ELSIF (lparen <= sym) & (sym <= ident) THEN err(comma)
				ELSE EXIT
				END ;
				Element(y, yy); HrB.Op(plus, x, y); xx := B.Dyadic(T.plusfn, xx, yy, bkmrk)
			END
		ELSE x := HrB.EmptySet(); xx := T.ThisSet({}); xx.pos := HrM.startpos + 1
		END ;
		CheckSym(rbrace)
	END Sets;
	
	PROCEDURE Factor(VAR x: HrCPT.Node; OUT xx: T.Expression);
		VAR id: HrCPT.Object;
			qd: T.Object; des: T.Designator; proc: T.Proc; __: BOOLEAN; ___: POINTER TO ARRAY OF T.Expression;
			stat: T.Stat; bkmrk: INTEGER;
			r: REAL; long: LONGINT; str: ARRAY 64 OF CHAR;
	BEGIN
		IF sym < not THEN err(13);
			REPEAT HrS.Get(sym) UNTIL sym >= lparen
		END ;
		IF sym = ident THEN bkmrk := HrM.startpos;
			qualident(id, qd); x := HrB.NewLeaf(id); 
			IF qd # NIL THEN
				WITH qd: T.SProc DO ASSERT((x.class = Nproc) & (x.obj.mode = SProc), 40); 
					StandProcCall(x, qd.num, xx, stat);
					ASSERT(~((xx # NIL) & (xx.type # T.errtp) & (stat # NIL)), 60);
					IF (xx # NIL) & (xx.type # T.errtp) THEN xx.pos := bkmrk END
				ELSE
					des := B.Qualident(qd, proc); des.pos := bkmrk;
					selector(x, des, TRUE, proc, __, ___); xx := des; 
					
				END
			ELSE xx := T.errdes
			END
		ELSIF sym = number THEN
			CASE HrS.numtyp OF
			   char:
				x := HrB.NewIntConst(HrS.intval); x.typ := HrCPT.char8typ;
				IF HrS.intval > 255 THEN x.typ := HrCPT.char16typ END
			| integer: x := HrB.NewIntConst(HrS.intval)
			| int64: x := HrB.NewLargeIntConst(HrS.intval, HrS.realval)
			| real: x := HrB.NewRealConst(HrS.realval, NIL)
			| real32: x := HrB.NewRealConst(HrS.realval, HrCPT.real32typ)
			| real64: x := HrB.NewRealConst(HrS.realval, HrCPT.real64typ)
			END ;
			CASE HrS.numtyp OF
			| char: xx := T.ThisChar(CHR(HrS.intval))
			| integer, int64: 
				(*r := x.conval.intval + x.conval.realval; long := ENTIER(x.conval.realval); Strings.IntToString(long, str);
				IF CpjsS.intval < 0 THEN xx := T.ThisInt(MIN(LONGINT))
				ELSE*) xx := T.ThisInt(HrS.intval + ENTIER(HrS.realval));
					IF HrS.numtyp = integer THEN ASSERT(xx.type.what = T.fInt32) END
				(*END*)
			| real, real32, real64: xx := T.ThisReal(HrS.realval)
			END; xx.pos := HrM.errpos;
			HrS.Get(sym)
		ELSIF sym = string THEN
			x := HrB.NewString(HrS.str, HrS.lstr, HrS.intval);
			xx := T.ThisString(HrS.str, HrS.lstr, HrS.intval); xx.pos := HrM.errpos;
			HrS.Get(sym)
		ELSIF sym = nil THEN xx := T.ThisNil(); xx.pos := HrM.errpos;
			ASSERT(xx.type.what = T.fNil);
			x := HrB.Nil(); HrS.Get(sym)
		ELSIF sym = lparen THEN
			HrS.Get(sym); Expression(x, xx); CheckSym(rparen)
		ELSIF sym = lbrak THEN
			HrS.Get(sym); err(lparen); Err(-lparen, 1); Expression(x, xx); CheckSym(rparen)
		ELSIF sym = lbrace THEN HrS.Get(sym); Sets(x, xx)
		ELSIF sym = not THEN
			HrS.Get(sym); Factor(x, xx); HrB.MOp(not, x); xx := B.Monadic(T.notfn, xx)
		ELSE err(13); HrS.Get(sym); x := NIL; Err(-13, 0); xx := T.errex
		END ;
		IF x = NIL THEN x := HrB.NewIntConst(1); x.typ := HrCPT.undftyp END;
		(*IF xx = NIL THEN xx := T.dir.Int(); xx.type := T.errtp END;*)
		ASSERT(xx # NIL, 60)
	END Factor;

	PROCEDURE Term(VAR x: HrCPT.Node; OUT xx: T.Expression);
		VAR y: HrCPT.Node; mulop: BYTE; yy: T.Expression; bkmrk: INTEGER;
	BEGIN Factor(x, xx);
		WHILE (times <= sym) & (sym <= and) DO bkmrk := HrM.errpos;
			mulop := sym; HrS.Get(sym);
			Factor(y, yy); HrB.Op(mulop, x, y);
			xx := B.Dyadic(OpToFn(mulop), xx, yy, bkmrk)
		END
	END Term;

	PROCEDURE SimpleExpression(VAR x: HrCPT.Node; OUT xx: T.Expression);
		VAR y: HrCPT.Node; addop: BYTE; r: REAL; yy: T.Expression; bkmrk: INTEGER;
	BEGIN 
		IF sym = minus THEN HrS.unaryMinus := TRUE; HrS.Get(sym); Term(x, xx); HrS.unaryMinus := FALSE;
			IF ~(xx IS T.Literal) THEN xx := B.Monadic(T.minusfn, xx) END;
			IF (x.class # Nconst) OR ~(x.typ.form IN {Int8, Int16, Int32, Int64, Real32, Real64}) THEN 
				HrB.MOp(minus, x) 
			END
		ELSIF sym = plus THEN HrS.Get(sym); Term(x, xx); 
			(*IF (x.class = Nconst) & (x.conval.intval < 0) THEN err(230); x.conval.intval := 0; END;*)
			xx := B.Monadic(T.plusfn, xx); HrB.MOp(plus, x)
		ELSE Term(x, xx); (*IF CpjsS.intval < 0 THEN err(-203) END*)
		END ;
		WHILE (plus <= sym) & (sym <= or) DO bkmrk := HrM.errpos;
			addop := sym; HrS.Get(sym); Term(y, yy); 
			IF x.typ.form = Pointer THEN HrB.DeRef(x) END;
			IF (x.typ.comp IN {Array, DynArr}) & (x.typ.BaseTyp.form IN charSet) (* OR (x.typ.sysflag = jstr) *) THEN
				HrB.StrDeref(x)
			END;
			IF y.typ.form = Pointer THEN HrB.DeRef(y) END;
			IF (y.typ.comp IN {Array, DynArr}) & (y.typ.BaseTyp.form IN charSet) (* OR (y.typ.sysflag = jstr) *) THEN
				HrB.StrDeref(y)
			END;
			HrB.Op(addop, x, y);
			xx := B.Dyadic(OpToFn(addop), xx, yy, bkmrk)
		END
	END SimpleExpression;

	PROCEDURE Expression(VAR x: HrCPT.Node; OUT xx: T.Expression);
		VAR y, pre, last: HrCPT.Node; obj: HrCPT.Object; relation: BYTE;
			qd: T.Object; yy: T.Expression; bkmrk: INTEGER; _: T.Proc;
	BEGIN 
		SimpleExpression(x, xx); bkmrk := HrM.errpos;
		IF (eql <= sym) & (sym <= geq) THEN
			relation := sym; HrS.Get(sym); SimpleExpression(y, yy);
			pre := NIL; last := NIL;
			IF (x.typ.comp IN {Array, DynArr}) & (x.typ.BaseTyp.form IN charSet) THEN
				HrB.StrDeref(x)
			END;
			IF (y.typ.comp IN {Array, DynArr}) & (y.typ.BaseTyp.form IN charSet) THEN
				HrB.StrDeref(y)
			END;
			(*CpjsB.CheckBuffering(x, NIL, NIL, pre, last);
			CpjsB.CheckBuffering(y, NIL, NIL, pre, last);*)
			HrB.Op(relation, x, y); 
			xx := B.Dyadic(OpToFn(relation), xx, yy, bkmrk);
			(*IF pre # NIL THEN CpjsB.Construct(Ncomp, pre, x); pre.typ := x.typ; x := pre END*)
		ELSIF sym = in THEN
			HrS.Get(sym); SimpleExpression(y, yy); HrB.In(x, y); xx := B.Dyadic(T.infn, xx, yy, bkmrk)
		ELSIF sym = is THEN bkmrk := HrM.errpos; 
			HrS.Get(sym);
			IF sym = ident THEN
				qualident(obj, qd); qd.textpos := HrM.errpos; xx := B.Is(xx(T.Designator), qd, bkmrk);	(* caution with typeguard... *)
				(*yy := B.Qualident(qd, _); yy.pos := HrM.errpos;
				IF yy = NIL THEN Err(-52, 1); xx := T.errex ELSE xx := B.Is(T.isfn, xx, yy, bkmrk) END;*)
				IF obj.mode = Typ THEN HrB.TypTest(x, obj, FALSE)
				ELSE err(52)
				END
			ELSE err(ident)
			END
		END
		;ASSERT(xx # NIL)
	END Expression;
	
	PROCEDURE ProcedureDeclaration(VAR x: HrCPT.Node; OUT p: T.Proc);
		VAR proc, fwd: HrCPT.Object;
			name: HrCPT.Name;
			mode: BYTE;
			forward: BOOLEAN;
			sys: SHORTINT;
			oproc: T.TProc;
			fw: T.Proc; obj: T.Object; pos: INTEGER; nam, lib, ent: POINTER TO T.Name; attr: SET;
			mods: POINTER TO ARRAY OF POINTER TO T.String; dup: BOOLEAN;
			fpar: POINTER TO ARRAY OF T.Par; ftype: T.Type; rcv: T.Par; rec: T.Record; base: T.TProc;
			symbol: INTEGER;

			VAR baseProc, o: HrCPT.Object;
				objTyp, recTyp: HrCPT.Struct;
				objMode, objVis: BYTE;
				objName: HrCPT.Name;
				pnode: HrCPT.Node;
				fwdAttr: SET;
				
		PROCEDURE GetCode;
			VAR ext: HrCPT.ConstExt; n, c, i: INTEGER; s: POINTER TO ARRAY OF SHORTCHAR;
				cx: HrCPT.Node; _: T.Expression; __: T.Literal;
			
			PROCEDURE EnsureLen(len: INTEGER);
				VAR j: INTEGER; s2: POINTER TO ARRAY OF SHORTCHAR;
			BEGIN
				IF len > LEN(s) THEN	(* if overflow then increase size of array s *)
					NEW(s2, LEN(s) * 2); FOR j := 0 TO n - 1 DO s2[j] := s[j] END; s := s2
				END
			END EnsureLen;
		BEGIN
			n := 0; NEW(s, 64);
			WHILE (sym # semicolon) & (sym # eof) DO
				ConstExpression(cx, _, __);
				IF cx.typ.form IN {Int8, Int16, Int32, Char8, Char16} THEN c :=cx.conval.intval; EnsureLen(n + 1);
					IF (0 <= c) & (c <= 255) THEN s[n] := SHORT(CHR(c)); INC(n)
					ELSE err(63)
					END
				ELSIF cx.typ.form = String8 THEN c := cx.conval.intval2 - 1 (*exclude 0X*); EnsureLen(n + c);
					FOR i := 0 TO c - 1 DO s[n + i] := cx.conval.ext[i] END;
					INC(n, c)
				ELSE (* Int64, Real32, Real64, String16, Bool, etc. *) err(63)
				END ;
				IF sym = comma THEN HrS.Get(sym); IF sym = semicolon THEN err(13) END
				ELSIF sym # semicolon THEN err(comma)
				END
			END;
			IF n # 0 THEN NEW(ext, n); i := 0;
				WHILE i < n DO ext[i] := s[i]; INC(i) END;
			ELSE ext := NIL
			END;
			proc.conval.ext := ext;
			INCL(proc.conval.setval, hasBody);
			WITH p: T.CProc DO p.code := ext ELSE HALT(20) END
		END GetCode;

		PROCEDURE GetParams;
			VAR name: HrCPT.String; checkRet: BOOLEAN; len, i: INTEGER;
		BEGIN
			proc.mode := mode; proc.typ := HrCPT.notyp;
			proc.sysflag := SHORT(sys);
			proc.conval.setval := {};
			(*IF sym = lparen THEN*) checkRet := sym = lparen;
				(*CpjsS.Get(sym); *)
				FormalParameters(proc.link, proc.typ, name, rcv, ~dup & ((fw # NIL) OR (base # NIL)), fpar, ftype, fw # NIL, p.par, p.type);
				IF p.par # NIL THEN len := LEN(p.par); i := 0; WHILE i < len DO T.Append(p.par[i]); INC(i) END END;
				IF (name # NIL) & checkRet THEN err(0) END;
			(*ELSIF rcv # NIL THEN NEW(p.par, 1); p.par[0] := rcv*)
			(*END;*)
			CheckForwardTypes; userList := NIL;
			IF fwd # NIL THEN
				HrB.CheckParameters(proc.link, fwd.link, TRUE);
				IF ~HrCPT.EqualType(proc.typ, fwd.typ) THEN err(117) END ;
				proc := fwd; HrCPT.topScope := proc.scope;
				IF mode = IProc THEN proc.mode := IProc END
			END
		END GetParams;

		PROCEDURE Body;
			VAR procdec, statseq: HrCPT.Node; c: INTEGER;
		BEGIN
			c := HrM.errpos;
			INCL(proc.conval.setval, hasBody);
			CheckSym(semicolon); Block(procdec, statseq, p.body); 
			HrB.Enter(procdec, statseq, proc); x := procdec;
			x.conval := HrCPT.NewConst(); x.conval.intval := c; x.conval.intval2 := HrM.startpos;
			CheckSym(end);
			IF sym = ident THEN
				IF HrS.name # proc.name^ THEN err(4) END ;
				HrS.Get(sym)
			ELSE err(ident)
			END
		END Body;

		(*TProcDecl*)
	
	BEGIN proc := NIL; forward := FALSE; x := NIL; mode := LProc; sys := 0;
		dup := FALSE; 
		IF (sym # ident) & (sym # lparen) THEN
			CheckSysFlag(sys, HrM.GetProcSysFlag);
			IF sys # 0 THEN
				IF ODD(sys DIV HrM.CProcFlag) THEN mode := CProc END;
			ELSE
				IF sym = times THEN	(* mode set later in CpjsB.CheckAssign *)
				ELSIF sym = arrow THEN forward := TRUE
				ELSE err(ident)
				END;
				HrS.Get(sym)
			END
		END ;
		IF sym = lparen THEN symbol := sTProc;
			HrS.Get(sym); Receiver(objMode, objVis, objName, objTyp, recTyp, rcv, rec)
		ELSE symbol := sXProc
		END;
		IF sym = ident THEN name := HrS.name; nam := ThisName(name); pos := HrM.startpos;
			IF (rcv # NIL) & (rcv.type = T.errtp) THEN p := T.dir.TProc(); dup := TRUE; proc := HrCPT.NewObj(); 
				proc.mode := TProc; proc.conval := HrCPT.NewConst(); proc.leaf := TRUE
				(* error in reciever: set up vars to parse & skip procedure declaration *)
			ELSIF rcv # NIL THEN HrCPT.FindField(name, recTyp, fwd); HrCPT.FindBaseField(name, recTyp, baseProc);
				IF (baseProc # NIL) & (baseProc.mode # TProc) THEN baseProc := NIL; err(1201) END ;
				IF fwd = baseProc THEN fwd := NIL END ;
				IF (fwd # NIL) & (fwd.mnolev # level) THEN fwd := NIL END ;
				IF (fwd # NIL) & (fwd.mode = TProc) & (fwd.conval.setval * {hasBody, absAttr, empAttr} = {}) THEN
					(* there exists a corresponding forward declaration *)
					proc := HrCPT.NewObj(); proc.leaf := TRUE;
					proc.mode := TProc; proc.conval := HrCPT.NewConst(); proc.textpos := HrM.startpos;
					(*CheckMark(proc, oproc);
					IF fwd.vis # proc.vis THEN err(118) END;*)
					fwdAttr := fwd.conval.setval
				ELSE
					IF fwd # NIL THEN err(1301); fwd := NIL END ;
					HrCPT.InsertField(name, recTyp, proc);
					proc.mode := TProc; proc.conval := HrCPT.NewConst(); proc.textpos := HrM.startpos;
					(*CheckMark(proc, oproc);*)
					IF recTyp.strobj # NIL THEN	(* preserve declaration order *)
						o := recTyp.strobj.link;
						IF o = NIL THEN recTyp.strobj.link := proc
						ELSE
							WHILE o.nlink # NIL DO o := o.nlink END;
							o.nlink := proc
						END
					END
				END;
				obj := T.FindInRec(rec, nam^ );
				IF obj = NIL THEN p := T.dir.TProc(); (*IF forward THEN INCL(flags, fwdFlag) END*)
				ELSIF ~(obj IS T.TProc) THEN err(1101); p := T.dir.TProc(); dup := TRUE
				ELSIF obj(T.TProc).par[0].type # rcv.type THEN 
					base := obj(T.TProc); p := T.dir.TProc(); fpar := base.par; ftype := base.type
				ELSIF ~(fwdFlag IN obj.flags) THEN err(1101); p := T.dir.TProc(); dup := TRUE
				ELSE fw := obj(T.Proc); fpar := fw.par; ftype := obj.type; p := T.dir.TProc(); IF forward THEN err(297) END
				END
			ELSE HrCPT.Find(HrS.name, fwd); 
				IF (fwd # NIL) & ((fwd.mnolev # level) OR (fwd.mode = SProc)) THEN fwd := NIL END ;
				IF (fwd # NIL) & (fwd.mode IN {LProc, XProc}) & ~(hasBody IN fwd.conval.setval) THEN
					(* there exists a corresponding forward declaration *)
					proc := HrCPT.NewObj(); proc.leaf := TRUE;
					proc.mode := mode; proc.conval := HrCPT.NewConst(); 
					(*CheckMark(proc, sProc, flags, lib, ent, mods);
					IF fwd.vis # proc.vis THEN err(118) END;*)
				ELSE
					IF fwd # NIL THEN err(1); fwd := NIL END ;
					HrCPT.Insert(name, proc);
					proc.mode := mode; proc.conval := HrCPT.NewConst();
					(*CheckMark(proc, sProc, flags, lib, ent, mods)*)
				END ;
				IF ODD(sys DIV HrM.CProcFlag) THEN p := T.dir.CProc(); IF level > 0 THEN err(73) END
				ELSE p := T.dir.XProc()
				END;
				obj := T.Find(HrS.name); 
				(*ASSERT(fwd = NIL);*)
				IF obj # NIL THEN 
					IF obj IS T.Proc THEN 		(* looks like in XProc branch dup = (fw = NIL) *)
						IF fwdFlag IN obj.flags THEN fw := obj(T.Proc); fpar := fw.par; ftype := fw.type 
						ELSIF forward THEN err(-297); dup := TRUE
						ELSE err(-1001); dup := TRUE
						END
					ELSE err(-1001); dup := TRUE 
					END
				END
			END;
			(* p # NIL; base # NIL => redefinition; check against fpars, ftype; p IS T.TProc => rcv # NIL *)


			(*ASSERT((rec # NIL) & (rec.typeName.name # "S"))*);
			p.textpos := pos; p.name := nam;
			CheckMark(proc, symbol, p.flags, lib, ent, mods);
			IF ODD(sys DIV HrM.CProcFlag) THEN p.flags := p.flags + T.codeF END;
			IF sys = 0 THEN
			ELSIF sys = -10 THEN p.flags := p.flags + T.ccallF
			ELSIF sys = -14 THEN p.flags := p.flags + T.ms64F
			ELSIF sys = -16 THEN p.flags := p.flags + T.sysvF		(* do away with magic numbers... *)
			ELSIF sys = 1 THEN p.flags := p.flags + T.codeF
			ELSE HALT(126)
			END;
			IF (fwd # NIL) & (fwd.vis # proc.vis) THEN err(118) END;
			IF forward THEN INCL(p.flags, fwdFlag) END;
			WITH p: T.CProc DO p.lib := lib; p.ent := ent; p.modifiers := mods
			| p: T.XProc DO p.lib := lib; p.ent := ent; p.modifiers := mods
			| p: T.TProc DO p.lib := lib; p.ent := ent; p.modifiers := mods
			| p: T.FProc DO (* do smth with lib, ent, mods of a forward decl? *)
			ELSE
			END;
			IF (fw # NIL) & ((fw.flags * T.visMask) # (p.flags * T.visMask)) THEN err(-118) END;
			IF (level > 0) & (({ T.exported } = (p.flags * T.visMask)) OR (p IS T.TProc)) THEN err(-73) END; 
			WITH p: T.TProc DO IF (rcv.type # T.errtp) & (fw = NIL) THEN T.AppendProc(rec, p) END
			ELSE IF ~dup & (fw = NIL) THEN T.AppendThis(p, nam, pos) END
			END;
			(*HALT(0);*)
			T.OpenProc(p);
			IF (proc.vis # internal) & (mode = LProc) THEN mode := XProc END ;
			IF (mode # LProc) & (level > 0) THEN err(73) END ;
			INC(level); 
			HrCPT.OpenScope(level, proc);
			; IF p IS T.TProc THEN ASSERT(proc.mode = TProc, 58) END;
			IF p IS T.TProc THEN HrCPT.Insert(objName, proc.link); 
				proc.link.mode := objMode; proc.link.vis := objVis; proc.link.typ := objTyp;
				ASSERT(HrCPT.topScope # NIL);
				mode := TProc
			ELSE proc.link := NIL
			END;
			GetParams;	(* may change proc := fwd !!! *)
			; IF p IS T.TProc THEN ASSERT(proc.mode = TProc, 59) END;
			IF fw # NIL THEN p := fw END;
			p.startPos := procStartPos;
			WITH p: T.TProc DO
				ASSERT(HrCPT.topScope # NIL);
				GetAttributes(proc, baseProc, recTyp, p, base, rec);
				(*HALT(3);*)
				IF (fwd # NIL) & (fwdAttr / proc.conval.setval * {absAttr, empAttr, extAttr} # {}) THEN err(271) 
				END;
				CheckOverwrite(proc, baseProc, recTyp, base);
				IF ~forward THEN
					IF empAttr IN proc.conval.setval THEN	(* insert empty procedure *)
						pnode := NIL; HrB.Enter(pnode, NIL, proc);
						pnode.conval := HrCPT.NewConst();
						pnode.conval.intval := HrM.errpos;
						pnode.conval.intval2 := HrM.errpos;
						x := pnode;
					ELSIF HrM.noCode IN HrM.options THEN INCL(proc.conval.setval, hasBody)
					ELSIF ~(absAttr IN proc.conval.setval) & ((proc.modifiers = NIL) OR (proc.modifiers[0] # HrM.JS))
					THEN Body
					END;
					proc.adr := 0
				ELSE
					proc.adr := HrM.errpos;
					IF proc.conval.setval * {empAttr, absAttr} # {} THEN err(184) END
				END;
			ELSE
				IF (mode = CProc) OR (p IS T.CProc) THEN GetCode
				ELSIF HrM.noCode IN HrM.options THEN INCL(proc.conval.setval, hasBody)	(*?*)
				ELSIF ~forward & ((proc.modifiers = NIL) OR (proc.modifiers[0] # HrM.JS)) THEN Body; proc.adr := 0
				ELSE proc.adr := HrM.errpos
				END ;
			END;
			p.endPos := HrM.startpos;
			DEC(level); HrCPT.CloseScope; (*CheckForwardProcedures;*) T.CloseProc;
			IF fw # NIL THEN EXCL(fw.flags, fwdFlag) END
			; IF p IS T.TProc THEN ASSERT(proc.mode = TProc, 60) END;
		ELSE err(ident)
		END
		(*;ASSERT(fw = NIL);
		IF fwd # NIL THEN ASSERT(p.name^ # "Q") END;*)
	(* T.TProc.num is left undefined! I wonder if it is assigned a value in RTTI-related procedures *)
	END ProcedureDeclaration;
	
	PROCEDURE CaseLabelList(VAR lab, root: HrCPT.Node; LabelForm: SHORTINT; VAR min, max: INTEGER;
		case: T.Case
	);
		CONST ints = {T.fByte, T.fInt8, T.fInt16, T.fInt32}; 
		VAR x, y: HrCPT.Node; f: SHORTINT; xval, yval: INTEGER;
			from, to: INTEGER; fromex, toex: T.Expression; val: T.Literal;
		
		PROCEDURE Insert(VAR n: HrCPT.Node);	(* build binary tree of label ranges *)	(* !!! *)
		BEGIN
			IF n = NIL THEN
				IF x.hint # 1 THEN n := x END
			ELSIF yval < n.conval.intval THEN Insert(n.left)
			ELSIF xval > n.conval.intval2 THEN Insert(n.right)
			ELSE err(63)
			END
		END Insert;
		
		PROCEDURE CheckConst (x: T.Literal): INTEGER;
			VAR res, casewhat, w: INTEGER;
		BEGIN
			w := x.type.what;
			WITH x: T.Int DO res := SHORT(x.long)
			| x: T.Char DO res := ORD(x.ch)
			(*IF w IN ints THEN res := CpjsB.EvalInt(x)
			ELSIF w IN {T.fChar8, T.fChar16} THEN res := ORD(CpjsB.EvalChar(x))*)
			ELSE Err(-61, 1)	(*$127*)
			END;
			casewhat := case.what.type.what;
			IF (w IN ints) & (casewhat IN T.chars) THEN Err(-61, 3)	(*$127*)
			ELSIF (w IN T.chars) & (casewhat IN ints) THEN Err(-61, 2)	(*$127*)
			END;
			RETURN res
		END CheckConst;
		
	BEGIN lab := NIL;
		LOOP ConstExpression(x, fromex, val); f := x.typ.form;
			from := CheckConst(val);
			IF f IN {Int8..Int32} + charSet THEN  xval := x.conval.intval
			ELSE err(61); xval := 1
			END ;
			IF (f IN {Int8..Int32}) # (LabelForm IN {Int8..Int32}) THEN err(60) END;
			IF sym = upto THEN
				HrS.Get(sym); ConstExpression(y, toex, val); to := CheckConst(val); yval := y.conval.intval;
				IF (y.typ.form IN {Int8..Int32}) # (LabelForm IN {Int8..Int32}) THEN err(60) END;
				IF yval < xval THEN err(63); yval := xval END;
				IF to < from THEN Err(-61, 4); to := from END	(*$127*)
			ELSE yval := xval; to := from
			END ;
			B.AppendCaseLabel(case, from, to);	(* this should preserve the label expressions, not just label values!! *)
			x.conval.intval2 := yval;
			IF xval < min THEN min := xval END;
			IF yval > max THEN max := yval END;
			IF lab = NIL THEN lab := x; Insert(root)
			ELSIF yval < lab.conval.intval - 1 THEN x.link := lab; lab := x; Insert(root)
			ELSIF yval = lab.conval.intval - 1 THEN x.hint := 1; Insert(root); lab.conval.intval := xval
			ELSIF xval = lab.conval.intval2 + 1 THEN x.hint := 1; Insert(root); lab.conval.intval2 := yval
			ELSE
				y := lab;
				WHILE (y.link # NIL) & (xval > y.link.conval.intval2 + 1) DO y := y.link END;
				IF y.link = NIL THEN y.link := x; Insert(root)
				ELSIF yval < y.link.conval.intval - 1 THEN x.link := y.link; y.link := x; Insert(root)
				ELSIF yval = y.link.conval.intval - 1 THEN x.hint := 1; Insert(root); y.link.conval.intval := xval
				ELSIF xval = y.link.conval.intval2 + 1 THEN x.hint := 1; Insert(root); y.link.conval.intval2 := yval
				END
			END;
			IF sym = comma THEN HrS.Get(sym)
			ELSIF (sym = number) OR (sym = ident) THEN err(comma)
			ELSE EXIT
			END
		END
	END CaseLabelList;
	
	PROCEDURE StatSeq(VAR stat: HrCPT.Node; OUT stats: T.Stat);
		VAR fpar, id, t: HrCPT.Object; idtyp: HrCPT.Struct; e: BOOLEAN;
				s, x, y, z, apar, last, lastif, pre, lastp: HrCPT.Node; pos, p: INTEGER;
			qd, tn: T.Object; xx, yy, zz: T.Expression; des: T.Designator; stmt, lastst: T.Stat; call: T.Call;
			properp: T.Proc; isSuper: BOOLEAN; ap: POINTER TO ARRAY OF T.Expression; _: T.Literal;
			bkmrk, bkmrk2: INTEGER; ifst, elsest: T.If; orig: T.Type; clause: T.WithClause; withst: T.With;
			bound: T.Var; (* for FOR *)
			
		PROCEDURE CasePart(VAR x: HrCPT.Node; OUT stmt: T.Stat);
			VAR low, high: INTEGER; e: BOOLEAN; cases, lab, y, lastcase, root: HrCPT.Node; 
				case: T.Case; what: T.Expression; from: INTEGER;
		BEGIN case := B.NewCase(); 
			Expression(x, what); 
			(*IF what IS T.ProcDes THEN Err(-126, 5); what := T.errdes	(*$127*)
			ELS*)IF what.type.what = T.fInt64 THEN Err(-260, 1)	(*$127*)
			ELSIF ~(what.type.what IN T.integers + T.chars) THEN IF what.type # T.errtp THEN Err(-125, 1) END	(*$127*)
			END;
			case.what := what;
			IF (x.class = Ntype) OR (x.class = Nproc) THEN err(126)
			ELSIF x.typ.form = Int64 THEN err(260)
			ELSIF ~(x.typ.form IN {Int8..Int32} + charSet) THEN err(125)
			END ;
			CheckSym(of); cases := NIL; lastcase := NIL; root := NIL;
			low := MAX(INTEGER); high := MIN(INTEGER);
			LOOP
				IF sym < bar THEN from := case.labels;
					CaseLabelList(lab, root, x.typ.form, low, high, case);
					CheckSym(colon); StatSeq(y, stmt);
					WHILE from < case.labels DO case.stmts[from] := stmt; INC(from) END;
					HrB.Construct(Ncasedo, lab, y); HrB.Link(cases, lastcase, lab)
				ELSE stmt := NIL
				END ;
				IF sym = bar THEN HrS.Get(sym) ELSE EXIT END
			END;
			e := sym = else;
			IF e THEN HrS.Get(sym); StatSeq(y, case.else) ELSE y := NIL END ;
			HrB.Construct(Ncaselse, cases, y); HrB.Construct(Ncase, x, cases);
			cases.conval := HrCPT.NewConst();
			cases.conval.intval := low; cases.conval.intval2 := high;
			IF e THEN cases.conval.setval := {1} ELSE cases.conval.setval := {} END;
			HrB.OptimizeCase(root); cases.link := root;	(* !!! *)
			stmt := B.OptimizeCase(case)
		END CasePart;
		
		PROCEDURE SetPos(x: HrCPT.Node);
		BEGIN
			x.conval := HrCPT.NewConst(); x.conval.intval := pos
		END SetPos;

		PROCEDURE CheckBool(VAR x: HrCPT.Node);
		BEGIN
			IF (x.class = Ntype) OR (x.class = Nproc) THEN err(126); x := HrB.NewBoolConst(FALSE)
			ELSIF x.typ.form # Bool THEN err(120); x := HrB.NewBoolConst(FALSE)
			END
		END CheckBool;
		
		PROCEDURE CheckBool2 (ex: T.Expression): T.Expression;
			VAR res: T.Expression;
		BEGIN res := T.errex; 
			IF ex.type = T.errtp THEN
			(*ELSIF ex IS T.ProcDes THEN Err(-126, 4)*)
			ELSIF ex.type.what # T.fBool THEN Err(-120, 1)
			ELSE res := ex
			END;
			RETURN res
		END CheckBool2;

	BEGIN stat := NIL; last := NIL;
		LOOP x := NIL; stmt := NIL;
			IF sym < ident THEN err(14); Err(-14, sym);
				REPEAT HrS.Get(sym) UNTIL sym >= ident
			END ;
			pos := HrM.startpos;
			IF sym = ident THEN bkmrk := HrM.errpos;
				qualident(id, qd); x := HrB.NewLeaf(id); 
				des := B.Qualident(qd, properp); des.pos := HrM.errpos;
				selector(x, des, FALSE, properp, isSuper, ap);
				IF sym = becomes THEN 
					HrS.Get(sym); Expression(y, yy);
					IF (y.typ.form = Pointer) & (x.typ.form = Comp) THEN HrB.DeRef(y) END;
					pre := NIL; lastp := NIL;
					(*HrB.CheckBuffering(y, x, NIL, pre, lastp);*) (* for js/java, string handling is done by js/java *)
					HrB.Assign(x, y); 
					stmt := B.Assign(des, yy); stmt.textpos := pos;
					(*IF pre # NIL THEN SetPos(x); HrB.Construct(Ncomp, pre, x); x := pre END;*)
				ELSIF sym = eql THEN
				ELSIF (x.class = Nproc) & (x.obj.mode = SProc) (*qd IS T.SProc*) THEN
					xx := NIL; stmt := NIL; StandProcCall(x, qd(T.SProc).num, xx, stmt);
					IF (xx # NIL) & (stmt # NIL) THEN Err(-55, 1) END; 
					IF stmt = T.errstat THEN stmt := NIL ELSIF stmt # NIL THEN stmt.textpos := pos END;
					IF (x # NIL) & (x.typ # HrCPT.notyp) THEN err(55) END;
					IF (x # NIL) & (x.class = Nifelse) THEN	(* error pos for ASSERT *)
						SetPos(x.left); SetPos(x.left.right)
					END
				ELSIF x.class = Ncall THEN err(55)
				ELSE
					ASSERT((properp # NIL) OR (des # NIL), 60);
					IF (properp # NIL) OR (des.type IS T.Procedure) THEN
						Call(x, des, properp, isSuper, ap, stmt); 
						IF stmt # NIL THEN stmt.textpos := pos END; (* old text *)
					ELSIF des # T.errdes THEN err(-538)
					END
				END
			ELSIF sym = if THEN bkmrk := HrM.startpos;
				HrS.Get(sym); pos := HrM.startpos; 
				Log.String("IF:"); Log.Int(bkmrk); Log.Ln;
				Expression(x, xx); CheckBool(x); xx := CheckBool2(xx); CheckSym(then); StatSeq(y, stmt); 
				ifst := B.If(xx, stmt); ifst.textpos := bkmrk; elsest := ifst;
				HrB.Construct(Nif, x, y); SetPos(x); lastif := x;
				WHILE sym = elsif DO bkmrk := HrM.startpos;
					HrS.Get(sym); pos := HrM.startpos; bkmrk := HrM.startpos;
					Expression(y, yy); CheckBool(y); yy := CheckBool2(yy); CheckSym(then); 
					StatSeq(z, stmt); elsest.next := B.If(yy, stmt); elsest := elsest.next; elsest.textpos := bkmrk;
					HrB.Construct(Nif, y, z); SetPos(y); HrB.Link(x, lastif, y)
				END ;
				pos := HrM.startpos;
				IF sym = else THEN bkmrk := HrM.startpos; HrS.Get(sym); StatSeq(y, stmt);
					elsest.next := B.If(NIL, stmt); elsest.next.textpos := bkmrk
				ELSE y := NIL END ;
				HrB.Construct(Nifelse, x, y); CheckSym(end); HrB.OptIf(x);
				stmt := ifst
			ELSIF sym = case THEN
				HrS.Get(sym); pos := HrM.startpos; CasePart(x, stmt); CheckSym(end)
			ELSIF sym = while THEN bkmrk := HrM.startpos; 
				HrS.Get(sym); pos := HrM.startpos; 
				Expression(x, xx); CheckBool(x); xx := CheckBool2(xx); 
				CheckSym(do); StatSeq(y, stmt); stmt := B.While(xx, stmt); stmt.textpos := bkmrk;
				HrB.Construct(Nwhile, x, y); CheckSym(end)
			ELSIF sym = repeat THEN bkmrk := HrM.startpos;
				HrS.Get(sym); StatSeq(x, stmt);
				IF sym = until THEN HrS.Get(sym); pos := HrM.startpos; Expression(y, yy); 
					CheckBool(y); yy := CheckBool2(yy); stmt := B.Repeat(stmt, yy); stmt.textpos := bkmrk
				ELSE err(43); Err(-43, 0)
				END ;
				HrB.Construct(Nrepeat, x, y)
			ELSIF sym = for THEN bkmrk := HrM.startpos; 
				HrS.Get(sym); pos := HrM.startpos; 
				IF sym = ident THEN qualident(id, qd); des := B.Qualident(qd, properp); des.pos := HrM.errpos;
					(* for error handling, it may be better to parse a designator and make sure it's just a var  *)
					IF ~(id.typ.form IN intSet) THEN err(68) END ;
					IF ~(des.type.what IN T.integers) THEN err(-68) END;
					CheckSym(becomes); bkmrk2 := HrM.errpos; Expression(y, xx); xx.pos := bkmrk2;
					x := HrB.NewLeaf(id); HrB.Assign(x, y); SetPos(x);
					CheckSym(to); pos := HrM.startpos; bkmrk2 := HrM.startpos; 
					Expression(y, yy); yy.pos := bkmrk2;
					IF y.class # Nconst THEN
						T.MakeAppendTempVar(des.type, bkmrk2, bound);
						HrB.GetTempVar("@for", x.left.typ, t);
						z := HrB.NewLeaf(t); HrB.Assign(z, y); SetPos(z); HrB.Link(stat, last, z);
						y := HrB.NewLeaf(t)
					ELSE
						HrB.CheckAssign(x.left.typ, y)
					END ;
					HrB.Link(stat, last, x);
					p := HrM.startpos; 
					IF sym = by THEN HrS.Get(sym); ConstExpression(z, zz, _); IF zz # NIL THEN zz.pos := HrM.errpos END
					ELSE z := HrB.NewIntConst(1) ; zz := NIL
					END ;
					x := HrB.NewLeaf(id);
					IF z.conval.intval > 0 THEN HrB.Op(leq, x, y)
					ELSIF z.conval.intval < 0 THEN HrB.Op(geq, x, y)
					ELSE err(63); Err(-63, 1); HrB.Op(geq, x, y)
					END ;
					CheckSym(do); StatSeq(s, stmt);
					y := HrB.NewLeaf(id); HrB.StPar1(y, z, incfn); pos := HrM.startpos; SetPos(y);
					IF s = NIL THEN s := y
					ELSE z := s;
						WHILE z.link # NIL DO z := z.link END ;
						z.link := y
					END ;
					CheckSym(end); HrB.Construct(Nwhile, x, s); pos := p;
					stmt := B.For(des, xx, yy, zz, bound, stmt); stmt.textpos := bkmrk
				ELSE err(ident); Err(-ident, 0)
				END
			ELSIF sym = loop THEN
				HrS.Get(sym); INC(LoopLevel); StatSeq(x, stmt); stmt := B.Loop(stmt); DEC(LoopLevel);
				HrB.Construct(Nloop, x, NIL); CheckSym(end)
			ELSIF sym = with THEN withst := B.NewWith(); withst.textpos := HrM.startpos; clause := T.dir.WithClause();
				HrS.Get(sym); idtyp := NIL; x := NIL;
				LOOP
					IF sym < bar THEN
						pos := HrM.startpos; 
						IF withst.clauses = NIL THEN 
							clause := T.dir.WithClause(); withst.clauses := clause; stmt := withst; stmt.textpos := HrM.startpos
						ELSE clause.next := T.dir.WithClause(); clause := clause.next
						END;
						clause.pos := HrM.startpos; clause.is := T.errex;
						IF sym = ident THEN
							qualident(id, qd); orig := qd.type;
							WITH qd: T.Var DO des := B.Qualident(qd, properp); des.pos := HrM.errpos;
							ELSE Err(-130, 2); des := T.errdes	(*$132*)
							END;
							clause.var := des;
							y := HrB.NewLeaf(id);
							IF (id # NIL) & (id.typ.form = Pointer) & ((id.mode = VarPar) OR ~id.leaf) THEN
								err(-302)	(* warning 302 *)
							END ;
							WITH des: T.VarDes DO 
								IF (des.type IS T.Pointer) &
									(~(des.var IS T.Par) & (des.var.flags * T.visMask # T.domestic)
									OR (des.var IS T.Par) & (des.var.flags * T.outK # {} ) )
								THEN Err(-302, 1) (* warning 302 *)	(* what is this warning and condition all about? *)
								END
							ELSE
							END;
							CheckSym(colon); bkmrk := HrM.startpos + 1;
							IF sym = ident THEN qualident(t, tn); 
								WITH tn: T.TypeName DO clause.typeName := tn; 
									IF clause.var # T.errdes THEN clause.is := B.Is(clause.var, tn, bkmrk);
									(*IF qd.ptype = NIL THEN qd.ptype := qd.type END;*) qd.type := tn.type
									END
									(* Actually, I cannot figure out why .ptyp (.ptype) is needed. It is set here *)
								ELSE Err(-130, 1)	(*$132*)
								END;
								IF t.mode = Typ THEN
									IF id # NIL THEN
										idtyp := id.typ; HrB.TypTest(y, t, FALSE); id.typ := t.typ;
										IF id.ptyp = NIL THEN id.ptyp := idtyp END
									ELSE err(130)
									END
								ELSE err(52)
								END
							ELSE err(ident); Err(-130, 1) 	(*$132*)
							END
						ELSE err(ident); Err(-ident, 0)
						END ;
						CheckSym(do); StatSeq(s, clause.do); 
						HrB.Construct(Nif, y, s); SetPos(y);
						IF idtyp # NIL THEN
							IF id.ptyp = idtyp THEN id.ptyp := NIL END;
							id.typ := idtyp; idtyp := NIL
						END ;
						qd.type := orig;
						IF x = NIL THEN x := y; lastif := x ELSE HrB.Link(x, lastif, y) END
					END;
					IF sym = bar THEN HrS.Get(sym) ELSE EXIT END
				END;
				e := sym = else; pos := HrM.startpos;
				IF e THEN HrS.Get(sym); StatSeq(s, withst.else) ELSE s := NIL END ;
				HrB.Construct(Nwith, x, s); CheckSym(end);
				IF e THEN x.subcl := 1 END
			ELSIF sym = exit THEN
				HrS.Get(sym);
				IF LoopLevel = 0 THEN err(46); Err(-46, 1) END ;
				HrB.Construct(Nexit, x, NIL);
				stmt := B.Exit(); stmt.textpos := pos
			ELSIF sym = return THEN HrS.Get(sym);
				IF sym < semicolon THEN Expression(x, xx) END ;
				IF level > 0 THEN HrB.Return(x, HrCPT.topScope.link);
				ELSE (* not standard Oberon *) HrB.Return(x, NIL);
				END;
				stmt := B.Return(xx, T.nowScope); stmt.textpos := pos; 
				hasReturn := TRUE
			END ;
			IF x # NIL THEN SetPos(x); HrB.Link(stat, last, x) END ;
			IF stmt # NIL THEN 
				IF stats = NIL THEN stats := stmt ELSE lastst.link := stmt END; 
				lastst := stmt; WHILE lastst.link # NIL DO lastst := lastst.link END; 	(* some stmts may actually *)
			END;
			IF sym = semicolon THEN HrS.Get(sym)
			ELSIF (sym <= ident) OR (if <= sym) & (sym <= return) THEN err(semicolon); Err(-49, 1)	(*$*)
			ELSE EXIT
			END
		END
	END StatSeq;

	PROCEDURE Block (VAR procdec, statseq: HrCPT.Node; OUT body: T.Stat);
		CONST sConst = 1;
		VAR typ: HrCPT.Struct;
			obj, first, last, o: HrCPT.Object;
			x, lastdec: HrCPT.Node;
			i: SHORTINT;
			rname: HrCPT.Name;
			name: HrCPT.String;
			rec: Elem;
			ob: T.Object; tn, rn: T.TypeName; t: T.Type; nam: POINTER TO T.Name; what, pos, num: INTEGER;
			ee: T.Expression; c: T.Const; v, fv, lv: T.Var; p: T.Proc;
	BEGIN
		unknownTypesAllowed := TRUE; topRecs := 0;
		IF ((sym < begin) OR (sym > var)) & (sym # procedure) & (sym # end) & (sym # close) THEN err(36) END;
		first := NIL; last := NIL; userList := NIL; recList := NIL;
		LOOP
			IF sym = const THEN
				HrS.Get(sym);
				WHILE sym = ident DO
					HrCPT.Insert(HrS.name, obj);
					pos := HrM.startpos; nam := ThisName(HrS.name); c := T.dir.Const();
					obj.textpos := pos;
					obj.mode := Con; CheckMark(obj, sConst, c.flags, c.lib, c.ent, c.modifiers);
					obj.typ := HrCPT.int8typ; obj.mode := Var;	(* Var to avoid recursive definition *)
					T.AppendThis(c, nam, pos);
					IF sym = eql THEN
						HrS.Get(sym); ConstExpression(x, c.ex, c.val)
					ELSIF sym = becomes THEN
						err(eql); HrS.Get(sym); ConstExpression(x, c.ex, c.val)
					ELSE err(eql); x := HrB.NewIntConst(1); c.val := T.errcon; c.ex := T.errex
					END ;
					c.type := c.val.type;
					obj.mode := Con; obj.typ := x.typ; obj.conval := x.conval; (* ConstDesc is not copied *)
					CheckSym(semicolon)
				END
			END ;
			IF sym = type THEN
				HrS.Get(sym); num := -1;
				WHILE sym = ident DO t := NIL; INC(num);
					HrCPT.Insert(HrS.name, obj); obj.mode := Typ; obj.typ := HrCPT.undftyp;
					obj.textpos := HrM.startpos;
					tn := T.dir.TypeName(); nam := ThisName(HrS.name); pos := HrM.startpos;
					CheckMark(obj, sType, tn.flags, tn.lib, tn.ent, tn.modifiers); obj.mode := -1;
					tn.name := nam; tn.textpos := pos;
					IF sym # eql THEN err(eql) END;
					IF (sym = eql) OR (sym = becomes) OR (sym = colon) THEN
						HrS.Get(sym); Type(obj.typ, name, t, tn); SetType(NIL, obj, obj.typ, name)
					ELSE t := T.errtp
					END;
					SetType2(NIL, tn, t, name);
					(*T.AppendThis(tn, nam, pos) (* AFTER tn.type has been set! *);*)
					obj.mode := Typ;
					IF obj.typ.form IN {Byte..Set, Char16, Int64} THEN	(* make alias structure *)
						typ := HrCPT.NewStr(obj.typ.form, Basic); i := typ.ref;
						typ^ := obj.typ^; typ.ref := i; typ.strobj := NIL; typ.mno := 0; typ.txtpos := HrM.errpos;
						typ.BaseTyp := obj.typ; (* this is how alias types are linked!! *)
						obj.typ := typ;
					END;	
					(* So, A = CHAR and B = CHAR => A.type # B.type - can distinguish alias types!!! But, is it needed? *)
					IF obj.typ.strobj = NIL THEN obj.typ.strobj := obj END ;
					IF tn.type.typeName = NIL THEN tn.type.typeName := tn END;
					IF obj.typ.form = Pointer THEN	(* !!! *)
						typ := obj.typ.BaseTyp;
						IF (typ # NIL) & (typ.comp = Record) & (typ.strobj = NIL) THEN
							(* pointer to unnamed record: name record as "pointerName^" *)
							rname := obj.name$; i := 0;
							WHILE rname[i] # 0X DO INC(i) END;
							rname[i] := "^"; rname[i+1] := 0X;
							HrCPT.Insert(rname, o); o.mode := Typ; o.typ := typ; typ.strobj := o
						END
					END;
					WITH t: T.RecPtr DO 
						ASSERT(t.base.typeName # NIL);
						IF T.exported IN tn.flags THEN 
							INCL(t.base.typeName.flags, T.exported); INCL(t.flags, T.exported) 
						END;	(* is T.exported used with T.Type? *)
					| t: T.Record DO IF T.exported IN tn.flags THEN INCL(t.flags, T.exported) END
					ELSE 
					END;
					IF obj.vis # internal THEN
						typ := obj.typ;
						IF typ.form = Pointer THEN typ := typ.BaseTyp END;
						IF typ.comp = Record THEN typ.exp := TRUE END
					END;
					CheckSym(semicolon)
		;IF nam^ = "Str" THEN ASSERT(~(T.open IN tn.type.flags), 77) END;
				END
			END ;
			IF sym = var THEN
				HrS.Get(sym);
				WHILE sym = ident DO
					LOOP
						IF sym = ident THEN
							HrCPT.Insert(HrS.name, obj);
							obj.mode := Var; obj.link := NIL; obj.leaf := obj.vis = internal; obj.typ := HrCPT.undftyp;
							pos := HrM.startpos; 
							obj.textpos := pos;
							IF level = 0 THEN v := T.dir.GlobVar(); what := sGVar ELSE v := T.dir.LocVar(); what := sLVar END;
							v.name := ThisName(HrS.name); v.textpos := pos;
							CheckMark(obj, what, v.flags, mark.lib, mark.ent, mark.mod);
							WITH v: T.GlobVar DO v.lib := mark.lib; v.ent := mark.ent; v.modifiers := v.modifiers ELSE END;
							IF fv = NIL THEN fv := v END; IF lv # NIL THEN lv.next := v END; lv := v;
							IF first = NIL THEN first := obj END ;
							IF last = NIL THEN HrCPT.topScope.scope := obj ELSE last.link := obj END ;
							last := obj
						ELSE err(ident)
						END ;
						IF sym = comma THEN HrS.Get(sym)
						ELSIF sym = ident THEN err(comma)
						ELSE EXIT
						END
					END ;
					CheckSym(colon); Type(typ, name, t, NIL);
					CheckAlloc(typ, FALSE, HrM.errpos);	(*?*)
					WHILE first # NIL DO SetType(NIL, first, typ, name); first := first.link END ;
					v := fv; WHILE v # NIL DO SetType2(NIL, v, t, name); 
						IF fv.next = NIL THEN fv := NIL ELSE fv := fv.next(T.Var) END; 
						v.next := NIL; T.Append(v); v := fv
					END;
					CheckSym(semicolon)
				END
			END ;
			IF (sym < const) OR (sym > var) THEN EXIT END ;
		END ;
		CheckForwardTypes;
		CheckForwardTypes2; fixups := NIL;
		userList := NIL; rec := recList; recList := NIL;
		HrCPT.topScope.adr := HrM.errpos;
		unknownTypesAllowed := FALSE;
		procdec := NIL; lastdec := NIL;
		IF (sym # procedure) & (sym # begin) & (sym # end) & (sym # close) THEN err(37) END;
		WHILE sym = procedure DO procStartPos := HrM.startpos;
			HrS.Get(sym); ProcedureDeclaration(x, p);
			IF x # NIL THEN
				IF lastdec = NIL THEN procdec := x ELSE lastdec.link := x END ;
				lastdec := x
			END ;
			CheckSym(semicolon)
		END ;
		IF HrM.noerr & ~(HrM.oberon IN HrM.options) THEN CheckRecords(rec) END;
		CheckForwardProcedures;
		hasReturn := FALSE;
		IF (sym # begin) & (sym # end) & (sym # close) THEN err(38) END;
		IF sym = begin THEN HrS.Get(sym); StatSeq(statseq, body)
		ELSE statseq := NIL
		END ;
		IF (HrCPT.topScope.link # NIL) & (HrCPT.topScope.link.typ # HrCPT.notyp)
			& ~hasReturn & (HrCPT.topScope.link.sysflag = 0) THEN err(133) END;
		IF (level = 0) & (TDinit # NIL) THEN
			lastTDinit.link := statseq; statseq := TDinit	(*?*)
		END
	END Block;
	
	PROCEDURE Utf8ToString (IN utf: ARRAY OF SHORTCHAR; OUT str: ARRAY OF CHAR);
		VAR res: INTEGER;
	BEGIN Utf.Utf8ToString(utf, str, res); ASSERT(res = 0, 20)
	END Utf8ToString;

	PROCEDURE Module* (VAR prog: HrCPT.Node; top: T.Segraph);
		VAR impName, aliasName: HrCPT.Name;
				procdec, statseq: HrCPT.Node;
				c: INTEGER; done: BOOLEAN; link: HrCPT.Object;
				m: T.Mod;
	BEGIN ASSERT(HrCPT.nofGmod # 0);
		HrS.Init; LoopLevel := 0; level := 0; HrS.Get(sym); 
		IF sym = module THEN HrS.Get(sym) ELSE err(16) END ;
		IF sym = ident THEN
			HrCPT.Open(HrS.name); 
			m := top.mod[0]; m.name := ThisName(HrS.name);
			T.nowScope := m;
			HrS.Get(sym);
			HrCPT.libName := "";
			IF sym = lbrak THEN
				INCL(HrM.options, HrM.interface); HrS.Get(sym);
				IF sym = eql THEN HrS.Get(sym)
				ELSE INCL(HrM.options, HrM.noCode)
				END;
				IF sym = string THEN HrCPT.libName := HrS.str$; 
					NEW(m.lib); m.lib^ := HrS.str$;
					HrS.Get(sym)
				ELSE err(string)
				END;
				CheckSym(rbrak)
			END;
			CheckSym(semicolon);
			IF sym = import THEN HrS.Get(sym);
				LOOP
					IF sym = ident THEN
						aliasName := HrS.name$; impName := aliasName$; HrS.Get(sym);
						IF sym = becomes THEN HrS.Get(sym);
							IF sym = ident THEN impName := HrS.name$; HrS.Get(sym) ELSE err(ident) END
						END ;
						IF impName = "JAVASCRIPT" THEN INCL(HrM.options, HrM.javascript);
							ASSERT(HrCPT.nofGmod # 0);
							link := T.BuildJSMod(-HrCPT.nofGmod) 
						END;
						HrCPT.Import(aliasName, impName, link, done);
						T.Import(top, impName$, aliasName$, done); IF ~done THEN Err(151, 1) END;
					ELSE err(ident)
					END ;
					IF sym = comma THEN HrS.Get(sym)
					ELSIF sym = ident THEN err(comma)
					ELSE EXIT
					END
				END ;
				CheckSym(semicolon)
			END ;
			IF HrM.noerr THEN TDinit := NIL; lastTDinit := NIL; c := HrM.errpos;
				Block(procdec, statseq, m.begin); 
				IF m.begin # NIL THEN m.endBegPos := HrM.startpos END;
				HrB.Enter(procdec, statseq, NIL); prog := procdec;
				prog.conval := HrCPT.NewConst(); prog.conval.intval := c; prog.conval.intval2 := HrM.startpos;
				IF sym = close THEN HrS.Get(sym); StatSeq(prog.link, m.close); m.endClosePos := HrM.startpos END;
				prog.conval.realval := HrM.startpos;
				CheckSym(end);
				IF sym = ident THEN
					IF HrS.name # HrCPT.SelfName THEN err(4) END ;
					HrS.Get(sym)
				ELSE err(ident)
				END;
				IF sym # period THEN err(period) END
			END
		ELSE err(ident)
		END ;
		TDinit := NIL; lastTDinit := NIL;
		HrS.str := NIL
	END Module;
