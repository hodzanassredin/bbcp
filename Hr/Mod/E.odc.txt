	(*HrE assumes the following memory map for a BlackBox module:
	
	Meta - the metadata block created by the compiler and written into OCF
	Proxies - references to imported procedures and, possibly, variables (TBD); filled out by the BB loader (or the host loader for ELFs) based on the UseBlk in the OCF (prepared by the compiler)
	Code - the machine text, or code, generated by the compiler and written into OCF
	Desc - the descriptor of the module - this resilient metadata it is not deallocated when the module is unloaded; written into OCF
	Vars - global variables of the module; allocated by the loader, not present in OCF
	
	In order to be (easily) loadable from ELF (and probably PE), Meta, Code and Desc start on AMD64 page boundaries.
	Cf. PROCEDURE AllocateBlocks
	*)

	PROCEDURE GetLongWords* (con: DevCPT.Const; OUT hi, low: INTEGER);
		CONST N = 4294967296.0; (* 2^32 *)
		VAR rh, rl: REAL;
	BEGIN
		rl := con.intval; rh := con.realval / N;
		IF rh >= MAX(INTEGER) + 1.0 THEN rh := rh - 1; rl := rl + N
		ELSIF rh < MIN(INTEGER) THEN rh := rh + 1; rl := rl - N
		END;
		hi := SHORT(ENTIER(rh));
		rl := rl + (rh - hi) * N;
		IF rl < 0 THEN hi := hi - 1; rl := rl + N
		ELSIF rl >= N THEN hi := hi + 1; rl := rl - N
		END;
		IF rl >= MAX(INTEGER) + 1.0 THEN rl := rl - N END;
		low := SHORT(ENTIER(rl))
(*
		hi := SHORT(ENTIER((con.realval + con.intval) / 4294967296.0));
		r := con.realval + con.intval - hi * 4294967296.0;
		IF r > MAX(INTEGER) THEN r := r - 4294967296.0 END;
		low := SHORT(ENTIER(r))
*)
	END GetLongWords;
	
	PROCEDURE GetRealWord* (con: DevCPT.Const; OUT x: INTEGER);
		VAR r: SHORTREAL;
	BEGIN
		r := SHORT(con.realval); x := SYSTEM.VAL(INTEGER, r)
	END GetRealWord;
		
	PROCEDURE GetRealWords* (con: DevCPT.Const; OUT hi, low: INTEGER);
		TYPE A = ARRAY 2 OF INTEGER;
		VAR a: A;
	BEGIN
		a := SYSTEM.VAL(A, con.realval);
		IF DevCPM.LEHost THEN hi := a[1]; low := a[0] ELSE hi := a[0]; low := a[1] END
	END GetRealWords;
		
	PROCEDURE IsSame (x, y: REAL): BOOLEAN;
	BEGIN
		RETURN  (x = y) & ((x #  0.) OR (1. / x = 1. / y))
	END IsSame;
	
	PROCEDURE AllocConst* (con: DevCPT.Const; form: BYTE; VAR obj: DevCPT.Object; VAR adr: INTEGER);
		VAR c: DevCPT.Const;
	BEGIN
		INCL(con.setval, form);
		CASE form OF
		| String8:
			obj := Const8; c := obj.conval;
			WHILE (c # NIL) & ((con.setval # c.setval) OR (con.intval2 # c.intval2) OR (con.ext^ # c.ext^)) DO c := c.link END;
			IF c = NIL THEN adr := idx8; INC(idx8, (con.intval2 + 3) DIV 4 * 4) END
		| String16:
			obj := Const16; c := obj.conval;
			WHILE (c # NIL) & ((con.setval # c.setval) OR (con.intval2 # c.intval2) OR (con.ext^ # c.ext^)) DO c := c.link END;
			IF c = NIL THEN adr := idx16; INC(idx16, (con.intval2 + 1) DIV 2 * 4) END
		| Int64:
			obj := Const64; c := obj.conval;
			WHILE (c # NIL) & ((con.setval # c.setval) OR (con.intval # c.intval2) OR (con.realval # c.realval)) DO
				c := c.link
			END;
			IF c = NIL THEN con.intval2 := con.intval; adr := idx64; INC(idx64, 8) END
		| Real32:
			obj := Const32; c := obj.conval;
			WHILE (c # NIL) & ((con.setval # c.setval) OR ~IsSame(con.realval, c.realval)) DO c := c.link END;
			IF c = NIL THEN adr := idx32; INC(idx32, 4) END
		| Real64:
			obj := Const64; c := obj.conval;
			WHILE (c # NIL) & ((con.setval # c.setval) OR ~IsSame(con.realval, c.realval)) DO c := c.link END;
			IF c = NIL THEN adr := idx64; INC(idx64, 8) END
		| Guid:
			obj := Const32; c := obj.conval;
			WHILE (c # NIL) & ((con.setval # c.setval) OR (con.intval2 # c.intval2) OR (con.ext^ # c.ext^)) DO c := c.link END;
			IF c = NIL THEN adr := idx32; INC(idx32, 16) END
		END;
		IF c = NIL THEN con.link := obj.conval; obj.conval := con ELSE adr := c.intval END;
		con.intval := adr
	END AllocConst;

	PROCEDURE DoAllocLiteral (VAR list: T.Literal; VAR lastAddr: INTEGER; size: INTEGER; lit: T.Literal): T.Literal;
		VAR q, s: T.Literal; w, sw: INTEGER;
	BEGIN 
		w := lit.type.what; q := list;
		WHILE q # NIL DO s := q; sw := s.type.what;
			WITH s: Y.Real DO
				IF (sw = w) & s.Equals(lit) THEN q := NIL ELSE q := s.link; s := NIL END
			| s: Y.Int DO IF (sw = w) & s.Equals(lit) THEN q := NIL ELSE q := s.link; s := NIL END
			| s: Y.StringLit DO IF (sw = w) & s.Equals(lit) THEN q := NIL ELSE q := s.link; s := NIL END
			ELSE HALT(60)
			END
		END;
		IF s = NIL THEN
			DEC(lastAddr, size); 
			WITH lit: Y.Real DO lit.adr := lastAddr; lit.link := list
			| lit: Y.Int DO lit.adr := lastAddr; lit.link := list
			| lit: Y.StringLit DO lit.adr := lastAddr; lit.link := list
			ELSE HALT(126)
			END;
			list := lit
		ELSE lit := s
		END;
		RETURN lit
	END DoAllocLiteral;

	PROCEDURE EAllocLiteral* (VAR con: T.Literal; what: INTEGER; (*VAR o: T.Object; *)VAR adr: INTEGER);
		VAR s, q: T.Literal; fnd: BOOLEAN; w: INTEGER;
	BEGIN
		(*INCL(con.setval, form);	(* ? what? *)*)
		w := con.type.what;
		WITH 
		| con: Y.StringLit DO
			IF w = T.fString8 THEN con := DoAllocLiteral(strings, last.str, con.len + 1, con)(Y.StringLit)
			ELSIF w = T.fString16 THEN HALT(126)
			ELSE HALT(20)
			END
		| con: Y.Real DO 
			IF w = T.fReal32 THEN con := DoAllocLiteral(doubles, last.dbl, 4, con)(Y.Real) 
			ELSIF w = T.fReal64 THEN con := DoAllocLiteral(quads, last.quad, 8, con)(Y.Real)
			ELSE HALT(20) 
			END
		| con: Y.Int DO
			IF w = T.fInt64 THEN con := DoAllocLiteral(quads, last.quad, 8, con)(Y.Int) 
			ELSIF w = T.fInt32 THEN con := DoAllocLiteral(doubles, last.dbl, 4, con)(Y.Int)
			ELSE HALT(20) 
			END
		ELSE HALT(126)
		END
	END EAllocLiteral;

	PROCEDURE PutByte* (a, x: INTEGER);
	BEGIN
		code[a DIV CodeLength]^[a MOD CodeLength] := SHORT(CHR(x MOD 256))
	END PutByte;
	
	PROCEDURE PutShort* (a, x: INTEGER);
	BEGIN
		IF bigEndian THEN
			PutByte(a, x DIV 256); PutByte(a + 1, x)
		ELSE
			PutByte(a, x); PutByte(a + 1, x DIV 256)
		END
	END PutShort;
	
	PROCEDURE PutWord* (a, x: INTEGER);
	BEGIN
		IF bigEndian THEN
			PutByte(a, x DIV 1000000H); PutByte(a + 1, x DIV 10000H);
			PutByte(a + 2, x DIV 256); PutByte(a + 3, x)
		ELSE
			PutByte(a, x); PutByte(a + 1, x DIV 256);
			PutByte(a + 2, x DIV 10000H); PutByte(a + 3, x DIV 1000000H)
		END
	END PutWord;
	
	PROCEDURE ThisByte* (a: INTEGER): INTEGER;
	BEGIN
		RETURN ORD(code[a DIV CodeLength]^[a MOD CodeLength])
	END ThisByte;
	
	PROCEDURE ThisShort* (a: INTEGER): INTEGER;
	BEGIN
		IF bigEndian THEN
			RETURN ThisByte(a) * 256 + ThisByte(a+1)
		ELSE
			RETURN ThisByte(a+1) * 256 + ThisByte(a)
		END
	END ThisShort;
	
	PROCEDURE ThisWord* (a: INTEGER): INTEGER;
	BEGIN
		IF bigEndian THEN
			RETURN ((ThisByte(a) * 256 + ThisByte(a+1)) * 256 + ThisByte(a+2)) * 256 + ThisByte(a+3)
		ELSE
			RETURN ((ThisByte(a+3) * 256 + ThisByte(a+2)) * 256 + ThisByte(a+1)) * 256 + ThisByte(a)
		END
	END ThisWord;
	
	PROCEDURE GenByte* (x: INTEGER);
	BEGIN
		(*ASSERT(pc # 12);*)
		IF actIdx >= CodeLength THEN
			IF blkIdx < CodeBlocks THEN
				NEW(actual); code[blkIdx] := actual; INC(blkIdx); actIdx := 0
			ELSE
				IF ~CodeOvF THEN DevCPM.err(210); CodeOvF := TRUE END;
				actIdx := 0; pc := 0
			END
			;SYSTEM.MOVE(SYSTEM.ADR(actual), SYSTEM.ADR(peep), SIZE(CodeBlock))
		END;
		actual[actIdx] := SHORT(CHR(x MOD 256)); INC(actIdx); INC(pc)
	END GenByte;
	
	PROCEDURE GenShort* (x: INTEGER);
	BEGIN
		IF bigEndian THEN
			GenByte(x DIV 256); GenByte(x)
		ELSE
			GenByte(x); GenByte(x DIV 256)
		END
	END GenShort;
	
	PROCEDURE GenWord* (x: INTEGER);
	BEGIN
		IF bigEndian THEN
			GenByte(x DIV 1000000H); GenByte(x DIV 10000H); GenByte(x DIV 256); GenByte(x)
		ELSE
			GenByte(x); GenByte(x DIV 256); GenByte(x DIV 10000H); GenByte(x DIV 1000000H)
		END
	END GenWord;
	
	PROCEDURE WriteCode;
		VAR i, j, k, n: INTEGER; b: CodeBlock;
	BEGIN
		j := 0; k := 0;
		WHILE j < pc DO
			n := pc - j; i := 0; b := code[k];
			IF n > CodeLength THEN n := CodeLength END;
			WHILE i < n DO M.ObjW(b^[i]); INC(i) END;
			INC(j, n); INC(k)
		END
	END WriteCode;
	
	PROCEDURE Snatch* (OUT a: ARRAY OF SHORTCHAR; OUT vWRTcode: INTEGER);
		VAR m, i, j, k, n: INTEGER; b: CodeBlock;
	BEGIN ASSERT(LEN(a) >= pc, 20);
		j := 0; k := 0; m := 0; 
		WHILE j < pc DO
			n := pc - j; i := 0; b := code[k];
			IF n > CodeLength THEN n := CodeLength END;
			WHILE i < n DO a[m] := b^[i]; INC(i); INC(m) END;
			INC(j, n); INC(k)
		END;
		vWRTcode := map.eVars
	END Snatch;


	PROCEDURE OffsetLink* (obj: DevCPT.Object; offs: INTEGER): DevCPT.LinkList;
		VAR link: DevCPT.LinkList; m: DevCPT.Object;
	BEGIN
		ASSERT((obj.mode # Typ) OR (obj.typ # DevCPT.int32typ));
		ASSERT((obj.mode # Typ) OR (obj.typ # DevCPT.iunktyp) & (obj.typ # DevCPT.guidtyp));
		IF obj.mnolev >= 0 THEN	(* not imported *)
			CASE obj.mode OF
			| Typ: IF obj.links = NIL THEN obj.link := descList; descList := obj END
			| TProc: IF obj.adr = -1 THEN obj := obj.nlink ELSE offs := offs + obj.adr; obj := Code END
			| Var: offs := offs + dsize; obj := Data
			| Con:
			| IProc, XProc, LProc: HALT(126) (* check to see when procedure's fixups are requested *)
			END
		ELSIF obj.mode = Typ THEN
			IF obj.typ.untagged THEN	(* add desc for imported untagged types *)
				IF obj.links = NIL THEN obj.link := descList; descList := obj END
			ELSE
				m := DevCPT.GlbMod[-obj.mnolev];
				IF m.library # NIL THEN RETURN NIL END	(* type import from dll *)
			END
		END;
		link := obj.links;
		WHILE (link # NIL) & (link.offset # offs) DO link := link.next END;
		IF link = NIL THEN
			NEW(link); link.offset := offs; link.linkadr := 0;
			link.next := obj.links; obj.links := link
		END;
		RETURN link
	END OffsetLink;

	PROCEDURE EOffsetLink* (obj: DevCPT.Object; offs: INTEGER): DevCPT.LinkList;
	(** Find the LinkList (head of fixup bookmark chain) for object identified by offs among fixup bookmark lists anchored in obj; create one if necessary *)
		VAR link: DevCPT.LinkList; m: DevCPT.Object;
	BEGIN
		ASSERT((obj.mode # Typ) OR (obj.typ # DevCPT.int32typ));
		ASSERT((obj.mode # Typ) OR (obj.typ # DevCPT.iunktyp) & (obj.typ # DevCPT.guidtyp));
		IF obj.mnolev >= 0 THEN	(* not imported *)
			CASE obj.mode OF
			| Typ: IF obj.links = NIL THEN obj.link := descList; descList := obj END
			| TProc: IF obj.adr = -1 THEN obj := obj.nlink ELSE offs := offs + obj.adr; obj := Code END
			| Var: offs := offs + dsize; obj := Data
			| Con, IProc, XProc, LProc:
			END
		ELSIF obj.mode = Typ THEN
			IF obj.typ.untagged THEN	(* add desc for imported untagged types *)
				IF obj.links = NIL THEN obj.link := descList; descList := obj END
			ELSE
				m := DevCPT.GlbMod[-obj.mnolev];
				IF m.library # NIL THEN RETURN NIL END	(* type import from dll *)
			END
		END;
		link := obj.links;
		WHILE (link # NIL) & (link.offset # offs) DO link := link.next END;
		IF link = NIL THEN
			NEW(link); link.offset := offs; link.linkadr := 0;
			link.next := obj.links; obj.links := link
		END;
		RETURN link
	END EOffsetLink;


	PROCEDURE TypeObj* (typ: DevCPT.Struct): DevCPT.Object;	(* => EGetTypeName *)

	PROCEDURE Align (n: INTEGER);
		VAR p: INTEGER;
	BEGIN
		p := DevCPM.ObjLen();
		DevCPM.ObjWBytes(zero, (-p) MOD n)
	END Align;
	
	PROCEDURE LAlign (n: INTEGER);
		VAR p: INTEGER;
	BEGIN
		p := M.ObjLen();
		M.ObjWBytes(zero, (-p) MOD n)
	END LAlign;
	
	PROCEDURE OutName (IN name: ARRAY OF SHORTCHAR);
		VAR ch: SHORTCHAR; i: SHORTINT;
	BEGIN i := 0;
		REPEAT ch := name[i]; DevCPM.ObjW(ch); INC(i) UNTIL ch = 0X
	END OutName;
	
	PROCEDURE EOutName (IN name: ARRAY OF CHAR);
		VAR ch: CHAR; i: SHORTINT; buf: ARRAY LEN(T.Name) * 2 OF SHORTCHAR; res: INTEGER;
	BEGIN i := 0;
		Utf.StringToUtf8(name, buf, res);
		REPEAT ch := buf[i]; ASSERT(ch <= MAX(SHORTCHAR), 126); M.ObjW(SHORT(ch)); INC(i) UNTIL ch = 0X
	END EOutName;
	
	PROCEDURE Out2 (x: INTEGER);	(* byte ordering must correspond to target machine *)
	BEGIN
		IF bigEndian THEN
			M.ObjW(SHORT(CHR(x DIV 256))); M.ObjW(SHORT(CHR(x)))	(* value out of range *)
		ELSE
			M.ObjCH(CHR(x))
		END
	END Out2;
	
	PROCEDURE Out4 (x: INTEGER);	(* byte ordering must correspond to target machine *)
	BEGIN
		IF bigEndian THEN
			M.ObjW(SHORT(CHR(x DIV 1000000H))); M.ObjW(SHORT(CHR(x DIV 10000H)));
			M.ObjW(SHORT(CHR(x DIV 256))); M.ObjW(SHORT(CHR(x)))
		ELSE
			M.ObjWLInt(x)
		END
	END Out4;
	
	PROCEDURE Out8 (long: LONGINT);
		TYPE A8 = ARRAY 8 OF SHORTCHAR; 
		VAR a8: A8; ch: SHORTCHAR;
	BEGIN
		a8 := SYSTEM.VAL(A8, long);
		IF M.LEHost = bigEndian THEN
			ch := a8[0]; a8[0] := a8[7]; a8[7] := ch;
			ch := a8[1]; a8[1] := a8[6]; a8[6] := ch;
			ch := a8[2]; a8[2] := a8[5]; a8[5] := ch;
			ch := a8[3]; a8[3] := a8[4]; a8[4] := ch
		END;
		M.ObjWBytes(a8, 8)
	END Out8;
	
	PROCEDURE ThisChain (VAR(*why VAR?*) head: Chain; offs: INTEGER): Chain;
	(* Looks like ThisChain should map objects → chains, i.e. it should identify an object's chain *)
		VAR res: Chain;
	BEGIN res := head;
		WHILE (res # NIL) & (res.offset # offs) DO res := res.next END;
		IF res = NIL THEN
			NEW(res); res.offset := offs; res.last := 0;
			IF head = NIL THEN head := res ELSE res.next := head; head := res END
		END;
		RETURN res
	END ThisChain;
	
	PROCEDURE DoOutBookmark (VAR last: INTEGER; typ: INTEGER);
	BEGIN
		Out4(typ * 1000000H + last MOD 1000000H);
		IF typ = absolute64 THEN Out4(11223344H); last := -(M.ObjLen() - headSize - 8)
		ELSE last := -(M.ObjLen() - headSize - 4)	(* '-' means that it is an offset into Meta. What about desc? *)
		END;
	END DoOutBookmark;
	
	PROCEDURE EOutBookmark (VAR head: Chain; offs, typ: INTEGER);
	(** Output a new fixup bookmark of type typ for object identified by offs anchored in head *)
		VAR chain: Chain;
	BEGIN ASSERT((typ = absolute) OR (typ = absolute64), 126);
		chain := ThisChain(head, offs);
		IF chain # NIL THEN DoOutBookmark(chain.last, typ) ELSIF typ = absolute64 THEN Out8(0) ELSE Out4(0) END;
	END EOutBookmark;
	
	PROCEDURE OutReference (*→EOutBookmark, EOutDescBookmark*) (obj: DevCPT.Object; offs, typ: INTEGER);
		VAR link: DevCPT.LinkList;
	BEGIN
		link := OffsetLink(obj, offs);
		IF link # NIL THEN
			Out4(typ * 1000000H + link.linkadr MOD 1000000H);
			link.linkadr := -(M.ObjLen() - headSize - 4)
		ELSE Out4(0)
		END
	END OutReference;

	PROCEDURE IncThisType (type: T.Type; VAR typ: Y.Type);
	(** Assuming typ is type's typ, include type into the list of proper or imported types *)
		VAR last: T.Type;
	BEGIN
		IF typ.link = NIL THEN
			IF T.IsProper(type) THEN 
				ASSERT(~nomorepropertypes, 20);
				IF types.last = terminator THEN types.proper := type; types.last := type
				ELSE last := types.last;
					WITH last: Y.Record DO last.typ.link := type| last: Y.RecPtr DO last.typ.link := type
					| last: Y.Array DO last.typ.link := type (*| last: Y.ArrayPtr DO last.link := type*)
					ELSE HALT(126)
					END;
					types.last := type
				END;
				typ.link := terminator
			ELSE typ.link := types.imp; types.imp := type; INC(impsCount)
			END
		END
	END IncThisType;

		END
	END InsOffsBkmrk;

	PROCEDURE FindPtrs (typ: DevCPT.Struct; adr: INTEGER; ip: BOOLEAN; VAR num: INTEGER);
		VAR fld: DevCPT.Object; btyp: DevCPT.Struct; i, n: INTEGER;
	BEGIN
		IF typ.form = Pointer THEN
			IF ip & (typ.sysflag = interface)
				OR ~ip & ~typ.untagged THEN Out4(adr); INC(num) END
		ELSIF (typ.comp = Record) & (typ.sysflag # union) THEN
			btyp := typ.BaseTyp;
			IF btyp # NIL THEN FindPtrs(btyp, adr, ip, num) END ;
			fld := typ.link;
			WHILE (fld # NIL) & (fld.mode = Fld) DO
				IF ip & (fld.name^ = DevCPM.HdUtPtrName) & (fld.sysflag = interface)
					OR ~ip & (fld.name^ = DevCPM.HdPtrName) THEN Out4(fld.adr + adr); INC(num)
				ELSE FindPtrs(fld.typ, fld.adr + adr, ip, num)
				END;
				fld := fld.link
			END
		ELSIF typ.comp = Array THEN
			btyp := typ.BaseTyp; n := typ.n;
			WHILE btyp.comp = Array DO n := btyp.n * n; btyp := btyp.BaseTyp END ;
			IF (btyp.form = Pointer) OR (btyp.comp = Record) THEN
				i := num; FindPtrs(btyp, adr, ip, num);
				IF num # i THEN i := 1;
					WHILE i < n DO
						INC(adr, btyp.size); FindPtrs(btyp, adr, ip, num); INC(i)
					END
				END
			END
		END
	END FindPtrs;
	
	PROCEDURE DoFindPtrs (typ: T.Type; adr: INTEGER; ip: BOOLEAN; VAR num: INTEGER);
		VAR btyp: DevCPT.Struct; i, n: INTEGER;
			f: POINTER TO ARRAY OF T.Field; fld: T.Field; elt: T.Type;
	BEGIN
		WITH 
		| typ: T.Basic DO
		| typ: T.Procedure DO
		| typ: T.Record DO
			IF typ.flags * T.strucMask # T.unionF THEN
				IF typ.base # NIL THEN DoFindPtrs(typ.base, adr, ip, num) END;
				f := typ.fields; IF f = NIL THEN n := 0 ELSE n := LEN(f) END;
				i := 0; WHILE i < n DO fld := f[i];
					IF ip & (fld.name^ = M.HdUtPtrName) & (fld.flags * T.strucMask = T.interfaceF)
						OR ~ip & (fld.name^ = M.HdPtrName) THEN Out4(fld.offs + adr); INC(num)
					ELSE DoFindPtrs(fld.type, fld.offs + adr, ip, num)
					END;
					INC(i)
				END
			END
		| typ: T.Pointer DO 
			IF ip & (T.interfaceF = typ.flags * T.strucMask) OR ~ip & ~(T.untagged IN typ.flags) THEN 
				Out4(adr); INC(num) 
			END
		| typ: T.Array DO 
			REPEAT elt := typ.elTyp; WITH elt: T.Array DO typ := elt ELSE typ := NIL END UNTIL typ = NIL;
			DoFindPtrs(elt, adr, ip, num);
		ELSE HALT(126)
		END
	END DoFindPtrs;
	
	PROCEDURE EFindPtrs (typ: T.Type; ip: BOOLEAN; OUT num: INTEGER);
	(** Find num of pointers in a value of type typ and, if ip, out the offset of each pointer within the value *)
	BEGIN num := 0; DoFindPtrs(typ, 0, ip, num)
	END EFindPtrs;

	PROCEDURE OutRefs* (obj: DevCPT.Object);
		VAR f: BYTE;
	BEGIN
		IF outRef & (obj # NIL) THEN
			OutRefs(obj.left);
			IF ((obj.mode = Var) OR (obj.mode = VarPar)) & (obj.history # removed) & (obj.name[0] # "@") THEN
				f := obj.typ.form;
				IF (f IN {Byte .. Set, Pointer, ProcTyp, Char16, Int64}) 
						OR outURef & (obj.typ.comp # DynArr)
						OR outAllRef & ~obj.typ.untagged
						OR (obj.typ.comp = Array) & (obj.typ.BaseTyp.form = Char8) THEN
					IF obj.mode = Var THEN DevCPM.ObjW(0FDX) ELSE DevCPM.ObjW(0FFX) END;
					IF obj.typ = DevCPT.anyptrtyp THEN DevCPM.ObjW(SHORT(CHR(mAnyPtr)))
					ELSIF obj.typ = DevCPT.anytyp THEN DevCPM.ObjW(SHORT(CHR(mAnyRec)))
					ELSIF obj.typ = DevCPT.sysptrtyp THEN DevCPM.ObjW(SHORT(CHR(mSysPtr)))
					ELSIF f = Char16 THEN DevCPM.ObjW(SHORT(CHR(mChar16)))
					ELSIF f = Int64 THEN DevCPM.ObjW(SHORT(CHR(mInt64)))
					ELSIF obj.typ = DevCPT.guidtyp THEN DevCPM.ObjW(SHORT(CHR(mGuid)))
					ELSIF obj.typ = DevCPT.restyp THEN DevCPM.ObjW(SHORT(CHR(mResult)))
					ELSIF f = Pointer THEN
						IF obj.typ.sysflag = interface THEN DevCPM.ObjW(SHORT(CHR(mInterface)))
						ELSIF obj.typ.untagged THEN DevCPM.ObjW(SHORT(CHR(mSysPtr)))
						ELSE DevCPM.ObjW(10X); OutReference(TypeObj(obj.typ), 0, absolute)
						END
					ELSIF (f = Comp) & outAllRef & (~obj.typ.untagged OR outURef & (obj.typ.comp # DynArr)) THEN
						DevCPM.ObjW(10X); OutReference(TypeObj(obj.typ), 0, absolute)
					ELSIF f < Int8 THEN DevCPM.ObjW(SHORT(CHR(f - 1)))
					ELSE DevCPM.ObjW(SHORT(CHR(f)))
					END;
					IF obj.mnolev = 0 THEN DevCPM.ObjWNum(obj.adr + dsize) ELSE DevCPM.ObjWNum(obj.adr) END;
					OutName(obj.name)
				END
			END ;
			OutRefs(obj.right)
		END
	END OutRefs;
	
	PROCEDURE EOutVarRef (obj: T.Var);
		CONST mmComp = 15; (* it seems Comp=15 was outed for ARRAY [N] OF SHORTCHAR *)
		VAR t: T.Type; x, w: INTEGER;
	BEGIN
		t := obj.type; w := t.what;
		IF (w IN T.basics + T.pointers) 
				OR outURef & ~((w = T.fArray) & (T.open IN t.flags))
				OR outAllRef & ~(T.untagged IN t.flags)
				OR (w = T.fArray) & (t(T.Array).elTyp.what = T.fChar8) THEN	(* special *)
			IF obj IS T.Par THEN IF T.kindMask * obj.flags = T.valK THEN x := 0FDH ELSE x := 0FFH END
			ELSE x := 0FDH 
			END;
			M.ObjW(SHORT(CHR(x))); 
			x := -2;
			CASE w OF
			| T.fInt8: x := mInt8 | T.fInt16: x := mInt16 | T.fInt32: x := mInt32 | T.fInt64: x := mInt64
			| T.fChar8: x := mChar8 | T.fChar16: x := mChar16 | T.fBool: x := mBool | T.fSet: x := mSet
			| T.fReal32: x := mReal32 | T.fReal64: x := mReal64 | T.fSysPtr: x := mSysPtr
			| T.fRecPtr, T.fArrayPtr:
				IF t = T.anyptrtp THEN x := mAnyPtr
				ELSIF T.interfaceF = T.strucMask * t.flags THEN x := mInterface
				ELSIF T.untagged IN t.flags THEN x := mSysPtr
				ELSE x := -1; M.ObjW(10X); EOutDescBookmark(t)
				END
			| T.fArray, T.fRecord:
				IF t = T.anyrectp THEN x := mAnyRec
				ELSE
					IF outAllRef & 
					(~(T.untagged IN t.flags) OR outURef & ~((w = T.fArray) & (T.open IN t.flags))) THEN
						x := -1; M.ObjW(10X); EOutDescBookmark(t)
					ELSIF (w = T.fArray) & (t(T.Array).elTyp.what = T.fChar8) THEN x := mmComp 
					END
				END
			(*ELSIF t = T.guidtyp THEN x := mGuid
			ELSIF t = T.restyp THEN x := mResult*)
			| T.fProcedure: x := mProcedure	(*not sure...*)
			ELSE HALT(126)
			END;
			ASSERT(x # -2, 100);
			IF x # -1 THEN M.ObjW(SHORT(CHR(x))) END;
			IF obj.owner.nesting = 0 THEN M.ObjWNum(obj.adr + dsize) ELSE M.ObjWNum(obj.adr) END;
			EOutName(obj.name)
		ELSE HALT(126)
		END
	END EOutVarRef;

	(** Out tree of objects starting with obj? Used to out local var refs of a procedure *)

	PROCEDURE OutGuid* (VAR str: ARRAY OF SHORTCHAR);
		
		PROCEDURE Copy (n: INTEGER);
			VAR x, y: INTEGER;
		BEGIN
			x := ORD(str[n]); y := ORD(str[n + 1]);
			IF x >= ORD("a") THEN DEC(x, ORD("a") - 10)
			ELSIF x >= ORD("A") THEN DEC(x, ORD("A") - 10)
			ELSE DEC(x, ORD("0"))
			END;
			IF y >= ORD("a") THEN DEC(y, ORD("a") - 10)
			ELSIF y >= ORD("A") THEN DEC(y, ORD("A") - 10)
			ELSE DEC(y, ORD("0"))
			END;
			DevCPM.ObjW(SHORT(CHR(x * 16 + y)))
		END Copy;
		
	BEGIN
		IF bigEndian THEN
			Copy(1); Copy(3); Copy(5); Copy(7); Copy(10); Copy(12); Copy(15); Copy(17)
		ELSE
			Copy(7); Copy(5); Copy(3); Copy(1); Copy(12); Copy(10); Copy(17); Copy(15)
		END;
		Copy(20); Copy(22); Copy(25); Copy(27); Copy(29); Copy(31); Copy(33); Copy(35)
	END OutGuid;

	PROCEDURE OutConst (obj: DevCPT.Object);
		TYPE A4 = ARRAY 4 OF SHORTCHAR; A8 = ARRAY 8 OF SHORTCHAR;
		VAR a, b, c: DevCPT.Const; r: SHORTREAL; lr: REAL; a4: A4; a8: A8; ch: SHORTCHAR; i, x, hi, low: INTEGER;
	BEGIN
		a := obj.conval; b := NIL;
		WHILE a # NIL DO c := a; a := c.link; c.link := b; b := c END;
		WHILE b # NIL DO
			IF String8 IN b.setval THEN
				DevCPM.ObjWBytes(b.ext^, b.intval2);
				Align(4)
			ELSIF String16 IN b.setval THEN
				i := 0; REPEAT DevCPM.GetUtf8(b.ext^, x, i); Out2(x) UNTIL x = 0;
				Align(4)
			ELSIF Real32 IN b.setval THEN
				r := SHORT(b.realval); a4 := SYSTEM.VAL(A4, r);
				IF DevCPM.LEHost = bigEndian THEN
					ch := a4[0]; a4[0] := a4[3]; a4[3] := ch;
					ch := a4[1]; a4[1] := a4[2]; a4[2] := ch
				END;
				DevCPM.ObjWBytes(a4, 4)
			ELSIF Real64 IN b.setval THEN
				a8 := SYSTEM.VAL(A8, b.realval);
				IF DevCPM.LEHost = bigEndian THEN
					ch := a8[0]; a8[0] := a8[7]; a8[7] := ch;
					ch := a8[1]; a8[1] := a8[6]; a8[6] := ch;
					ch := a8[2]; a8[2] := a8[5]; a8[5] := ch;
					ch := a8[3]; a8[3] := a8[4]; a8[4] := ch
				END;
				DevCPM.ObjWBytes(a8, 8)
			ELSIF Int64 IN b.setval THEN
				(* intval moved to intval2 by AllocConst *)
				x := b.intval; b.intval := b.intval2; GetLongWords(b, hi, low); b.intval := x;
				IF bigEndian THEN Out4(hi); Out4(low) ELSE Out4(low); Out4(hi) END
			ELSIF Guid IN b.setval THEN
				OutGuid(b.ext)
			END;
			b := b.link
		END
	END OutConst;
	
	PROCEDURE OutReal (short: BOOLEAN; real: REAL);
		TYPE A4 = ARRAY 4 OF SHORTCHAR; A8 = ARRAY 8 OF SHORTCHAR;
		VAR r: SHORTREAL; a4: A4; a8: A8; ch: SHORTCHAR; 
	BEGIN
		IF short THEN
			r := SHORT(real); a4 := SYSTEM.VAL(A4, r);
			IF M.LEHost = bigEndian THEN
				ch := a4[0]; a4[0] := a4[3]; a4[3] := ch;
				ch := a4[1]; a4[1] := a4[2]; a4[2] := ch
			END;
			M.ObjWBytes(a4, 4)
		ELSE
			a8 := SYSTEM.VAL(A8, real);
			IF M.LEHost = bigEndian THEN
				ch := a8[0]; a8[0] := a8[7]; a8[7] := ch;
				ch := a8[1]; a8[1] := a8[6]; a8[6] := ch;
				ch := a8[2]; a8[2] := a8[5]; a8[5] := ch;
				ch := a8[3]; a8[3] := a8[4]; a8[4] := ch
			END;
			M.ObjWBytes(a8, 8)
		END
	END OutReal;
	
	PROCEDURE EOutLiterals;	(* merge doubles and quads output *)
		VAR gap, i, len: INTEGER; s: POINTER TO T.String; next, lit: T.Literal;
	BEGIN
		gap := (blockAlign - (-(last.str + last.quad + last.dbl)) MOD blockAlign) MOD blockAlign;
		IF gap > 0 THEN M.ObjWBytes(zero, gap) END;
		lit := strings; WHILE lit # NIL DO
			WITH lit: Y.StringLit DO i := 0; len := lit.len; s := lit.str; next := lit.link;
				CASE lit.type.what OF
				| T.fString8: WHILE i <= len DO M.ObjW(SHORT(s[i])); INC(i) END
				| T.fString16: WHILE i <= len DO Out2(ORD(s[i])); INC(i) END
				ELSE HALT(20)
				END
			ELSE HALT(60)
			END;
			(*LAlign(8);*)
			lit := next
		END;
		lit := doubles; WHILE lit # NIL DO
			WITH lit: Y.Real DO ASSERT(lit.type.what = T.fReal32); OutReal(TRUE, lit.real); next := lit.link 
			| lit: Y.Int DO Out4(SHORT(lit.long)); next := lit.link 
			ELSE HALT(20) 
			END;
			lit := next
		END;
		lit := quads; WHILE lit # NIL DO
			WITH lit: Y.Real DO ASSERT(lit.type.what = T.fReal64); OutReal(FALSE, lit.real); next := lit.link 
			| lit: Y.Int DO Out8(lit.long); next := lit.link 
			ELSE HALT(20) 
			END;
			lit := next
		END;
		ASSERT(M.ObjLen() MOD blockAlign = 0)
	END EOutLiterals;
	
	PROCEDURE OutStruct (typ: DevCPT.Struct; unt: BOOLEAN);
	BEGIN
		IF typ = NIL THEN Out4(0)
		ELSIF typ = DevCPT.sysptrtyp THEN Out4(mSysPtr)
		ELSIF typ = DevCPT.anytyp THEN Out4(mAnyRec)
		ELSIF typ = DevCPT.anyptrtyp THEN Out4(mAnyPtr)
		ELSIF typ = DevCPT.guidtyp THEN Out4(mGuid)
		ELSIF typ = DevCPT.restyp THEN Out4(mResult)
		ELSE
			CASE typ.form OF
			| Undef, Byte, String8, NilTyp, NoTyp, String16: Out4(0)
			| Bool, Char8: Out4(typ.form - 1)
			| Int8..Set: Out4(typ.form)
			| Char16: Out4(mChar16)
			| Int64: Out4(mInt64)
			| ProcTyp: OutReference(TypeObj(typ), 0, absolute)
			| Pointer:
				IF typ.sysflag = interface THEN Out4(mInterface)
				ELSIF typ.untagged THEN Out4(mSysPtr)
				ELSE OutReference(TypeObj(typ), 0, absolute)
				END
			| Comp:
				IF ~typ.untagged OR (outURef & unt) THEN OutReference(TypeObj(typ), 0, absolute)
				ELSE Out4(0)
				END
			END
		END
	END OutStruct;
	
	PROCEDURE EOutTypeRef (typ: T.Type; unt: BOOLEAN);
	BEGIN
		IF typ = NIL THEN Out8(0)
		ELSIF typ = T.sysptrtp THEN Out8(mSysPtr)
		ELSIF typ = T.anyrectp THEN Out8(mAnyRec)
		ELSIF typ = T.anyptrtp THEN Out8(mAnyPtr)
		ELSIF typ = T.guidtyp THEN Out8(mGuid)
		ELSIF typ = T.restyp THEN Out8(mResult)
		ELSE
			CASE typ.what OF
			| (*Undef, *) T.fByte, T.fString8, T.fNil, (*NoTyp, *) T.fString16: Out8(0)
			| T.fBool: Out8(mBool)
			| T.fChar8: Out8(mChar8)
			| T.fInt8: Out8(mInt8)
			| T.fInt16: Out8(mInt16)
			| T.fInt32: Out8(mInt32)
			| T.fReal32: Out8(mReal32)
			| T.fReal64: Out8(mReal64)
			| T.fSet: Out8(mSet)
			| T.fChar16: Out8(mChar16)
			| T.fInt64: Out8(mInt64)
			(*| T.fProcTyp: OutReference(TypeObj(typ), 0, absolute)*)
			(*| T.fPointer:
				IF typ.sysflag = interface THEN Out4(mInterface)
				ELSIF typ.untagged THEN Out4(mSysPtr)
				ELSE OutReference(TypeObj(typ), 0, absolute)
				END*)
			| T.fSysPtr: Out8(mSysPtr)
			| T.fArrayPtr, T.fRecPtr: IF T.untagged IN typ.flags THEN Out8(mSysPtr) ELSE EOutDescBookmark(typ) END
			| T.fArray, T.fRecord:
				IF ~(T.untagged IN typ.flags) OR (outURef & unt) THEN EOutDescBookmark(typ) ELSE Out8(0) END
			ELSE HALT(126)
			END
		END
	END EOutTypeRef;
	
	PROCEDURE NameIdx (obj: DevCPT.Object): INTEGER;
		VAR n: INTEGER;
	BEGIN

				WHILE obj.name[n] # 0X DO INC(n) END;
				INC(namex, n + 1);
				obj.nlink := nameList; nameList := obj
			END;
			n := obj.num;
		END;
		RETURN n
	END NameIdx;

	PROCEDURE OutSignature (par: DevCPT.Object; retTyp: DevCPT.Struct; OUT pos: INTEGER);
		VAR p: DevCPT.Object; n, m: INTEGER;
	BEGIN
		pos := DevCPM.ObjLen() - headSize;
		OutStruct(retTyp, TRUE);
		p := par; n := 0;
		WHILE p # NIL DO INC(n); p := p.link END;
		Out4(n); p := par;
		WHILE p # NIL DO
			IF p.mode # VarPar THEN m := mValue
			ELSIF p.vis = inPar THEN m := mInPar
			ELSIF p.vis = outPar THEN m := mOutPar
			ELSE m := mVarPar
			END;
			Out4(NameIdx(p) * 256 + m);
			OutStruct(p.typ, TRUE);
			p := p.link
		END
	END OutSignature;
	
	PROCEDURE EOutSignature (proc: T.Proc; OUT pos: INTEGER);
		VAR i, n, m: INTEGER; par: POINTER TO ARRAY OF T.Par; p: T.Par; kind: SET;
	BEGIN
		pos := M.ObjLen() - headSize;	(* this is some mess!! position in Meta segment is output to obj.textpos?! this pos is actually the value for the head of bookmark list *)
		EOutTypeRef(proc.type, TRUE);
		par := proc.par;
		IF par = NIL THEN n := 0 ELSE n := LEN(par) END;
		Out4(n);
		i := 0; WHILE i < n DO p := par[i]; kind := p.flags * T.kindMask;
			IF kind = T.valK THEN m := mValue
			ELSIF kind = T.inK THEN m := mInPar
			ELSIF kind = T.outK THEN m := mOutPar
			ELSE m := mVarPar
			END;
			Out4(ENameIdx(p) * 256 + m);
			EOutTypeRef(p.type, TRUE);
			INC(i)
		END
	END EOutSignature;

	PROCEDURE PrepDesc (desc: DevCPT.Struct);
		VAR fld: DevCPT.Object; n: INTEGER; l: DevCPT.LinkList; b: DevCPT.Struct;
	BEGIN
		IF desc.comp = Record THEN	(* write field list *)
			desc.strobj.adr := DevCPM.ObjLen() - headSize;
			n := 0; fld := desc.link;
			WHILE (fld # NIL) & (fld.mode = Fld) DO
				IF expAllFields OR (fld.vis # internal) THEN INC(n) END;
				fld := fld.link
			END;
			Out4(n); fld := desc.link;
			WHILE (fld # NIL) & (fld.mode = Fld) DO
				IF expAllFields OR (fld.vis # internal) THEN
					OutObject(mField, 0, fld.adr, fld.typ, fld)
				END;
				fld := fld.link
			END
		ELSIF (desc.form = ProcTyp) & outSignatures THEN	(* write param list *)
			OutSignature(desc.link, desc.BaseTyp, desc.n)
		END;
		(* assert name and base type are included *)
		IF desc.untagged THEN n := NameIdx(untgd)
		ELSE n := NameIdx(desc.strobj)
		END;
		IF desc.form # ProcTyp THEN b := desc.BaseTyp;
			IF (b # NIL) & (b # DevCPT.anytyp) & (b # DevCPT.anyptrtyp) & (b.form IN {Pointer, Comp, ProcTyp})
					& (b.sysflag # interface) & (b # DevCPT.guidtyp)
					& (~b.untagged OR outURef & (b.form = Comp)) THEN
				l := OffsetLink(TypeObj(b), 0)
			END
		END
	END PrepDesc;

	PROCEDURE IncludeType (t: T.Type);
	(** Include t in descriptored/imported type lists in case it satisfies inclusion criteria *)
	BEGIN	(* criteria adapted from end of OutDirSign *)
		IF (t # T.anyrectp) & (t # T.anyptrtp) & (t.what IN descriptored)
			& ~(T.interfaceF = t.flags * T.strucMask) & (t # T.guidtp)
			& (~(T.untagged IN t.flags) OR outURef & (t.what IN T.structures)) 
			& T.IsProper(t)
		THEN
			WITH t: Y.Record DO IncThisType(t, t.typ) 
			| t: Y.Array DO IncThisType(t, t.typ)
			| t: Y.RecPtr DO IncThisType(t, t.typ)
			ELSE HALT(20) 
			END
		END
	END IncludeType; 
	
	PROCEDURE EOutFields (t: Y.Record);
		VAR len, n, i: INTEGER; fld: T.Field;
	BEGIN
		Out4(14022001H);
		t.fieldsOffs := M.ObjLen() - pos.meta;	(* field directory position, used in EOutRecDesc *)
		IF t.fields = NIL THEN len := 0 ELSE len := LEN(t.fields) END;
		IF expAllFields THEN n := len 
		ELSE
			n := 0; 
			i := 0; WHILE i < len DO IF t.fields[i].flags * T.visMask # T.domestic THEN INC(n) END; INC(i) END
		END;
		Out4(n);
		i := 0; WHILE i < len DO fld := t.fields[i];
			IF expAllFields OR (fld.flags * T.visMask # T.domestic) THEN 
				EOutObject(mField, 0, fld.offs, fld.type, fld)
			END;
			INC(i)
		END
	END EOutFields;
	
	PROCEDURE NumMeth (root: DevCPT.Object; num: INTEGER): DevCPT.Object;
		VAR obj: DevCPT.Object;
	BEGIN
		IF (root = NIL) OR (root.mode = TProc) & (root.num = num) THEN RETURN root END;
		obj := NumMeth(root.left, num);
		IF obj = NIL THEN obj := NumMeth(root.right, num) END;
		RETURN obj
	END NumMeth;
	
	PROCEDURE OutDesc (desc: DevCPT.Struct);
		VAR m: DevCPT.Object; i, nofptr, flddir, size: INTEGER; t, xb: DevCPT.Struct; form, lev, attr: BYTE;
	BEGIN
		ASSERT(~desc.untagged);
		IF desc.comp = Record THEN
			xb := desc; flddir := desc.strobj.adr;
			REPEAT xb := xb.BaseTyp UNTIL (xb = NIL) OR (xb.mno # 0) OR xb.untagged;
			Out4(-1); i := desc.n;
			WHILE i > 0 DO DEC(i); t := desc;
				REPEAT
					m := NumMeth(t.link, i); t := t.BaseTyp
				UNTIL (m # NIL) OR (t = xb);
				IF m # NIL THEN 
					IF absAttr IN m.conval.setval THEN Out4(0)
					ELSE OutReference(m, 0, absolute)
					END
				ELSIF (xb = NIL) OR xb.untagged THEN Out4(0)	(* unimplemented ANYREC method *)
				ELSE OutReference(xb.strobj, -4 - 4 * i, copy)
				END
			END;
			desc.strobj.adr := DevCPM.ObjLen() - headSize;	(* desc adr *)
			Out4(desc.size);
			OutReference(Mod, 0, absolute);
			IF desc.untagged THEN m := untgd ELSE m := desc.strobj END;
			IF desc.attribute = extAttr THEN attr := 1
			ELSIF desc.attribute = limAttr THEN attr := 2
			ELSIF desc.attribute = absAttr THEN attr := 3
			ELSE attr := 0
			END;
			Out4(mRecord + attr * 4 + desc.extlev * 16 + NameIdx(m) * 256); i := 0;
			WHILE i <= desc.extlev DO
				t := desc;
				WHILE t.extlev > i DO t := t.BaseTyp END;
				IF t.sysflag = interface THEN Out4(0)
				ELSIF t.untagged THEN OutReference(TypeObj(t), 0, absolute)
				ELSIF (t.mno = 0) THEN OutReference(t.strobj, 0, absolute)
				ELSIF t = xb THEN OutReference(xb.strobj, 0, absolute)
				ELSE OutReference(xb.strobj, 12 + 4 * i, copy)
				END;
				INC(i)
			END;
			WHILE i <= DevCPM.MaxExts DO Out4(0); INC(i) END;
			OutReference(Meta, flddir, absolute);	(* ref to field list *)
			nofptr := 0; FindPtrs(desc, 0, FALSE, nofptr);
			Out4(-(4 * nofptr + 4));
			nofptr := 0; FindPtrs(desc, 0, TRUE, nofptr);
			Out4(-1)
		ELSE
			desc.strobj.adr := DevCPM.ObjLen() - headSize;
			lev := 0; size := 0;
			IF desc.comp = Array THEN
				size := desc.n; form := mArray
			ELSIF desc.comp = DynArr THEN
				form := mArray; lev := SHORT(SHORT(desc.n + 1))
			ELSIF desc.form = Pointer THEN
				form := mPointer
			ELSE ASSERT(desc.form = ProcTyp);
				DevCPM.FPrint(size, XProc); DevCPT.FPrintSign(size, desc.BaseTyp, desc.link); form := mProctyp;
			END;
			Out4(size);
			OutReference(Mod, 0, absolute);
			IF desc.untagged THEN m := untgd ELSE m := desc.strobj END;
			Out4(form + lev * 16 + NameIdx(m) * 256);
			IF desc.form # ProcTyp THEN OutStruct(desc.BaseTyp, TRUE)
			ELSIF outSignatures THEN OutReference(Meta, desc.n, absolute)	(* ref to par list *)
			END
		END
	END OutDesc;

	PROCEDURE EOutRecDesc (rec: Y.Record);
		VAR base: ARRAY M.MaxExts OF T.Record; t, bottom: T.Record; proc: T.TProc; 
			nofptr, attr, i, addr: INTEGER; nm: T.TypeName; flags: SET;
	BEGIN
		bottom := rec;
		REPEAT bottom := bottom.base 
		UNTIL (bottom = NIL) OR (bottom.typeName.imno # 0) OR (T.untagged IN bottom.flags);
		Out4(-1); 
		IF rec.proc # NIL THEN i := LEN(rec.proc) ELSE i := 0 END;
		WHILE i > 0 DO HALT(126); (* loop body not tested, because bound procedures not available yet *)
			DEC(i); 
			t := rec; REPEAT proc := t.proc[i]; t := t.base UNTIL (proc # NIL) OR (t = bottom);
			IF proc # NIL THEN 
				IF T.abstract IN proc.flags THEN addr := 0 ELSE addr := proc.adr END
			ELSIF (bottom = NIL) OR (T.untagged IN bottom.flags) THEN addr := 0
				(* unimplemented ANYREC procedure *)
			ELSE HALT(126) (* out a reference for a non-redefined bound procedure *)
			END;
			Out4(addr); Out4(28051978H(*magic*))
		END;
		rec.typ.desc := M.ObjLen() - pos.desc;	(* holds rec desc's address WRT DescBlk *)
		Out4(rec.size);	(* size4 *)
		DoOutBookmark(modDescBkmrk, absolute64);	(* bookmark for module descriptor *)
		IF T.extensible IN rec.flags THEN attr := 1
		ELSIF T.limited IN rec.flags THEN attr := 2
		ELSIF T.abstract IN rec.flags THEN attr := 3
		ELSE attr := 0
		END;
		IF T.untagged IN rec.flags THEN nm := Euntgd 
		ELSE nm := rec.typeName; IF nm = NIL THEN nm := Eanon END
		END;
		Out4(mRecord + attr * 4 + rec.level * 16 + ENameIdx(nm) * 256); 	(* id4 *)
		i := rec.level; t := rec; REPEAT base[i] := t; t := t.base; DEC(i) UNTIL i < 0;
		i := 0; WHILE i <= rec.level DO
			t := base[i]; flags := t.flags;
			IF T.interface IN flags THEN Out8(0)
			ELSIF T.untagged IN flags THEN HALT(126) 
			ELSIF T.IsProper(t) THEN EOutDescBookmark(t)
			(*ELSIF t = xb THEN OutReference(xb.strobj, 0, absolute)*)
			ELSE HALT(126)
			END;
			INC(i)
		END;
		WHILE i <= DevCPM.MaxExts DO Out8(0); INC(i) END;
		EOutBookmark(EMeta, rec.fieldsOffs, absolute64);		(* ref to field list *)
		EFindPtrs(rec, FALSE, nofptr);	(* it may be more efficient to leave a placeholder then reposition and write out then reposition again *)
		Out4(-(4 * nofptr + 4));	(* what is this? *)
		EFindPtrs(rec, TRUE, nofptr);
		Out4(-1)
		;Log.String("EOutRecDesc: "); 
		IF rec.typeName # NIL THEN Log.String(rec.typeName.name) ELSE Log.String("anonymous") END; 
		Log.Int(rec.level); Log.Ln
	END EOutRecDesc;
	
	PROCEDURE EOutDesc (type: T.Type);
	(** Out type descriptor. descriptored types are granted a descriptor *)
		VAR m: T.Object; i, nofptr, flddir, size: INTEGER; offs, form, lev, attr: INTEGER; base: T.Type;
	BEGIN
		ASSERT(~(T.untagged IN type.flags));	(*? maybe i do want to include descriptors for untagged types... cf. CheckInType *)
		WITH type: Y.Record DO EOutRecDesc(type)
			;ASSERT(type.typ.desc # Y.none, 112)
		| type: T.Record DO HALT(20)
		ELSE
			offs := M.ObjLen() - pos.desc;
			lev := 0; size := 0;
			WITH type: Y.Array DO form := mArray; base := type.elTyp; type.typ.desc := offs;
				IF T.open IN type.flags THEN lev := type.len + 1 ELSE size := type.len END
			| type: Y.RecPtr DO type.typ.desc := offs; form := mPointer; base := type.base
			(*| type: Y.ArrayPtr DO type.type := offs; form := mPointer; base := type.base*)
			ELSE HALT(126)
			END;
			Out4(size);
			DoOutBookmark(modDescBkmrk, absolute64);		(* bookmark for module descriptor *)
			IF T.untagged IN type.flags THEN m := Euntgd 
			ELSE m := type.typeName; IF m = NIL THEN m := Eanon END
			END;
			Out4(form + lev * 10H + ENameIdx(m) * 100H);	(* id4 *)
			IF type.what # T.fProcedure THEN EOutTypeRef(base, TRUE)
			ELSIF outSignatures THEN HALT(126)
			END
		END
	END EOutDesc;

	PROCEDURE OutModDesc (nofptr, refSize, namePos, ptrPos, expPos, impPos: INTEGER; IN name: T.Name);
		VAR i: INTEGER; t: Dates.Time; d: Dates.Date;
	BEGIN

	PROCEDURE OutExport (obj: DevCPT.Object);
		VAR num: INTEGER;
	BEGIN

			OutExport(obj.left);
			IF (obj.history # removed) & ((obj.vis # internal) OR
						(obj.mode = Typ) & (obj.typ.strobj = obj) & (obj.typ.form = Comp)) THEN
				DevCPT.FPrintObj(obj);
				IF obj.mode IN {LProc, XProc, IProc} THEN
					IF procVarIndirect THEN
						ASSERT(obj.nlink = NIL);
						num := obj.num; obj.num := 0;
						OutObject(mProc, obj.fprint, num, NIL, obj);
						obj.num := num
					ELSE
						OutObject(mProc, obj.fprint, obj.adr, NIL, obj)

					OutObject(mTyp, obj.typ.pbfp, obj.typ.pvfp, obj.typ, obj)
				ELSE ASSERT(obj.mode IN {Con, CProc});
					OutObject(mConst, obj.fprint, 0, NIL, obj)

			OutExport(obj.right)
		END

		VAR num: INTEGER; outType: BOOLEAN; type: T.Type;

			EOutExport(obj.left, cnt);
			WITH obj: T.TypeName DO outType := (obj.type.typeName = obj) & (obj.type.what IN T.structures)
				(* this makes sure ALL structured types are listed in ExpBlk... why exactly? *)
			ELSE outType := FALSE
			END;
			IF (obj.history # removed) & ((obj.flags * T.visMask # T.domestic) OR outType) THEN
				T.FPrintObj(obj);
				WITH
				| obj: T.XProc DO
					IF procVarIndirect THEN HALT(126)
					ELSE EOutObject(mProc, obj.fprint, obj.adr, NIL, obj)
					END
				| obj: T.Var DO EOutObject(mVar, obj.fprint, dsize + obj.adr, obj.type, obj)
				| obj: T.TypeName DO type := obj.type; EOutObject(mTyp, type.pbfp, type.pvfp, type, obj)
				ELSE HALT(126)
				END;
				INC(cnt);
				;Log.String("EOutExport "); Log.String(obj.name); Log.Int(obj.fprint); Log.Ln;
			END;
			EOutExport(obj.right, cnt)
		END
	END EOutExport;

	PROCEDURE OutImport (obj: DevCPT.Object);
		VAR typ: DevCPT.Struct; strobj: DevCPT.Object; opt: INTEGER;
	BEGIN
		IF obj # NIL THEN
			OutImport(obj.left);
			IF obj.mode = Typ THEN typ := obj.typ;
				IF obj.used OR
					(typ.form IN {Pointer, Comp}) & (typ.strobj = obj) &
						((obj.links # NIL) OR (obj.name # DevCPT.null) & (typ.pvused OR typ.pbused)) THEN
					DevCPT.FPrintStr(typ);
					DevCPM.ObjW(SHORT(CHR(mTyp))); OutName(obj.name^);
					IF obj.used THEN opt := 2 ELSE opt := 0 END;
					IF (typ.form = Comp) & ((typ.pvused) OR (obj.name = DevCPT.null)) THEN
						DevCPM.ObjWNum(typ.pvfp); DevCPM.ObjW(SHORT(CHR(opt + 1)));
						IF obj.history = inconsistent THEN DevCPT.FPrintErr(obj, 249) END
					ELSE DevCPM.ObjWNum(typ.pbfp); DevCPM.ObjW(SHORT(CHR(opt)))
					END;
					OutLink(obj.links)
				END
			ELSIF obj.used THEN
				DevCPT.FPrintObj(obj);
				IF obj.mode = Var THEN
					DevCPM.ObjW(SHORT(CHR(mVar))); OutName(obj.name^);
					DevCPM.ObjWNum(obj.fprint); OutLink(obj.links)
				ELSIF obj.mode IN {XProc, IProc} THEN
					DevCPM.ObjW(SHORT(CHR(mProc))); OutName(obj.name^);
					DevCPM.ObjWNum(obj.fprint); OutLink(obj.links)
				ELSE ASSERT(obj.mode IN {Con, CProc});
					DevCPM.ObjW(SHORT(CHR(mConst))); OutName(obj.name^); DevCPM.ObjWNum(obj.fprint)
				END
			END;
			OutImport(obj.right)
		END
	END OutImport;

		m := dllList;
		WHILE m # NIL DO

				IF obj.mode = Var THEN DevCPM.ObjW(SHORT(CHR(mVar)))
				ELSE DevCPM.ObjW(SHORT(CHR(mProc)))
				END;
				IF obj.entry # NIL THEN OutName(obj.entry)
				ELSE OutName(obj.name);
				END;
				DevCPT.FPrintObj(obj); DevCPM.ObjWNum(obj.fprint); OutLink(obj.links);
				obj := obj.nlink
			END;
			DevCPM.ObjW(0X); m := m.link
		END;
		i := 1;
		WHILE i < DevCPT.nofGmod DO
			obj := DevCPT.GlbMod[i];
			IF obj.library = NIL THEN OutImport(obj.right); DevCPM.ObjW(0X) END;
			INC(i)
		END;
	END OutUseBlock;

	PROCEDURE EOutUseBlock (IN mod: ARRAY OF T.Mod);
		VAR addr, i: INTEGER;
			o: T.Object; e: ExtLib; entry: POINTER TO T.Name; cod: INTEGER; m: T.Mod;
			check: RECORD imps, exts: INTEGER END; next, type: T.Type;
	BEGIN
		check.imps := 0; check.exts := 0;
		e := extLibs;
		WHILE e # NIL DO
			o := e.head;
			WHILE o # NIL DO
				WITH 
				| o: T.GlobVar DO cod := mVar; entry := o.ent; addr := o.adr; HALT(126)
				| o: Y.XProc DO cod := mProc; entry := o.ent; addr := o.adr
				ELSE HALT(20) 
				END;
				INC(check.exts);
				M.ObjW(SHORT(CHR(cod)));
				IF (entry # NIL) & (entry^ # '') THEN EOutName(entry^) ELSE EOutName(o.name) END;
				entry := NIL;
				(*126 T.Fingerprint(o);*) M.ObjWNum(o.fprint); 
				(* instead, the new UseBlk has this in place of a bookmark chain head: addr 0 0X, where addr is the proxy address in the ProxyBlk (byte offset into PROXY table) *)
				M.ObjWNum(addr); M.ObjWNum(0); M.ObjW(0X);
				
				o := o.next
			END;
			M.ObjW(0X); e := e.link
		END;
		ASSERT(check.exts = extsCount, 100);
		i := 2; WHILE (i < LEN(mod)) & (mod[i] # NIL) DO
			m := mod[i]; 
			IF (m.lib = NIL) & ~(T.pseudo IN m.flags) THEN 
				o := m.head; 
				WHILE o # NIL DO
					IF T.used IN o.flags THEN
						WITH o: Y.XProc DO EOutImport(o, mProc, o.adr, 0, check.imps)
						| o: Y.GlobVar DO HALT(126)
						| o: T.Const DO HALT(126)
						| o: T.TypeName DO	(* imported type references are outed later *)
						ELSE HALT(20)
						END
					END;
					o := o.next
				END;
				type := types.imp; 
				WHILE type # terminator DO
					ASSERT(type.mod # NIL);
					WITH type: Y.Record DO 
						IF type.mod = m THEN
							EOutImport(type.typeName, mTyp, type.typ.desc, type.typ.bkmrks, check.imps) 
						END;
						next := type.typ.link
					| type: Y.RecPtr DO 
						IF type.mod = m THEN
							EOutImport(type.typeName, mTyp, type.typ.desc, type.typ.bkmrks, check.imps) 
						END;
						next := type.typ.link
					ELSE HALT(126)
					END;
					type := next
				END;
				M.ObjW(0X) 
			END;
			INC(i)
		END;
		IF newRec # NIL THEN INC(check.imps) END; IF newArr # NIL THEN INC(check.imps) END;
		ASSERT(check.imps = impsCount, 101);
		(*IF kernel # NIL THEN EOutImport(kernel.head); M.ObjW(0X) END*)
	END EOutUseBlock;
	
	PROCEDURE CollectDll (obj: DevCPT.Object; mod: DevCPT.String);
		VAR name: DevCPT.String; dll: DevCPT.Object;
	BEGIN
		IF obj # NIL THEN
			CollectDll(obj.left, mod);
			IF obj.used & (obj.mode IN {Var, XProc, IProc}) THEN
				IF obj.library # NIL THEN name := obj.library
				ELSE name := mod
				END;
				dll := dllList;
				WHILE (dll # NIL) & (dll.library^ # name^) DO dll := dll.link END;
				IF dll = NIL THEN
					NEW(dll); dll.library := name; INC(imports);
					IF dllList = NIL THEN dllList := dll ELSE dllLast.link := dll END;
					dllLast := dll; dll.left := dll;
				END;
				dll.left.nlink := obj; dll.left := obj
			END;
			CollectDll(obj.right, mod)
		END
	END CollectDll;

	PROCEDURE ECollectExtLibs (o: T.Object; defName: POINTER TO T.Name);
	(** External library (DLL and .so) handling. 
	An interface module may be a collection of objects from different external libraries. When a module's header is written, the number of imports must be calculated; this requires special analysis for interface modules, since one interface module may in fact refer to multiple external libraries, thus increasing the number of imports by more than one. This analysis was done in CollectDlls.
	At the end of OCF text, in the UseBlk, all the imported names have to be also spelled out. CollectDll used to reorganize interface module's objects into a collection of lists of objects, one collection per every interfaces library, and each collection containing objects imported from the corresponding library. These collections are represented with type ExtLib. The restructuring and counting is done in ECollectExtLibs. *)
	(* Rebuilds .next list!!! *)
		VAR handle: BOOLEAN; libName: POINTER TO T.Name; e: ExtLib; next: T.Object;
	BEGIN
		WHILE o # NIL DO 
			next := o.next; handle := TRUE;
			IF T.used IN o.flags THEN
				WITH 
				| o: T.GlobVar DO libName := o.lib
				| o: T.XProc DO libName := o.lib
				ELSE handle := FALSE
				END;
				IF handle THEN
					IF libName = NIL THEN libName := defName END; 
					e := extLibs; WHILE (e # NIL) & (e.name^ # libName^) DO e := e.link END;
					IF e = NIL THEN 
						NEW(e); e.name := libName; INC(imports); 
						IF extLibs # NIL THEN e.link := extLibs END; extLibs := e
					END;
					ASSERT(e # NIL);
					o.next := e.head; e.head := o;
					libName := NIL
				END
			END;
			o := next
		END
	END ECollectExtLibs;

	PROCEDURE LOutHeader* (top: T.Segraph); 
		VAR i: INTEGER; m: T.Mod; e: ExtLib; 
	BEGIN
		M.ObjWLInt(processor);	(* processor type *)
		M.ObjWLInt(0); 
		M.ObjWLInt(0); 
		M.ObjWLInt(0);
		M.ObjWLInt(0); M.ObjWLInt(0);	(* sizes *)
		(*M.ObjW(0X);*)
		M.ObjWLInt(-1); M.ObjWLInt(-1);	(* placeholders for NewRec/NewArr proxy indices *)
		M.ObjWLInt(0); M.ObjWLInt(0);	(* placeholders for extsCount and impsCount *)
		imports := 0; i := 0;
		REPEAT INC(i); m := top.mod[i];
			IF m # NIL THEN
				IF m.lib # NIL THEN	(* dll import *)
					ECollectExtLibs(m.head, m.lib)
				ELSIF (m.name[0] # '@') & ~(T.pseudo IN m.flags) THEN INC(imports)
					(* module import, except @UNIVERSE and SYSTEM aren't counted *)
				END
			END
		UNTIL m = NIL;
		M.ObjWNum(imports);	(* num of import *)
		EOutName(top.mod[0].name); 
		e := extLibs; WHILE e # NIL DO M.ObjW("$"); EOutName(e.name^); e := e.link END;
		i := 1;
		REPEAT m := top.mod[i];
			IF (m # NIL) & (m.lib = NIL) & (m.name[0] # '@') & ~(T.pseudo IN m.flags) THEN EOutName(m.realName) END;	(* exclude  *)
			INC(i)
		UNTIL m = NIL;
		LAlign(16); headSize := M.ObjLen();
		IF procVarIndirect THEN
			HALT(126);
			i := 0; EnumXProc(DevCPT.topScope.right, i)
		END
	END LOutHeader;
	
	PROCEDURE Aligned (x, modulus: INTEGER): INTEGER;
	BEGIN RETURN (x + modulus - 1) DIV modulus * modulus
	END Aligned;
	
	PROCEDURE AdAligned (x: INTEGER): INTEGER;
	BEGIN RETURN (x + M.pointerSize - 1) DIV M.pointerSize * M.pointerSize
	END AdAligned;
	
	PROCEDURE PageAligned (x: INTEGER): INTEGER;
	BEGIN RETURN (x + M.pageSize - 1) DIV M.pageSize * M.pageSize
	END PageAligned;
	
	PROCEDURE AllocateBlocks (ms, ds, cs, vs: INTEGER);
		VAR ps: INTEGER;
	BEGIN 
		map.desc := PageAligned(cs);
		map.eVars := Aligned(map.desc + ds + vs, M.pointerSize);	(* End of Var block WRT code *)
		ASSERT(map.eVars MOD M.pointerSize = 0);
		ps := M.pointerSize * (impsCount + extsCount);
		map.meta := -PageAligned(AdAligned(ms) + ps);
		map.proxies := map.meta + AdAligned(ms);
		map.quads := map.proxies;
		map.doubles := map.quads + last.quad;
		map.strings := map.doubles + last.dbl
	END AllocateBlocks;
	
	PROCEDURE FixupChain (bkmrk, wrtCode: INTEGER);
	(** fixup chain starting at bkmrk using wrtCode as offset with respect to base of code block *)
		VAR next, type, pc: INTEGER;
	BEGIN
		WHILE bkmrk # 0 DO
			next := ThisWord(bkmrk);
			type := next DIV 1000000H; ASSERT(type >= ripBased);
			pc := bkmrk + (type - ripBased);
			PutWord(bkmrk, wrtCode - pc -4);
			bkmrk := next MOD 1000000H
		END
	END FixupChain;
	
	PROCEDURE ApplyToBkmrks (IN b: Y.Bkmrks; base: INTEGER);
		VAR i: INTEGER;
	BEGIN i := 0; WHILE b[i].offset # 0 DO FixupChain(b[i].last, base + b[i].offset); INC(i) END
	END ApplyToBkmrks;
	
	PROCEDURE ApplyGlobVarFixups (o: Y.GlobVar; base: INTEGER);
		VAR more: Y.BkmrksPtr;
	BEGIN
		FixupChain(o.last, o.adr + base);
		ApplyToBkmrks(o.bkmrks, base); more := o.moreBkmrks;
		WHILE more # NIL DO ApplyToBkmrks(more.bkmrks, base); more := more.more END
	END ApplyGlobVarFixups;
	
	PROCEDURE ApplyRIPFixupsToList (o: T.Symbol);
		VAR next: T.Symbol; base: INTEGER;
	BEGIN
		WHILE o # NIL DO
			WITH 
			| o: T.Mod DO 
				IF o.lib = NIL THEN ApplyRIPFixupsToList(o.head) 
				ELSE (* libs are fixuped directly from ApplyRIPFixups *) 
				END; 
				next := o.next 
			| o: T.Object DO
				IF T.used IN o.flags THEN 
					(* IF HrSettings.debug THEN Log.String("Used: "); Log.String(o.name); Log.Ln END; *)
					WITH 
					| o: Y.GlobVar DO 
						IF o.imno = 0 THEN base := map.eVars 	(* domestic var *)
						ELSE base := map.proxies; HALT(126)	(* imported var *)
						END;
						ApplyGlobVarFixups(o, base)
					| o: Y.XProc DO 
						IF o.imno # 0 THEN FixupChain(o.last, map.proxies + o.adr) 	(* imprtd proc *)
						ELSE FixupChain(o.last, o.adr)	(* domestic forward proc *)
						END
					| o: T.CProc DO
					| o: T.Const DO
					| o: T.TypeName DO
					ELSE HALT(126)
					END
				(* ELSIF HrSettings.debug THEN Log.String("Unused: "); Log.String(o.name); Log.Ln *)
				END;
				next := o.next
			| o: Y.StringLit DO FixupChain(o.bkmrk, map.proxies + o.adr); next := o.link
				;HALT(9)
			ELSE
			END;
			o := next
		END
	END ApplyRIPFixupsToList;
	
	PROCEDURE FixupReals (head: T.Literal; wrt: INTEGER);	(* Or FixupLiterals? *)
	(** Fixup the references to the literals in the list head assuming their offsets are relative to wrt  *)
		VAR next: T.Literal; bkmrk, addr: INTEGER;
	BEGIN
		WHILE head # NIL DO
			WITH head: Y.Real DO bkmrk := head.bkmrk; addr := head.adr; next := head.link 
			| head: Y.Int DO bkmrk := head.bkmrk; addr := head.adr; next := head.link 
			ELSE HALT(20) END;
			FixupChain(bkmrk, wrt + addr);
			head := next
		END
	END FixupReals;

	PROCEDURE FixupLiterals (head: T.Literal; wrt: INTEGER);	(* Or FixupLiterals? *)
	(** Fixup the references to the literals in the list head assuming their offsets are relative to wrt  *)
		VAR next: T.Literal; bkmrk, addr: INTEGER;
	BEGIN
		WHILE head # NIL DO
			WITH head: Y.Real DO bkmrk := head.bkmrk; addr := head.adr; next := head.link 
			| head: Y.Int DO bkmrk := head.bkmrk; addr := head.adr; next := head.link 
			| head: Y.StringLit DO bkmrk := head.bkmrk; addr := head.adr; next := head.link
			ELSE HALT(20) 
			END;
			FixupChain(bkmrk, wrt + addr);
			head := next
		END
	END FixupLiterals;

	PROCEDURE FixupTypes (types: T.Type);

		END;

	END ApplyRIPFixups;

Structures and fixups
Structured and structure pointer types are granted a descriptor (descriptored). IncludeType defines the criteria for whether a type should or should not have a descriptor (i.e. ANYREC doesn't get a descriptor, etc.)
Referencing a type T descriptor from an instruction (from code block):
	L.Item: .type := T; .mode := L.Abs; .what := T.fType;
	L.GenRef
		E.AppendTypeRIP
Referencing a type T descriptor from meta/desc block:
	EOutDescBookmark

Every type referenced (thru AppendTypeRIP or EOutDescBookmark) is added to a list: types.proper for proper types, types.imp for imported types. Both lists are terminated with terminator. Types are prepended to types.imp; and appended to types.proper; the tail of types.proper is in types.last.

All named exported types (basic and structured alike) are outed into the Exports block; the export entry for a type contains a bookmark for the descriptored type.

After synthesis is complete, all exported types are added to types.proper (some exported types may not be used in subject module and thus would not be referenced during synthesis).
After that, each type T in types.proper is traversed to make sure all other descriptored types referenced in T (field types, base types and elementary types) are also added to types.proper.
This guarantees completeness of types.proper and types.imp: they include all types necessary for subject module to be compiled, loaded, linked and executed.
Types may be anonymous. It is not clear yet how anonymous implicitly exported types can be referenced in importing modules (i.e. when VAR v*: POINTER TO RECORD END is imported and NEW(v) is called).

Descriptored types (Y.Record, Y.RecPtr, Y.Array, Y.ArrayPtr) hold a Y.Type record:
	Type = RECORD 
		link: T.Type;	links types in types.proper and types.imp
		rips, bkmrks,	RIP and bookmark chain heads
		desc: INTEGER	(* proxy offset for imported types; descriptor offset WRT DescBlk for proper types *)
	END

RIPs (RIP fixups) are necessary when type descriptors or proxy entries for imported type descriptors are referenced from the code block.
Bookmarks are necessary ONLY when type descriptors of both proper and imported types are referenced from Meta and Desc blocks.

Fixups

Machine code synthesized by the compiler cannot be executed as synthesized, but rather needs fixups.

In machine text, objects are represented in machine memory and referenced by their address. These cannot always be evaluated at code synthesis, and thus arises the need for fixups - corrections to addresses in machine text.

Three cases when an address cannot be evaluated at synthesis are:
* RIP addressing
* Base address changes
* Importing objects from other modules

1. RIP addressing
AMD64 instructions use RIP-relative addressing: the address of instruction operands is evaluated as the sum of instruction's address (stored in RIP register - instruction pointer; in fact, the address used is the address following instruction's last byte) and the operand's offset with respect to the instruction:
	EA = RIP + OFFSET
where EA is operand's effective address, RIP is the address of byte immediately following instruction's last byte, OFFSET is the operand offset encoded in the instruction.
	Consider the following:
	VAR x: INTEGER; 	(* global variable *)
	...
	x := 1978
	This text translates into
	MOV [x], 1978
	where 1978 is an immediate operand, and [x] is the offset of VAR x with respect to the current instruction's address. During synthesis, the compiler counts instruction bytes and thus knows the address of the MOV instruction in focus; also, at the time this MOV is synthesized, the address of x in the data block is also known. What is not known, and cannot be, is the size of the code block. The latter is only available when the whole module is compiled and the code block is complete; at this point, the compiler may properly allocate the code and data blocks (and other blocks holding metadata) and finally calculate the offset needed for [x] in the aforementioned MOV instruction. It will be:
	[x] = (size of code block) - (address of MOV) + (address of x in data block)
	in case the data block follows immediately after the code block
	Thus, the machine code for MOV has to be corrected - hence the first kind of fixups, the RIP fixups.
	These fixups can be performed by the compiler before the machine code and other module blocks are written to the executable format.

2. Base fixups
After a module has been loaded, it's metadata is a complex data structure containing pointers. Consider this:
	TYPE R = RECORD (Base) END;
	PROCEDURE P;
		VAR x: INTEGER; r: R
	The metadata for this example passage shall list, among other things, P's local variables x and r, with their respective types. r's description shall hold a pointer to the descriptor of R, which in turn shall point to the descriptor of Base. 
	The metadata is formed by the compiler, except for the pointers. For the latter to be formed, the base address of the meta block is needed; it is only determined at load time; thus it is the loader's job to perform base fixups.
	The compiler leaves data for the loader to perform base fixups. Let's say R's descriptor is placed by the compiler at offset [R] with respect to meta block base address; also, R is referenced twice in the meta block. (This means the metadata shall have two pointers to R). Let the first reference (future pointer to R) occur at offset [R1] with respect to meta block, and the second at offset [R2], R1 < R2. 
	In this case, the compiler shall output a zero address for [R1], and the value R2 for the second reference [R2]; we call these bookmarks. Shall there be another reference to R at offset [R3], the compiler shall output the value R2 for the reference at [R3] - make another bookmark. Thus, all references to R form a traversible chain of bookmarks. 
	Given the meta block base address (determined at load time), the offset [R] and the chain head [R3], the loader can traverse the cain of bookmarks (references to R) and replace them with actual pointers to R.
	Heads of chains of bookmarks are outed by the compiler in the special Fixup block, which is module metadata not represented at run-time.
	
3. Import fixups
Consider this:
	TYPE R = RECORD (Exporter.Base) END;
	The metadata shall contain a descriptor for R, which shall have a pointer to the descriptor for Exporter.Base. This type is defined in another module called Exporter. It is the linker's job to find module Exporter, look up it's metadata for the descriptor of Base, and form the proper pointer in R's descriptor.
	The compiler forms all references to imported objects in much the same manner as with base fixups: references to the same object are organized in one chain, and the head of this chain is outed with the name of the imported object. Given this information, the linker can traverse the chain of bookmarks and replace them with actual pointers to proper objects.
	Imported objects chain bookmarks are outed by the compiler in the special Uses block, which is module metadata not represented at run-time.
	The most often used linker is part of the linking loader of modules; another less often used is the binder, or static linker, that links several modules together to form an executable format of a host operating system.
