MODULE ObxProbe34;

	IMPORT SYSTEM, Console, Strings;

	VAR bad: INTEGER;

	PROCEDURE CHK (IN s: ARRAY OF CHAR; got, exp: LONGINT);
		VAR t: ARRAY 40 OF CHAR;
	BEGIN
		IF got # exp THEN
			INC(bad); Console.WriteStr("BAD "); Console.WriteStr(s);
			Strings.IntToString(got, t); Console.WriteStr(" got="); Console.WriteStr(t);
			Strings.IntToString(exp, t); Console.WriteStr(" exp="); Console.WriteStr(t);
			Console.WriteLn
		END
	END CHK;

	PROCEDURE Go*;
		VAR x: LONGINT; i, n: INTEGER;
	BEGIN
		bad := 0;
		(* константный сдвиг *)
		x := 1;
		CHK("lsh con +", SYSTEM.LSH(x, 40), 10000000000L);	(* 2^40 *)
		CHK("lsh con -", SYSTEM.LSH(x, -40), 0);
		x := 8000000000000000L;	(* MIN(LONGINT) *)
		CHK("lshr sign", SYSTEM.LSH(x, -63), 1);	(* логический: знак не тиражируется *)
		(* ROT константный *)
		x := 8000000000000001L;
		CHK("rot con +1", SYSTEM.ROT(x, 1), 3);
		CHK("rot con -1", SYSTEM.ROT(x, -1), 0C000000000000000L);
		(* переменный сдвиг *)
		n := 40; x := 1;
		CHK("lsh var +", SYSTEM.LSH(x, n), 10000000000L);
		n := -40;
		CHK("lsh var -", SYSTEM.LSH(x, n), 0);
		x := 8000000000000000L; n := 63;
		CHK("lshr var sign", SYSTEM.LSH(x, -n), 1);
		x := 8000000000000001L; n := 1;
		CHK("rot var +1", SYSTEM.ROT(x, n), 3);
		n := -1;
		CHK("rot var -1", SYSTEM.ROT(x, n), 0C000000000000000L);
		(* INTEGER не сломан *)
		i := 1; n := 20;
		IF SYSTEM.LSH(i, n) # 100000H THEN INC(bad); Console.WriteStr("BAD int lsh"); Console.WriteLn END;
		i := -2147483648;
		IF SYSTEM.LSH(i, -31) # 1 THEN INC(bad); Console.WriteStr("BAD int lshr"); Console.WriteLn END;
		IF bad = 0 THEN Console.WriteStr("Probe34 OK")
		ELSE Console.WriteStr("Probe34 FAILURES") END;
		Console.WriteLn
	END Go;

END ObxProbe34.
