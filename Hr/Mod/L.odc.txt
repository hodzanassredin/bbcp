	PROCEDURE AllocConst* (VAR x: Item; con: DevCPT.Const; form: BYTE);
		VAR r: REAL; short: SHORTREAL;
	BEGIN
		IF form IN {Real32, Real64} THEN
			r := con.realval;
			IF ABS(r) <= MAX(SHORTREAL) THEN
				short := SHORT(r);
				IF short = r THEN form := Real32	(* a shortreal can represent the exact value *)
				ELSE form := Real64	(* use a real *)
				END
			ELSE form := Real64	(* use a real *)
			END
		ELSIF form IN {String8, String16, Guid} THEN
			x.index := con.intval2	(* string length *)
		END;
		DevCPE.AllocConst(con, form, x.obj, x.offset);
		x.form := form; x.mode := Abs; x.scale := 0
	END AllocConst;

	PROCEDURE LAllocLiteral* (OUT x: Item; lit: T.Literal; what: INTEGER);
		VAR real: REAL; short: SHORTREAL;
	BEGIN
		IF what IN T.reals THEN ASSERT(what = lit.type.what)
		ELSIF what = Guid THEN HALT(126)
		ELSIF what IN T.strings + { Guid } THEN
			x.index := lit(T.StringLit).len + 1	(* string length *)
		END;
		E.EAllocLiteral(lit, what, (*x.o, *)x.offset);
		x.what := what; x.mode := Abs; x.scale := 0; x.lit := lit
	END LAllocLiteral;

	PROCEDURE SetLabel* (VAR L: Label);
		VAR link, typ, disp, x: INTEGER;
	BEGIN
		ASSERT(L <= 0); link := -L;
		WHILE link # 0 DO
			typ := link DIV 1000000H; link := link MOD 1000000H;
			IF typ = short THEN
				disp := DevCPE.pc - link - 1; ASSERT(disp < 128);
				DevCPE.PutByte(link, disp); link := 0
			ELSIF typ = relative THEN
				x := DevCPE.ThisWord(link); DevCPE.PutWord(link, DevCPE.pc - link - 4); link := x
			ELSE
				x := DevCPE.ThisWord(link); DevCPE.PutWord(link, DevCPE.pc + typ * 1000000H); link := x
			END
		END;		
		L := DevCPE.pc;
		a1.mode := 0; a2.mode := 0
	END SetLabel;
	
	PROCEDURE LSetLabel* (VAR L: Label);
		VAR link, typ, disp, x: INTEGER;
	BEGIN
		ASSERT(L <= 0); link := -L;
		WHILE link # 0 DO
			typ := link DIV 1000000H; link := link MOD 1000000H;
			IF typ = short THEN
				disp := E.pc - link - 1; ASSERT(disp < 128);
				E.PutByte(link, disp); link := 0
			ELSIF typ = relative THEN
				x := E.ThisWord(link); E.PutWord(link, E.pc - link - 4); link := x
			ELSE
				x := E.ThisWord(link); E.PutWord(link, E.pc + typ * 1000000H); link := x
			END
		END;		
		L := E.pc;
		a1.mode := 0; a2.mode := 0
	END LSetLabel;

	PROCEDURE GenWord (x: INTEGER);
	BEGIN
		E.GenByte(x); E.GenByte(x DIV 256)
	END GenWord;

	PROCEDURE GenDbl (x: INTEGER);
	BEGIN
		E.GenByte(x); E.GenByte(x DIV 256); E.GenByte(x DIV 10000H); E.GenByte(x DIV 1000000H)
	END GenDbl;
	
	PROCEDURE GenQuad (a, b: INTEGER);
	BEGIN GenDbl(a); GenDbl(b)
	END GenQuad;
	
	PROCEDURE CaseEntry* (tab, from, to: INTEGER);
		VAR a, e: INTEGER;
	BEGIN
		a := tab + 4 * from; e := tab + 4 * to;
		WHILE a <= e DO
			DevCPE.PutByte(a, DevCPE.pc);
			DevCPE.PutByte(a + 1, DevCPE.pc DIV 256);
			DevCPE.PutByte(a + 2, DevCPE.pc DIV 65536);
			INC(a, 4)
		END;
		a1.mode := 0; a2.mode := 0
	END CaseEntry;

	PROCEDURE GenLinked (VAR x: Item; type: BYTE);	(** emit reference for a global variable *)
		VAR link: DevCPT.LinkList;
	BEGIN
		IF x.obj = NIL THEN GenDbl(SHORT(x.offset))
		ELSE
			link := DevCPE.OffsetLink(x.obj, SHORT(x.offset));
			IF link # NIL THEN
				GenDbl(type * 1000000H + link.linkadr MOD 1000000H);
				link.linkadr := DevCPE.pc - 4
			ELSE GenDbl(0)
			END
		END
	END GenLinked;
	
	PROCEDURE LGenRef (VAR x: Item; pos, type: INTEGER; OUT val: INTEGER);
	(* type is a bad parameter name: the word usually refers to CP types, here it refers to the kind of bookmark *)
	(** generate (but don't emit) reference value: base or binding relocation bookmark for a global var/proc; typically used as the machine operation Displacement. pos is the position in the machine text where the reference will be emitted *)
		VAR link: DevCPT.LinkList; 
			lit: T.Literal; last: INTEGER; o: T.Object; t: T.Type;
	BEGIN val := -1;
		o := x.o;
		IF o =	(* without fixups *) NIL THEN	(* for literals *)
			ASSERT(type >= ripBased);
			IF x.what IN T.strings + T.reals + { T.fInt64, T.fInt32 } THEN 	(* merge braches *)
				E.BookmarkLiteral2(x.lit, pos, last); val := type * 1000000H + last MOD 1000000H
			(*ELSIF x.what IN T.reals + T.integers THEN
				E.BookmarkLiteral2(x.lit, pos, last); val := type * 1000000H + last MOD 1000000H*)
			ELSIF x.what = T.fType THEN ASSERT(x.type # NIL, 20); ASSERT(x.type IS T.Record, 126);
				(*IF x.type.typeName # NIL THEN ASSERT(x.type.typeName.imno = 0) END;*)
				t := x.type; 
				WITH t: T.Record DO 
					E.AppendTypeRIP(t, pos, last); val := type * 1000000H + last MOD 1000000H
				ELSE HALT(126)
				END
			ELSE HALT(126)
				(*(* without fixups: *)
				val := -(pos + 4 + (type - ripBased)) + x.offset*)
			END
		(*not needed since CTyp now requests type descriptors thru .what = T.fType  ELSIF o IS T.TypeName THEN 
			E.BookmarkType(o(Y.TypeName), pos, last); val := type * 1000000H + last MOD 1000000H*)
		ELSIF type >= ripBased THEN (* let's assume that the code segment immediately follows the data segment;  *)
			ASSERT(o.owner.nesting = 0);	(* only global objects *)
			IF (o IS T.XProc) & (o.imno = 0) & (o(T.XProc).adr > 0) THEN	(* domestic global procedure - out address immediately *)
				val := o(T.XProc).adr - (pos + 4 + (type - ripBased)) 	(* +4 is the size of Disp field being written *)
			ELSE 	(* imported global object or domestic global var *)
				ASSERT((o IS T.XProc) OR (o IS T.GlobVar), 126);
				E.BookmarkThis(o, x.offset, pos, last);
				val := type * 1000000H + last MOD 1000000H
			END
		ELSE HALT(126);	(* this branch is for non-RIP-relative fixup types *)
			E.EOutCodeBookmark(x.o, absolute)
		END
		;ASSERT(val # -1, 60)
	END LGenRef;
	
	PROCEDURE LGenLinked (VAR x: Item; type: INTEGER);
		VAR val: INTEGER;
	BEGIN LGenRef(x, E.pc, type, val); GenDbl(val)
	END LGenLinked;
	
	PROCEDURE CheckSize (form: BYTE; VAR w: INTEGER);
	BEGIN
		IF form IN {Int16, Char16} THEN DevCPE.GenByte(66H); w := 1
		ELSIF form >= Int32 THEN ASSERT(form IN {Int32, Set, NilTyp, Pointer, ProcTyp}); w := 1
		ELSE w := 0
		END
	END CheckSize;
	
	PROCEDURE LCheckSize (what: INTEGER; long: BOOLEAN; VAR w, byteLen: INTEGER);
		(*CONST REXW = 48H;	(* REX prefix with W bit set, to indicate 64-bit operand size *)*)
	BEGIN
		HALT(9);	(*deprecated*)
		IF what IN words THEN E.GenByte(66H); w := 1; byteLen := 2	(* operand size override for 16bit *)
		ELSIF what IN doubles THEN w := 1; byteLen := 4
		ELSIF what IN { T.fInt64 } + T.pointers THEN w := 1; byteLen := 4; IF long THEN (*E.GenByte(REXW) *)END
		ELSIF (*unreachable branch*) what IN quads THEN (*E.GenByte(REXW);*) w := 1; byteLen := 8
		ELSIF what IN bytes THEN w := 0; byteLen := 1	(* 8-bit operands *)
		ELSE HALT(20)
		END
		(* w = 0 results in 0B0H for MOV, w = 1 results in 0B8H for MOV *)
	END LCheckSize;
	(* CheckSize in CP2 generates the operand size override prefix. This is fine in amd64 as long as the extended features like the extra registers (GPR) are used. Because referencig R8..R16 is done thru a bit in the REX prefix; at this point, in CheckSize, this information is not available, thus the proper prefix cannot be written. Some other structure will be required. *)
	
	PROCEDURE LCheckPrefix (what: INTEGER; long: BOOLEAN; VAR w, byteLen: INTEGER);
		CONST REXW = 48H;	(* REX prefix with W bit set, to indicate 64-bit operand size *)
	BEGIN
		SIZEPREF := FALSE; REX := {};
		IF what IN words THEN SIZEPREF := TRUE; w := 1; byteLen := 2	(* operand size override for 16bit *)
		ELSIF what IN doubles THEN w := 1; byteLen := 4
		ELSIF what IN { T.fInt64 } + T.pointers THEN w := 1; byteLen := 4; IF long THEN INCL(REX, RexW) END
		ELSIF (*unreachable branch*) what IN quads THEN INCL(REX, RexW); w := 1; byteLen := 8
		ELSIF what IN bytes THEN w := 0; byteLen := 1	(* 8-bit operands *)
		ELSE HALT(20)
		END
		(* w = 0 results in 0B0H for MOV, w = 1 results in 0B8H for MOV *)
	END LCheckPrefix;

	PROCEDURE LCheckPrefixes (what: INTEGER; VAR opts: SET);
	BEGIN 			(* could INCL op1 in the first place and then exclude it in one case *)
		IF (what = T.fReal64) & (xmm IN opts) THEN INCL(opts, RexW)
		ELSIF what IN T.reals THEN
		ELSIF what IN words THEN opts := opts + { szPref, op1 } 	(* operand size override for 16bit *)
		ELSIF what IN doubles THEN INCL(opts, op1)
		ELSIF what IN { T.fInt64, T.fType } + T.pointers THEN opts := opts + { op1, RexW }
		ELSIF what IN bytes THEN	(* 8-bit operands - op1 not included *)
		ELSIF what IN { T.fRecord, T.fArray, T.fArrayPtr } + T.strings THEN INCL(opts, RexW)	(* LEA operations? *)
		ELSE HALT(20)
		END
	END LCheckPrefixes;

	PROCEDURE CheckForm (form: BYTE; VAR mf: INTEGER);
	BEGIN
		IF form = Real32 THEN mf := 0
		ELSIF form = Real64 THEN mf := 4
		ELSIF form = Int32 THEN mf := 2
		ELSE ASSERT(form = Int16); mf := 6
		END
	END CheckForm;

		ELSIF what = T.fInt32 THEN mf := 2
		ELSIF what = T.fChar16 THEN mf := 6	(*experimental, cf. B.CheckCharConv*)

	PROCEDURE DoGenConst (VAR x: Item; short, emit: BOOLEAN; VAR immLen: INTEGER);
		(** emit code for an immediate value *)
	BEGIN	ASSERT(x.what IN T.integers + T.chars + { T.fBool, T.fSet} + T.pointers, 20);
		ASSERT(x.o = NIL, 21);
		IF (x.what IN bytes) OR (short & (x.offset >= -128) & (x.offset < 128)) THEN
			immLen := 1; IF emit THEN E.GenByte(x.offset) END
		ELSIF x.what IN {T.fInt16, T.fChar16} THEN immLen := 2; IF emit THEN GenWord(x.offset) END
		ELSIF (x.what IN doubles) OR (x.what IN quads) & (short & ((x.scale = 0) OR (x.scale = -1))) THEN 
			immLen := 4; IF emit THEN GenDbl(x.offset) END
		ELSIF x.what IN quads THEN immLen := 8; IF emit THEN GenQuad(x.offset, x.scale) END
		ELSE HALT(22)
		END
	END DoGenConst;

	PROCEDURE LCheckImm (IN x: Item; tryShort: BOOLEAN; OUT opFix, immLen: INTEGER);
	BEGIN	ASSERT(x.what IN T.integers + T.chars + { T.fBool, T.fSet} + T.pointers + T.strings, 20);
		ASSERT(x.o = NIL, 21);
		opFix := 0;
		IF (x.what IN bytes) OR (tryShort & (x.offset >= -128) & (x.offset < 128)) THEN
			immLen := 1; 
			IF (x.what IN T.integers + { T.fChar16 } + T.sets + T.pointers) THEN opFix := 2 END
		ELSIF x.what IN {T.fInt16, T.fChar16} THEN immLen := 2
		ELSIF (x.what IN doubles) OR (x.what IN quads) & (tryShort & ((x.scale = 0) OR (x.scale = -1))) THEN immLen := 4
		ELSIF x.what IN quads THEN immLen := 8
		ELSE HALT(22)
		END
	END LCheckImm;
	
	PROCEDURE GenConst (VAR x: Item; short: BOOLEAN);	(* emit code for an immediate value *)
		VAR _: INTEGER;
	BEGIN 	ASSERT(x.what IN T.integers + T.chars + { T.fBool, T.fSet} + T.pointers + T.strings, 20);
		IF x.o # NIL THEN LGenLinked(x, ripBased)
		ELSE DoGenConst(x, short, TRUE, _)
		END
	END GenConst;
	
	PROCEDURE DoGenConst2 (IN x: Item; len: INTEGER);
	BEGIN

	PROCEDURE GenCExt (code: INTEGER; VAR x: Item);
		VAR disp, mod, base, scale: INTEGER;
	BEGIN
		ASSERT(x.mode IN {Reg, Ind, Abs, Stk});
		ASSERT((code MOD 8 = 0) & (code < 64));
		disp := x.offset; base := x.reg; scale := x.scale;
		IF x.mode = Reg THEN mod := 0C0H; scale := 0
		ELSIF x.mode = Stk THEN base := SP; mod := 0; disp := 0; scale := 0
		ELSIF x.mode = Abs THEN
			IF scale = 1 THEN base := x.index; mod := 80H; scale := 0
			ELSE base := BP; mod := 0
			END
		ELSIF (disp = 0) & (base # BP) THEN mod := 0
		ELSIF (disp >= -128) & (disp < 128) THEN mod := 40H
		ELSE mod := 80H
		END;
		IF scale # 0 THEN
			DevCPE.GenByte(mod + code + 4); base := base + x.index * 8;
			IF scale = 8 THEN DevCPE.GenByte(0C0H + base);
			ELSIF scale = 4 THEN DevCPE.GenByte(80H + base);
			ELSIF scale = 2 THEN DevCPE.GenByte(40H + base);
			ELSE ASSERT(scale = 1); DevCPE.GenByte(base);
			END;
		ELSE
			DevCPE.GenByte(mod + code + base);
			IF (base = SP) & (mod <= 80H) THEN DevCPE.GenByte(24H) END
		END;
		IF x.mode = Abs THEN GenLinked(x, absolute)
		ELSIF mod = 80H THEN GenDbl(disp)
		ELSIF mod = 40H THEN DevCPE.GenByte(disp)
		END
	END GenCExt;

	PROCEDURE GenMove* (VAR from, to: Item);
		VAR w: INTEGER;
	BEGIN
		ASSERT(Size[from.form] = Size[to.form]);
		IF to.mode = Reg THEN
			IF from.mode = Con THEN
				IF to.reg = AX THEN

					IF (a1.mode = Con) & (from.offset = a1.offset) & (from.obj = a1.obj) & (from.form = a1.form) THEN
						RETURN
					END;

					a1 := from; a2.mode := 0
				END;
				CheckSize(from.form, w);
				IF (from.offset = 0) & (from.obj = NIL) THEN
					DevCPE.GenByte(30H + w); DevCPE.GenByte(0C0H + 9 * to.reg) (* XOR r,r *)
				ELSE
					DevCPE.GenByte(0B0H + w * 8 + to.reg); GenConst(from, FALSE)
				END;
			ELSIF (to.reg = AX) & (from.mode = Abs) & (from.scale = 0) THEN

				IF (a1.mode = Abs) & (from.offset = a1.offset) & (from.obj = a1.obj) & (from.form = a1.form)
					OR (a2.mode = Abs) & (from.offset = a2.offset) & (from.obj = a2.obj) & (from.form = a2.form) THEN
					RETURN
				END;

				a1 := from; a2.mode := 0;
				CheckSize(from.form, w);
				DevCPE.GenByte(0A0H + w); GenLinked(from, absolute);
			ELSIF (from.mode # Reg) OR (from.reg # to.reg) THEN
				IF to.reg = AX THEN
					IF (from.mode = Ind) & (from.scale = 0) & ((from.reg = BP) OR (from.reg = BX)) THEN

						IF (a1.mode = Ind) & (from.offset = a1.offset) & (from.reg = a1.reg) & (from.form = a1.form)
							OR (a2.mode = Ind) & (from.offset = a2.offset) & (from.reg = a2.reg) & (from.form = a2.form) THEN
							RETURN
						END;

						a1 := from
					ELSE a1.mode := 0
					END;
					a2.mode := 0
				END;
				CheckSize(from.form, w);
				DevCPE.GenByte(8AH + w); GenDExt(to, from)
			END 
		ELSE
			CheckSize(from.form, w);
			IF from.mode = Con THEN
				DevCPE.GenByte(0C6H + w); GenCExt(0, to); GenConst(from, FALSE);
				a1.mode := 0; a2.mode := 0
			ELSIF (from.reg = AX) & (to.mode = Abs) & (to.scale = 0) THEN
				DevCPE.GenByte(0A2H + w); GenLinked(to, absolute);
				a2 := to
			ELSE
				DevCPE.GenByte(88H + w); GenDExt(from, to);
				IF from.reg = AX THEN
					IF (to.mode = Ind) & (to.scale = 0) & ((to.reg = BP) OR (to.reg = BX)) THEN a2 := to END
				ELSE a1.mode := 0; a2.mode := 0
				END
			END
		END
	END GenMove;

	PROCEDURE LGenOpReg (op, reg: INTEGER);
	(** Generate instruction with opcode op with reg encoded into op and REX; based on global instruction variables, and reset the latter *)
	BEGIN
		IF reg >= 8 THEN INCL(REX, RexB) END;
		IF REX # {} THEN E.GenByte(40H + ORD(REX)); REX := {} END;
		IF SIZEPREF THEN E.GenByte(66H); SIZEPREF := FALSE END;
		IF op < 100H THEN E.GenByte(op) ELSIF op < 10000H THEN GenWord(op) ELSE HALT(20) END;
		IF hasMRM THEN E.GenByte(MRM); hasMRM := FALSE END;
		IF hasSIB THEN E.GenByte(SIB); hasSIB := FALSE END;
		CASE DISPLEN OF
		| 0:
		| 1: E.GenByte(DISP)
		| 4: GenDbl(DISP)
		ELSE HALT(20)
		END;
		DISPLEN := 0
	END LGenOpReg;

	PROCEDURE LGenOp (op: INTEGER);
	(** Generate instruction with opcode op based on global instruction variables, and reset the latter *)
	BEGIN LGenOpReg(op, 0)
	END LGenOp;
	
	PROCEDURE DoGenOp2 (opcode, reg: INTEGER; VAR x, imm: Item; opts: SET);
	(** Generate an operation. fix in fix1 or noFixOp or other *)
		CONST rex = { RexW, RexB, RexR, RexX };
		VAR _, w, opFix, immLen, pos: INTEGER;
	BEGIN 
		IF addReg IN opts THEN ASSERT(reg = ignored, 20) END;
		pos := E.pc; 
		LCheckPrefixes(x.what, opts); 
		IF addReg IN opts THEN reg := x.reg; IF (8 <= reg) & (reg < 16) THEN INCL(opts, RexB) END
		ELSIF ~(noReg IN opts) THEN
			IF (8 <= reg) & (reg < 16) THEN INCL(opts, RexR) END
		ELSIF reg = ignored THEN
		ELSE ASSERT((0 <= reg) & (reg < 8))
		END;
		IF (x.mode = Reg) & (8 <= x.reg) & (x.reg < 16) THEN INCL(opts, RexB) END; (*!*)
		IF forceRexW IN opts THEN INCL(opts, RexW) ELSIF noRexW IN opts THEN EXCL(opts, RexW) END;
		(* eliminate forceRexW? now RexW can be directly included in opts *)
		(*REX := REX - REXIGNORE; REXIGNORE := {};*)
		IF xmm IN opts THEN E.GenByte(66H); INC(pos) END;
		IF rex * opts # {} THEN INC(pos); E.GenByte(40H + ORD(rex * opts)) END;
		IF ~(xmm IN opts) & (szPref IN opts) & (fix1 IN opts) THEN E.GenByte(66H); INC(pos) END; 
		IF op1 IN opts THEN w := 1 ELSE w := 0 END;
		IF fix1 IN opts THEN (*w := w*) ELSIF fix8 IN opts THEN w := w * 8 ELSE w := 0 END;
		INC(opcode, w);
		IF addReg IN opts THEN 	(* reg[0..2] is added to opcode; op operates on that reg and a imm (no MRM) *)
			INC(opcode, reg MOD 8)
		ELSIF reg = ignored THEN reg := 0
		ELSE reg := reg MOD 8 * 8	(* reg -> regfld, shift 3 bits left *)
		END;
		
		IF imm.what # T.fUndef THEN LCheckImm(imm, fixShort IN opts, opFix, immLen) ELSE opFix := 0; immLen := 0 END;
		IF fixShort IN opts THEN INC(opcode, opFix) END;
		
		IF opcode < 0 THEN HALT(20) ELSIF opcode > 0FFFFH THEN HALT(21) END;
		IF opcode > 0FFH THEN E.GenByte(opcode DIV 100H); INC(pos) END;
		E.GenByte(opcode); INC(pos);
		
		IF imm.what # T.fUndef THEN w := ImmLen(imm, fixShort IN opts) ELSE w := 0 END;
		IF ~(addReg IN opts) & ~(noMRM IN opts) THEN
			LGenMRMSIB(pos, reg, x, immLen);	(* => DISP *)
			IF hasMRM THEN E.GenByte(MRM); hasMRM := FALSE END;
			IF hasSIB THEN E.GenByte(SIB); hasSIB := FALSE END;
			CASE DISPLEN OF
			| 0:
			| 1: E.GenByte(DISP)
			| 4: GenDbl(DISP)
			ELSE HALT(20)
			END;
			DISPLEN := 0
		END;
		
		IF imm.what # T.fUndef THEN DoGenConst2(imm, immLen) END
	END DoGenOp2;

	PROCEDURE LGenOp2 (opcode, fix, reg: INTEGER; VAR x: Item);
		VAR opts: SET;
	BEGIN 
		IF fix = fix1 THEN opts := { fix1 } ELSE opts := {} END;
		DoGenOp2(opcode, reg, x, unused, opts)
	END LGenOp2;

	PROCEDURE LGenOpLong (opcode, fix, reg: INTEGER; VAR x: Item);
	(** Generate an operation that defaults to 8-byte operands (i.e. no REX.W prefix) and doesn't have imm field *)
	BEGIN REXIGNORE := { RexW }; LGenOp2(opcode, fix, reg, x); REXIGNORE := {}
		;HALT(125);
	END LGenOpLong;
	
	PROCEDURE LGenOpConst (opcode, fix: INTEGER; VAR const: Item);
	(** Generate an operation that doesn't have MRMSIB bytes: one of it's operands is encoded in opcode, and the other is the constant *)
		VAR w: INTEGER;
	BEGIN
		IF (fix # noFixOp) & ~(const.what IN bytes) THEN w := fix ELSE w := 0 END;
		E.GenByte(opcode + w); GenConst(const, FALSE)
	END LGenOpConst;
	
	PROCEDURE LGenOpImm (opcode, regfld: INTEGER; VAR x, const: Item; opts: SET);
	(* Generate an operation opcode with an operand x, an immediate operand const and REG field regfld in the MRM byte taking into account options opts (IN fix1, short). short IN opts => generate a 1-byte imm operand  *)
	BEGIN
		DoGenOp2(opcode, regfld, x, const, opts)
	END LGenOpImm;

	PROCEDURE LGenMove* (VAR from, to: Item);
		CONST i64ptrs = { T.fInt64 } + T.pointers;
		VAR w, bytes: INTEGER; vv: T.Object; second, dbl: BOOLEAN; z, four, base: Item;
	BEGIN 
		ASSERT(SSize[from.what] = SSize[to.what]);
		IF to.mode = Reg THEN
			IF from.mode = Con THEN
				IF to.reg = AX THEN
					IF (a1.mode = Con) & (from.offset = a1.offset) & (from.obj = a1.obj) & (from.what = a1.what) THEN
						RETURN
					END;
					a1 := from; a2.mode := 0
				END;
				IF (from.offset = 0) & (from.what IN doubles) OR (from.what IN quads) & (from.offset = 0) & (from.scale = 0)
				& (from.obj = NIL) THEN	(* ? zero literal? *)
					LGenOpRMR(32H, to.reg, to, { fix1 })	 (* XOR r,r *)
				ELSE
					DoGenOp2(0B0H, ignored, to, from, { fix8, addReg })	(* MOV reg8-64, imm8-64 *)
				END;
			ELSIF (to.reg = AX) & (from.mode = Abs) & (from.scale = 0)(*?non-index*) THEN
				IF (a1.mode = Abs) & (from.offset = a1.offset) & (from.obj = a1.obj) & (from.what = a1.what)
					OR (a2.mode = Abs) & (from.offset = a2.offset) & (from.obj = a2.obj) & (from.what = a2.what) THEN
					RETURN
				END;
				a1 := from; a2.mode := 0;
				LGenOpRMR(8AH, to.reg, from, { fix1 }); 	(* MOV reg, reg/mem *)
			ELSIF (from.mode # Reg) OR (from.reg # to.reg) THEN
				IF to.reg = AX THEN
					IF (from.mode = Ind) & (from.scale = 0) & ((from.reg = BP) OR (from.reg = BX)) THEN
						IF (a1.mode = Ind) & (from.offset = a1.offset) & (from.reg = a1.reg) & (from.what = a1.what)

							RETURN
						END;
						a1 := from
					ELSE a1.mode := 0
					END;
					a2.mode := 0
				END;
				LGenOpRMR(8AH, to.reg, from, { fix1 })	(* MOV reg, reg/mem *)

				dbl := from.what IN { T.fInt64 } + T.pointers; IF dbl THEN from.what := T.fInt32 END;
				second := dbl &~((from.scale = 0) & (from.offset >= 0)); IF second THEN to.what := T.fInt32 END;

					IF to.scale = 0 THEN INC(to.offset, 4) 
					ELSE LMakeConst(four, 4, T.fInt64); LMakeReg(base, to.reg, T.fInt64); LGenAdd(four, base, FALSE)
					END;
					from.offset := from.scale;
					DoGenOp2(0C6H, ignored, to, from, { fix1 });
				END;
				a1.mode := 0; a2.mode := 0
			ELSIF (from.reg = AX) & (to.mode = Abs) & (to.scale = 0) THEN
				LGenOpRMR(88H, from.reg, to, { fix1 });	(* MOV r/m, reg *) 
				a2 := to

				LGenOpRMR(88H, from.reg, to, { fix1 });	(* MOV r/m, reg *) 
				IF from.reg = AX THEN
					IF (to.mode = Ind) & (to.scale = 0) & ((to.reg = BP) OR (to.reg = BX)) THEN a2 := to END
				ELSE a1.mode := 0; a2.mode := 0
				END
			END
		END
	END LGenMove;

		IF (from.mode = Reg) & (from.reg = AX) & (to.reg = DX) THEN
			DevCPE.GenByte(99H)	(* cdq *)
		ELSE
			GenMove(from, to);	(* mov to, from *)
			DevCPE.GenByte(0C1H); GenCExt(38H, to); DevCPE.GenByte(31)	(* sar to, 31 *)
		END
	END GenSignExt;
	
	PROCEDURE GenLoadAdr* (VAR from, to: Item);
	BEGIN
		ASSERT(to.form IN {Int32, Pointer, ProcTyp});
		IF (from.mode = Abs) & (from.scale = 0) THEN
			DevCPE.GenByte(0B8H + to.reg); GenLinked(from, absolute)
		ELSIF from.mode = Stk THEN
			DevCPE.GenByte(89H); GenCExt(SP * 8, to)
		ELSIF (from.mode # Ind) OR (from.offset # 0) OR (from.scale # 0) THEN
			DevCPE.GenByte(8DH); GenDExt(to, from)	(* LEA *)
		ELSIF from.reg # to.reg THEN
			DevCPE.GenByte(89H); GenCExt(from.reg * 8, to)
		ELSE RETURN
		END;
		IF to.reg = AX THEN a1.mode := 0; a2.mode := 0 END
	END GenLoadAdr;

	PROCEDURE GenConOp (op: INTEGER; VAR src, dst: Item);
		VAR w, s: INTEGER;
	BEGIN
		ASSERT(Size[src.form] = Size[dst.form]);
		CheckSize(src.form, w);
		CheckConst(src, s);
		IF (dst.mode = Reg) & (dst.reg = AX) & (s = 0) THEN
			DevCPE.GenByte(op + 4 + w); GenConst(src, FALSE)
		ELSE
			DevCPE.GenByte(80H + s + w); GenCExt(op, dst); GenConst(src, TRUE)
		END
	END GenConOp;
	
	PROCEDURE LGenConOp (op: INTEGER; VAR src, dst: Item);	(* Generate operation with a constant *)
		VAR s: INTEGER; 
	BEGIN
		ASSERT(SSize[src.what] = SSize[dst.what]);
		(*LCheckSize(src.what, TRUE, w, _);*)	(* LCheckPrefixes is later called from LGenOp2. Is this optimizable? *)
		LCheckConst(src, s);
		IF src.what IN quads THEN ASSERT((src.scale = 0) OR (src.scale = -1) OR (src.mode = Reg)) END;
		IF (dst.mode = Reg) & (dst.reg = AX) & (s = 0) THEN
			LGenOpImm(op DIV 8 + 4, ignored, src, src, { fix1, noMRM })
		ELSE
			LGenOpImm(80H, op DIV 8, dst, src, { fix1, fixShort })
		END
	END LGenConOp;
	
	PROCEDURE GenDirOp (op: INTEGER; VAR src, dst: Item);
		VAR w: INTEGER;
	BEGIN
		ASSERT(Size[src.form] = Size[dst.form]);
		CheckSize(src.form, w);
		IF dst.mode = Reg THEN
			DevCPE.GenByte(op + 2 + w); GenDExt(dst, src)
		ELSE
			DevCPE.GenByte(op + w); GenDExt(src, dst)
		END
	END GenDirOp;

	PROCEDURE LGenMul* (VAR src, dst: Item; ovflchk: BOOLEAN);
		CONST LEA = 8DH; SHL = 0C1H; 
		VAR pref, fixup, w, s, val, f2, f5, f9: INTEGER; prefixed: BOOLEAN;
	BEGIN 
		ASSERT((dst.mode = Reg) & (SSize[src.what] = SSize[dst.what]));
		prefixed := src.what = T.fInt64; IF prefixed THEN pref := 48H (* or other val for R8..R16 *) END;
		IF (src.mode = Con) & (src.offset = 1) THEN RETURN END;
		IF src.what IN bytes THEN
			ASSERT(dst.reg = 0);
			E.GenByte(0F6H); GenCExt(28H, src)	(* IMUL AL, r/m8 *)
		ELSIF src.mode = Con THEN
			val := src.offset;
			IF (src.obj = NIL) & (val # 0) & ~ovflchk THEN
				f2 := 0; f5 := 0; f9 := 0;
				WHILE ~ODD(val) DO val := val DIV 2; INC(f2) END;
				WHILE val MOD 9 = 0 DO val := val DIV 9; INC(f9) END;
				WHILE val MOD 5 = 0 DO val := val DIV 5; INC(f5) END;
				IF ABS(val) <= 3 THEN
					WHILE f9 > 0 DO
						IF prefixed THEN E.GenByte(pref) END;
						E.GenByte(LEA); E.GenByte(dst.reg * 8 + 4); E.GenByte(0C0H + dst.reg * 9); DEC(f9)
					END;
					WHILE f5 > 0 DO 
						IF prefixed THEN E.GenByte(pref) END;
						E.GenByte(LEA); E.GenByte(dst.reg * 8 + 4); E.GenByte(80H + dst.reg * 9); DEC(f5)
					END;
					IF ABS(val) = 3 THEN 
						IF prefixed THEN E.GenByte(pref) END;
						E.GenByte(LEA); E.GenByte(dst.reg * 8 + 4); E.GenByte(40H + dst.reg * 9)
					END;
					IF f2 > 1 THEN
						IF prefixed THEN E.GenByte(pref) END;
						E.GenByte(0C1H); E.GenByte(0E0H + dst.reg); E.GenByte(f2) 	(* SHL *)
					ELSIF f2 = 1 THEN 
						IF prefixed THEN E.GenByte(pref) END;
						E.GenByte(1); E.GenByte(0C0H + dst.reg * 9)	(* ADD r/m, reg *)
					END;
					IF val < 0 THEN 
						LGenOpRMR(0F7H, 3, dst, { noReg })	(* NEG r/m *)
					END;
					IF dst.reg = AX THEN a1.mode := 0; a2.mode := 0 END;
					RETURN		(* ARRRRGGGGGHHHH!!!!! *)
				END
			END;
			LGenOpImm(69H, dst.reg, dst, src, { fixShort })	(* IMUL reg, r/m, imm *)	(*$138*)
		ELSE
			LGenOpRMR(0FAFH, dst.reg, src, {})	(* IMUL reg, r/m *)	(*$139*)
		END;
		IF ovflchk THEN E.GenByte(0CEH) END;
		IF dst.reg = AX THEN a1.mode := 0; a2.mode := 0 END
	END LGenMul;
	
	PROCEDURE GenMul* (VAR src, dst: Item; ovflchk: BOOLEAN);
		VAR w, s, val, f2, f5, f9: INTEGER;
	BEGIN
		ASSERT((dst.mode = Reg) & (Size[src.form] = Size[dst.form]));
		IF (src.mode = Con) & (src.offset = 1) THEN RETURN END;
		IF src.form <= Int8 THEN
			ASSERT(dst.reg = 0);
			DevCPE.GenByte(0F6H); GenCExt(28H, src)
		ELSIF src.mode = Con THEN
			val := src.offset;
			IF (src.obj = NIL) & (val # 0) & ~ovflchk THEN
				f2 := 0; f5 := 0; f9 := 0;
				WHILE ~ODD(val) DO val := val DIV 2; INC(f2) END;
				WHILE val MOD 9 = 0 DO val := val DIV 9; INC(f9) END;
				WHILE val MOD 5 = 0 DO val := val DIV 5; INC(f5) END;
				IF ABS(val) <= 3 THEN
					WHILE f9 > 0 DO
						DevCPE.GenByte(8DH);
						DevCPE.GenByte(dst.reg * 8 + 4);
						DevCPE.GenByte(0C0H + dst.reg * 9);
						DEC(f9)
					END;
					WHILE f5 > 0 DO
						DevCPE.GenByte(8DH);
						DevCPE.GenByte(dst.reg * 8 + 4);
						DevCPE.GenByte(80H + dst.reg * 9);
						DEC(f5)
					END;
					IF ABS(val) = 3 THEN
						DevCPE.GenByte(8DH); DevCPE.GenByte(dst.reg * 8 + 4); DevCPE.GenByte(40H + dst.reg * 9)
					END;
					IF f2 > 1 THEN DevCPE.GenByte(0C1H); DevCPE.GenByte(0E0H + dst.reg); DevCPE.GenByte(f2)
					ELSIF f2 = 1 THEN DevCPE.GenByte(1); DevCPE.GenByte(0C0H + dst.reg * 9)
					END;
					IF val < 0 THEN DevCPE.GenByte(0F7H); GenCExt(18H, dst) END;
					IF dst.reg = AX THEN a1.mode := 0; a2.mode := 0 END;
					RETURN
				END
			END;
			CheckSize(src.form, w); CheckConst(src, s);
			DevCPE.GenByte(69H + s); GenDExt(dst, dst); GenConst(src, TRUE)
		ELSE
			CheckSize(src.form, w);
			DevCPE.GenByte(0FH); DevCPE.GenByte(0AFH); GenDExt(dst, src)
		END;
		IF ovflchk THEN DevCPE.GenByte(0CEH) END;
		IF dst.reg = AX THEN a1.mode := 0; a2.mode := 0 END
	END GenMul;
	
	PROCEDURE GenDiv* (VAR src: Item; mod, pos: BOOLEAN);
		VAR w, rem: INTEGER;
	BEGIN
		ASSERT(src.mode = Reg);
		IF src.form >= Int32 THEN DevCPE.GenByte(99H) (* cdq *)
		ELSIF src.form = Int16 THEN DevCPE.GenByte(66H); DevCPE.GenByte(99H) (* cwd *)
		ELSE DevCPE.GenByte(66H); DevCPE.GenByte(98H) (* cbw *)
		END;
		CheckSize(src.form, w); DevCPE.GenByte(0F6H + w); GenCExt(38H, src); (* idiv src *)
		IF src.form > Int8 THEN rem := 2 (* edx *) ELSE rem := 4 (* ah *) END;
		IF pos THEN (* src > 0 *)
			CheckSize(src.form, w); DevCPE.GenByte(8 + w); DevCPE.GenByte(0C0H + 9 * rem); (* or rem,rem *)
			IF mod THEN
				DevCPE.GenByte(79H); DevCPE.GenByte(2);	(* jns end *)
				DevCPE.GenByte(2 + w); GenCExt(8 * rem, src); (* add rem,src *)
			ELSE
				DevCPE.GenByte(79H); DevCPE.GenByte(1);	(* jns end *)
				DevCPE.GenByte(48H);					(* dec eax *)
			END
		ELSE
			CheckSize(src.form, w); DevCPE.GenByte(30H + w); GenCExt(8 * rem, src); (* xor src,rem *)
			IF mod THEN
				DevCPE.GenByte(79H);	(* jns end *)
				IF src.form = Int16 THEN DevCPE.GenByte(9); DevCPE.GenByte(66H) ELSE DevCPE.GenByte(8) END;
				DevCPE.GenByte(8 + w); DevCPE.GenByte(0C0H + 9 * rem); (* or rem,rem *)
				DevCPE.GenByte(74H); DevCPE.GenByte(4);	(* je end *)
				DevCPE.GenByte(30H + w); GenCExt(8 * rem, src); (* xor src,rem *)
				DevCPE.GenByte(2 + w); GenCExt(8 * rem, src); (* add rem,src *)
			ELSE
				DevCPE.GenByte(79H);	(* jns end *)
				IF src.form = Int16 THEN DevCPE.GenByte(6); DevCPE.GenByte(66H) ELSE DevCPE.GenByte(5) END;
				DevCPE.GenByte(8 + w); DevCPE.GenByte(0C0H + 9 * rem); (* or rem,rem *)
				DevCPE.GenByte(74H); DevCPE.GenByte(1);	(* je end *)
				DevCPE.GenByte(48H);					(* dec eax *)
			END
(*
			CheckSize(src.form, w); DevCPE.GenByte(3AH + w); GenCExt(8 * rem, src); (* cmp rem,src *)
			IF mod THEN
				DevCPE.GenByte(72H); DevCPE.GenByte(4);	(* jb end *)
				DevCPE.GenByte(7FH); DevCPE.GenByte(2);	(* jg end *)
				DevCPE.GenByte(2 + w); GenCExt(8 * rem, src); (* add rem,src *)
			ELSE
				DevCPE.GenByte(72H); DevCPE.GenByte(3);	(* jb end *)
				DevCPE.GenByte(7FH); DevCPE.GenByte(1);	(* jg end *)
				DevCPE.GenByte(48H);					(* dec eax *)
			END
*)
		END;
		a1.mode := 0; a2.mode := 0
	END GenDiv;

	PROCEDURE ^ GenJump* (cc: INTEGER; VAR L: Label; shortjmp: BOOLEAN);	(* unforward! *)
	PROCEDURE LGenDiv* (VAR src: Item; mod, pos: BOOLEAN);
		VAR _, w, rem: INTEGER;	(* src is the divisor; the dividend is already in AX *)
			i64: BOOLEAN; end2, end: Label; remr: Item;
	BEGIN
		ASSERT(src.mode = Reg);	(* why is this required? *)
		i64 := src.what = T.fInt64;
		IF i64 THEN E.GenByte(48H); E.GenByte(99H)	(* CQO RAX->RDX:RAX *)
		ELSIF src.what = T.fInt32 THEN E.GenByte(99H) (* cdq *)
		ELSIF src.what = T.fInt16 THEN E.GenByte(66H); E.GenByte(99H) (* cwd *)
		ELSE E.GenByte(66H); E.GenByte(98H) (* cbw *)
		END;
		LGenOpRMR(0F6H, 7, src, { fix1, noReg }); (* idiv src *)
		IF src.what # T.fInt8 THEN rem := 2 (* edx *) ELSE rem := 4 (* ah *) END;
		(* because rem is RDX, EDX or AH, only REX.W prefix is needed, so it can be outed directly, beside GenOpRMR *)
		IF src.what IN bytes THEN w := 0 ELSE w := 1 END;
		LMakeReg(remr, rem, src.what);
		end := NewLbl; 
		IF pos THEN (* src > 0 *)
			(*LCheckSize(src.what, TRUE, w, _); *)E.GenByte(8 + w); E.GenByte(0C0H + 9 * rem); (* or rem,rem *)
			IF mod THEN
				GenJump(ccNS, end, TRUE);	(* jns end *)
				LGenOpRMR(2, rem, src, { fix1 }) (* add rem,src *)
			ELSE
				GenJump(ccNS, end, TRUE);	(* jns end *)
				IF src.what = T.fInt64 THEN E.GenByte(48H) END;
				E.GenByte(0FFH); E.GenByte(0C8H) 					(* dec eax/rax *)
			END
		ELSE
			end2 := NewLbl;
			LGenOpRMR(30H, rem, src, { fix1 }); (* xor src,rem *)
			IF mod THEN
				GenJump(ccNS, end, TRUE);
				LGenOpRMR(8, rem, remr, { fix1 });	(* or rem,rem *)
				GenJump(ccE, end2, TRUE); 	(* je end *)
				LGenOpRMR(30H, rem, src, { fix1 }); (* xor src,rem *)
				LGenOpRMR(2, rem, src, { fix1 }) (* add rem,src *)
			ELSE
				GenJump(ccNS, end, TRUE);	(* jns end *)
				LGenOpRMR(8, rem, remr, { fix1 }); (* or rem,rem *)
				GenJump(ccE, end2, TRUE);	(* je end *)
				IF src.what = T.fInt64 THEN E.GenByte(48H) END;
				E.GenByte(0FFH); E.GenByte(0C8H) 					(* dec eax/rax *)
			END;
			LSetLabel(end2)
		END;
		LSetLabel(end); 
		a1.mode := 0; a2.mode := 0
	END LGenDiv;

	PROCEDURE GenShiftOp* (op: INTEGER; VAR cnt, dst: Item);
		VAR w: INTEGER;
	BEGIN
		CheckSize(dst.form, w);
		IF cnt.mode = Con THEN
			ASSERT(cnt.offset >= 0); ASSERT(cnt.obj = NIL);
			IF cnt.offset = 1 THEN
				IF (op = 10H) & (dst.mode = Reg) THEN (* shl r *)
					DevCPE.GenByte(w); GenDExt(dst, dst) (* add r, r *)
				ELSE
					DevCPE.GenByte(0D0H + w); GenCExt(op, dst)
				END
			ELSIF cnt.offset > 1 THEN
				DevCPE.GenByte(0C0H + w); GenCExt(op, dst); DevCPE.GenByte(cnt.offset)
			END
		ELSE
			ASSERT((cnt.mode = Reg) & (cnt.reg = CX));
			DevCPE.GenByte(0D2H + w); GenCExt(op, dst)
		END;
		IF (dst.mode # Reg) OR (dst.reg = AX) THEN a1.mode := 0; a2.mode := 0 END
	END GenShiftOp;
	
	PROCEDURE LGenShiftOp* (op: INTEGER; VAR cnt, dst: Item);
		VAR _, w: INTEGER;
	BEGIN
		(*LCheckSize(dst.what, TRUE, w, _);*)
		IF cnt.mode = Con THEN ASSERT(cnt.what = T.fInt32);
			ASSERT(cnt.offset >= 0); ASSERT(cnt.obj = NIL);
			IF cnt.offset = 1 THEN
				IF (op = 10H) & (dst.mode = Reg) THEN (* shl r *)
					LGenOpRMR(0, dst.reg, dst, { fix1 }) (* add r, r *)
				ELSE
					LGenOpRMR(0D0H, op DIV 8, dst, { fix1, noReg })		(* remove DIV after correcting constants *)
				END
			ELSIF cnt.offset > 1 THEN
				LGenOpImm(0C0H, op DIV 8, dst, cnt, { fix1, noReg })
			END
		ELSE
			ASSERT((cnt.mode = Reg) & (cnt.reg = CX));
			LGenOpRMR(0D2H, op DIV 8, dst, { fix1, noReg }) 
		END;
		IF (dst.mode # Reg) OR (dst.reg = AX) THEN a1.mode := 0; a2.mode := 0 END
	END LGenShiftOp;
	
	PROCEDURE GenBitOp* (op: INTEGER; VAR num, dst: Item);
	BEGIN
		DevCPE.GenByte(0FH);
		IF num.mode = Con THEN
			ASSERT(num.obj = NIL);
			DevCPE.GenByte(0BAH); GenCExt(op, dst); DevCPE.GenByte(num.offset)
		ELSE
			ASSERT((num.mode = Reg) & (num.form = Int32));
			DevCPE.GenByte(83H + op); GenDExt(num, dst)
		END;
		IF (dst.mode # Reg) OR (dst.reg = AX) THEN a1.mode := 0; a2.mode := 0 END
	END GenBitOp;

		ELSE
			ASSERT((num.mode = Reg) & (num.what = T.fInt32));
			LGenOpRMR(0F83H + op, num.reg, dst, {})
		END;
		IF (dst.mode # Reg) OR (dst.reg = AX) THEN a1.mode := 0; a2.mode := 0 END
	END LGenBitOp;

	PROCEDURE GenFLoad* (VAR src: Item);
		VAR mf: INTEGER;
	BEGIN
		IF src.mode = Con THEN (* predefined constants *)
			DevCPE.GenByte(0D9H); DevCPE.GenByte(0E8H + src.offset)
		ELSIF src.form = Int64 THEN
			DevCPE.GenByte(0DFH); GenCExt(28H, src)
		ELSE
			CheckForm(src.form, mf);
			DevCPE.GenByte(0D9H + mf); GenCExt(0, src)
		END
	END GenFLoad;

	PROCEDURE LGenRMove* (VAR src, dst: Item);	(* Generate Real move (XMM) *)
	BEGIN
		IF dst.mode = Xmm THEN 	(* load into XMM register - for ms64 calling convention *)
			(*ASSERT(dst.reg < 0);	(* XMM register numbers are negative: for XMMi, i = (-src.reg) DIV 2; src.reg # 0 *)*)
			ASSERT(src.mode IN { Abs, Ind, Stk });
			IF (src.mode = Abs) & (src.lit # NIL) THEN src.type := src.lit.type; src.what := src.type.what END;
			LGenOpRMR(0F6EH, dst.reg, src, { xmm }); 		(* MOVD xmm, mem/reg *)
		ELSIF dst.mode IN { Ind, Abs, Stk } THEN	(* store into memory from XMM register *)
			ASSERT(src.mode = Xmm);
			LGenOpRMR(0F7EH, src.reg, dst, { xmm })	(* MOVD mem/reg, xmm *)
		ELSE HALT(126)
		END
	END LGenRMove;

	PROCEDURE LGenFLoad* (VAR src: Item);
		VAR mf: INTEGER;
	BEGIN
		IF src.mode = Con THEN (* predefined constants *)
			E.GenByte(0D9H); E.GenByte(0E8H + src.offset)
		ELSIF src.what = T.fInt32 THEN LGenOpRMR(0DBH, 0, src, {})
		ELSIF src.what = T.fInt64 THEN LGenOpRMR(0DFH, 5, src, { noRexW })
		ELSE
			LGenOpRMR(0D9H + FloatOpFix(src), 0, src, { })	(* ? FLD mem32real, FLD mem64real *)
			;(*ASSERT(src.lit(T.Real).real = 3.14, 4);*)
		END
	END LGenFLoad;
	
	PROCEDURE GenFStore* (VAR dst: Item; pop: BOOLEAN);
		VAR mf: INTEGER;
	BEGIN
		IF dst.form = Int64 THEN ASSERT(pop);
			DevCPE.GenByte(0DFH); GenCExt(38H, dst); DevCPE.GenByte(9BH)	(* wait *)
		ELSE
			CheckForm(dst.form, mf); DevCPE.GenByte(0D9H + mf);
			IF pop THEN GenCExt(18H, dst); DevCPE.GenByte(9BH)	(* wait *)
			ELSE GenCExt(10H, dst)
			END
		END;
		a1.mode := 0; a2.mode := 0
	END GenFStore;

	PROCEDURE LGenFStore* (VAR dst: Item; pop: BOOLEAN);
		VAR oce (* opcode extension *), mf: INTEGER;
	BEGIN
		IF dst.what = T.fInt64 THEN ASSERT(pop);
			LGenOpRMR(0DFH, 7, dst, {}); E.GenByte(9BH)	(* wait *)
		ELSE
			IF pop THEN oce := 3 ELSE oce := 2 END;	(* FSTP/FST *)
			LGenOpRMR(0D9H + FloatOpFix(dst), oce, dst, { noReg }); (* F(I)ST(P) mem32real, FST mem64real *)
			IF pop THEN E.GenByte(9BH) END	(* wait *)
		END;
		a1.mode := 0; a2.mode := 0
	END LGenFStore;

	PROCEDURE GenJump* (cc: INTEGER; VAR L: Label; shortjmp: BOOLEAN);
	BEGIN
		IF cc # ccNever THEN
			IF shortjmp OR (L > 0) & (E.pc + 2 - L <= 128) & (cc # ccCall) THEN
				IF cc = ccAlways THEN E.GenByte(0EBH)
				ELSE E.GenByte(70H + cc)
				END;
				IF L > 0 THEN E.GenByte(L - E.pc - 1)
				ELSE ASSERT(L = 0); L := -(E.pc + short * 1000000H); E.GenByte(0)
				END
			ELSE
				IF cc = ccAlways THEN E.GenByte(0E9H)
				ELSIF cc = ccCall THEN E.GenByte(0E8H)
				ELSE E.GenByte(0FH); E.GenByte(80H + cc)
				END;
				IF L > 0 THEN GenDbl(L - E.pc - 4)
				ELSE GenDbl(-L); L := -(E.pc - 4 + relative * 1000000H)
				END
			END
		END
	END GenJump;
	
	PROCEDURE GenExtJump* (cc: INTEGER; VAR dst: Item);
	BEGIN
		IF cc = ccAlways THEN DevCPE.GenByte(0E9H)
		ELSE DevCPE.GenByte(0FH); DevCPE.GenByte(80H + cc)
		END;
		dst.offset := 0; GenLinked(dst, relative)
	END GenExtJump;
	
	PROCEDURE GenIndJump* (VAR dst: Item);

		DevCPE.GenByte(0FFH); GenCExt(20H, dst)
	END GenIndJump;

	PROCEDURE GenCaseEntry* (VAR L: Label; last: BOOLEAN);
		VAR typ: INTEGER;
	BEGIN
		IF last THEN typ := tableend * 1000000H ELSE typ := table * 1000000H END;
		IF L > 0 THEN GenDbl(L + typ) ELSE GenDbl(-L); L := -(DevCPE.pc - 4 + typ) END
	END GenCaseEntry;
	
	PROCEDURE GenCall* (VAR dst: Item);
	BEGIN
		IF dst.mode IN {LProc, XProc, IProc} THEN
			DevCPE.GenByte(0E8H);
			IF dst.obj.mnolev >= 0 THEN (* local *)
				IF dst.obj.adr > 0 THEN GenDbl(dst.obj.adr - DevCPE.pc - 4)
				ELSE GenDbl(-dst.obj.adr); dst.obj.adr := -(DevCPE.pc - 4 + relative * 1000000H)
				END
			ELSE (* imported *)
				dst.offset := 0; GenLinked(dst, relative)
			END
		ELSE DevCPE.GenByte(0FFH); GenCExt(10H, dst)
		END;
		a1.mode := 0; a2.mode := 0
	END GenCall;
	
	PROCEDURE LGenCall* (VAR dst: Item);
		VAR p: T.Proc; _: INTEGER;
	BEGIN
		IF (dst.mode = XProc) & (dst.o(T.Proc).imno = 0) THEN p := dst.o(T.Proc);
			E.GenByte(0E8H);	(* CALL near *)
			IF p.imno = 0 THEN (* local *)		(* eliminate this IF! It's always this branch, see above *)
				IF p.adr > 0 THEN GenDbl(p.adr - E.pc - 4)
				ELSE GenDbl(-p.adr); p.adr := -(E.pc - 4 + relative * 1000000H)
				END
			ELSE (* imported *)HALT(20) (* imported procedures: MOV [reg64], fixup; CALL [reg64] *)
			END
		ELSE 
			IF dst.mode = XProc THEN 
				dst.mode := Abs; dst.what := T.fProcedure; dst.offset := 0; dst.index := 0; dst.scale := 0 
			END;
			LGenOpRMR(0FFH, 2, dst, { noRexW })	 (* FF /2 CALL r/m16/64, doesn't need REX.W prefix *)
		END;
		a1.mode := 0; a2.mode := 0
	END LGenCall;
	
	PROCEDURE GenAssert* (cc, no: INTEGER);
	BEGIN
		IF cc # ccAlways THEN
			IF cc >= 0 THEN
				DevCPE.GenByte(70H + cc); (* jcc end *)
				IF no < 0 THEN DevCPE.GenByte(2) ELSE DevCPE.GenByte(3) END
			END;
			IF no < 0 THEN
				DevCPE.GenByte(8DH); DevCPE.GenByte(0E0H - no)
			ELSE
				DevCPE.GenByte(8DH); DevCPE.GenByte(0F0H); DevCPE.GenByte(no)
			END
		END
	END GenAssert;
	
	PROCEDURE LGenAssert* (cc, no: INTEGER);
	BEGIN
		IF cc # ccAlways THEN
			IF cc >= 0 THEN
				E.GenByte(70H + cc); (* jcc end *)
				IF no < 0 THEN E.GenByte(2) ELSE E.GenByte(3) END
			END;
			IF no < 0 THEN
				E.GenByte(8DH); E.GenByte(0E0H - no)
			ELSE
				E.GenByte(8DH); E.GenByte(0F0H); E.GenByte(no)
			END
		END
	END LGenAssert;
	
	PROCEDURE GenReturn* (val: INTEGER);
	BEGIN
		IF val = 0 THEN DevCPE.GenByte(0C3H)
		ELSE DevCPE.GenByte(0C2H); GenWord(val)
		END;
		a1.mode := 0; a2.mode := 0
	END GenReturn;

	PROCEDURE LoadStr (size: INTEGER);
	BEGIN
		IF size = 2 THEN DevCPE.GenByte(66H) END;
		IF size <= 1 THEN DevCPE.GenByte(0ACH) ELSE DevCPE.GenByte(0ADH) END (* lods *)
	END LoadStr;
	
	PROCEDURE LLoadStr (size: INTEGER);
	BEGIN
		IF size = 8 THEN E.GenByte(REXW)	(* is this reachable? *)
		ELSIF size = 2 THEN E.GenByte(66H) 
		END;
		IF size <= 1 THEN E.GenByte(0ACH) ELSE E.GenByte(0ADH) END (* lods *)
	END LLoadStr;
	
	PROCEDURE StoreStr (size: INTEGER);

	PROCEDURE ScanStr (size: INTEGER; rep: BOOLEAN);

		IF rep THEN DevCPE.GenByte(0F2H) END;
		IF size <= 1 THEN DevCPE.GenByte(0AEH) ELSE DevCPE.GenByte(0AFH) END (* scas *)
	END ScanStr;

	END LScanStr;
	
	PROCEDURE TestNull (size: INTEGER);
	BEGIN
		IF size = 2 THEN DevCPE.GenByte(66H) END;
		IF size <= 1 THEN DevCPE.GenByte(8); DevCPE.GenByte(0C0H); (* or al,al *)
		ELSE DevCPE.GenByte(9); DevCPE.GenByte(0C0H); (* or ax,ax *)
		END
	END TestNull;
	
	PROCEDURE LTestNull (size: INTEGER);
		VAR w: INTEGER;
	BEGIN w := 1;
		IF size = 8 THEN E.GenByte(REXW)	(* is this reachable? *)
		ELSIF size = 2 THEN E.GenByte(66H) 
		ELSIF size <= 1 THEN w := 0
		END;
		E.GenByte(8 + w); E.GenByte(0C0H); (* or ax,ax *)
	END LTestNull;
	
	PROCEDURE GenBlockMove* (wsize, len: INTEGER);	(* len = 0: len in ECX *)
		VAR w: INTEGER;
	BEGIN
		IF len = 0 THEN (* variable size move *)
			IF wsize = 4 THEN w := 1 ELSIF wsize = 2 THEN w := 1; DevCPE.GenByte(66H) ELSE w := 0 END;
			DevCPE.GenByte(0F3H); DevCPE.GenByte(0A4H + w); (* rep:movs *)
		ELSE (* fixed size move *)
			len := len * wsize;
			IF len >= 16 THEN
				DevCPE.GenByte(0B9H); GenDbl(len DIV 4); (* ld ecx,len/4 *)
				DevCPE.GenByte(0F3H); DevCPE.GenByte(0A5H); (* rep:movs long*)
				len := len MOD 4
			END;
			WHILE len >= 4 DO DevCPE.GenByte(0A5H); DEC(len, 4) END; (* movs long *);
			IF len >= 2 THEN DevCPE.GenByte(66H); DevCPE.GenByte(0A5H) END; (* movs word *);
			IF ODD(len) THEN DevCPE.GenByte(0A4H) END; (* movs byte *)
		END
	END GenBlockMove;
	
	PROCEDURE LGenBlockMove* (wsize, len: INTEGER);	(* len = 0: len in ECX *)
		VAR w: INTEGER;
	BEGIN
		IF len = 0 THEN (* variable size move *)
			E.GenByte(0F3H); 
			IF wsize = 8 THEN E.GenByte(REXW); w := 1 
			ELSIF wsize = 4 THEN w := 1 
			ELSIF wsize = 2 THEN w := 1; E.GenByte(66H) 
			ELSIF wsize = 1 THEN w := 0 
			ELSE HALT(20)
			END;
			E.GenByte(0A4H + w); (* rep:movs *)
		ELSE (* fixed size move *)
			len := len * wsize;
			IF len >= 16 THEN
				E.GenByte(0B9H); GenDbl(len DIV 4); (* ld ecx,len/4 *)
				E.GenByte(0F3H); E.GenByte(0A5H); (* rep:movs long*)
				len := len MOD 4
			END;
			WHILE len >= 8 DO E.GenByte(48H); E.GenByte(0A5H); DEC(len, 8) END; (* movs quad *);
			IF len >= 4 THEN E.GenByte(0A5H); DEC(len, 4) END; (* movs long *);
			IF len >= 2 THEN E.GenByte(66H); E.GenByte(0A5H) END; (* movs word *);
			IF ODD(len) THEN E.GenByte(0A4H) END; (* movs byte *)
		END
	END LGenBlockMove;
	
	PROCEDURE GenBlockStore* (wsize, len: INTEGER);	(* len = 0: len in ECX *)
		VAR w: INTEGER;
	BEGIN
		IF len = 0 THEN (* variable size move *)
			IF wsize = 4 THEN w := 1 ELSIF wsize = 2 THEN w := 1; DevCPE.GenByte(66H) ELSE w := 0 END;
			DevCPE.GenByte(0F3H); DevCPE.GenByte(0AAH + w); (* rep:stos *)
		ELSE (* fixed size move *)
			len := len * wsize;
			IF len >= 16 THEN
				DevCPE.GenByte(0B9H); GenDbl(len DIV 4); (* ld ecx,len/4 *)
				DevCPE.GenByte(0F3H); DevCPE.GenByte(0ABH); (* rep:stos long*)
				len := len MOD 4
			END;
			WHILE len >= 4 DO DevCPE.GenByte(0ABH); DEC(len, 4) END; (* stos long *);
			IF len >= 2 THEN DevCPE.GenByte(66H); DevCPE.GenByte(0ABH) END; (* stos word *);
			IF ODD(len) THEN DevCPE.GenByte(0ABH) END; (* stos byte *)
		END
	END GenBlockStore;
	
	PROCEDURE GenBlockComp* (wsize, len: INTEGER);	(* len = 0: len in ECX *)
		VAR w: INTEGER;
	BEGIN
		ASSERT(len >= 0);
		IF len > 0 THEN DevCPE.GenByte(0B9H); GenDbl(len) END; (* ld ecx,len *)
		IF wsize = 4 THEN w := 1 ELSIF wsize = 2 THEN w := 1; DevCPE.GenByte(66H) ELSE w := 0 END;
		DevCPE.GenByte(0F3H); DevCPE.GenByte(0A6H + w) (* repe:cmps *)
	END GenBlockComp;

	PROCEDURE LGenBlockComp* (wsize, len: INTEGER);	(* len = 0: len in ECX *)
		VAR w: INTEGER;
	BEGIN
		ASSERT(len >= 0);
		IF len > 0 THEN E.GenByte(0B9H); GenDbl(len) END; (* ld ecx,len *)
		IF wsize = 4 THEN w := 1 ELSIF wsize = 2 THEN w := 1; E.GenByte(66H) ELSE w := 0 END;
		E.GenByte(0F3H); E.GenByte(0A6H + w) (* repe:cmps *)
	END LGenBlockComp;
	
	PROCEDURE GenStringMove* (excl: BOOLEAN; wsize, dsize, len: INTEGER);
	(*
	len = 0: len in ECX, len = -1: len undefined; wsize # dsize -> convert; size = 0: opsize = 1, incsize = 2; excl: don't move 0X
	*)
		VAR loop, end: Label;
	BEGIN
		IF len > 0 THEN DevCPE.GenByte(0B9H); GenDbl(len) END; (* ld ecx,len *)
		(* len >= 0: len IN ECX *)
		IF (dsize = 2) & (wsize < 2) THEN DevCPE.GenByte(31H); DevCPE.GenByte(0C0H) END; (* xor eax,eax *)
		loop := NewLbl; end := NewLbl;
		SetLabel(loop); LoadStr(wsize);
		IF wsize = 0 THEN DevCPE.GenByte(46H) END; (* inc esi *)
		IF len < 0 THEN (* no limit *)
			StoreStr(dsize); TestNull(wsize); GenJump(ccNE, loop, TRUE);
			IF excl THEN (* dec edi *)
				DevCPE.GenByte(4FH);
				IF dsize # 1 THEN DevCPE.GenByte(4FH) END
			END;
		ELSE	(* cx limit *)
			IF excl THEN TestNull(wsize); GenJump(ccE, end, TRUE); StoreStr(dsize)
			ELSE StoreStr(dsize); TestNull(wsize); GenJump(ccE, end, TRUE)
			END;
			DevCPE.GenByte(49H); (* dec ecx *)
			GenJump(ccNE, loop, TRUE);
			GenAssert(ccNever, copyTrap); (* trap *)
			SetLabel(end)
		END;
		a1.mode := 0; a2.mode := 0
	END GenStringMove;
	
	PROCEDURE LGenStringMove* (excl: BOOLEAN; wsize, dsize, len: INTEGER);
	(*
	len = 0: len in ECX, len = -1: len undefined; wsize # dsize -> convert; size = 0: opsize = 1, incsize = 2; excl: don't move 0X
	*)
		VAR loop, end: Label;
	BEGIN
		IF len > 0 THEN E.GenByte(REXW); E.GenByte(0C7H); E.GenByte(0C1H); GenDbl(len) END; (* ld rcx,len *)
		(* len >= 0: len IN RCX *)
		IF (dsize = 2) & (wsize < 2) THEN (*E.GenByte(REXW); *)E.GenByte(31H); E.GenByte(0C0H) END; (* xor eax,eax *)
		loop := NewLbl; end := NewLbl;
		LSetLabel(loop); LLoadStr(wsize);	(* LODS *)
		IF wsize = 0 THEN E.GenByte(REXW); E.GenByte(0FFH); E.GenByte(0C6H) END;	 (* inc rsi *)
		IF len < 0 THEN (* no limit *) 
			LStoreStr(dsize); LTestNull(wsize); GenJump(ccNE, loop, TRUE);
			IF excl THEN (* dec edi *)
				E.GenByte(REXW); 
				IF dsize = 1 THEN E.GenByte(0FFH); E.GenByte(0CFH)	(* DEC RDI *)
				ELSE E.GenByte(83H); E.GenByte(0EFH); E.GenByte(dsize)
				END
			END;
		ELSE	(* cx limit *)
			IF excl THEN LTestNull(wsize); GenJump(ccE, end, TRUE); LStoreStr(dsize)	(* STOS *)
			ELSE LStoreStr(dsize); LTestNull(wsize); GenJump(ccE, end, TRUE)
			END;
			(*E.GenByte(REXW); *)E.GenByte(0FFH); E.GenByte(0C9H);	  (* dec rcx *)
			GenJump(ccNE, loop, TRUE);
			LGenAssert(ccNever, copyTrap); (* trap *)
			LSetLabel(end)
		END;
		a1.mode := 0; a2.mode := 0
	END LGenStringMove;
	
	PROCEDURE GenStringComp* (wsize, dsize: INTEGER);
	(* wsize # dsize -> convert; size = 0: opsize = 1, incsize = 2; *)
		VAR loop, end: Label;
	BEGIN
		IF (dsize = 2) & (wsize < 2) THEN DevCPE.GenByte(31H); DevCPE.GenByte(0C0H); (* xor eax,eax *) END;
		loop := NewLbl; end := NewLbl;
		SetLabel(loop); LoadStr(wsize);
		IF wsize = 0 THEN DevCPE.GenByte(46H) END; (* inc esi *)
		ScanStr(dsize, FALSE); GenJump(ccNE, end, TRUE);
		IF dsize = 0 THEN DevCPE.GenByte(47H) END; (* inc edi *)
		TestNull(wsize); GenJump(ccNE, loop, TRUE);
		SetLabel(end);
		a1.mode := 0; a2.mode := 0
	END GenStringComp;

	PROCEDURE GenStringLength* (wsize, len: INTEGER);	(* len = 0: len in ECX, len = -1: len undefined *)
	BEGIN
		DevCPE.GenByte(31H); DevCPE.GenByte(0C0H); (* xor eax,eax *)
		IF len # 0 THEN DevCPE.GenByte(0B9H); GenDbl(len) END; (* ld ecx,len *)
		ScanStr(wsize, TRUE);
		a1.mode := 0; a2.mode := 0
	END GenStringLength;

		LScanStr(wsize, TRUE);
		a1.mode := 0; a2.mode := 0

	PROCEDURE GenStrStore* (size: INTEGER);
		VAR w: INTEGER;
	BEGIN
		IF size # 0 THEN
			IF size MOD 4 = 0 THEN w := 1; size := size DIV 4
			ELSIF size MOD 2 = 0 THEN w := 2; size := size DIV 2
			ELSE w := 0
			END;
			DevCPE.GenByte(0B9H); GenDbl(size); (* ld ecx,size *)
			IF w = 2 THEN DevCPE.GenByte(66H); w := 1 END
		ELSE w := 0
		END;
		DevCPE.GenByte(0F3H); DevCPE.GenByte(0AAH + w); (* rep:stos *)
		a1.mode := 0; a2.mode := 0
	END GenStrStore;

	PROCEDURE LGenAddChar* (chr: INTEGER; short, last: BOOLEAN);
		VAR ok: Label;
	BEGIN
		ok := NewLbl;
		IF last THEN
			E.GenByte(083H); E.GenByte(0E9H); E.GenByte(2);	(* SUB ecx, 2 *)
			GenJump(ccGE, ok, TRUE)
		ELSE E.GenShort(0C9FFH);	(* DEC ecx *)
			GenJump(ccNE, ok, TRUE)
		END;
		E.GenShort(0E88DH);	(* TRAP -8 *)
		LSetLabel(ok);
		E.GenByte(0B8H); E.GenWord(chr);	(* MOV eax, chr *)
		IF last = short THEN E.GenShort(0AB66H)	(* STOSW *)
		ELSIF short THEN E.GenByte(0AAH)	(* STOSB *)
		ELSE E.GenByte(0ABH)	(* STOSD *)
		END
	END LGenAddChar;
			(* cx=0 is ok if last, otherwise not ok *)
