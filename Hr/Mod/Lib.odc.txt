MODULE HrLib;
	(** The Librarian locates module source, object and symbol hypostases  *)

	IMPORT Files, TextModels, TextViews;
	
	TYPE
		Name* = ARRAY LEN(Files.Name) OF CHAR;
		Librarian* = POINTER TO ABSTRACT RECORD END;
		FakeLibrarian = POINTER TO RECORD (Librarian) END;
	
	VAR 
		fake-: Librarian;
		
	PROCEDURE (lib: Librarian) OldHrSym* (mod: Name; rd: Files.Reader): Files.Reader, NEW, ABSTRACT;
	(** Returns a reader for an existing Herschel symbol file for module mod, if any, recycling rd *)
	
	PROCEDURE (lib: Librarian) NewHrSym* (IN mod: ARRAY OF CHAR): Files.File, NEW, ABSTRACT;
	(** Returns a reader for a new Herschel symbol file for module mod, if any, recycling rd *)
	
	PROCEDURE (lib: Librarian) RegisterHrSym* (sym: Files.File; IN mod: ARRAY OF CHAR), NEW, ABSTRACT;
	
	PROCEDURE (lib: Librarian) NewCPSym* (IN mod: ARRAY OF CHAR): Files.File, NEW, ABSTRACT;
	(** Returns a new CP2 symbol file for module mod *)
	
	PROCEDURE (lib: Librarian) OldCPSym* (mod: Name): Files.File, NEW, ABSTRACT;
	
	PROCEDURE (lib: Librarian) RegisterCPSym* (sym: Files.File; IN mod: ARRAY OF CHAR), NEW, ABSTRACT;
	
	PROCEDURE (lib: Librarian) NewCode* (IN mod: ARRAY OF CHAR; wr: Files.Writer): Files.Writer, NEW, ABSTRACT;
	
	PROCEDURE (lib: Librarian) OldCode* (mod: Name; rd: Files.Reader): Files.Reader, NEW, ABSTRACT;
	
	PROCEDURE (lib: Librarian) RegisterCode* (code: Files.File; IN mod: ARRAY OF CHAR), NEW, ABSTRACT;
	
	PROCEDURE (l: Librarian) OldSource* (IN mod: ARRAY OF CHAR; OUT text: TextModels.Model; OUT beg, end: INTEGER), NEW, ABSTRACT;
	
	PROCEDURE (l: Librarian) OldSourceView* (IN mod: ARRAY OF CHAR; OUT text: TextViews.View; OUT beg, end: INTEGER; OUT loc: Files.Locator; OUT name: Files.Name), NEW, ABSTRACT;

	
	
	
	
	
	PROCEDURE (lib: FakeLibrarian) OldHrSym* (mod: Name; rd: Files.Reader): Files.Reader;
	BEGIN HALT(126); RETURN NIL
	END OldHrSym;
	
	PROCEDURE (lib: FakeLibrarian) NewHrSym* (IN mod: ARRAY OF CHAR): Files.File;
	BEGIN RETURN Files.dir.Temp()
	END NewHrSym;

	PROCEDURE (lib: FakeLibrarian) RegisterHrSym* (sym: Files.File; IN mod: ARRAY OF CHAR), EMPTY;

	PROCEDURE (lib: FakeLibrarian) NewCPSym* (IN mod: ARRAY OF CHAR): Files.File;
	BEGIN RETURN Files.dir.Temp()
	END NewCPSym;
	
	PROCEDURE (lib: FakeLibrarian) OldCPSym* (mod: Name): Files.File;
	BEGIN RETURN NIL
	END OldCPSym;
	
	PROCEDURE (lib: FakeLibrarian) RegisterCPSym* (sym: Files.File; IN mod: ARRAY OF CHAR), EMPTY;
	
	PROCEDURE (lib: FakeLibrarian) NewCode* (IN mod: ARRAY OF CHAR; wr: Files.Writer): Files.Writer;
	BEGIN HALT(126); RETURN NIL
	END NewCode;
	
	PROCEDURE (lib: FakeLibrarian) OldCode* (mod: Name; rd: Files.Reader): Files.Reader;
	BEGIN HALT(126); RETURN NIL
	END OldCode;
	
	PROCEDURE (lib: FakeLibrarian) RegisterCode* (sym: Files.File; IN mod: ARRAY OF CHAR), EMPTY;

	PROCEDURE (l: FakeLibrarian) OldSource* (IN mod: ARRAY OF CHAR; OUT text: TextModels.Model; OUT beg, end: INTEGER);
	BEGIN HALT(126)
	END OldSource;

	PROCEDURE (l: FakeLibrarian) OldSourceView* (IN mod: ARRAY OF CHAR; OUT text: TextViews.View; OUT beg, end: INTEGER; OUT loc: Files.Locator; OUT name: Files.Name);
	BEGIN HALT(126)
	END OldSourceView;


	PROCEDURE Init;
		VAR f: FakeLibrarian;
	BEGIN NEW(f); fake := f
	END Init;

BEGIN Init
END HrLib.
