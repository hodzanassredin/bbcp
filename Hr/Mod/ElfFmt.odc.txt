MODULE HrElfFmt;
	
	IMPORT SYSTEM, Files, Utf, 
		B := HrBinder;
	
	CONST
		pageSize = 1000H; 
		v32 = 0; v64 = 1;
		 version = v64;	(* should be a variable when/if 32bit is implemented *)
	CONST PT_LOAD = 1; PT_DYNAMIC = 2; (*PT_INTERP = 3;*) PT_PHDR = 6;
		(*SHT_STRTAB = 3; SHT_DYNSYM = 11; SHT_RELA = 4; SHT_DYNAMIC = 6; SHT_PROGBITS = 1;*)
		SHF_ALLOC = 1;
		PF_X = 0; PF_W = 1; PF_R = 2; 
		rwProt = { PF_R, PF_W }; rProt = { PF_R }; rxProt = { PF_R, PF_X };
		pageAligned = 1X; posAligned = 2X; adAligned = 3X;	(** alignment types *)
		linuxInterpreter = "/lib/ld-linux.so.2"; (* ELF32 x86 *)
		linuxInterpreter64 = "/lib64/ld-linux-x86-64.so.2";
		openBSDInterpreter = "/usr/libexec/ld.so"; (* ELF32, ELF64 *)
		freeBSDInterpreter = "/libexec/ld-elf.so.1";

		(* OS ABI  *)
			ELFOSABI_NONE* = 0X;
			ELFOSABI_NETBSD* = 2X;
			ELFOSABI_LINUX* = 3X;
			ELFOSABI_FREEBSD* = 9X;
			ELFOSABI_OPENBSD* = 0BX;
			ELFOSABI_ARM* = 61X;
		

	TYPE
		Formatter = POINTER TO RECORD (B.Formatter) END;
		Section = RECORD
			pos, 	(** position in file *)
			len,	(** length in file and size in memory match *)
			addr: INTEGER;	(** address in memory *)
			nameInd: INTEGER;	(* name index *)
			flags: SET;
		END;
		
	VAR
		out: Files.Writer;

		zero: ARRAY 4096 OF BYTE;
		segments: INTEGER;	(* number of CP module segments written to ELF; counted in DoAddOcf for OutTables *)
		
		iinterpreter: ARRAY 2 OF ARRAY 256 OF SHORTCHAR;	(* elf interpreter path *)
		machine,
		symEntSize,
		relEntSize,
		dynEntSize,
		zeroBase, 		(* default load address for first segment *)
		addrSize, 
		elfHeaderSize, phtEntSize, shtEntSize: ARRAY 2 OF INTEGER;
		
		shtOffs: INTEGER;	(* values for ELF header: program and section header tables file offsets *)
		sections: INTEGER;	(* counters for segments and sections *)
		
		m64: BOOLEAN;	(* m64 = version = v64 *)
		availAddr, availPos: INTEGER;	(* first available virtual address and file position *)
		
		formatter: Formatter;


	PROCEDURE OutSString (IN s: ARRAY OF SHORTCHAR);
		VAR len: INTEGER;
	BEGIN len := LEN(s$) + 1; out.WriteBytes(SYSTEM.THISARRAY(SYSTEM.ADR(s[0]), len), 0, len)
	END OutSString;
	
	PROCEDURE OutInt (x: INTEGER);
	BEGIN out.WriteBytes(SYSTEM.THISARRAY(SYSTEM.ADR(x), 4), 0, 4)
	END OutInt;
	
	PROCEDURE OutLong (x: LONGINT);
	BEGIN out.WriteBytes(SYSTEM.THISARRAY(SYSTEM.ADR(x), 8), 0, 8)
	END OutLong;
	
	PROCEDURE OutShort (x: INTEGER);
	BEGIN out.WriteBytes(SYSTEM.THISARRAY(SYSTEM.ADR(x), 2), 0, 2)
	END OutShort;
	
	PROCEDURE OutIntOrLong (x: LONGINT);
	BEGIN IF m64 THEN OutLong(x) ELSE OutInt(SHORT(x)) END
	END OutIntOrLong;




	PROCEDURE PosAligned (pos, x: INTEGER): INTEGER;
		VAR res, modulo: INTEGER;
	BEGIN 
		modulo := pos MOD pageSize;
		IF modulo = 0 THEN res := pos ELSE res := pos + pageSize END;
		IF res < x THEN res := B.PageAligned(x) + modulo END;
		RETURN res
	END PosAligned;
	

	PROCEDURE SkipTo (pos: INTEGER); 
	BEGIN DEC(pos, out.Pos()); ASSERT(pos >= 0, 20); IF pos > 0 THEN out.WriteBytes(zero, 0, pos) END
	END SkipTo;
	
	PROCEDURE OutSeg (IN s: B.Segment; a: ARRAY OF BYTE; b: POINTER TO ARRAY OF BYTE);
	(** Output data in a and b (if any) into segment s *)
	BEGIN
		SkipTo(s.pos); ASSERT(s.pos = out.Pos());
		out.WriteBytes(a, 0, LEN(a)); IF b # NIL THEN out.WriteBytes(b^, 0, LEN(b)) END;
		ASSERT(out.Pos() = s.pos + s.len)
	END OutSeg;
	
	PROCEDURE DefAllocSec (VAR s: Section; len: INTEGER; flags: SET);
	BEGIN
		s.pos := B.AdAligned(availPos); s.addr := B.AdAligned(availAddr); ASSERT(s.pos - availPos = s.addr - availAddr);
		s.len := len; availPos := s.pos + len; availAddr := s.addr + len; s.flags := flags
	END DefAllocSec;
	

	PROCEDURE PositionSeg (VAR s: B.Segment; len: INTEGER; align: CHAR);
	BEGIN 
		IF align = pageAligned THEN s.pos := B.PageAligned(availPos) 
		ELSIF align = adAligned THEN s.pos := B.AdAligned(availPos) 
		ELSE HALT(20)
		END;
		availPos := s.pos + len; s.len := len
	END PositionSeg;
	
	PROCEDURE AllocSeg (VAR s: B.Segment; size: INTEGER; align: CHAR);
	BEGIN 
		IF align = pageAligned THEN s.addr := B.PageAligned(availAddr) 
		ELSIF align = adAligned THEN s.addr := B.AdAligned(availAddr) 
		ELSIF align = posAligned THEN s.addr := PosAligned(s.pos, availAddr)
		ELSE HALT(20)
		END;
		availAddr := s.addr + size; s.size := size
	END AllocSeg;

	PROCEDURE (elf: Formatter) AllocateMod- (m: B.Module);
	BEGIN
		(* allocate segments' virtual base addresses *)
		B.DefSeg(m.meta, PT_LOAD, pageSize, rwProt);
		PositionSeg(m.meta, B.AdAligned(m.ms) + m.imps * addrSize[version], pageAligned);	(* file:  *)
		AllocSeg(m.meta, B.AdAligned(m.ms) + (1 + m.imps + m.exts) * addrSize[version], pageAligned); 
		
		B.DefSeg(m.code, PT_LOAD, pageSize, rxProt);
		PositionSeg(m.code, m.cs, pageAligned);
		AllocSeg(m.code, m.cs, pageAligned);
		
		B.DefSeg(m.vars, PT_LOAD, pageSize, rwProt);
		PositionSeg(m.vars, m.ds, pageAligned);	(* in file, Vars holds DescBlk *)
		AllocSeg(m.vars, B.AdAligned(m.ds) + B.AdAligned(m.vs), pageAligned);	(* in memory, Vars holds  *)
	END AllocateMod;


	PROCEDURE OutExternals (VAR ext: Section);
		VAR i: INTEGER; tag: BYTE;
	BEGIN
		ext.pos := out.Pos();
		DefAllocSec(ext, B.extIndex * symEntSize[version], {SHF_ALLOC});
		SkipTo(ext.pos);
		out.WriteBytes(zero, 0, symEntSize[version]);	(* first entry has to be null (for the sake of Hash table) *)
		i := 1; WHILE i < B.extIndex DO
			OutInt(B.externals[i].nsoffs); ASSERT(B.externals[i].nsoffs # 0);
			CASE B.externals[i].type OF B.mProc: tag := 12H ELSE HALT(126) END;
			out.WriteByte(tag);
			out.WriteBytes(zero, 0, 19);
			INC(i)
		END;
		ASSERT((out.Pos() - ext.pos) MOD symEntSize[version] = 0)
	END OutExternals;
		
	PROCEDURE OutBindings (VAR bnd: Section);	 (** write out relocations for external symbols *)
		CONST R_X86_64_64 = 1;	(* relocation type *)
		VAR i, j, k: INTEGER; r: POINTER TO B.RelocArr;
	BEGIN
		bnd.pos := out.Pos();
		DefAllocSec(bnd, relEntSize[version] * (B.relPage * LEN(B.RelocArr) + B.relModulus), {SHF_ALLOC});
		SkipTo(bnd.pos);
		k := LEN(B.RelocArr);
		i := 0; WHILE i <= B.relPage DO 
			r := B.elfReloc[i]; ASSERT((r # NIL) OR (B.relPage = 0) & (B.relModulus = 0));
			IF i = B.relPage THEN k := B.relModulus END;
			j := 0; WHILE j < k DO
				IF m64 THEN
					OutLong(r[j].virtAddr); OutInt(R_X86_64_64); OutInt(r[j].symInd); OutLong(0)
				ELSE HALT(126)
				END;
				INC(j)
			END;
			INC(i)
		END;
		ASSERT((out.Pos() - bnd.pos) MOD relEntSize[version] = 0)
	END OutBindings;
	
	PROCEDURE OutDir (tag, val: INTEGER);
	BEGIN IF m64 THEN OutLong(tag); OutLong(val) ELSE HALT(126) END
	END OutDir;
	
	PROCEDURE OutDirectives (IN dir: B.Segment; IN str, ext, bnd: Section);	
	(** Output loader directives = ELF DYNAMIC segment *)
		CONST DT_NULL = 0; DT_NEEDED = 1; DT_STRTAB = 5; DT_SYMTAB = 6; 
			DT_REL = 17; DT_RELSZ = 18; DT_RELENT = 19; DT_STRSZ = 10; DT_SYMENT = 11;
			DT_RELA = 7; DT_RELASZ = 8; DT_RELAENT = 9; 
		VAR m: B.Module; blanks: INTEGER;
	BEGIN
		blanks := 1;
		ASSERT(out.Pos() = dir.pos);
		m := B.modules; WHILE m # NIL DO
			IF (m.name[0] = '$') & m.used THEN OutDir(DT_NEEDED, m.nameInd) END;
			m := m.link
		END;
		IF ~B.terminal THEN HALT(126) END;
		OutDir(DT_STRTAB, str.addr);
		IF B.extIndex > 1 THEN 
			OutDir(DT_SYMTAB, ext.addr); 
			OutDir(DT_SYMENT, symEntSize[version])
		ELSE INC(blanks, 2) 
		END;
		IF bnd.len > 0 THEN 
			OutDir(DT_RELA, bnd.addr);
			OutDir(DT_RELASZ, bnd.len);
			OutDir(DT_RELAENT, relEntSize[version])
		ELSE INC(blanks, 3)
		END;
		OutDir(DT_STRSZ, str.len);
		REPEAT OutDir(DT_NULL, 0); DEC(blanks) UNTIL blanks = 0
	END OutDirectives;
	
	PROCEDURE OutSegHdr (IN s: B.Segment); 
	BEGIN 
		OutInt(s.type); 
		IF m64 THEN OutInt(ORD(s.flags)) END;
		OutIntOrLong(s.pos); OutIntOrLong(s.addr); OutIntOrLong(s.addr);	 (* sysical addr, p_paddr *)
		OutIntOrLong(s.len); OutIntOrLong(s.size); 
		IF ~m64 THEN OutInt(ORD(s.flags)) END;
		OutIntOrLong(s.align)
	END OutSegHdr;
	
	PROCEDURE AssertPosAddr (s: B.Segment);
	BEGIN ASSERT(s.pos MOD pageSize = s.addr MOD pageSize)
	END AssertPosAddr;

	PROCEDURE OutPHT (VAR dir, interp: B.Segment; OUT phtPos: INTEGER);
		CONST PT_INTERP = 3;
		VAR meta, phdr: B.Segment; x: INTEGER; m: B.Module; dyn: BOOLEAN;
	BEGIN
		dyn := B.extIndex > 1;
		IF dyn THEN INC(segments, 4) ELSE INC(segments) END;
		B.DefSeg(phdr, PT_PHDR, addrSize[version], rProt);
		x := segments * phtEntSize[version];
		PositionSeg(phdr, x, adAligned); AllocSeg(phdr, x, adAligned); 
		SkipTo(phdr.pos); OutSegHdr(phdr);
		phtPos := phdr.pos;
		AssertPosAddr(phdr);
		IF dyn THEN B.DefSeg(interp, PT_INTERP, 1, rProt); OutSegHdr(interp); AssertPosAddr(interp) END;
		(* out module segment headers *)
		m := B.modules; WHILE m # NIL DO
			IF m.name[0] # '$' THEN OutSegHdr(m.meta); OutSegHdr(m.code); OutSegHdr(m.vars) END;
			AssertPosAddr(m.meta); AssertPosAddr(m.code); AssertPosAddr(m.vars);
			m := m.link
		END;
		IF dyn THEN
			B.DefSeg(meta, PT_LOAD, dir.align, rwProt);
			meta.pos := dir.pos; meta.addr := dir.addr; 
			meta.len := phdr.pos + phdr.len - meta.pos; meta.size := meta.len;
			dir.len := meta.len; dir.size := meta.size; 
			AssertPosAddr(meta);
			OutSegHdr(meta);
			OutSegHdr(dir)
		END
	END OutPHT;
	
	PROCEDURE OutStrings (VAR str, ext, bnd: Section; VAR dir, interp: B.Segment);
		VAR res, j, k, i, base: INTEGER; m: B.Module; n: ARRAY 512 OF SHORTCHAR; sh: SHORTCHAR;
	BEGIN
		base := out.Pos(); 
		out.WriteByte(0);	(* empty string at index 0 in string section *)
		(* names of external libraries *)
		m := B.modules; WHILE m # NIL DO
			IF (m.name[0] = '$') & (m.used) THEN m.nameInd := out.Pos() - base;
				Utf.StringToUtf8(m.name, n, res); ASSERT(res = 0);
				j := 0; k := 1; REPEAT sh := n[k]; n[j] := sh; INC(j); INC(k) UNTIL sh = 0X;
				OutSString(n)
			END;
			m := m.link
		END;
		(* external symbol names *)
		i := 1; WHILE i < B.extIndex DO
			B.SetExtNmPos(i, out.Pos() - base);
			Utf.StringToUtf8(B.externals[i].name, n, res); ASSERT(res = 0);
			OutSString(n);
			INC(i)
		END;
		(* section names *)
		IF ~B.terminal THEN
			str.nameInd := out.Pos() - base; OutSString(".dynstr");
			ext.nameInd := out.Pos() - base; OutSString(".dynsym");
			bnd.nameInd := out.Pos() - base; OutSString(".rela")
		END;
		str.pos := base; str.len := out.Pos() - str.pos;
		str.addr := dir.addr + dir.size;
		interp.pos := out.Pos(); OutSString(iinterpreter[version]); interp.len := out.Pos() - interp.pos;
		interp.addr := str.addr + (interp.pos - str.pos); interp.size := interp.len
	END OutStrings;
	
	PROCEDURE OutMeta (OUT phtPos: INTEGER); 	(* ElfMeta = DYNAMIC STRTAB SYMTAB REL  *)
		VAR x, end: INTEGER; dir, int: B.Segment; str, ext, bnd: Section;
	BEGIN
		B.DefSeg(dir, PT_DYNAMIC, pageSize, rwProt);
		x := dynEntSize[version] * ((B.extIndex - 1) + 8); 	(* extIndex - 1, because extIndex 0 is blank *)
		IF ~B.terminal THEN INC(x) (* for DT_HASH directive *) END;
		PositionSeg(dir, x, pageAligned);
		AllocSeg(dir, x, posAligned);
		AssertPosAddr(dir);
		SkipTo(dir.pos); SkipTo(dir.pos + dir.len);	(* placeholder to be filled out later *)
		OutStrings(str, ext, bnd, dir, int);	(* has to fill out addr! *)
		x := out.Pos() - availPos; INC(availPos, x); INC(availAddr, x);	(* accomodate strings *)
		OutExternals(ext);
		OutBindings(bnd);
		IF ~B.terminal THEN HALT(126)END;
		OutPHT(dir, int, phtPos);
		end := out.Pos();
		out.SetPos(dir.pos); OutDirectives(dir, str, ext, bnd); out.SetPos(end)
	END OutMeta;
	
	PROCEDURE OutHeader (phtPos: INTEGER);
		CONST ELFCLASS32 = 1X; ELFCLASS64 = 2X; ELFDATA2LSB = 1X; EV_CURRENT = 1X;
		CONST ET_REL = 1; ET_EXEC = 2; ET_DYN = 3; 
		VAR e_ident : ARRAY 16 OF SHORTCHAR; i: INTEGER;
	BEGIN
		ASSERT(B.entryMod # NIL, 20);
		i := 0; WHILE i < LEN(e_ident) DO e_ident[i] := 0X; INC(i) END;
		IF m64 THEN e_ident := 7FX + 'ELF'+ ELFCLASS64 + ELFDATA2LSB + EV_CURRENT
		ELSE e_ident := 7FX + 'ELF'+ ELFCLASS32 + ELFDATA2LSB + EV_CURRENT
		END;
		e_ident[7] := 0X; (*LB.opt.OSABI;*)
		e_ident[8] := 0X;
		out.WriteBytes(SYSTEM.THISARRAY(SYSTEM.ADR(e_ident[0]), 16), 0, 16); 
		IF B.terminal THEN OutShort(ET_EXEC) ELSE HALT(126) END;
		OutShort(machine[version]); 
		OutInt(1); (* version  *)
		OutIntOrLong(B.entryMod.code.addr); 
		OutIntOrLong(phtPos); 
		OutIntOrLong(shtOffs); 
		OutInt(0); (* flags  *)
		OutShort(elfHeaderSize[version]); 
		OutShort(phtEntSize[version]); 
		OutShort(segments); 
		OutShort(0); 
		OutShort(sections); 
		IF B.terminal THEN OutShort(0) ELSE HALT(126) END; 	(* SHT names table *)
		ASSERT(out.Pos() = elfHeaderSize[version], 101)
	END OutHeader;

	





	PROCEDURE (elf: Formatter) NewJob- (f: Files.File);
	BEGIN
		m64 := TRUE;
		out := f.NewWriter(out);
		availPos := elfHeaderSize[version]; availAddr := zeroBase[version] + availPos;
		segments := 0; sections := 0; 
		shtOffs := 0;
		out.SetPos(0); out.WriteBytes(zero, 0, elfHeaderSize[version]);	(* placeholder for the header *)
		out.WriteByte(-1); (* otherwise readelf complains segment size is beyond file size and doesn't show directives *)
	END NewJob;
	
	PROCEDURE (elf: Formatter) WriteMod- (m: B.Module; meta, imps, code, desc: POINTER TO ARRAY OF BYTE);
	BEGIN
		OutSeg(m.meta, meta, imps); 
		
		OutSeg(m.code, code, NIL); OutSeg(m.vars, desc, NIL);
		
		INC(segments, 3);	(* may change if some segments are allowed to be missing *)
	END WriteMod;
	
	PROCEDURE (elf: Formatter) CloseJob-;
		VAR phtPos: INTEGER;
	BEGIN
		OutMeta(phtPos); out.WriteByte(-1); out.SetPos(0); OutHeader(phtPos);
	END CloseJob;
	
	PROCEDURE Install*; BEGIN B.elfFmt := formatter END Install;

	PROCEDURE InitMod;
		CONST EM_386 = 3; EM_X86_64 = 62;
		VAR i: INTEGER;
	BEGIN
		i := 0; WHILE i < LEN(zero) DO zero[i] := 0; INC(i) END;
		elfHeaderSize[v32] := 52; elfHeaderSize[v64] := 64;
		(*phtOffset[v32] := 52; phtOffset[v64] := 64;*)
		phtEntSize[v32] := 32; phtEntSize[v64] := 56; 
		shtEntSize[v32] := 32; shtEntSize[v64] := 64; 
		(*phtSize[v32] := phtEntSize[v32] * segmentCount; phtSize[v64] := phtEntrySize[v64] * segmentCount; *)
		zeroBase[v32] := 8048000H; zeroBase[v64] := 404000H;
		ASSERT((zeroBase[v32] + elfHeaderSize[v32]) MOD pageSize = elfHeaderSize[v32] MOD pageSize);
		ASSERT((zeroBase[v64] + elfHeaderSize[v64]) MOD pageSize = elfHeaderSize[v64] MOD pageSize);
		machine[v32] := EM_386; machine[v64] := EM_X86_64;
		relEntSize[v32] := 8; relEntSize[v64] := 24;
		addrSize[v32] := 4; addrSize[v64] := 8;
		symEntSize[v32] := 16; symEntSize[v64] := 24;
		dynEntSize[v32] := 8; dynEntSize[v64] := 16;
		
		iinterpreter[v32] := linuxInterpreter; iinterpreter[v64] := linuxInterpreter64;
		NEW(formatter)
	END InitMod;

BEGIN InitMod
END HrElfFmt.
