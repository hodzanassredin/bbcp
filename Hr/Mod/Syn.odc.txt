		VAR res: BOOLEAN;

	END Equals;
	
	PROCEDURE (s: StringLit) Equals* (q: T.Literal): BOOLEAN, NEW;
		VAR res: BOOLEAN;
	BEGIN 
		WITH q: T.StringLit DO res := (s.type.what = q.type.what) & (s.len = q.len) & (s.str^ = q.str^) 
		ELSE res := FALSE 
		END;
		RETURN res
	END Equals;
	
	PROCEDURE AllocSysV (p: Par; VAR intAvail, realAvail: INTEGER);
		CONST maxReal = 0; maxInt = 6; 
		VAR w, pass: INTEGER; kind: SET;
	BEGIN
		w := p.type.what; kind := p.flags * T.kindMask;
		IF (w = T.fArray) & (kind = T.valK) THEN pass := banned
		ELSE
			IF w = T.fArray THEN w := T.fSysPtr END;
			IF (w IN T.reals) & (kind = T.valK) THEN HALT(126); 	(* allocate floating point reg *)
				pass := callingConventions[sysvCall].realRegs[realAvail]; 
				IF realAvail < callingConventions[sysvCall].maxRealParReg THEN INC(realAvail) END
			ELSIF w IN T.basics + T.pointers THEN	 (* allocate integer reg *)
				pass := callingConventions[sysvCall].intParRegs[intAvail]; 
				IF intAvail < callingConventions[sysvCall].maxIntParReg THEN INC(intAvail) END
			ELSE HALT(126)
			END
		END;
		p.pass := pass
	END AllocSysV;
	
	PROCEDURE AllocMs (p: Par; VAR intAvail, realAvail: INTEGER);
		CONST maxReal = 0; maxInt = 6; 
		VAR w, pass: INTEGER; kind: SET;
	BEGIN
		w := p.type.what; kind := p.flags * T.kindMask;
		IF (w = T.fArray) & (kind = T.valK) THEN pass := banned
		ELSE
			IF w = T.fArray THEN w := T.fSysPtr END;
			IF (w IN T.reals) & (kind = T.valK) THEN  	(* allocate floating point reg *)
				pass := callingConventions[ms64Call].realRegs[realAvail];
				IF pass # stk THEN INC(realAvail) END
			ELSIF w IN T.basics + T.pointers THEN	 (* allocate integer reg *)
				pass := callingConventions[ms64Call].intParRegs[intAvail]; 
				IF intAvail < callingConventions[ms64Call].maxIntParReg THEN INC(intAvail) END
			ELSE HALT(126)
			END
		END;
		p.pass := pass
	END AllocMs;
	
	PROCEDURE AllocPar* (conv: INTEGER; p: Par; VAR intAvail, realAvail: INTEGER);
	BEGIN
		CASE conv OF
		| sysvCall: AllocSysV(p, intAvail, realAvail)
		| ms64Call: AllocMs(p, intAvail, realAvail)
		| hrCall: p.pass := stk
		ELSE HALT(126)
		END
	END AllocPar;
