MODULE ObxProbe10;

	IMPORT Kernel, Console, Strings;

	TYPE FR = POINTER TO RECORD x: INTEGER END;

	VAR keep: ARRAY 64 OF FR; finCount: INTEGER;

	PROCEDURE (f: FR) FINALIZE;
	BEGIN
		INC(finCount)
	END FINALIZE;

	PROCEDURE T*;
		VAR i, s: INTEGER; f: FR; t: ARRAY 32 OF CHAR;
	BEGIN
		finCount := 0;
		FOR i := 0 TO 1999 DO
			NEW(f); f.x := i;
			IF i MOD 32 = 0 THEN keep[i DIV 32] := f END
		END;
		Kernel.Collect; Kernel.Collect;
		s := 0;
		FOR i := 0 TO 63 DO
			IF keep[i] # NIL THEN
				IF keep[i].x # i * 32 THEN Console.WriteStr("P10 CORRUPT at "); Strings.IntToString(i, t); Console.WriteStr(t); Console.WriteLn; RETURN END;
				INC(s)
			END
		END;
		(* вторая волна: переиспользование памяти после collect *)
		FOR i := 0 TO 1999 DO NEW(f); f.x := i END;
		Kernel.Collect;
		FOR i := 0 TO 63 DO
			IF (keep[i] # NIL) & (keep[i].x # i * 32) THEN Console.WriteStr("P10 CORRUPT2"); Console.WriteLn; RETURN END
		END;
		Console.WriteStr("P10 OK fin="); Strings.IntToString(finCount, t); Console.WriteStr(t); Console.WriteLn
	END T;

END ObxProbe10.
