	PROCEDURE ReadInt (r: Files.Reader; VAR i: INTEGER);
		VAR b: BYTE; x: INTEGER;
	BEGIN
		r.ReadByte(b); x := b MOD 256;
		r.ReadByte(b); x := x + 100H * (b MOD 256);
		r.ReadByte(b); x := x + 10000H * (b MOD 256);
		r.ReadByte(b); i := x + 1000000H * b;
	END ReadInt;
	
	PROCEDURE ReadAsChar (rd: Files.Reader; VAR ch: CHAR);
		VAR b: BYTE;
	BEGIN rd.ReadByte(b); ch := CHR(b - ASH(b, -7) * 100H)
	END ReadAsChar;

	PROCEDURE ReadNum (r: Files.Reader; VAR i: INTEGER);
		VAR b: BYTE; s, y: INTEGER;
	BEGIN
		s := 0; y := 0; r.ReadByte(b);
		WHILE b < 0 DO INC(y, ASH(b + 128, s)); INC(s, 7); r.ReadByte(b) END;
		i := ASH((b + 64) MOD 128 - 64, s) + y;
	END ReadNum;
	
	PROCEDURE Align (rd: Files.Reader; bits: INTEGER);	(** Align rd.Pos() to 2^bits boundary *)
		VAR buf: ARRAY 32 OF BYTE; x: INTEGER;
	BEGIN ASSERT((0 < bits) & (bits <= 5), 20);
		x := ASH(1, bits);
		x := (x - 1) DIV x * x; rd.ReadBytes(buf, 0, x);
		WHILE x > 0 DO DEC(x); ASSERT(buf[x] = 0) END
	END Align;
	
	PROCEDURE ReadName (rd: Files.Reader; OUT name: ARRAY OF CHAR);
		VAR b: BYTE; i: INTEGER;
	BEGIN
		i := 0; REPEAT rd.ReadByte(b); name[i] := CHR(b + b DIV 80H * 100H); INC(i) UNTIL b = 0
	END ReadName;
	
	PROCEDURE Type (rd: Files.Reader; VAR fmt: TextMappers.Formatter; OUT any: BOOLEAN);
		VAR x: INTEGER;
	BEGIN 
		ReadInt(rd, x);
		any := x # 0;
		CASE x OF
		| 0: 
		| mBool: fmt.WriteString("BOOLEAN")
		| mChar8: fmt.WriteString("SHORTCHAR")
		| mChar16: fmt.WriteString("CHAR")
		| mInt8:  fmt.WriteString("BYTE")
		| mInt16: fmt.WriteString("SHORTINT")
		| mInt32: fmt.WriteString("INTEGER")
		| mInt64: fmt.WriteString("LONGINT")
		| mSet: fmt.WriteString("SET")
		| mAnyRec: fmt.WriteString("ANYREC")
		| mAnyPtr: fmt.WriteString("ANYPTR")
		| mSysPtr: fmt.WriteString("SYSTEM.PTR")
		| mReal32: fmt.WriteString("SHORTREAL")
		| mReal64:  fmt.WriteString("REAL")
		ELSE
		END
	END Type;
		
	
	PROCEDURE Signatures (rd: Files.Reader; VAR fmt: TextMappers.Formatter);
		VAR _, any: BOOLEAN; parId, n: INTEGER;
	BEGIN n := bufText.Length(); IF n > 0 THEN bufText.Delete(0, n) END;
		Type(rd, bfmt, any);
		ReadInt(rd, n);
		IF n > 0 THEN fmt.WriteChar('(');
			REPEAT ReadInt(rd, parId);
				CASE parId MOD 0FFH OF
				| mValue: 
				| mInPar: fmt.WriteString("IN ")
				| mVarPar: fmt.WriteString("VAR ")
				| mOutPar: fmt.WriteString("OUT ")
				ELSE HALT(20)
				END;
				fmt.WriteChar('^'); fmt.WriteInt(parId DIV 100H);
				fmt.WriteString(": "); Type(rd, fmt, _); fmt.WriteString("; ");
				DEC(n)
			UNTIL n = 0;
			fmt.WriteChar(')')
		END;
		IF any THEN fmt.WriteString(": "); fmt.rider.Base().Append(bufText); 
			fmt.SetPos(fmt.rider.Base().Length()); fmt.WriteChar(';'); fmt.WriteLn
		END
	END Signatures;

	PROCEDURE RLink (rd: Files.Reader);
		VAR x: INTEGER;
	BEGIN
		ReadNum(rd, x);
		WHILE x # 0 DO ReadNum(rd, x); ReadNum(rd, x) END
	END RLink;

	PROCEDURE GetTypeName (sym: Symbolist; disp: LONGINT; OUT name: Name);
			(*skip 12 bytes, read 4-byte id, decode name offset, get name from names *)
		TYPE Desc = POINTER TO RECORD size: INTEGER; modRef: LONGINT; id: INTEGER END;
		VAR desc: Desc; _, addr, j: INTEGER; utf: ARRAY 256 OF SHORTCHAR; 
			sh: SHORTCHAR;
	BEGIN
		desc := SYSTEM.VAL(Desc, SYSTEM.ADR(sym.desc[disp]));
		addr := sym.namesBlk + desc.id DIV 100H; 
		j := 0;
		REPEAT 
			SYSTEM.GET(addr, sh); utf[j] := sh; INC(addr); INC(j) 
		UNTIL sh = 0X;
		Utf.Utf8ToString(utf, name, _)
	END GetTypeName;
		
	PROCEDURE WriteLocVars (VAR fmt: TextMappers.Formatter; VAR vs: HrOcf.VarScanner; sym: Symbolist);
		VAR descBase: ADDRESS; name: Name; n: INTEGER;
	BEGIN
		descBase := HrOcf.LONGADDR(SYSTEM.ADR(sym.desc[0]));
		REPEAT vs.ScanVar;
			IF ~vs.eos THEN
				fmt.WriteTab; 
				CASE vs.mode OF HrOcf.varpartg: fmt.WriteString("VAR ") 
				| HrOcf.valpartg: fmt.WriteString("VAL ")
				| HrOcf.locvartg: fmt.WriteString("LOC ")
				ELSE HALT(20)
				END;
				fmt.WriteString(vs.name); fmt.WriteString(': '); 
				IF vs.type = 10H THEN 
					IF vs.typeaddr < 6666666600000000L THEN 
						GetTypeName(sym, vs.typeaddr - descBase, name);
						fmt.WriteString(name);
						fmt.WriteString(" (DESC:"); fmt.WriteInt(vs.typeaddr - descBase); fmt.WriteChar(")")
					ELSIF vs.typeaddr < 7777777700000000L THEN
						fmt.WriteString(sym.shadows[vs.typeaddr MOD 10000000H])
					ELSE n := sym.proxies[vs.typeaddr MOD 1000000H];
						fmt.WriteString(sym.imports[n]); fmt.WriteChar('.'); fmt.WriteString(sym.names[n])
					END
				ELSE fmt.WriteInt(vs.type)
				END;
				fmt.WriteString(" @"); fmt.WriteInt(vs.offs);
				fmt.WriteLn
			END
		UNTIL vs.eos
	END WriteLocVars;
	
	PROCEDURE DecodeRefs (VAR rs: HrOcf.RefScanner; VAR fmt: TextMappers.Formatter; sym: Symbolist);
		VAR p, i: INTEGER; g, f: StdFolds.Fold; 
	BEGIN
		OpenFold(fmt, StdFolds.expanded, '', g); p := fmt.rider.Pos();
		i := 0; REPEAT rs.ScanProc;
			IF ~rs.eos THEN fmt.WriteString("PROCEDURE "); fmt.WriteString(rs.name); 
				fmt.WriteChar(' '); fmt.WriteInt(rs.endpos);
				OpenFold(fmt, StdFolds.expanded, srcRefsLabel, f);
				REPEAT rs.src.ScanSourceRef;
					IF ~rs.src.eos THEN fmt.WriteInt(rs.src.da); fmt.WriteChar("↔"); fmt.WriteInt(rs.src.dt); 
					fmt.WriteTab 
					ELSE fmt.WriteLn
					END
				UNTIL rs.src.eos;
				CloseFold(fmt, f);
				fmt.WriteLn; 
				WriteLocVars(fmt, rs.var, sym)
			END;
			INC(i)
		UNTIL rs.eos
		;fmt.WriteString("Total procedures: "); fmt.WriteInt(MAX(0, i - 1)); fmt.WriteLn;
		CloseFold(fmt, g);
		g.Flip;
		fmt.rider.SetPos(p);
		Bold(fmt, TRUE);
		fmt.WriteString("SOURCE AND VAR REFERENCES");
		Bold(fmt, FALSE);
		fmt.rider.SetPos(fmt.rider.Pos() + 1)
	END DecodeRefs;
	
	PROCEDURE WriteMeta (IN a: ARRAY OF BYTE; VAR fmt: TextMappers.Formatter);
	(** Write a as a two-column hex/char dump *)
		VAR j, i: INTEGER; ch: CHAR; k: BYTE;
	BEGIN
		fmt.WriteView(HrOcfTargets.NewTarget(metaTgt));
		i := 0; WHILE i < LEN(a) DO
			IF i MOD 16 = 0 THEN 
				fmt.WriteChar('['); fmt.WriteInt(i); fmt.WriteChar(']'); fmt.WriteTab
			END;
			fmt.WriteIntForm(a[i], TextMappers.hexadecimal, 2, '0', ~TextMappers.showBase);
			IF (i MOD 16 = 15) OR (i = LEN(a) - 1) THEN
				fmt.WriteTab;
				j := i - i MOD 16; 
				WHILE j <= i DO k := a[j];
					ch := CHR(k - k DIV 80H * 100H); 
					IF (ch < ' ') OR (ch = 7FX) THEN ch := '‧' END;
					fmt.WriteChar(ch);
					INC(j)
				END;
				fmt.WriteLn 
			ELSIF i MOD 4 = 3 THEN fmt.WriteTab
			ELSE fmt.WriteChar(' ')
			END;
			INC(i)
		END
	END WriteMeta;
	
	PROCEDURE InsRel (VAR rel: ARRAY OF Rel; adr: INTEGER);
		VAR a, i, j, free: INTEGER;
	BEGIN
		Log.Int(adr); Log.Ln;
		i := 1; j := 0; free := rel[0].left; 
		REPEAT a := rel[i].adr;
			IF adr < a THEN j := rel[i].left; IF j = -1 THEN rel[i].left := free END
			ELSIF a < adr THEN j := rel[i].right; IF j = -1 THEN rel[i].right := free END
			ELSE HALT(20)	(* same address cannot be relocated twice => no duplicates in tree *)
			END;
			IF j = -1 THEN 
				rel[free].adr := adr; rel[free].left := -1; rel[free].right := -1; INC(rel[0].left)
			ELSE i := j
			END
		UNTIL j = -1
			
	END InsRel;

	PROCEDURE (VAR us: UseScanner) Resolve (OUT x: ADDRESS);
		CONST none = -1;
		VAR n: INTEGER;
	BEGIN
		CASE us.type OF 
		| HrOcf.mProc, HrOcf.mType:
			us.fmt.WriteLn; us.fmt.WriteTab; us.fmt.WriteChar('.'); 
			us.fmt.WriteString(us.name); us.fmt.WriteString(" #"); us.fmt.WriteInt(us.fprint);
			IF us.chainHead # none THEN 
				n := us.chainHead DIV 8;
				us.imports[n] := us.name$; us.proxies[n] := us.imp;
				us.fmt.WriteTab; us.fmt.WriteString("=> PROXY["); us.fmt.WriteInt(n); us.fmt.WriteChar(']');
				IF us.type = HrOcf.mType THEN x := n + 7777777700000000L ELSE x := 0 END
			ELSE 
				us.shadows[us.shadowCnt] := us.provider + "." + us.name; 
				IF us.type = HrOcf.mType THEN x := us.shadowCnt + 6666666600000000L ELSE x := 0 END;
				INC(us.shadowCnt)
			END
		ELSE HALT(126)
		END
	END Resolve;

	PROCEDURE AdAligned (x: INTEGER): INTEGER;
	BEGIN RETURN (x + addrSize[version] - 1) DIV addrSize[version] * addrSize[version]
	END AdAligned;
	
	PROCEDURE PageAligned (x: INTEGER): INTEGER;
	BEGIN RETURN (x + pageSize - 1) DIV pageSize * pageSize
	END PageAligned;
	
	PROCEDURE AllocateBlocks (ms, ds, cs, vs, ps: INTEGER; OUT mo, po, do, vo: INTEGER);
	(** Keep in sync with HrE.AllocateBlocks!!! mo, po, do, vo: block offsets wrt code block base. *)
	BEGIN 
		do := PageAligned(cs);
		vo := AdAligned(do + ds + vs) - vs;
		mo := -PageAligned(AdAligned(ms) + ps);
		po := mo + AdAligned(ms)
	END AllocateBlocks;
	
	PROCEDURE InitRel* (VAR rel: ARRAY OF Rel);
		(* In a tree/array of Rel, [0].left is the index of first free element, [1].adr = 0 is the root *)
	BEGIN rel[0].left := 2; rel[1].adr := 0; rel[1].left := -1; rel[1].right := -1
	END InitRel;

	PROCEDURE ReadImports (os: HrOcf.OcfScanner; OUT imports: POINTER TO ARRAY OF Name);
	(** Read in imported modules list*)
		VAR _, i: INTEGER;
	BEGIN
		IF os.nofImports > 0 THEN
			NEW(imports, os.nofImports);
			os.rider.SetPos(os.importsPos);
			i := 0; WHILE i < os.nofImports DO HrOcf.ReadName(os.rider, imports[i], _); INC(i) END
		END
	END ReadImports;

	PROCEDURE Decode* (ocf: Files.Reader;  src: TextModels.Reader; fmt: POINTER TO TextMappers.Formatter );
	(** Read OCF from rd, prepare it for decoding: allocate and relocate all structures except VarBlk *)
		CONST litTag = 6F4F4346H; bigTag = 46434F6FH;
		VAR 
			res, (*processor, tag, *)
			hs, (*ms, ds, cs, vs,*) ps, mo, po, do, vo, mb, cb, db, vb, pb,
				(* rewrite: offsets are with respect to code base; bases are wrt mem *)
			(*extsProx, impsProx, nofImp*)
			i, cnt	: INTEGER; 
			name: Name; ch: CHAR;
			buf: ARRAY 256 OF CHAR; (*imps, names: POINTER TO ARRAY OF Name;*)
			(*proxies: POINTER TO ARRAY OF INTEGER;*)
			x: INTEGER; tmp: LONGINT;
			uses: TextMappers.Formatter; us: UseScanner;
			_: INTEGER; __: Kernel.Utf8Name;
			os: HrOcf.OcfScanner; bs: HrOcf.BookmarkScanner; es: HrOcf.ExpScanner;
			(*mem, *)(*meta, code, desc: POINTER TO ARRAY OF BYTE;*)
	BEGIN
		os.ConnectTo(ocf);
		IF os.res = HrOcf.OK THEN
			IF os.processor = HrOcf.amd64 THEN version := 1;
				ps := (os.exts + os.imps) * 8;
				AllocateBlocks(os.ms, os.ds, os.cs, os.vs, ps, mo, po, do, vo);
				(*NEW(mem, vo - mo); *)mb := 0;
				cb := -mo + mb; db := (*cb +*) do; vb := cb + vo; pb := cb + po;
				NEW(os.meta, os.ms); ocf.SetPos(os.metaPos); ocf.ReadBytes(os.meta, 0, os.ms);
				NEW(os.desc, os.ds); ocf.SetPos(os.descPos); ocf.ReadBytes(os.desc, 0, os.ds);
				NEW(os.code, os.cs); ocf.SetPos(os.codePos); ocf.ReadBytes(os.code, 0, os.cs);
				os.vars := SYSTEM.VAL(SYSTEM.PTR, 78190528H);
				NEW(sym.rs); os.ConnectRefScanner(sym.rs)
			ELSE HALT(126)
			END;
			(*fmt.WriteString("Module memory map:"); fmt.WriteLn;*) fmt.WriteTab;
			fmt.WriteChar('['); fmt.WriteInt(mo); 
			fmt.WriteString(']‧‧Meta('); fmt.WriteInt(os.ms); fmt.WriteString(')‧‧['); 
			fmt.WriteInt(mo + os.ms); fmt.WriteString(']‧‧Proxies('); fmt.WriteInt(ps);
			fmt.WriteString(')‧‧['); fmt.WriteInt(po + ps); fmt.WriteString(']'); fmt.WriteLn; fmt.WriteTab;
			fmt.WriteString('[0]‧‧Code‧‧['); fmt.WriteInt(os.cs); fmt.WriteChar(']'); fmt.WriteLn; fmt.WriteTab;
			fmt.WriteChar('['); fmt.WriteInt(do); fmt.WriteString(']‧‧Desc('); fmt.WriteInt(os.ds);
			fmt.WriteString(')‧‧['); fmt.WriteInt(do + os.ds); fmt.WriteChar(']');
			IF do + os.ds # vo THEN fmt.WriteTab; fmt.WriteChar('['); fmt.WriteInt(vo); fmt.WriteChar(']') END;
			fmt.WriteString('‧‧Vars('); fmt.WriteInt(os.vs); fmt.WriteString(')‧‧['); fmt.WriteInt(vo + os.vs);
			fmt.WriteChar(']'); fmt.WriteLn;
			fmt.WriteLn;
			
			ReadImports(os, sym.imports);
			IF sym.imports # NIL THEN fmt.WriteString("IMPORT ");
				i := 0; WHILE i < LEN(sym.imports) DO IF i > 0 THEN fmt.WriteString(", ") END;
					fmt.WriteString(sym.imports[i]);
					INC(i)
				END;
				fmt.WriteChar(";"); fmt.WriteLn;
			END;
			cnt := os.imps + os.exts; 
			IF os.newRec >= 0 THEN DEC(cnt) END; IF os.newArr >= 0 THEN DEC(cnt) END;
			ASSERT(cnt >= 0);
			IF cnt > 0 THEN
				NEW(sym.names, cnt); NEW(sym.proxies, cnt)
			END;
			IF os.processor = HrOcf.amd64 THEN 
				sym.newRec := os.newRec; sym.newArr := os.newArr;
				fmt.WriteString("NEW(rec) = "); fmt.WriteInt(os.newRec);
				fmt.WriteString(" NEW(arr) = "); fmt.WriteInt(os.newArr);
				fmt.WriteLn
			END;
			
			(*ocf.SetPos(hs + ms + ds + cs);*)	(* at fixups *)
			NEW(sym.rel, (ocf.Base().Length() - ocf.Pos()) DIV 3);	(* wild guess *)
			InitRel(sym.rel);
			(* skip NewRec, then NewArr fixup heads *)

	PROCEDURE (sym: Symbolist) WriteDisp (disp: LONGINT);
		VAR mode, _: SHORTCHAR; __: Kernel.Type; ___, ref, adr: INTEGER; 
			utf: Kernel.Utf8Name; name: Name;
			l, r: StdLinks.Link; buf: ARRAY 16 OF CHAR; suppress: BOOLEAN;
	BEGIN
		suppress := FALSE;
		IF (sym.lb <= disp) & (disp < sym.le) THEN sym.out.WriteString("CONST:"); DEC(disp, sym.lb);
			Strings.IntToString(disp, buf);
			MakeLinks("HrOcfExplorer.JumpToConst(" + buf + ")", l, r)
		ELSIF (sym.pb <= disp) & (disp < sym.pe) THEN sym.out.WriteString("PROXY:"); DEC(disp, sym.pb);
			IF disp = sym.newRec THEN sym.out.WriteString("NEW(rec)")
			ELSIF disp = sym.newArr THEN sym.out.WriteString("NEW(arr)")
			ELSE (*IF sym.names # NIL THEN*) sym.out.WriteString(sym.names[disp DIV 8]) (*END;*)
			END;
			sym.out.WriteChar(":")
		ELSIF (sym.cb <= disp) & (disp < sym.ce) THEN (*DEC(disp, sym.cb); sym.cb=0*)
			sym.rs.Reset;
			REPEAT sym.rs.ScanProc UNTIL sym.rs.eos OR (sym.rs.endpos >= disp);
			IF ~sym.rs.eos & (sym.rs.endpos = disp) THEN suppress := TRUE;
				sym.rs.ScanProc; 
				Bold(sym.out, TRUE); sym.out.WriteString(sym.rs.name); Bold(sym.out, FALSE)
			ELSE sym.out.WriteString('CODE:')
			END
		ELSIF (sym.vb <= disp) & (disp < sym.ve) THEN sym.out.WriteString("VAR:"); DEC(disp, sym.vb);
			ref := sym.globals; 
			REPEAT Kernel.GetRefVar(ref, mode, _, __, adr, utf) UNTIL (mode # 1X) OR (disp = adr);
			IF mode = 1X THEN 
				Utf.Utf8ToString(utf, name, ___); sym.out.WriteString(name); sym.out.WriteChar(":") 
			END
		ELSIF (sym.db <= disp) & (disp < sym.de) THEN sym.out.WriteString("DESC:"); DEC(disp, sym.db);
			GetTypeName(sym, disp, name); sym.out.WriteString(name); sym.out.WriteChar(":")
		ELSE sym.out.WriteString("UNKNWN:") (* this works for Jcc <byte>. Should a different proc be called? *)
		END;
		IF l # NIL THEN sym.out.WriteView(l); sym.out.rider.SetAttr(sym.link) END;
		IF ~suppress THEN sym.out.WriteInt(disp) END;
		IF r # NIL THEN sym.out.rider.SetAttr(sym.default); sym.out.WriteView(r) END
	END WriteDisp;

	PROCEDURE (sym: Symbolist) WriteDisp2 (adr: INTEGER; disp: LONGINT);
	BEGIN
		IF sym.IsReloc(adr) THEN
			IF (sym.lb <= disp) & (disp < sym.le) THEN sym.out.WriteString("CONST:"); DEC(disp, sym.le)
			ELSIF (sym.pb <= disp) & (disp < sym.pe) THEN sym.out.WriteString("PROXY:"); DEC(disp, sym.pb)
			ELSIF (sym.cb <= disp) & (disp < sym.ce) THEN sym.out.WriteString("CODE:"); DEC(disp, sym.cb)
			ELSIF (sym.vb <= disp) & (disp < sym.ve) THEN sym.out.WriteString("VAR:"); DEC(disp, sym.vb)
			ELSE sym.out.WriteString("U:")
			END
		END;
		sym.out.WriteInt(disp);
	END WriteDisp2;

	PROCEDURE (sym: Symbolist) IsReloc (adr: INTEGER): BOOLEAN;
		VAR i, a: INTEGER;
	BEGIN i := 1;
		REPEAT a := sym.rel[i].adr; 
			IF adr < a THEN i := sym.rel[i].left ELSIF a < adr THEN i := sym.rel[i].right END
		UNTIL (i = -1) OR (a = adr);
		RETURN i # -1
	END IsReloc;

	PROCEDURE WriteSrcLink (sym: Symbolist; from, to: INTEGER; prologue: BOOLEAN);															(* based on DevDecoder486.WriteSrcLink *)
		VAR ch: CHAR; left, right: StdLinks.Link; out: POINTER TO TextMappers.Formatter;
			src: TextModels.Reader;
			fromStr, toStr: ARRAY 20 OF CHAR;
	BEGIN
		out := sym.out; src := sym.src;
		out.rider.SetAttr(sym.link);
		IF insLinks THEN
			Strings.IntToString(from, fromStr);
			Strings.IntToString(to, toStr);
			left := StdLinks.dir.NewLink("DevDecoder386.ShowSrc('" + sym.mod + "', "
				+ fromStr + ", " + toStr + ")");
			right := StdLinks.dir.NewLink("");
			out.WriteView(left)
		END;
		IF src # NIL THEN
			src.SetPos(from);
			WHILE src.Pos() < to DO
				src.ReadChar(ch);
				IF ch = 09X THEN out.WriteString("     ") (* a TAB would use the ruler's tab positions *)
				ELSIF ch = 0DX THEN
					WHILE (ch <= " ") & (src.Pos() < to) DO src.ReadChar(ch) END; (* trim left *)
					IF ch > " " THEN out.WriteLn; out.WriteChar(ch) END
				ELSIF ch >= " " THEN out.WriteChar(ch)
				END
			END
		ELSIF prologue THEN out.WriteString("BEGIN")
		ELSE out.WriteString("END")
		END;
		IF insLinks THEN out.WriteView(right) END;
		out.rider.SetAttr(sym.default);
		out.WriteLn
	END WriteSrcLink;
	
	PROCEDURE (sym: Symbolist) NextOp (pc: INTEGER);
		VAR prologue: BOOLEAN; from: INTEGER;
	BEGIN 
		IF ~sym.refs.eos THEN
			prologue := pc = sym.refs.endpos;
			IF prologue THEN 
				sym.out.rider.SetAttr(sym.signature);
				sym.refs.ScanProc; 
				IF ~sym.refs.eos THEN
					sym.out.WriteLn;
					sym.out.WriteString("PROCEDURE "); sym.out.WriteString(sym.refs.name);
					sym.out.rider.SetAttr(sym.link);
					sym.out.WriteLn; 
					WriteLocVars(sym.out, sym.refs.var, sym);
					sym.out.WriteString("BEGIN"); sym.out.WriteLn; 
					sym.out.rider.SetAttr(sym.default);
					IF pc # 0 THEN sym.refs.src.ScanSourceRef; INC(sym.next.pos, sym.refs.src.dt)	(* skip *) END
				END
			END;
			IF pc = sym.next.pc THEN
				IF ~sym.refs.src.eos THEN sym.refs.src.ScanSourceRef END;
				IF sym.refs.src.eos THEN 
					sym.next.pc := sym.refs.endpos;
					sym.out.rider.SetAttr(sym.signature);
					sym.out.WriteString("END"); sym.out.WriteLn;
					sym.out.rider.SetAttr(sym.default)
				ELSE INC(sym.next.pc, sym.refs.src.da); 
					from := sym.next.pos; INC(sym.next.pos, sym.refs.src.dt);
					IF ~prologue THEN WriteSrcLink(sym, from, sym.next.pos, TRUE) END
				END
			ELSIF pc > sym.next.pc THEN HALT(20)
			END
		END
	END NextOp;
		
	PROCEDURE AdaptAttr* (wr: TextModels.Writer);
		(** Change wr's attributes to those at wr.Pos(),
			if available, otherwise to system defaults *)

	PROCEDURE JumpToConst* (addr: INTEGER);
	(** Jump to specified addr within Meta block *)
		VAR rd: TextModels.Reader; buf: ARRAY 16 OF CHAR; done: BOOLEAN; line, pos: INTEGER;
			ch: CHAR; v: Views.View; fmt: TextMappers.Formatter; default: TextModels.Attributes;
			backtag: INTEGER; op: Stores.Operation; txt: TextModels.Model; tv: TextViews.View;
	BEGIN done := FALSE;
		IF (StdLinks.par # NIL) & (StdLinks.par.context IS TextModels.Context) THEN
			rd := StdLinks.par.context(TextModels.Context).ThisModel().NewReader(NIL);
			REPEAT rd.ReadView(v) 
			UNTIL (v = NIL) OR (v IS HrOcfTargets.Target) & (v(HrOcfTargets.Target).tag = metaTgt);
			IF v # NIL THEN 
				line := addr DIV 16; pos := v.context(TextModels.Context).Pos() + 1;
				rd.SetPos(pos);
				(* skip line lines *)
				ch := "!";
				WHILE (line > 0) & (ch # 0X) DO
					rd.ReadChar(ch); 
					IF ch = TextModels.line THEN DEC(line) END
				END;
				IF ch # 0X THEN
					Strings.IntToString(addr DIV 16 * 16, buf);
					addr := addr MOD 16;
					pos := LEN(buf$) + 3 + addr*3(*each byte is 2 hex digits plus space/tab*) + rd.Pos();
					txt := rd.Base();
					IF pos < txt.Length() THEN 
						Models.BeginScript(txt, "Insert targets", op);
						fmt.ConnectTo(txt); 
						fmt.rider.SetPos(pos); AdaptAttr(fmt.rider); default := fmt.rider.attr;
						backtag := SHORT(ENTIER(ObxRandom.Uniform() * (MAX(INTEGER) - 100)) + 100);
						Strings.IntToString(backtag, buf);
						fmt.WriteView(StdLinks.dir.NewLink("HrOcfExplorer.JumpBack(" + buf + ")"));
						fmt.rider.SetAttr(TextModels.NewColor(default, Ports.blue));
						fmt.WriteChar('◆');
						fmt.rider.SetAttr(default);
						fmt.WriteView(StdLinks.dir.NewLink(''));
						(* insert the jump back target *)
						fmt.SetPos(StdLinks.par.context(TextModels.Context).Pos());
						AdaptAttr(fmt.rider);
						fmt.WriteView(HrOcfTargets.NewTarget(backtag));
						Models.EndScript(txt, op);
						tv := TextViews.Focus();
						IF tv.ThisModel() = txt THEN tv.ShowRange(pos, pos, TextViews.focusOnly)
						ELSE TextViews.ShowRange(txt, pos, pos, TextViews.focusOnly)
						END;
						done := TRUE
					END
				END
			END;
			IF ~done THEN Log.String("HrOcfExplorer.JumpToConst: cannot find the position"); Log.Ln END
		ELSE Log.String("HrOcfExplorer.JumpToConst: only works from within text links")
		END
	END JumpToConst;
	
	PROCEDURE JumpBack* (to: INTEGER);
		VAR rd: TextModels.Reader; ctx: TextModels.Context; done: BOOLEAN;
			txt: TextModels.Model; tv: TextViews.View; v: Views.View; pos: INTEGER; 
			back: HrOcfTargets.Target;
			op: Stores.Operation;
	BEGIN
		done := FALSE;
		IF (StdLinks.par # NIL) & (StdLinks.par.context IS TextModels.Context) THEN
			ctx := StdLinks.par.context(TextModels.Context);
			txt := ctx.ThisModel();
			rd := txt.NewReader(NIL);
			REPEAT rd.ReadView(v) 
			UNTIL (v = NIL) OR (v IS HrOcfTargets.Target) & (v(HrOcfTargets.Target).tag = to);
			IF v # NIL THEN back := v(HrOcfTargets.Target);
				rd.SetPos(ctx.Pos() + 1);
				REPEAT rd.ReadView(v) UNTIL (v = NIL) OR (v IS StdLinks.Link);
				IF v # NIL THEN 
					Models.BeginScript(txt, "Remove targets", op);
					txt.Delete(ctx.Pos(), v.context(TextModels.Context).Pos() + 1);
					pos := back.context(TextModels.Context).Pos();
					txt.Delete(pos, pos + 1);
					Models.EndScript(txt, op);
					tv := TextViews.Focus(); 
					IF tv.ThisModel() = txt THEN tv.ShowRange(pos, pos, TextViews.focusOnly)
					ELSE TextViews.ShowRange(txt, pos, pos, TextViews.focusOnly)
					END;
					done := TRUE
				END
			END;
			IF ~done THEN Log.String("Position(s) not found"); Log.Ln END
		ELSE Log.String("HrOcfExplorer.JumpBack: only works from within text links")
		END
	END JumpBack;

ObjFile	=	HeaderBlk MetaBlk DescBlk CodeBlk FixBlk UseBlk.

It seems that in order to extract complete meta information from OCF, you'd have to perform fixups first. That is, you have to load HeaderBlk, MetaBlk, DescBlk to perform fixups on them. But, then again, you have to load FixBlk that follows all of the above ad the CodeBlk, in order to get the heads of fixup chains. AAAA!!!

But in the loaded image of MetaBlk, quite a bit of information is represented as variable-size numbers, which has to be decoded again and again every time it is accessed.

So, maybe OCF file is optimized for smaller size; and thus the meta overhead in memory is also optimized for smaller size. But it does not lend itself to speed optimization. Something to think about.
