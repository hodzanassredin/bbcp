MODULE HrAlpha;	(** The first - Alpha - phase of the compiler.  *)
	
	IMPORT Files, 
		TextModels, 
		HrLib,
		(* Phase1 modules: *) HrM, HrS, HrT, HrP, HrB, 
		HrCPT;
	
	VAR dbgsubj: HrT.Segraph;	(* for debugging *)
	
	PROCEDURE Do* (subjRd: TextModels.Reader; lib: HrLib.Librarian; objDir: HrT.Dir;  log: TextModels.Model; opts: SET; OUT subj: HrT.Segraph);
		VAR _: HrCPT.Node; sym: Files.File; __: BOOLEAN;
	BEGIN
		ASSERT(subjRd # NIL, 20); ASSERT(subjRd.Pos() <= subjRd.Base().Length(), 21);
		ASSERT(lib # NIL, 22);
		IF objDir # NIL THEN HrT.SetDir(objDir) ELSE HrT.SetDir(HrT.stdDir) END;
		IF log = NIL THEN log := TextModels.dir.New() END;
		(* deal with options... *)
		HrCPT.Init({}); HrCPT.processor := 10(*obsolete const*);
		HrM.lib := lib;
		HrM.Init(subjRd, log); 
		HrB.typSize := NIL;
		HrT.OpenTop(subj); 
		HrP.Module(_, subj);
		HrT.CloseTop;
		;dbgsubj := subj;	(* temp; for debugging *)
		IF HrM.noerr THEN 
			sym := lib.NewHrSym(subj.mod[0].name);
			HrT.Externalize(subj, HrT.extExported, sym, 0); 
			lib.RegisterHrSym(sym, subj.mod[0].name);
			HrCPT.Export(__, __);
			HrM.RegisterNewSym;
		END
	END Do;
		
END HrAlpha.

HrCPT.Init({}); HrCPT.processor := HrV.processor; HrB.typSize := NIL;
