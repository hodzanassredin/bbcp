MODULE ObxProbe28;

	IMPORT Console, Strings, SYSTEM;

	VAR bad: INTEGER;

	PROCEDURE CHK (IN s: ARRAY OF CHAR; got, exp: LONGINT);
		VAR t: ARRAY 40 OF CHAR;
	BEGIN
		IF got # exp THEN
			INC(bad); Console.WriteStr("BAD "); Console.WriteStr(s);
			Strings.IntToString(got, t); Console.WriteStr(" got="); Console.WriteStr(t);
			Strings.IntToString(exp, t); Console.WriteStr(" exp="); Console.WriteStr(t);
			Console.WriteLn
		END
	END CHK;

	PROCEDURE Go*;
		VAR a, b: LONGINT; i, j: INTEGER;
	BEGIN
		bad := 0;
		a := 1234567890123; b := 10007; i := 5;
		CHK("ASH(a,3)", ASH(a, 3), 9876543120984);
		CHK("ASH(a,-2)", ASH(a, -2), 308641972530);
		CHK("ASH(a,i)", ASH(a, i), 39506172483936);
		CHK("ABS(-a)", ABS(-a), 1234567890123);
		CHK("MIN(a,b)", MIN(a, b), 10007);
		CHK("MAX(a,b)", MAX(a, b), 1234567890123);
		IF ~ODD(a) THEN INC(bad); Console.WriteStr("BAD ODD(a)"); Console.WriteLn END;
		a := -a;
		CHK("ASH(-a,2)", ASH(a, 2), -4938271560492);
		j := 123456789;
		CHK("LSH(j,3)", SYSTEM.LSH(j, 3), 987654312);
		CHK("LSH(j,-4)", SYSTEM.LSH(j, -4), 7716049);
		CHK("ROT(j,4)", SYSTEM.ROT(j, 4), 1975308624);
		IF bad = 0 THEN Console.WriteStr("Probe28 OK")
		ELSE Console.WriteStr("Probe28 FAILURES") END;
		Console.WriteLn
	END Go;

END ObxProbe28.
