		(* On naming of entities. I find the naming of entities in OP2/Dev confusing and obfuscating. What is in the LR called a type, in the compiler is called a Struct. Structures in LR are a case of types. A type is a set of values which a variable of this type may assume. Structured types are array and record types. In Wirth's compiler, the record type representing the type of a variable is called Type. So I will adopt the terminology from the LR, and call a type a Type, and the object that represents the named type declaration I will call a TypeObj. Other options for the latter would be TypeDecl, TypeName, NamedType. *)

	PROCEDURE (c: Literal) Type (): Type, NEW, EXTENSIBLE; BEGIN HALT(125); RETURN NIL END Type;
	(* Since Literal now is an extension of Expression, it has a .type field that should be used instead of Type(); so Type() is deprecated. When a literal is created, however, these procedures can be used to identify the actual type of the literal. It is not always minimal: i.e., default for integer types is fInt32 *)
	PROCEDURE (c: Nil) Type (): Pointer; BEGIN RETURN niltp END Type;
	PROCEDURE (c: Int) Type (): Basic;
		VAR res: Basic;
	BEGIN
		IF (MIN(INTEGER) <= c.long) & (c.long <= MAX(INTEGER)) THEN res := inttp ELSE res := longtp END;
		RETURN res
	END Type;
	
	PROCEDURE (c: Char) Type (): Type;
		VAR res: Type;
	BEGIN IF c.ch <= MAX(SHORTCHAR) THEN res := shortchartp ELSE res := chartp END; RETURN res
	END Type;
	
	PROCEDURE (c: Set) Type (): Type; BEGIN RETURN settp END Type;
	PROCEDURE (c: Bool) Type (): Type; BEGIN RETURN booltp END Type;
	
	PROCEDURE (c: Real) Type (): Type;
		VAR res: Type; short: SHORTREAL;
	BEGIN 
		res := realtp;
		IF ABS(c.real) <= MAX(SHORTREAL) THEN
			short := SHORT(c.real);
			IF short = c.real THEN res := shortrealtp END
		END;
		RETURN res
	END Type;
	
	PROCEDURE (c: StringLit) Type (): Array;
		VAR i: INTEGER; res: Array;
	BEGIN
		i := 0; WHILE (c.str[i] # 0X) & (c.str[i] <= MAX(SHORTCHAR)) DO INC(i) END;
		IF c.str[i] = 0X THEN res := shortstrtp ELSE res := strtp END;
		RETURN res
	END Type;

	PROCEDURE ^ EqualTypes* (a, b: Type): BOOLEAN;
	PROCEDURE MatchingPars* (ap, bp: POINTER TO ARRAY OF Par; ares, bres: Type; checkNames: BOOLEAN ): BOOLEAN;
		VAR i, al, bl: INTEGER; f, a: Par; res: BOOLEAN;
	BEGIN 
		IF ap = NIL THEN al := 0 ELSE al := LEN(ap) END; IF bp = NIL THEN bl := 0 ELSE bl := LEN(bp) END;
		res := al = bl;
		i := 0; WHILE res & (i < al) DO
			f := ap[i]; a := bp[i];
			IF ~EqualTypes(f.type, a.type) THEN err(-115); res := FALSE
			ELSIF f.flags * kindMask # a.flags * kindMask THEN err(-290); res := FALSE
			ELSIF checkNames & (f.name^ # a.name^) THEN err(-291); res := FALSE
			END;
			INC(i)
		END;
		RETURN res
	END MatchingPars;
	
	PROCEDURE EqualTypes* (a, b: Type): BOOLEAN;
		VAR eq: BOOLEAN;
	BEGIN
		eq := SameType(a, b);
		IF ~eq THEN
			WITH a: Array DO WITH b: Array DO 
				eq := (open IN a.flags) & (open IN b.flags) & EqualTypes(a.elTyp, b.elTyp) 
			ELSE END ELSE END;
			IF ~eq THEN
				WITH a: Procedure DO WITH b: Procedure DO
					eq := MatchingPars(a.par, b.par, a.res, b.res, FALSE)
				ELSE END ELSE END;
				IF ~eq THEN 
					WITH a: ArrayPtr DO WITH b: ArrayPtr DO
						eq := EqualTypes(a.base, b.base)
					ELSE END ELSE END;
					IF ~eq THEN 
						WITH a: RecPtr DO WITH b: RecPtr DO
							eq := EqualTypes(a.base, b.base)
						ELSE END ELSE END
					END
				END
			END
		END;
		RETURN eq
	END EqualTypes;
	
	PROCEDURE Is* (ext, base: Type): BOOLEAN;
		VAR is: BOOLEAN; a, b: Record;
	BEGIN is := FALSE;
		IF (ext IS RecPtr) & (base IS RecPtr) OR (ext IS Record) & (base IS Record) THEN
			WITH ext: RecPtr DO a := ext.base | ext: Record DO a := ext ELSE END;
			WITH base: RecPtr DO b := base.base | base: Record DO b := base ELSE END;
			IF b # NIL THEN WHILE (a # NIL) & (a # b) DO a := a.base END; is := a # NIL END
		END;
		RETURN is
	END Is;
	
	PROCEDURE Includes* (a, b: INTEGER): BOOLEAN;
		(** Implements Langage Report p. 6.1 term "includes", plus for other types Includes = (a = b). WITH the significant distinction: Reals do not include Integers. Includes is used to determine whether a type conversion is necessary *)
		(* this may have to be extensible to allow different backends to inform the *)
		VAR res: BOOLEAN;
	BEGIN	(* it's strange that chars are always included in numbers... this is against LR. What is it for? *)
		CASE a OF
		| fChar8: res := b IN {fChar8}
		| fChar16: res := b IN {fChar8, fChar16}
		| (*Byte, *)fInt8: res := b IN {(*Byte, *)fInt8}
		| fInt16: res := b IN {fChar8, (*Byte, *)fInt8, fInt16}
		| fInt32: res := b IN {fChar8, fChar16, (*Byte, *)fInt8, fInt16, fInt32}
		| fInt64: res := b IN {fChar8, fChar16, (*Byte, *)fInt8, fInt16, fInt32, fInt64}
		| fReal32: res := b IN {(*fChar8, fChar16, (*Byte, *)fInt8, fInt16, fInt32, fInt64, *)fReal32}
		| fReal64: res := b IN {(*fChar8, fChar16, (*Byte, *)fInt8, fInt16, fInt32, fInt64,*) fReal32, fReal64}
		(*| fString16: res := b IN {fString8, fString16}	(* is this needed? String cannot be a target *)
		if i decide to uncomment this, check out V.VActualPar - the branch for strings relies on string16 not including string8 *)
		ELSE res := a = b
		END;
		RETURN res
	END Includes;

	PROCEDURE NewString (utf: POINTER TO ARRAY OF SHORTCHAR): POINTER TO String;
		VAR _, len: INTEGER; s, t: POINTER TO String; buf: ARRAY 1024 OF CHAR; fits: BOOLEAN;
	BEGIN 
		IF (utf # NIL) & (utf^ # '') THEN 
			len := LEN(utf^); fits := len <= LEN(buf);
			IF fits THEN Utf.Utf8ToString(utf, buf, _); len := LEN(buf$) + 1
			ELSE NEW(t, len); Utf.Utf8ToString(utf, t, _); len := LEN(t$) + 1
			END;
			NEW(s, len);
			IF fits THEN s^ := buf$ ELSE s^ := t$ END
		END;
		RETURN s
	END NewString;

	PROCEDURE ThisConstVal (val: T.Const; type: T.Struct): Literal;
		VAR int: Int; bool: Bool; char: Char; str: StringLit; real: Real; set: Set; 
			res: Literal;
	BEGIN
		CASE type.form OF
		| Byte, Int8, Int16, Int32, Int64: int := dir.Int(); res := int;
			IF type.form = Int64 THEN int.long := ENTIER(val.realval) + LONG(val.intval)
			ELSE int.long := val.intval
			END
		| cBool: bool := dir.Bool(); res := bool; 
			IF val.intval = 1 THEN bool.bool := TRUE ELSIF val.intval = 0 THEN bool.bool := FALSE
			ELSE HALT(20)
			END
		| Char8, Char16: char := dir.Char(); res := char; 
			IF (0 <= val.intval) & (val.intval <= ORD(MAX(CHAR))) THEN char.ch := CHR(val.intval)
			ELSE HALT(20)
			END
		| Real32, Real64: real := dir.Real(); res := real; real.real := val.realval
		| cSet: set := dir.Set(); res := set; set.set := val.setval
		| String8, String16: str := dir.StringLit(); res := str; 
			IF val.ext # NIL THEN str.str := NewString(val.ext) END
		| NilTyp: res := nil
		ELSE HALT(20)
		END;
		RETURN res
	END ThisConstVal;

	PROCEDURE GetObjFlags (o: Object; p: T.Object);
		CONST none = 0; nil = 1; in = 2; out = 4; new = 5; iid = 6;
		VAR set: SET;
	BEGIN ASSERT(o # NIL, 20); ASSERT(p # NIL, 21);
		WITH o: Par DO ASSERT(o.flags - kindMask = {}) ELSE o.flags := {} END;
		IF p.used THEN INCL(o.flags, used) END;
		IF p.fpdone THEN INCL(o.flags, fpdone) END;
		IF p.leaf THEN
			IF (o IS LProc) OR (o IS XProc) OR (o IS IProc) OR (o IS TProc) OR (o IS SProc) 
				OR (o IS TypeName) OR (o IS Const) OR (o IS Field) OR (o IS Mod)
			THEN INCL(o.flags, leaf)
			ELSIF o IS Var THEN	(* ok *)
			ELSE HALT(22)
			END
		ELSIF o IS Var THEN INCL(o.flags, ref)
		END;
		IF o IS TProc THEN o.flags := o.flags + p.conval.setval
		ELSIF (o IS Par) & (o.flags * kindMask # valK) THEN 
			CASE p.sysflag OF none: set := {}
			| nil: set := nilPF | in: set := inPF | out: set := outPF | new: set := newPF | iid: set := iidPF
			ELSE HALT(25)
			END;
			o.flags := o.flags + set
		END;
		IF o IS Proc THEN o.flags := o.flags + ThisProcSysflag(p.sysflag) END;
		CASE p.vis OF 
		| internal: set := domestic
		| external:
			IF (o IS GlobVar) OR (o IS Field) OR (o IS Const) OR (o IS TypeName) OR (o IS XProc) 
				OR (o IS TProc)
			THEN set := { exported }
			ELSE HALT(23)
			END
		| externalR: 
			IF (o IS GlobVar) OR (o IS Field) OR (o IS TypeName) OR (o IS TProc) THEN set := { readOnly }
			ELSE HALT(23)
			END
		|  inPar, outPar: ASSERT((o IS Par) & (o.flags * kindMask # valK), 23)
		ELSE HALT(23)
		END;
		o.flags := o.flags + set;
		IF (p.modifiers # NIL) & (p.modifiers[0] = HrM.JS) THEN 
			IF (o IS GlobVar) OR (o IS Field) OR (o IS Proc) THEN
				WITH o: Proc DO IF o.level = 0 THEN INCL(o.flags, js) ELSE HrM.Mark(512, o.textpos) END
				ELSE INCL(o.flags, js)
				END
			ELSE HALT(24)
			END
		END;
		ASSERT(o.flags - { 0..29 } = {}, 60)
	END GetObjFlags;

	PROCEDURE GetStrucFlags (t: Type; s: T.Struct);
		CONST 	(* values from CpjsM.Get?SysFlag procedures *)
			none = 0; untagged = 1; handle = 2; noalign = 3; align2 = 4; align4 = 5; align8 = 6; union = 7;
			interface = 10; jint = -11; jarr = -12; jstr = -13; som = 20; ccall = -10; ccall16 = -12;
		VAR f: SET;
	BEGIN
		WITH t: Array DO ASSERT(t.flags - { open } = {}) ELSE t.flags := {} END;
		IF s.untagged THEN
			IF (t IS Array) OR (t IS Record) OR (t IS Pointer) THEN INCL(t.flags, untagged) ELSE HALT(23) 
			END
		END;
		IF s.pbused THEN INCL(t.flags, pbused) END;
		IF s.pvused THEN INCL(t.flags, pvused) END;
		IF s.exp THEN INCL(t.flags, exp) END;
		IF t IS Record THEN t.flags := t.flags + BITS(s.attribute) END;
		IF t IS Structure THEN
			CASE s.sysflag OF
			| none: f := {}
			| untagged: f := untaggedF | handle: f := handleF | noalign: f := noAlignF
			| align2: f := align2F | align4: f := align4F | align8: f := align8F | union: f := unionF
			| interface: f := interfaceF | jint: f := jintF | jstr: f := jstrF | som: f := somF
			| ccall: f := ccallF
			| jarr (* = ccall16 *): 
				IF t IS Array THEN f := jarrF ELSIF t IS Procedure THEN f := ccall16F ELSE HALT(21) END
			ELSE HALT(20)
			END;
			t.flags := t.flags + f
		END;
		ASSERT(t.flags - { 0..25 } = {}, 60)
	END GetStrucFlags;
	
	PROCEDURE Insert* (VAR tree: Object; o: Object);	(* eliminate recursion *)
	BEGIN
		IF tree = NIL THEN tree := o
		ELSIF o.name$ < tree.name$ THEN Insert(tree.left, o)
		ELSIF o.name$ > tree.name$ THEN Insert(tree.right, o)
		ELSE HrM.Mark(1, o.textpos)
		END
	END Insert;
	
	PROCEDURE InsertTypeName (VAR tree: Object; o: TypeName; OUT old: TypeName);
		(* eliminate recursion *)
	BEGIN 
		IF tree = NIL THEN tree := o
		ELSIF o.name$ < tree.name$ THEN InsertTypeName(tree.left, o, old)
		ELSIF o.name$ > tree.name$ THEN InsertTypeName(tree.right, o, old)
		ELSIF (tree IS TypeName) & (tree.type IS Forward) (*& ~(o.type IS Forward)*) THEN
			tree.type(Forward).actual := o.type; tree.type := o.type;
			tree.textpos := o.textpos; ASSERT(tree.imno = o.imno);
			old := tree(TypeName)
		ELSE HrM.Mark(1, o.textpos)
		END
	END InsertTypeName;
	
	PROCEDURE AppendTo* (o: Object; VAR scope: SCope);
	(** Appends o to scope. Could be eliminated for efficiency, would require Append to be identical text *)
		VAR old: TypeName; p: Object;
	BEGIN (* do bound procedures need to be in the scope's tree? Or just the list? 'cause *)
		IF scope.head = NIL THEN scope.head := o ELSE scope.tail.next := o END;
		scope.tail := o;
		o.imno := scope.imno; o.owner := scope;
		WITH o: Proc DO o.nesting := scope.nesting + 1 ELSE END;
		WITH o: TypeName DO
			InsertTypeName(scope.tree, o, old);
			IF old # NIL THEN
				IF scope.head = old THEN scope.head := old.next 
				ELSE p := scope.head; WHILE p.next # old DO p := p.next END; p.next := old.next
				END
			END
		ELSE IF ~(o IS TProc) THEN Insert(scope.tree, o) END
		END
	END AppendTo;
	
	PROCEDURE Append* (o: Object); 
		(** Appends object to current scope (nowScope). Special treatment of type  *)
	BEGIN AppendTo(o, nowScope)
	END Append;
	
	PROCEDURE AppendThis* (o: Object; n: POINTER TO Name; pos: INTEGER);

	PROCEDURE AppendProc* (r: Record; p: TProc);
		VAR proc, copy: POINTER TO ARRAY OF TProc; i, len: INTEGER;
	BEGIN proc := r.proc;
		IF proc = NIL THEN NEW(proc, capProc); len := capProc; r.proc := proc ELSE len := LEN(proc) END;
		IF r.nproc = len THEN NEW(copy, len + capProc); 
			i := 0; WHILE i < len DO copy[i] := proc[i]; INC(i) END; proc := copy; r.proc := proc
		END;
		proc[r.nproc] := p; Insert(r.tree, p); INC(r.nproc);
		Append(p)
	END AppendProc;

	PROCEDURE CountFieldsVerbs (o: T.Object; OUT fields, verbs: INTEGER);
		PROCEDURE Do (o: T.Object);
		BEGIN IF o.left # NIL THEN Do(o.left) END; IF o.right # NIL THEN Do(o.right) END;
			IF o.mode = cFld THEN INC(fields) ELSIF o.mode = cTProc THEN INC(verbs) ELSE HALT(20) END
		END Do;
	BEGIN fields := 0; verbs := 0; IF o # NIL THEN Do(o) END
	END CountFieldsVerbs;
	
	
	
	PROCEDURE BuildObjectTree (OUT tree: Object; o: T.Object);
		VAR obj: Object;
		PROCEDURE Traverse (o: T.Object);
		BEGIN 
			IF o # NIL THEN obj := ThisObject(o); IF obj # NIL THEN Insert(tree, obj) END;
				IF o.left # NIL THEN Traverse(o.left) END; IF o.right # NIL THEN Traverse(o.right) END
			END
		END Traverse;
	BEGIN Traverse(o)
	END BuildObjectTree;
	
	PROCEDURE GetMembers (r: Record; fields: T.Object);
		VAR i, j: INTEGER; f: POINTER TO ARRAY OF Field; p: POINTER TO ARRAY OF TProc; obj: ANYPTR;
			
		PROCEDURE Traverse (o: T.Object);
		BEGIN obj := ThisObject(o); ASSERT(obj # NIL, 60);
			WITH obj: Field DO f[i] := obj; INC(i) | obj: TProc DO p[j] := obj; INC(j) ELSE HALT(21) END;
			Insert(r.tree, obj(Object));
			IF o.left # NIL THEN Traverse(o.left) END; IF o.right # NIL THEN Traverse(o.right) END
		END Traverse;
	
	BEGIN ASSERT(r # NIL, 20);
		CountFieldsVerbs(fields, i, r.nproc);
		IF i > 0 THEN NEW(f, i) END; IF r.nproc > 0 THEN NEW(p, r.nproc) END; 
		IF i + r.nproc > 0 THEN i := 0; j := 0; Traverse(fields) END;
		r.fields := f; r.proc := p
	END GetMembers;


	
	



	PROCEDURE ^ ThisType (OUT type: Type; s: T.Struct; VAR rec: Record; VAR arr: Array);
	PROCEDURE GetNewTypeDef (OUT type: Type; s: T.Struct; VAR rec: Record; VAR arr: Array);
		VAR basic: Basic; proc: Procedure; i, len: INTEGER;
			par: T.Object; o: Object; mode: BYTE;
			sptr: SysPtr; arrPtr: ArrayPtr; recPtr: RecPtr;
			_: Type; __: Record; ___: Array;
	BEGIN
		CASE s.form OF
		| Byte, cBool, Char8, Char16, Int8, Int16, Int32, Real32, Real64, cSet, Int64, 
			String8, String16:
			basic := dir.Basic(); type := basic; basic.what := ThisForm(s.form);
		| fJString, fErr: basic := dir.Basic(); type := basic; basic.what := s.form
		| Undef, NilTyp: HALT(20)
		| ProcTyp: proc := dir.Procedure(); type := proc; proc.what := fProcedure; 
			IF s.BaseTyp.form # fNoTyp THEN ThisType(proc.res, s.BaseTyp, rec, arr) END;
			len := 0; par := s.link; 
			WHILE par # NIL DO INC(len); par := par.link END;
			IF len > 0 THEN NEW(proc.par, len);
				i := 0; par := s.link; WHILE par # NIL DO 
					INC(par.mode, parMagic); o := ThisObject(par); 
					WITH o: Par DO proc.par[i] := o; proc.par[i].num := i ELSE HALT(61) END;
					INC(i); par := par.link 
				END
			END
		| cPointer: ASSERT(s.BaseTyp # NIL);
			CASE s.BaseTyp.comp OF 
			| fcBasic: ASSERT(s.BaseTyp.form = fUndef, 20); 
				sptr := dir.SysPtr(); type := sptr; type.what := fSysPtr
			| fcArray, fcDynArr:
				arrPtr := dir.ArrayPtr(); type := arrPtr; arrPtr.what := fcArrPtr; 
				ThisType(_, s.BaseTyp, __, arrPtr.base)
			| fRecord: recPtr := dir.RecPtr(); type := recPtr; recPtr.what := fRecPtr;
				ThisType(_, s.BaseTyp, recPtr.base, ___)
			ELSE HALT(22)
			END;
		| Comp:
			CASE s.comp OF
			| cArray, DynArr: arr := dir.Array(); type := arr; arr.what := fArray; 
				INC(openDepth);
				ThisType(arr.elTyp, s.BaseTyp, __, ___);
				DEC(openDepth);
				IF s.comp = DynArr THEN INCL(arr.flags, open); arr.lenoffs := s.align; arr.len := s.n
				ELSE (*EXCL(arr.flags, open);*) arr.len := s.n
				END;
			| cRecord: rec := dir.Record(); type := rec; rec.what := fRecord; rec.nproc := s.n; 
				rec.level := s.extlev;
				rec.align := s.align; 
				IF s.BaseTyp # NIL THEN ThisType(_, s.BaseTyp, rec.base, ___) END;
				GetMembers(rec, s.link)
			ELSE HALT(21)
			END
		ELSE HALT(126) (* includes fNoTyp, which is eliminated from my structures *)
		END;
		ASSERT(type # NIL, 60);
		WITH type: Structure DO type.mmno := s.mno; type.pos := s.txtpos
		ELSE
		END;
		GetStrucFlags(type, s);
		type.pbfp := s.pbfp; type.pvfp := s.pvfp; type.size := s.size;
		IF (s.strobj # NIL) & (s.strobj.name^ # "") THEN GetType(type, s.strobj) END;
		AddType(type, s);
		ASSERT(type # NIL, 60)
	END GetNewTypeDef;
	(* Let's say scope S1 defines type T and proc P. P defines var V: T. It may be that as S1 is processed, V's definition will be encountered before T's. In this case, when V's typedef is processed, T is still not translated. Can it be translated at the spot? No, because there is no (cheap) way to figure out T's scope. So, either a multipass or a fixup approach is needed. I adopt the fixup approach. *)
	
	PROCEDURE ThisType (OUT type: Type; s: T.Struct; VAR rec: Record; VAR arr: Array);
		VAR res: Type; t: TypeName;
	BEGIN ASSERT(s # NIL, 20); 
		IF (s.strobj = NIL) OR (s.strobj.name^ = "") THEN GetNewTypeDef(type, s, rec, arr)
		ELSE t := Lookup(s.strobj);
			IF t = NIL THEN type := ThisObject(s.strobj)(TypeName).type
			ELSE 
				IF t.type # NIL THEN type := t.type 
				ELSE GetNewTypeDef(type, s, rec, arr); t.type := type 
				END
			END
		END;
		ASSERT(type # NIL, 60);
		WITH type: Record DO rec := type | type: Array DO arr := type ELSE END
	END ThisType;
	
	PROCEDURE ThisGenericType (s: T.Struct): Type;
		VAR res: Type; rec: Record; arr: Array;
	BEGIN ThisType(res, s, rec, arr); RETURN res
	END ThisGenericType;
	
	



	PROCEDURE FindInTree* (t: Object; IN n: Name): Object;	(* reimplement using Cmp *)
	BEGIN
		WHILE (t # NIL) & (t.name^ # n) DO
			IF n < t.name^ THEN t := t.left ELSE t := t.right END
		END;
		RETURN t
	END FindInTree;

	PROCEDURE FindDeepShort (s: SCope; IN n: ARRAY OF SHORTCHAR): Object;
		VAR o: Object; m: Name; _: INTEGER;
	BEGIN Utf.Utf8ToString(n, m, _);
		WHILE (s # NIL) & (o = NIL) DO
			o := FindInTree(s.tree, m);
			IF o = NIL THEN s := s.owner END 
		END;
		RETURN o

	PROCEDURE FindInRec* (rec: Record; IN n: ARRAY OF CHAR): Object;
		VAR o: Object;
	BEGIN ASSERT(rec # NIL, 20); ASSERT(n # '', 21);
		REPEAT o := FindInTree(rec.tree, n$); IF o = NIL THEN rec := rec.base END 
		UNTIL (rec = NIL) OR (o # NIL);
		RETURN o
	END FindInRec;

	PROCEDURE FindProc (o: T.Object): Proc;
		VAR res: Proc; type: Type; obj: Object; sc: SCope;
	BEGIN ASSERT(o # NIL, 20); 
		IF o.mode = cTProc THEN
			type := LookupType(o.link.typ); ASSERT(type # NIL);
			IF type IS RecPtr THEN type := type(RecPtr).base END;
			WITH type: Record DO obj := FindShortInRec(type, o.name) ELSE HALT(20) END
		ELSE 
			IF o.mnolev < 0 THEN sc := nowTop.mod[-o.mnolev + 1] ELSE sc := nowScope END; 
			obj := FindDeepShort(sc, o.name)
		END;
		ASSERT(obj # NIL); ASSERT(obj IS Proc);
		RETURN obj(Proc)
	END FindProc;

	PROCEDURE ThisScope (s: SCope; scope: T.Object);
		VAR _: INTEGER; o: T.Object; obj, p: Object; t: Type; n: Name;
	BEGIN s.owner := nowScope; (* or make it a param? *)
		WITH s: Mod DO nowTop.mod[s.imno] := s ELSE END; 
		nowScope := s; 
		o := scope.next;
		WHILE o # NIL DO 
			IF o.mode = cTProc THEN t := LookupType(o.link.typ);
				IF t IS RecPtr THEN t := t(RecPtr).base END; Utf.Utf8ToString(o.name^, n, _);
				WITH t: Record DO obj := FindInTree(t.tree, n) ELSE HALT(20) END; 
				ASSERT(obj # NIL, 20)
			ELSE obj := ThisObject(o)
			END;
			IF obj # NIL THEN IF ~(obj IS TProc) & (obj.name[0] # "@") THEN Insert(s.tree, obj) END; 
				IF p # NIL THEN p.next := obj ELSE s.head := obj END; 
				p := obj 
			END;
			o := o.next
		END;
		nowScope := s.owner
	END ThisScope;
	
	PROCEDURE IsPar (o: T.Object): BOOLEAN;
		VAR p: T.Object;
	BEGIN p := nowPar; WHILE (p # NIL) & (p # o) DO p := p.link END; RETURN p # NIL
	END IsPar;

	PROCEDURE ThisObject (o: T.Object): Object;
		CONST typed = { cVar, cCon };
			procedures = { cSProc, cTProc, cXProc, cCProc, cLProc, cIProc };
		VAR obj, oo: Object; con: Const; gvar: GlobVar; lvar: LocVar; par: Par;
			proc: Proc; lproc: LProc; xproc: XProc; sproc: SProc; cproc: CProc; tproc: TProc; opar: T.Object;
			i: INTEGER; kind: SET; isPar: BOOLEAN;
			typeName: TypeName; fld: Field; mod: Mod; _: Array; __: Record;
	BEGIN ASSERT(o # NIL, 20); typeName := Lookup(o);
		IF o.mode = cAttr THEN
		ELSIF typeName # NIL THEN obj := typeName	(* some typenames are created before  *)
		ELSE
			CASE o.mode OF
			| cCon: con := dir.Const(); obj := con; con.class := cCon; 
				con.val := ThisConstVal(o.conval, o.typ); (*not tested:*)con.ex := con.val
			| cTyp: typeName := dir.TypeName(); obj := typeName; typeName.class := cTypeName; 
				AddThis(o, typeName)
			| cFld: fld := dir.Field(); obj := fld; fld.class := cFld; fld.offs := o.adr;
			| cMod: mod := dir.Mod(); obj := mod; mod.class := cMod; mod.imno := -o.mnolev + 1;
				mod.name := NewName(o.name); 	(* enables ThisObj to identify by name the obj's mod *)
				ThisScope(mod, o.scope); mod.lib := NewName(o.library)
			| cVar, cVarPar, cVar + parMagic, cVarPar + parMagic: 
				isPar := o.mode > parMagic;
				IF isPar THEN DEC(o.mode, parMagic);
					par := dir.Par(); obj := par; par.class := cPar; lvar := par; par.adr := o.adr;
					par.num := none; (* par.num to be filled out later !!! *)
					IF o.mode = cVarPar THEN
						CASE o.vis OF inPar:  kind := inK | outPar: kind := outK | 0: kind := varK 
						ELSE HALT(23) 
						END
					ELSE kind := valK
					END;
					par.flags := kind
				ELSIF nowScope IS Mod THEN
					gvar := dir.GlobVar(); obj := gvar; gvar.class := cGVar; gvar.adr := o.adr; 
					gvar.ent := NewName(o.entry); gvar.lib := NewName(o.library); 
				ELSE lvar := dir.LocVar(); lvar.class := cLVar; obj := lvar; lvar.adr := o.adr
				END
			| cIProc: HALT(126)
			| cLProc, cXProc, cSProc, cCProc, cTProc:
				CASE o.mode OF cLProc: lproc := dir.LProc(); obj := lproc; lproc.class := cLProc; 
					LibEnt(o, lproc.lib, lproc.ent)
				| cXProc: xproc := dir.XProc(); obj := xproc; xproc.class := cXProc;
					LibEnt(o, xproc.lib, xproc.ent)
				| cSProc: sproc := dir.SProc(); obj := sproc; sproc.class := cSProc; 
					sproc.num := SHORT(SHORT(o.adr))
				| cCProc: cproc := dir.CProc(); obj := cproc; cproc.class := cCProc; 
					cproc.code := o.conval.ext
				| cTProc: tproc := dir.TProc(); obj := tproc; tproc.class := cTProc; 
					tproc.num := SHORT(o.num); LibEnt(o, tproc.lib, tproc.ent)
				ELSE HALT(61)
				END;
				proc := obj(Proc); proc.level := o.mnolev;
				i := 0; opar := o.link; 
				WHILE opar # NIL DO INC(i); INC(opar.mode, parMagic); opar := opar.link END; 
				proc.parlen := i;		GetObjFlags(obj, o);
				IF (o.scope # NIL) & (o.mode # cSProc) THEN ThisScope(proc, o.scope) 	
				END; 	(* SProc does not have a scope *)
				IF proc.parlen > 0 THEN NEW(proc.par, proc.parlen);
					i := 0; opar := o.link; WHILE i < proc.parlen DO ASSERT(opar # NIL, 21);
						IF proc # NIL THEN oo := FindShortInScope(proc, opar.name^) END;
						IF oo = NIL THEN oo := ThisObject(opar) END;	(* FINALIZE(this) is not  *)
						WITH oo: Par DO proc.par[i] := oo; proc.par[i].num := i ELSE HALT(22) END;
						INC(i); opar := opar.link 
					END
				END
			| cAttr: 	(* Attr objects are not propagated into the new symbol table *)
			ELSE HALT(23)
			END;
			ASSERT(obj # NIL, 60);
			IF obj.name = NIL THEN obj.name := NewName(o.name) END;
			obj.fprint := o.fprint; 
			WITH obj: Proc DO
				IF o.mnolev <= 0 THEN obj.imno := -o.mnolev + 1; obj.nesting := 1
				ELSE obj.imno := none; obj.nesting := o.mnolev - 1
				END
			| obj: Mod DO obj.nesting := 0
			ELSE
			END;
			obj.textpos := o.textpos;
			IF ~(obj IS Proc) THEN GetObjFlags(obj, o) END;
			obj.history := o.history;
			IF (o.typ.form = fJString) & (nowTop.mod[nowScope.imno].name^ # "JS") THEN 
				IF obj IS Par THEN ASSERT(nowScope IS Proc, 60); 
					IF ~(js IN nowScope.flags) THEN HALT(0); HrM.Mark(515, obj.textpos); o.typ.form := fErr END
				ELSIF ~(js IN obj.flags) THEN HrM.Mark(513, obj.textpos); o.typ.form := fErr 
				END;
				IF ~(obj IS GlobVar) & ~(obj IS Field) & ~(obj IS Par) THEN HrM.Mark(514, obj.textpos); 
					o.typ.form := fErr 
				END
			END;
			IF o.typ.form # fNoTyp THEN ThisType(obj.type, o.typ, __, _) END
		END;
		ASSERT((o.mode = cAttr) OR (obj # NIL), 63);
		RETURN obj
	END ThisObject;

		VAR i: INTEGER;
	BEGIN
		WITH
		| ex: ConDes DO SpellQualident(ex.const)
		| ex: VarDes DO SpellQualident(ex.var)
		| ex: FieldDes DO SpellEx(ex.rec); log.WriteChar('.'); log.WriteString(ex.field.name)
		| ex: Deref DO SpellEx(ex.what); log.WriteChar('^')
		| ex: Index DO SpellEx(ex.arr); log.WriteChar('['); SpellInd(ex.dim, ex.ipos); log.WriteChar(']')
		| ex: GuardDes DO SpellEx(ex.what); log.WriteChar('('); SpellQualident(ex.type.typeName); log.WriteChar(')')
		| ex: TypeDes DO SpellQualident(ex.name)
		| ex: ProcDes DO SpellQualident(ex.proc)
		| ex: CallDes DO SpellCall(ex.proc, ex.what, ex.apar, ex.type, ex.super)
		| ex: GuardEx DO SpellEx(ex.what); log.WriteChar('('); SpellQualident(ex.type.typeName); log.WriteChar(')')
		| ex: Op DO SpellOp(ex)
		| ex: IsOp DO SpellEx(ex.des); log.WriteString(" IS "); SpellQualident(ex.typeName)
		| ex: Literal DO SpellConstVal(ex)
		| ex: StringSel DO SpellDes(ex.what); log.WriteChar("$")
		| ex: ErrDes DO log.WriteString("bad!designator")
		| ex: ErrEx DO log.WriteString("bad!expression")
		ELSE log.WriteString("SpellEx NIY") ;HALT(1)
		END
	END SpellEx; 
	
	PROCEDURE SpellDim (IN dim: ARRAY OF Expression);
		VAR i: INTEGER;
	BEGIN i := 0; WHILE dim[i] # NIL DO Comma(i # 0); SpellEx(dim[i]); INC(i) END
	END SpellDim;
	
	PROCEDURE SpellInd (IN ind: ARRAY OF Expression; IN ipos: ARRAY OF INTEGER);
		VAR i: INTEGER;
	BEGIN i := 0; WHILE ind[i] # NIL DO Comma(i # 0); Pos(ipos[i]); SpellEx(ind[i]); INC(i) END
	END SpellInd;

	PROCEDURE SpellCall (proc: Proc; what: Designator; actuals: POINTER TO ARRAY OF Expression;
		type: Type; isSuper: BOOLEAN
	);
		VAR i, len: INTEGER;
	BEGIN
		IF proc # NIL THEN 
			IF proc IS TProc THEN i := 1; SpellEx(actuals[0]); log.WriteChar('.') ELSE i := 0 END;
			SpellQualident(proc); IF isSuper THEN log.WriteChar("^") END
		ELSE i := 0; SpellEx(what) END; 
		IF actuals # NIL THEN len := LEN(actuals) ELSE len := 0 END;
		IF i < len THEN log.WriteChar("("); 
			WHILE i < len DO IF actuals[i] # NIL THEN SpellEx(actuals[i]) ELSE log.WriteString("<NULL>") END;
				IF i < len - 1 THEN log.WriteString(", ") END; INC(i) 
			END;
			log.WriteChar(")")
		ELSIF type # NIL THEN log.WriteString("()")
		END
	END SpellCall;

	PROCEDURE SpellOp (op: Op);
		VAR what: INTEGER;
	BEGIN ASSERT(op # NIL, 20); 
		(*IF (op.what = bit) & (op.b # NIL) THEN what := sysbit
		ELS*)
		IF ((op.what = maxfn) OR (op.what = minfn)) & (op.a = NIL) & (op.b = NIL) THEN what := -op.what
		ELSE what := op.what 
		END;
		CASE what OF
		| -maxfn, -minfn:
			log.WriteString(opNames[-what]); log.WriteChar("("); SpellType(op.type, showAlias); log.WriteChar(")")
		| maxsetfn: log.WriteString("MAX(SET)")
		| minsetfn: log.WriteString("MIN(SET)")
		| absfn, capfn, oddfn, adrfn, ccfn, valfn, ashfn, lenfn, minfn, maxfn, lshfn, rotfn, typfn, sizefn:
			log.WriteString(opNames[what]); log.WriteChar("("); SpellEx(op.a); 
			IF op.b # NIL THEN log.WriteString(', '); SpellEx(op.b) END;
			log.WriteChar(")")
		| notfn, minusfn, isfn, timesfn, slashfn, divfn, modfn, andfn, plusfn, orfn, eqlfn, neqfn, lssfn, leqfn, gtrfn, geqfn, infn:
			IF (what = notfn) OR (what = minusfn) & (op.b = NIL) THEN
				log.WriteString(opNames[what]); SpellEx(op.a)
			ELSE SpellEx(op.a); log.WriteChar(' '); log.WriteString(opNames[what]); log.WriteChar(' '); SpellEx(op.b)
			END
		| bitfn, ssetel, range: log.WriteString("{ "); SpellEx(op.a); 
			IF op.b # NIL THEN log.WriteString(".."); SpellEx(op.b) END;
			log.WriteString(" }")
		| convfn: SpellEx(op.a); log.WriteChar("("); SpellType(op.type, showAlias); log.WriteChar(")") 
		| mskfn: log.WriteString("NIY")
		ELSE HALT(20)
		END
	END SpellOp;
	
	PROCEDURE SpellDes* (d: Designator);
	BEGIN SpellEx(d)
	END SpellDes;
	
	PROCEDURE SpellAssign (a: Assign);
	BEGIN
		CASE a.kind OF errorfn: log.WriteString("ERROR:=")
		| incfn, decfn, inclfn, exclfn: log.WriteString(opNames[a.kind]); log.WriteChar("("); SpellEx(a.des);
			IF a.expr # one THEN log.WriteString(", "); SpellEx(a.expr) END;
			log.WriteChar(")")
		ELSE SpellDes(a.des); log.WriteString(" := "); SpellEx(a.expr)
		END
	END SpellAssign;
	
	PROCEDURE (s: Stat) Spell* (VAR log: TextMappers.Formatter), NEW, EXTENSIBLE;
	BEGIN log.WriteString("NIY")
	END Spell;

	PROCEDURE SpellSeq* (s: Stat);
		VAR if: If; i, len: INTEGER; st: Stat; clause: WithClause; _: Type;
	BEGIN
		WHILE s # NIL DO 
			log.WriteLn;
			IF details THEN log.WriteChar("<"); log.WriteInt(s.textpos); log.WriteChar(">") END;
			WITH s: Assign DO SpellAssign(s)
			| s: New DO log.WriteString("NEW("); SpellDes(s.des); 
				IF s.dim[0] # NIL THEN log.WriteString(", "); SpellDim(s.dim) END; log.WriteChar(")")
			| s: Return DO log.WriteString("RETURN "); IF s.ex # NIL THEN SpellEx(s.ex) END
			| s: Exit DO log.WriteString("EXIT")
			| s: Trap DO 
				IF s.cond = NIL THEN log.WriteString("TRAP(") 
				ELSE log.WriteString("ASSERT("); SpellEx(s.cond); log.WriteString(", ")
				END; 
				SpellEx(s.num); log.WriteChar(")")
			| s: While DO log.WriteString("WHILE "); SpellEx(s.cond); log.WriteString(" DO"); 
				INC(logIndent); SpellSeq(s.do); DEC(logIndent); log.WriteLn; log.WriteString("END")
			| s: Repeat DO log.WriteString("REPEAT ");
				INC(logIndent); SpellSeq(s.do); DEC(logIndent); log.WriteLn; log.WriteString("UNTIL "); SpellEx(s.until)
			| s: Loop DO log.WriteString("LOOP");
				INC(logIndent); SpellSeq(s.do); DEC(logIndent); log.WriteLn; log.WriteString("END")
			| s: Call DO SpellCall(s.proc, s.what, s.actuals, _, s.super)
			| s: If DO if := s;
				WHILE if # NIL DO 
					IF if.cond # NIL THEN IF s = if THEN log.WriteString("IF ") ELSE log.WriteLn; log.WriteString("ELSIF ") END; 
						SpellEx(if.cond); log.WriteString(" THEN")
					ELSIF if.do # NIL THEN log.WriteLn; log.WriteString("ELSE ")
					END;
					IF if.do # NIL THEN INC(logIndent); SpellSeq(if.do); DEC(logIndent) END;
					if := if.next
				END;
				log.WriteLn; log.WriteString("END")
			| s: Case DO st := NIL; log.WriteString("CASE "); SpellEx(s.what); log.WriteString(" OF"); log.WriteLn;
				len := s.labels;
				i := 0; WHILE i < len DO
					IF (st # NIL) & (s.stmts[i] # st) THEN
						log.WriteChar(":"); INC(logIndent); SpellSeq(st); DEC(logIndent); 
						log.WriteLn; log.WriteString("| "); 
					ELSIF (st # NIL) THEN log.WriteString(", ")
					END;
					log.WriteInt(s.ranges[i].from); 
					IF s.ranges[i].to # s.ranges[i].from THEN log.WriteString(".."); log.WriteInt(s.ranges[i].to) END;
					st := s.stmts[i];
					INC(i)
				END;
				log.WriteChar(":"); INC(logIndent); SpellSeq(st); DEC(logIndent); 
				log.WriteLn;
				IF s.else # NIL THEN 
					log.WriteString("ELSE"); INC(logIndent); SpellSeq(s.else); DEC(logIndent); log.WriteLn
				END;
				log.WriteString("END")
			| s: With DO log.WriteString("WITH "); clause := s.clauses;
				REPEAT log.WriteLn; log.WriteString("| "); 
					SpellEx(clause.is);
					log.WriteString(" DO "); INC(logIndent); SpellSeq(clause.do); DEC(logIndent);
					clause := clause.next
				UNTIL clause = NIL;
				log.WriteLn;
				IF s.else # NIL THEN log.WriteString("ELSE "); 
					INC(logIndent); SpellSeq(s.else); DEC(logIndent); log.WriteLn
				END;
				log.WriteString("END")
			| s: PutReg DO 
				IF s.put THEN log.WriteString("SYSTEM.PUTREG(") ELSE log.WriteString("SYSTEM.GETREG(") END; SpellEx(s.reg); log.WriteString(", "); SpellEx(s.expr);
				log.WriteChar(")")
			ELSE s.Spell(log)
			END;
			log.WriteChar(';'); s := s.link
		END
	END SpellSeq;

		ELSIF readOnly IN flags THEN log.WriteChar("-")
		END
	END SpellExpMark;

	PROCEDURE SpellProcSysflags (f: SET);

	END SpellProcSysflags;

	PROCEDURE SpellSignature (par: POINTER TO ARRAY OF Par; first: INTEGER; res: Type);
		VAR len: INTEGER; hasParen, hasRes: BOOLEAN; 
	BEGIN
		hasRes := (res # NIL) & (res.what # fNoTyp);
		IF par = NIL THEN len := 0 ELSE len := LEN(par) END;
		hasParen := (first < len) OR hasRes;
		IF hasParen THEN log.WriteChar("(") END;
		WHILE first < len DO SpellObj(par[first]); IF first < len - 1 THEN log.WriteString("; ") END; INC(first) END;
		IF hasParen THEN log.WriteChar(")") END;
		IF hasRes THEN log.WriteString(": "); SpellType(res, showAlias) END
	END SpellSignature;

	PROCEDURE SpellType (t: Type; doShowAlias: BOOLEAN);
		VAR n: ARRAY 32 OF CHAR; elem: Array; base: Record; i, len: INTEGER;
	BEGIN ASSERT(t # NIL, 20);
		IF t IS Forward THEN log.WriteString("ERROR: Unresolved Forward Type Reference!")
		ELSIF doShowAlias & (t.typeName # NIL) THEN SpellQualident(t.typeName) 
		ELSIF doShowAlias & (t.qualident # NIL) THEN log.WriteString(t.qualident)
		ELSE
			WITH t: Basic DO
				CASE t.what OF 
				| fByte, fInt8: n := "BYTE" | fBool: n := "BOOLEAN" | fChar8: n := "SHORTCHAR"
				| fChar16: n := "CHAR" | fInt16: n := "SHORTINT" | fInt32: n := "INTEGER" 
				| fInt64: n := "LONGINT" | fReal32: n := "REAL" | fReal64:  n := "LONGREAL"
				| fSet: n := "SET"
				| fErr: n := "!#@^ERROR"
				ELSE HALT(21)
				END; log.WriteString(n)
			| t: Array DO log.WriteString("ARRAY");
				IF untagged IN t.flags THEN log.WriteString(" [untagged]") END;
				IF open IN t.flags THEN log.WriteString(" [open]") END;
				(*HALT(2);*)
				IF details THEN log.WriteString(" ["); log.WriteInt(t.len); log.WriteInt(t.lenoffs); log.WriteChar("]") END;
				elem := t;
				IF ~(open IN t.flags) THEN log.WriteChar(" "); log.WriteInt(t.len); 
					WHILE elem.elTyp IS Array DO elem := elem.elTyp(Array); log.WriteString(", "); log.WriteInt(elem.len)
					END
				ELSIF details THEN log.WriteChar("!"); log.WriteInt(t.len)
				END;
				log.WriteString(" OF "); SpellType(elem.elTyp, showAlias)
			| t: Pointer DO
				WITH 
				| t: SysPtr DO log.WriteString("SYSTEM.PTR")
				| t: ArrayPtr DO log.WriteString("POINTER"); 
					IF untagged IN t.flags THEN log.WriteString(" [untagged]") END;
					log.WriteString(" TO "); SpellType(t.base, showAlias)
				| t: RecPtr DO
					IF (t.base.typeName # NIL) & (t.base.typeName.name^ = "ANYREC") THEN
						log.WriteString("ANYPTR")
					ELSE log.WriteString("POINTER TO "); SpellType(t.base, showAlias)
					END
				ELSE HALT(126)
				END
			| t: Procedure DO log.WriteString("PROCEDURE "); 
				IF details THEN SpellProcSysflags(t.flags) END; SpellSignature(t.par, 0, t.res)
			| t: Record DO 
				IF absAttr IN t.flags THEN log.WriteString("ABSTRACT ")
				ELSIF limAttr IN t.flags THEN log.WriteString("LIMITED ")
				ELSIF extAttr IN t.flags THEN log.WriteString("EXTENSIBLE ")
				END;
				log.WriteString("RECORD ");
				IF untagged IN t.flags THEN log.WriteString("[untagged] ") END;
				i := t.level; base := t.base;
				WHILE i > 0 DO log.WriteChar("("); SpellType(base, showAlias); base := base.base; DEC(i) END;
				i := t.level; WHILE i > 0 DO log.WriteChar(")"); DEC(i) END;
				INC(logIndent); 
				IF t.fields = NIL THEN len := 0 ELSE len := LEN(t.fields) END;
				i := 0; WHILE i < len DO 
					log.WriteLn; SpellObj(t.fields[i]); IF i + 1 < len THEN log.WriteChar(";") END; INC(i)
				END;
				DEC(logIndent); log.WriteLn; log.WriteString("END");
				(*i := 0; WHILE i < t.nproc DO log.WriteLn; log.WriteLn; SpellObj(t.proc[i]); INC(i) END*)
			| t: AuxType DO IF t.what = fErr THEN log.WriteString("!#@^ERROR") ELSE HALT(20) END
			ELSE log.WriteString("Typedef NIY"); log.WriteInt(t.what)
			END
		END
	END SpellType;
	
	PROCEDURE SpellConstVal (c: Literal);
		VAR buf: ARRAY 32 OF CHAR;
	BEGIN
		WITH c: Int DO Strings.IntToString(c.long, buf); log.WriteChar(' '); log.WriteString(buf) 
		| c: Char DO log.WriteChar(c.ch) | c: Set DO log.WriteSet(c.set)
		| c: Bool DO log.WriteBool(c.bool) | c: Real DO log.WriteReal(c.real) 
		| c: StringLit DO log.WriteString(c.str)
		| c: Nil DO log.WriteString("NIL")
		| c: BadLiteral DO log.WriteString("bad!constant")
		ELSE HALT(20)
		END
	END SpellConstVal;
	
	PROCEDURE SpellLibEnt (o: Object);
		PROCEDURE Do (lib, ent: POINTER TO Name);
		BEGIN 
			IF (lib # NIL) & (lib^ # '') OR (ent # NIL) & (ent^ # '') THEN log.WriteChar("[");
				IF (lib = NIL) OR (lib^ = '') THEN log.WriteString('"", "') 
				ELSE log.WriteChar('"'); log.WriteString(lib); log.WriteString('", "')
				END;
				IF (ent # NIL) & (ent^ # '') THEN log.WriteString(ent) END;
				log.WriteString('"] ')
			END
		END Do;
	BEGIN
		WITH o: XProc DO Do(o.lib, o.ent) | o: LProc DO Do(o.lib, o.ent) | o: TProc DO Do(o.lib, o.ent)
		| o: Mod DO Do(o.lib, NIL) | o: Const DO Do(o.lib, o.ent) | o: GlobVar DO Do(o.lib, o.ent)
		| o: FProc DO IF o.actual # NIL THEN SpellLibEnt(o.actual) END | o: CProc DO Do(o.lib, o.ent)
		ELSE HALT(20)
		END
	END SpellLibEnt;
	
	PROCEDURE SpellBlock (o: SCope);
		VAR obj: Object;
	BEGIN
		obj := o.head; WHILE obj # NIL DO 
			IF ~(obj IS Par) THEN
				log.WriteLn; IF obj IS Proc THEN log.WriteLn END; SpellObj(obj); 
				IF ~(obj IS Proc) & ~(obj IS Mod) THEN log.WriteChar(";") END; 
			END;
			obj := obj.next
		END
	END SpellBlock;
	
	PROCEDURE ^ SpellTree (o: Object);
	PROCEDURE SpellObj* (o: Object);
		VAR len, i: INTEGER; hasParen, hasRes: BOOLEAN; kind: SET; obj: Object;
	BEGIN
		IF details & (o.textpos # 0) THEN
			log.WriteChar("@"); log.WriteInt(o.textpos); log.WriteChar(">"); log.WriteSet(o.flags)
		END;
		WITH 
		| o: Var DO
			WITH o: Par DO kind := kindMask * o.flags;
				IF kind = varK THEN log.WriteString("VAR ")
				ELSIF kind = inK THEN log.WriteString("IN ")
				ELSIF kind = outK THEN log.WriteString("OUT ")
				END
			ELSE
			END;
			log.WriteString(o.name); IF o IS GlobVar THEN SpellExpMark(o.flags) END;
			log.WriteString(": "); SpellType(o.type, showAlias)
		| o: Proc DO log.WriteString("PROCEDURE "); 
			IF (o IS FProc) OR (31 IN o.flags (*cheat*) ) THEN log.WriteString("^ ") END;
			IF o.flags * codeF = codeF THEN log.WriteString("[code] ") END;
			IF details THEN 
				log.WriteString("<@"); log.WriteInt(o.startPos); log.WriteString(".."); log.WriteInt(o.endPos); log.WriteString("> ");
				SpellProcSysflags(o.flags)
			END;
			WITH o: SProc DO log.WriteString(" ["); log.WriteInt(o.num); log.WriteString("]") ELSE END;
			IF o IS TProc THEN log.WriteChar("("); SpellObj(o.par[0]); log.WriteString(") "); i := 1 ELSE i := 0 END;
			log.WriteString(o.name); SpellExpMark(o.flags);
			SpellLibEnt(o);
			SpellSignature(o.par, i, o.type);
			WITH o: TProc DO
				IF newAttr IN o.flags THEN log.WriteString(", NEW") END;
				IF absAttr IN o.flags THEN log.WriteString(", ABSTRACT")
				ELSIF limAttr IN o.flags THEN log.WriteString(", LIMITED")
				ELSIF empAttr IN o.flags THEN log.WriteString(", EMPTY")
				ELSIF extAttr IN o.flags THEN log.WriteString(", EXTENSIBLE")
				END
			| o: CProc DO log.WriteChar('['); log.WriteInt(LEN(o.code)); log.WriteString(' bytes]')
			| o: IProc DO Log.String("Interrupt procedure NIY")
			ELSE
			END; log.WriteChar(";"); INC(logIndent); 
			SpellBlock(o); DEC(logIndent); log.WriteLn;
			IF ~(o IS FProc) THEN
				log.WriteString("BEGIN"); INC(logIndent); SpellSeq(o.body); DEC(logIndent); log.WriteLn;
				log.WriteString("END "); log.WriteString(o.name); log.WriteChar(";"); log.WriteLn
			END
		| o: TypeName DO log.WriteString(o.name); SpellExpMark(o.flags); 
			IF details THEN log.WriteString(" ["); log.WriteInt(o.type.size); log.WriteChar("]") END;
			log.WriteString(" = "); 
			IF o.type.typeName # o THEN SpellQualident(o.type.typeName)
			ELSE SpellType(o.type, ~showAlias); 
			END
		| o: Const DO log.WriteString("CONST "); log.WriteString(o.name); SpellExpMark(o.flags); 
			log.WriteString(" = "); SpellEx(o.ex)
		| o: Field DO log.WriteString(o.name); SpellExpMark(o.flags); 
			log.WriteString(": "); SpellType(o.type, showAlias)
		| o: Mod DO 
		ELSE log.WriteString("NIY Object: "); log.WriteString(o.name)
		END
	END SpellObj;
	
	PROCEDURE SpellSubj* (subj: Segraph);
		VAR m: Mod; len, i: INTEGER;
	BEGIN ASSERT(subj # NIL, 20); m := subj.mod[0];
		log.WriteString("MODULE "); log.WriteString(m.name); SpellLibEnt(m);
		IF m.lib # NIL THEN log.WriteString('["'); log.WriteString(m.lib); log.WriteString('"]') END; 
		log.WriteChar(';'); INC(logIndent); log.WriteLn;
		len := LEN(subj.mod);
		IF (len > 2) & (subj.mod[2] # NIL) THEN log.WriteString("IMPORT ");
			i := 2; WHILE (i < len) & (subj.mod[i] # NIL) DO
				Comma(i > 2); log.WriteString(subj.mod[i].name); log.WriteChar(' ');
				IF i < 22 THEN log.WriteChar(CHR(ORD('①') + i - 2)) ELSE log.WriteInt(i - 2) END;
				INC(i)
			END; log.WriteChar(';')
		END; 
		SpellBlock(m); DEC(logIndent); log.WriteLn;
		IF m.begin # NIL THEN
			log.WriteString("BEGIN"); INC(logIndent); SpellSeq(m.begin); DEC(logIndent); log.WriteLn
		END;
		IF m.close # NIL THEN
			log.WriteString("CLOSE"); INC(logIndent); SpellSeq(m.close); DEC(logIndent); log.WriteLn
		END;
		log.WriteString("END "); log.WriteString(m.name); log.WriteChar(".")
	END SpellSubj;
	
	PROCEDURE SpellTree (o: Object);
	BEGIN ASSERT(o # NIL, 20);
		IF o.left # NIL THEN SpellTree(o.left) END; SpellObj(o); log.WriteLn;
		IF o.right # NIL THEN SpellTree(o.right) END
	END SpellTree;

	PROCEDURE ThisExpr (nd: T.Node): Expression;
		VAR ex: Expression; o: Object; fpar: POINTER TO ARRAY OF Expression; 
			proc: Proc; rec: Record; guard: GuardEx; type: Type;
			op: Op; isOp: IsOp;
	BEGIN ASSERT(nd # NIL, 20);
		CASE nd.class OF
		| Nvar, Nvarpar, Nfield, Nderef, Nindex, Nguard, Ntype, Nproc: ex := ThisDesignator(nd)
		| Nconst: ex := ThisConstVal(nd.conval, nd.typ)
		| Ncall: ex := ThisDesignator(nd)
		(*| Nguard: guard := dir.Guard(); ex := guard; guard.what := ThisDesignator(nd.left); 
			guard.type := LookupType(nd.typ)*)
		| Nmop, Ndop, Nupto: 
			IF (nd.class = Nmop) & (nd.subcl = isfn) THEN isOp := dir.IsOp(); ex := isOp; 
				o := FindDeepShort(nowScope, nd.obj.name); ASSERT(o # NIL); 
				isOp.typeName := o(TypeName);
				isOp.des := ThisExpr(nd.left); isOp.type := ThisGenericType(nd.typ)
			ELSE op := dir.Op(); ex := op; 
				IF nd.class = Nupto THEN op.what := range ELSE op.what := nd.subcl END;
				op.a := ThisExpr(nd.left);
				IF nd.right # NIL THEN op.b := ThisExpr(nd.right) END;
				op.type := ThisGenericType(nd.typ)
			END
		| Ncomp: HALT(126)
		ELSE HALT(126)
		END;
		ASSERT(ex # NIL, 60); 
		WITH ex: CallDes DO 
			IF ex.proc # NIL THEN ASSERT(ex.type = ex.proc.type, 61) 
			ELSE ASSERT(ex.type = ex.what.type, 62)
			END
		ELSE ASSERT((ex IS ProcDes) OR (ex.type # NIL), 62) (*can't assert this: erroneous designators *)
		END;
		RETURN ex
	END ThisExpr;
	
	PROCEDURE ThisDesignator (nd: T.Node): Designator;
		VAR o: Object; var: VarDes; des: Designator; field: FieldDes; type: Type; deref: Deref; 
			index: Index; guard: GuardDes; typeDes: TypeDes; proc: ProcDes; call: CallDes; strsel: StringSel;
			i, j: INTEGER; idx: ARRAY 16 OF Expression;
	BEGIN ASSERT(nd # NIL, 20);
		CASE nd.class OF
		| Nproc : proc := dir.ProcDes(); des := proc; 
			proc.proc := FindDeepShort(nowScope, nd.obj.name)(Proc)
		| Nvar, Nvarpar: o := FindDeepShort(nowScope, nd.obj.name); ASSERT(o # NIL);
			IF nd.class = Nproc THEN 
			ELSIF (nd.class = Nvar) OR (nd.class = Nvarpar) THEN 
				var := dir.VarDes(); des := var; var.var := o(Var)
			END;
			des.type := o.type
				(* should GlobVar, LocVar and Par be distinguished? *)
		| Nfield: field := dir.FieldDes(); des := field; field.rec := ThisDesignator(nd.left); 
			type := LookupType(nd.left.typ);
			IF type IS RecPtr THEN type := type(RecPtr).base END;
			WITH type: Record DO field.field := FindShortInRec(type, nd.obj.name)(Field) 
			ELSE HALT(21) 
			END;
			field.type := field.field.type
		| Nderef: 
			IF nd.subcl = 0 THEN deref := dir.Deref(); des := deref; deref.what := ThisDesignator(nd.left);
				deref.type := deref.what.type(Pointer).Base()
			ELSIF nd.subcl = 1 THEN strsel := dir.StringSel(); des := strsel; 
				strsel.what := ThisDesignator(nd.left); strsel.type := strsel.what.type
			ELSE HALT(126)
			END
		| Nindex: index := dir.Index(); des := index; index.type := ThisGenericType(nd.typ);
			i := 0; REPEAT idx[i] := ThisExpr(nd.right); INC(i); nd := nd.left UNTIL nd.class # Nindex;
			j := 0; DEC(i); WHILE i >= 0 DO index.dim[j] := idx[i]; INC(j); DEC(i) END;
			index.arr := ThisDesignator(nd); 
		| Nguard: guard := dir.GuardDes(); des := guard; guard.what := ThisDesignator(nd.left); 
			guard.type := LookupType(nd.typ)
		| Ntype: typeDes := dir.TypeDes(); des := typeDes; typeDes.type := LookupType(nd.typ)
			(* represents the type reference in MAX(INTEGER), SYSTEM.TYP(Mod.Typ) *)
		| Ncall: call := dir.CallDes(); des := call; 
			GetThisCall(nd, call.proc, call.what, call.apar, call.type, call.super)
		ELSE HALT(126)
		END;
		ASSERT(des # NIL, 60);
		WITH des: Deref DO ASSERT((des.what.type IS SysPtr) OR (des.type # NIL))
		ELSE ASSERT((des IS ProcDes) OR (des.type # NIL), 61) (*can't assert this: erroneous *)
		END;
		RETURN des
	END ThisDesignator;

	PROCEDURE ThisAssign (nd: T.Node): Stat;
		VAR assign: Assign; put: Put; putreg: PutReg; move: Move; res: Stat; trap: Trap;
			e: T.Node; i: INTEGER; new: New;
	BEGIN ASSERT(nd.class = Nassign, 20);
		CASE nd.subcl OF
		| putfn: put := dir.Put(); res := put; put.adr := ThisExpr(nd.left); put.expr := ThisExpr(nd.right)
		| putrfn: putreg := dir.PutReg(); res := putreg; putreg.reg := ThisExpr(nd.left); 
			putreg.expr := ThisExpr(nd.right)
		| movefn: move := dir.Move(); res := move; move.from := ThisExpr(nd.left); 
			move.to := ThisExpr(nd.right);
			move.count := ThisExpr(nd.right.link)
		| assignfn, incfn, decfn, inclfn, exclfn: assign := dir.Assign(); res := assign; 
			assign.des := ThisDesignator(nd.left); assign.kind := nd.subcl;
			IF nd.right # NIL THEN assign.expr := ThisExpr(nd.right) END
		| newfn: new := dir.New(); res := new; new.des := ThisDesignator(nd.left);
			e := nd.right; i := 0; WHILE e # NIL DO new.dim[i] := ThisExpr(e); e := e.link; INC(i) END
		| assertfn: trap := dir.Trap(); res := trap; trap.cond := ThisExpr(nd.left); 
			IF nd.right # NIL THEN trap.num := ThisInt(nd.right.conval.intval)
			ELSE trap.num := zero
			END
		| copyfn, typfn: HALT(126)
		ELSE HALT(126)
		END;
		ASSERT(res # NIL, 60);
		RETURN res
	END ThisAssign;
	
	PROCEDURE ^ ThisStatSeq (nd: T.Node): Stat;
	PROCEDURE ThisStat (nd: T.Node): Stat;
		VAR res: Stat; return: Return; while: While; rpt: Repeat; loop: Loop; call: Call; _: BOOLEAN;
			__: Type; exit: Exit; trap: Trap; if, else: If; 
			case: Case; m, n: T.Node; i, ranges, cases: INTEGER; do: Stat;
			r: POINTER TO ARRAY OF Range; s: POINTER TO ARRAY OF Stat;
			with: With; clause: WithClause;
	BEGIN ASSERT(nd # NIL, 20);
		CASE nd.class OF
		| Nassign: res := ThisAssign(nd)
		| Nreturn: return := dir.Return(); res := return; 
			IF nd.left # NIL THEN return.ex := ThisExpr(nd.left) END; 
			return.proc := FindProc(nd.obj)
		| Nwhile: while := dir.While(); 
			res := while; while.cond := ThisExpr(nd.left); while.do := ThisStatSeq(nd.right)
		| Nrepeat: rpt := dir.Repeat(); res := rpt; rpt.until := ThisExpr(nd.right); rpt.do := ThisStatSeq(nd.left)
		| Nloop: loop := dir.Loop(); res := loop; loop.do := ThisStatSeq(nd.left)
		| Ncall: call := dir.Call(); res := call; GetThisCall(nd, call.proc, call.what, call.actuals, __, call.super)
		| Nexit: exit := dir.Exit(); res := exit
		| Ntrap: trap := dir.Trap(); res := trap; trap.num := ThisInt(nd.right.conval.intval)
		| Nifelse: if := dir.If(); res := if; if.cond := ThisExpr(nd.left.left); if.do := ThisStatSeq(nd.left.right);
			IF nd.right # NIL THEN else := dir.If(); else.do := ThisStatSeq(nd.right) END;
			m := nd.left.link;
			WHILE m # NIL DO if.next := dir.If(); if := if.next; 
				if.cond := ThisExpr(m.left); if.do := ThisStatSeq(m.right);
				m := m.link
			END;
			if.next := else
		| Ncase: case := dir.Case(); res := case; case.what := ThisExpr(nd.left); 
			IF nd.right.right # NIL THEN case.else := ThisStatSeq(nd.right.right) END;
			ranges := 0; cases := 0; n := nd.right.left;
			WHILE n # NIL DO INC(cases); 
				m := n.left; WHILE m # NIL DO INC(ranges); m := m.right END;
				n := n.link
			END;
			case.cases := cases;
			IF ranges > 0 THEN NEW(r, ranges); NEW(s, ranges); n := nd.right.left; i := 0;
				WHILE n # NIL DO ASSERT(n.class = Ncasedo, 100);
					do := ThisStatSeq(n.right); s[i] := do; m := n.left; 
					REPEAT ASSERT(m.class = Nconst, 100); 
						r[i].from := m.conval.intval; r[i].to := m.conval.intval2; INC(i); m := m.right
					UNTIL m = NIL;
					n := n.link
				END;
				case.ranges := r; case.stmts := s
			END
		| Nwith: with := dir.With(); res := with;
			IF nd.right # NIL THEN with.else := ThisStatSeq(nd.right) END;
			ASSERT(nd.left.class = Nif); ASSERT((nd.left.left.class = Nmop) & (nd.left.left.subcl = isfn));
			(*with.what := ThisDesignator(nd.left.left.left); *)
			with.clauses := dir.WithClause(); clause := with.clauses; m := nd.left;
			WHILE m # NIL DO clause.do := ThisStatSeq(m.right); 
				clause.var := ThisDesignator(m.left.left);
				clause.typeName := FindDeepShort(nowScope, m.left.obj.name)(TypeName); 
				ASSERT(clause.typeName # NIL, 20);
				IF m.link # NIL THEN clause.next := dir.WithClause(); clause := clause.next END;
				m := m.link
			END
		ELSE HALT(126)
		END;
		ASSERT(res # NIL, 60); res.textpos := nd.conval.intval;
		RETURN res
	END ThisStat;
	
	PROCEDURE ThisStatSeq (nd: T.Node): Stat;
		VAR res, s, last: Stat;
	BEGIN
		WHILE nd # NIL DO s := ThisStat(nd);
			IF res = NIL THEN res := s; last := res ELSE last.link := s; last := last.link END;
			nd := nd.link
		END;
		RETURN res
	END ThisStatSeq;

			nested := nested.link
		END
	END DoProcedures;

	PROCEDURE Cmp (IN a, b: ARRAY OF CHAR): INTEGER;
		CONST  eql = 0; less = 1; grtr = 2;
		VAR i, res: INTEGER;
	BEGIN i := 0; WHILE a[i] = b[i] DO INC(i) END;	
		IF a[i] < b[i] THEN res := less ELSIF a[i] = b[i] THEN res := eql ELSE res := grtr END;
		RETURN res
	END Cmp;

	PROCEDURE FindReference (IN qual: ARRAY OF CHAR): Type;
		CONST  eql = 0; less = 1; grtr = 2;
		VAR r: Typeref; res: Type; cmp: INTEGER;
	BEGIN r := refDict;
		WHILE r # NIL DO cmp := Cmp(qual, r.type.qualident);
			IF cmp = less THEN r := r.left ELSIF cmp = grtr THEN r := r.right ELSE res := r.type; r := NIL END
		END;
		RETURN res
	END FindReference;
	
	PROCEDURE InsertReference (s: Symbol; qual: POINTER TO String);
		CONST  eql = 0; less = 1; grtr = 2;
		VAR t, r, n: Typeref; cmp: INTEGER;
	BEGIN ASSERT(s IS Type); ASSERT(qual # NIL);
		r := refDict;
		WHILE r # NIL DO cmp := Cmp(qual, r.type.qualident);
			IF cmp = less THEN t := r; r := r.left 
			ELSIF cmp = grtr THEN t := r; r := r.right 
			ELSE HALT(20 (* duplicate qualident *) ) 
			END
		END;
		NEW(n); WITH s: Type DO n.type := s; s.qualident := qual END;
		IF t = NIL THEN refDict := n ELSIF cmp = less THEN t.left := n ELSE t.right := n END
	END InsertReference;

	PROCEDURE DoWrSym (VAR wr: SymbolWriter; s: Symbol; IN qual: ARRAY OF CHAR);
		CONST niltg = -1;
		VAR ind: INTEGER; ref, old: BOOLEAN; bkmrk, skip, what: INTEGER;
	BEGIN
		IF wr.typed THEN wr.rider.WriteByte(symboltg) END;
		IF s = NIL THEN wr.Int(niltg) 
		ELSE
			Register(s, ind, old); wr.Int(ind);
			IF ~old THEN ref := qual # '';
				IF ref THEN wr.Tag(reftg); wr.String(qual); bkmrk := wr.rider.Pos();
					wr.Int(none)
				END;
				what := WhatIs(s); wr.AsByte(what);
				IF ~((s IS Basic) & (s(Basic).what IN integers + reals + chars + booleans + sets)) THEN
					s.Externalize(wr); s.Externalize2(wr)	(* don't externalize basics, just their tag *)
				END;
				IF ref THEN skip := wr.rider.Pos(); wr.rider.SetPos(bkmrk); wr.Int(skip); wr.rider.SetPos(skip) 
				END
			END
		END;
		IF wr.typed THEN wr.rider.WriteByte(symbolendtg) END
	END DoWrSym;
	
	PROCEDURE WriteSymbol (VAR wr: SymbolWriter; s: Symbol);
	BEGIN DoWrSym(wr, s, '')
	END WriteSymbol;
	
	PROCEDURE (VAR wr: SymbolWriter) Symbol (s: Symbol), NEW;
	BEGIN DoWrSym(wr, s, '')
	END Symbol;
	
	PROCEDURE DoRdSym (VAR rd: SymbolReader; OUT s: Symbol; OUT qual: POINTER TO String);
		CONST niltg = -1;
		VAR what, ind, skip: INTEGER; tag: BYTE; isRef: BOOLEAN; xproc: XProc; gvar: GlobVar;
			oldmod: Mod;
	BEGIN
		IF rd.typed THEN rd.rider.ReadByte(tag)
			;IF tag # symboltg THEN rd.Err(eBadTag) END; (*ASSERT(tag = symboltg) *)
		END;
		IF rd.errTop = 0 THEN rd.Int(ind) END;
		IF (rd.errTop = 0) & (ind # niltg) THEN
			IF (0 <= ind) & (ind < LEN(symDict)) THEN s := symDict[ind];
				IF s = NIL (* 1st encounter of s in rd.Base() *) THEN
					rd.FromByte(what); 
					IF what = reftg THEN ReadAsStringPtr(rd, qual); rd.Int(skip); s := FindReference(qual) END;
					IF s # NIL THEN rd.rider.SetPos(skip)
					ELSE 
						CASE what OF 
						| gvartg: gvar := dir.GlobVar(); gvar.adr := -1;  (* for imported procedures and variables, -1 is required for HrE.BookmarkThis and HrE.AllocProxy to identify that XProc hasn't been assigned a proxy table entry *)
							s := gvar
						| lvartg: s := dir.LocVar() | partg: s := dir.Par() 
						| consttg: s := dir.Const() | fldtg: s := dir.Field() | typenametg: s := dir.TypeName()
						| lproctg: s := dir.LProc() 
						| xproctg: xproc := dir.XProc(); xproc.adr := -1; (* for imported procedures and variables, -1 is required for HrE.BookmarkThis and HrE.AllocProxy to identify that XProc hasn't been assigned a proxy table entry *)
							
							s := xproc
						| cproctg: s := dir.CProc()
						| sproctg: s := dir.SProc() | iproctg: s := dir.IProc() | tproctg: s := dir.TProc()
						| modtg: s := dir.Mod() | segtg: s := dir.Segraph()
						| sysptrtg: s := dir.SysPtr() | arrayptrtg: s := dir.ArrayPtr() | recptrtg: s := dir.RecPtr()
						| recordtg: s := dir.Record() | arraytg: s := dir.Array() | proceduretg: s := dir.Procedure()
						| basictg: HALT(26); s := dir.Basic()
						| int8tg: s := bytetp | int16tg: s := shorttp | int32tg: s := inttp | int64tg: s := longtp
						| real32tg: s := shortrealtp | real64tg: s := realtp | char8tg: s := shortchartp 
						| char16tg: s := chartp | booltptg: s := booltp | settptg: s := settp
						| inttg: s := dir.Int() | chartg: s := dir.Char() | settg: s := dir.Set() | booltg: s := dir.Bool()
						| realtg: s := dir.Real() | stringtg: s := dir.StringLit()
						| condestg: s := dir.ConDes()
						ELSE HALT(126)
						END;
						ASSERT(s # NIL, 60);
						symDict[ind] := s; IF what = reftg THEN InsertReference(s, qual) END;
						(*oldmod := rd.mod; WITH s: Mod DO rd.mod := s ELSE END;*)
						IF (s # NIL) & ~((s IS Basic) & (s(Basic).typeName # NIL)) THEN 
							s.Internalize(rd); s.Internalize2(rd) 
							(* don't internalize basic types - they are represented as tags only, no content *)
						END;
						(*rd.mod := oldmod*)
					END
				END
			ELSE HALT(20)
			END;
			ASSERT(s # NIL, 61)
		END;
		IF rd.typed THEN rd.rider.ReadByte(tag); IF tag # symbolendtg THEN rd.Err(eBadTag) END END
	END DoRdSym;
	
	PROCEDURE ReadSymbol (VAR rd: SymbolReader; OUT s: Symbol);
	(** Read a symbol; trap if a reference is encountered *)
		VAR qual: POINTER TO String;
	BEGIN DoRdSym(rd, s, qual); ASSERT(qual = NIL, 60 (* expected a symbol, but found a reference *))
	END ReadSymbol;
	
	PROCEDURE (VAR rd: SymbolReader) Symbol (OUT s: Symbol), NEW;
	BEGIN ReadSymbol(rd, s)
	END Symbol;
	
	PROCEDURE WriteType (VAR wr: SymbolWriter; t: Type);
	BEGIN
		IF t.typeName = NIL THEN wr.Symbol(t) 
		ELSIF t.typeName.imno = 0 THEN 
			wr.Symbol(t) (* and Internalize should put down the .typeName *)
		ELSIF t.typeName.imno > 0 THEN 
			DoWrSym(wr, t, wr.mod[t.typeName.imno].realName + "." + t.typeName.name)
		ELSIF t.qualident # NIL THEN DoWrSym(wr, t, t.qualident)
		ELSE HALT(60)
		END;
		
	END WriteType;
	
	PROCEDURE ReadType (VAR rd: SymbolReader; OUT t: Type);
		VAR s: Symbol; qual: POINTER TO String;
	BEGIN 
		
		DoRdSym(rd, s, qual);
		IF s # NIL THEN WITH s: Type DO t := s; t.qualident := qual ELSE rd.Err(eTypeExpected) END END
	END ReadType;
	
	
	(* Types are different from other symbols. Only 'domestic' symbols are put in a module's segraph - with the exception of types: all types reachable from a module are put in a module's segraph, even if they are not domestic. That includes types declared in directly and indirectly imported modules, regardless of whether their typenames are exported or not.
	Thus, there are three kinds of types: domestic, directly and indirectly imported.
	Domestic types are (.typeName = NIL) OR (.typeName.imno = 0). They are put down into the segraph entirely, and their typeName is put in, as well.
	Directly imported types are (.typeName.imno > 0) (maybe >1, 'cuase 1 is Universe; but there may also be other special modules, which require additional deliberation).
	Indirectly imported types are distinguished in some other (yet undefined) way. They do NOT have a .typeName: .typeName = NIL. This is because the .typeName variable is costly and holds information mostly useless for indirectly imported types. But, if an indirectly imported type has a name in it's defining module, this name has to be preserved so that all (other) references to this type, direct or indirect, would resolve to the same type variable.
	Anonymous types can be imported as well, however. It's hard to classify them as directly imported or indirectly imported, exactly because they are anonymous. It seems that they need to be externalized as symbols, not as references, because they can never be referenced other than at the spot they are declared. However, .typeName = NIL is not sufficient criteria that a type is anonymous: it might be indirectly imported. So this is TBD: representing module and name for indirectly imported types.
	All four kinds of types can be properly externalized and internalized by procedures WriteType and ReadType; they will be the incapsulators of all the complexity of externalizing types. Other linearization procedures may just use WriteType and ReadType, without considering the complexities of types.
	MODULE N;
	TYPE T* = RECORD f*: RECORD END END;
	END N;
	MODULE K; IMPORT N; VAR v*: N.T; END K. (* Here, the type of field .f is externalized as a reference;
		but this type is anonymous - hence the necessity to somehow process this case ) *)
	MODULE L; IMPORT N; VAR v*: N.T; END L.
	MODULE Subject; IMPORT K, L;
	BEGIN K.v.f := L.v.f
	END Subject.
	
	After a type is saved as a reference, it has to be restored with that same reference: the reference (the .qualident) has to be set for the type after it is internalized. This has to be done in SymbolReader.Symbol, because after it's finished, the fact that a type was read by reference cannot be established, and the reference qualident is lost as well. But SymbolReader.Symbol is not a good place to manipulate Type's specific fields, 'cause .Symbol is agnostic of type Symbol's extensions, and as long as it is, it can be actually moved to a different module, like HrM.
	Wait, no, it can't! Because Symbol is declared in HrT, and HrT imports HrM, not the other way around. 
	Maybe, to move SymbolReader to HrM, the procedure Symbol has to be unbound? And left in HrT. And then the SymbolReader is no longer a symbol reader, 'cause it's not aware of type Symbol. Well, maybe then it will just become a Reader: HrM.Reader. Because the main point of this reader/writer is type-safe linearization of foundational types (basic + strings) by means of type tagging.
	Another option: make Reader/Writer extensible, extend them in HrT and bind procedure Symbol to the extended type. The donwside would be implementation inheritance: the point of taking Reader/Writer out of HrT is to downsize HrT, so implementations would have to go into HrM, not just the interface, hence implementation inheritance. That is not a welcome practise, and what's the point anyway of binding procedure Symbol to the type? I mean, there is no extension planned... The only benefit would arguably be that you can write the reader/writer's variable name in front of the procedure name.. Well, it's probably not worth the effort.
	So: (be prepared to) take Reader/Writer out of HrT into HrM; unbind procedure Symbol from the types; fiddle with Type.qualident right in ReadSymbol.

	PROCEDURE ExtLibEnt (VAR wr: SymbolWriter; o: Object);
		VAR lib, ent: POINTER TO Name;
	BEGIN
		WITH o: GlobVar DO lib := o.lib; ent := o.ent | o: Const DO lib := o.lib; ent := o.ent 
		| o: LProc DO lib := o.lib; ent := o.ent | o: XProc DO lib := o.lib; ent := o.ent 
		| o: TProc DO lib := o.lib; ent := o.ent | o: Mod DO lib := o.lib
		ELSE
		END;
		IF lib = NIL THEN wr.String('') ELSE wr.String(lib) END;
		IF ent = NIL THEN wr.String('') ELSE wr.String(ent) END
	END ExtLibEnt;
	
	PROCEDURE IntLibEnt (VAR rd: SymbolReader; o: Object);
		VAR lib, ent: POINTER TO Name; empty: ARRAY 1 OF CHAR;
	BEGIN
		ReadAsNamePtr(rd, lib); IF lib = noname THEN lib := NIL END;
		ReadAsNamePtr(rd, ent); IF lib = noname THEN lib := NIL END;
		IF (lib # NIL) OR (ent # NIL) THEN
			WITH o: GlobVar DO o.lib := lib; o.ent := ent | o: Const DO o.lib := lib; o.ent := ent
			| o: LProc DO o.lib := lib; o.ent := ent | o: XProc DO o.lib := lib; o.ent := ent
			| o: TProc DO o.lib := lib; o.ent := ent | o: Mod DO o.lib := lib
			| o: TypeName DO o.lib := lib; o.ent := ent
			| o: Field DO
			| o: LocVar DO
			| o: Segraph DO
			ELSE HALT(126)
			END
		END
	END IntLibEnt;
	
	PROCEDURE (e: Expression) Externalize (VAR wr: SymbolWriter);
		VAR
	BEGIN
		wr.Symbol(e.type); wr.Int(e.pos); wr.Set(e.flags);
		WITH e: ConDes DO wr.Symbol(e.const)
		| e: VarDes DO wr.Symbol(e.var)
		| e: FieldDes DO wr.Symbol(e.rec); wr.Symbol(e.field)
		| e: Deref DO wr.Symbol(e.what)
		| e: StringSel DO wr.Symbol(e.what)
		| e: Nil DO (*!?!*) | e: Int DO wr.Long(e.long) | e: Char DO wr.Char(e.ch) | e: Set DO wr.Set(e.set) 
		| e: Bool DO wr.Bool(e.bool) | e: Real DO wr.Real(e.real) | e: StringLit DO wr.String(e.str)
		ELSE HALT(126)
		END
	END Externalize;
	
	PROCEDURE (e: Expression) Internalize (VAR rd: SymbolReader);
		VAR s: Symbol;
	BEGIN
		ReadType(rd, e.type); IF e.type = NIL THEN rd.Err(eTypeExpectedNilFound) END;
		rd.Int(e.pos); rd.Set(e.flags); IF e IS Literal THEN ASSERT(const IN e.flags) END;
		WITH 
		| e: ConDes DO rd.Symbol(s); 
			IF s = NIL THEN rd.Err(eConstExpectedNilFound) 
			ELSE WITH s: Const DO e.const := s ELSE rd.Err(eConstExpected) END
			END
		| e: Int DO rd.Long(e.long) | e: Real DO rd.Real(e.real) 
		| e: Char DO rd.Char(e.ch) | e: Set DO rd.Set(e.set) | e: Bool DO rd.Bool(e.bool)
		| e: StringLit DO ReadAsStringPtr(rd, e.str); 
			IF e.str = NIL THEN e.len := 0 ELSE e.len := LEN(e.str$) END
		ELSE HALT(126)
		END
	END Internalize;
	
	PROCEDURE (o: Object) Externalize (VAR wr: SymbolWriter);
		VAR len, i: INTEGER;
		PROCEDURE ScopeSymbols (o: Object);
		BEGIN 
			WHILE o # NIL DO 
				IF ~wr.exported OR (o.flags * visMask # domestic) OR (o IS Par) THEN wr.Symbol(o) END; 
				o := o.next 
			END; wr.Symbol(NIL)
		END ScopeSymbols;
	BEGIN
		IF o.name = NIL THEN wr.String(noname) ELSE wr.String(o.name) END;
		WITH
		| o: Par DO wr.AsByte(o.num)
		| o: Const DO wr.Symbol(o.ex); wr.Symbol(o.val)
		| o: Proc DO
			wr.AsByte(o.nesting); 
			WITH o: SProc DO wr.AsByte(o.num)
			| o: CProc DO ASSERT(o.code # NIL); swr.SString(o.code)
			| o: TProc DO wr.AsShort(o.num)
			ELSE
			END;
			IF o.par = NIL THEN len := 0 ELSE len := LEN(o.par) END; wr.AsByte(len); 
			ScopeSymbols(o.head);
			(*i := 0; WHILE i < len DO wr.Symbol(o.par[i]); INC(i) END;*)
			wr.Int(o.startPos); wr.Int(o.endPos); wr.Int(o.level); (* byte? *)
			(*wr.Symbol(o.owner)*)
		| o: Segraph DO ASSERT(o.mod # NIL);
			i := LEN(o.mod); REPEAT DEC(i) UNTIL o.mod[i] # NIL; 
			wr.Int(i); WHILE i > 1 DO wr.Symbol(o.mod[i]); DEC(i) END;
			wr.Symbol(o.mod[0]); ScopeSymbols(o.mod[0].head)
		| o: Mod DO (*wr.Symbol(o.owner)*)
		ELSE
		END;
		wr.Symbol(o.type); wr.Set(o.flags); 
		wr.Int(o.fprint); wr.AsByte(o.imno); wr.Int(o.textpos);
		ExtLibEnt(wr, o)
	END Externalize;
			
	PROCEDURE ReadMod (VAR rd: SymbolReader; OUT m: Mod);
		VAR s: Symbol; 
	BEGIN rd.Symbol(s);
		IF s = NIL THEN rd.Err(eModuleExpectedNilFound)
		ELSIF ~(s IS Mod) THEN rd.Err(eModuleExpected)
		ELSE m := s(Mod)
		END;
	END ReadMod;
	
	PROCEDURE (o: Object) Internalize (VAR rd: SymbolReader);
		VAR what, len, i: INTEGER; s: Symbol; empty: ARRAY 1 OF CHAR; 
			lit: Literal; oldmod: Mod;
		PROCEDURE ScopeSymbols (sc: SCope);
			VAR s: Symbol; o: Object; nsc: SCope;
		BEGIN nsc := nowScope; nowScope := sc;
			REPEAT rd.Symbol(s);
				IF s = NIL THEN o := NIL 
				ELSE WITH s: Object DO o := s ELSE rd.Err(eObjectExpected); o := NIL END 
				END;
				IF o # NIL THEN Insert(sc.tree, o); o.owner := sc; 
					IF sc.head = NIL THEN sc.head := o END;
					IF sc.tail # NIL THEN sc.tail.next := o END; sc.tail := o;
					WITH o: Par DO sc(Proc).par[o.num] := o ELSE END
				END
			UNTIL s = NIL;
			nowScope := nsc
		END ScopeSymbols;
	BEGIN
		ReadAsNamePtr(rd, o.name);
		WITH
		| o: Par DO rd.FromByte(o.num)
		| o: Const DO rd.Symbol(s);
			IF s = NIL THEN rd.Err(eExprExpectedNilFound)
			ELSE WITH s: Expression DO o.ex := s ELSE rd.Err(eExprExpected) END
			END;
			rd.Symbol(s);
			IF s = NIL THEN rd.Err(eLitExpectedNilFound)
			ELSE WITH s: Literal DO o.val := s ELSE rd.Err(eLitExpected) END
			END
		| o: Proc DO
			rd.FromByte(o.nesting);
			WITH o: SProc DO rd.FromByte(o.num)
			| o: CProc DO ReadAsSStringPtr(rd, o.code)
			| o: TProc DO rd.FromShort(o.num)
			ELSE
			END;
			o.owner := nowScope; nowScope := o;
			rd.FromByte(len); 
			IF len < 0 THEN rd.Err(eNegativeParamLen) ELSIF len > 0 THEN NEW(o.par, len) END;
			ScopeSymbols(o);
			rd.Int(o.startPos); rd.Int(o.endPos); rd.Int(o.level);
			nowScope := o.owner
		| o: Segraph DO 
			rd.Int(i); 
			IF (1 <= i) & (i < capImports) THEN NEW(o.mod, i + 1) ELSE rd.Err(eBadImportsNr) END;
			WHILE i > 1 DO ReadMod(rd, o.mod[i]); DEC(i) END;
			ReadMod(rd, o.mod[0]);
			IF (o.mod # NIL) & (o.mod[0] # NIL) THEN 
				o.mod[0].owner := o.mod[1]; 	(*???*)
				oldmod := rd.mod; rd.mod := o.mod[0]; ASSERT(rd.mod # NIL);
				ScopeSymbols(o.mod[0]);
				rd.mod := oldmod
			END
		| o: Mod DO (*rd.Symbol(s); 
			IF s # NIL THEN WITH s: SCope DO o.owner := s ELSE rd.Err(eModScopeExpected) END END*)
		ELSE
		END;
		ReadType(rd, o.type);
		rd.Set(o.flags); IF rd.resetUsed THEN EXCL(o.flags, used) END;
		(*WITH o: Mod DO IF o.owner = NIL THEN ASSERT(o.name^ = "@UNIVERSE") END ELSE END;*)
		rd.Int(o.fprint); rd.FromByte(o.imno); o.imno := rd.imno; rd.Int(o.textpos);
		IntLibEnt(rd, o)
	END Internalize;
	
	PROCEDURE (t: Type) Externalize (VAR wr: SymbolWriter);
		VAR i, j, len: INTEGER; f: Field; v: TProc;
	BEGIN 
		WITH t: Basic DO
		| t: ArrayPtr DO wr.Symbol(t.base)
		| t: RecPtr DO wr.Symbol(t.base)
		| t: Record DO wr.Symbol(t.base); 
			IF t.fields = NIL THEN len := 0 ELSE len := LEN(t.fields) END;
			IF wr.exported THEN i := 0; j := 0;
				WHILE i < len DO IF t.fields[i].flags * visMask # domestic THEN INC(j) END; INC(i) END
			ELSE j := len
			END;
			wr.AsShort(j);
			i := len - 1; WHILE i >= 0 DO f := t.fields[i];
				IF ~wr.exported OR (f.flags * visMask # domestic) THEN wr.Symbol(f) END; DEC(i) 
			END;
			len := t.nproc;
			IF wr.exported THEN i := 0; j := 0; 
				WHILE i < len DO IF t.proc[i].flags * visMask # domestic THEN INC(j) END; INC(i) END;
				wr.AsShort(j)
			ELSE wr.AsShort(len)
			END; 
			i := len - 1; WHILE i >= 0 DO v := t.proc[i];
				IF ~wr.exported OR (v.flags * visMask # domestic) THEN wr.Symbol(v) END; DEC(i)
			END;
			wr.AsByte(t.level); wr.AsByte(t.align)
		| t: Array DO wr.Int(t.len); wr.Int(t.lenoffs); wr.Symbol(t.elTyp)
		| t: Procedure DO wr.Symbol(t.res); 
			IF t.par = NIL THEN len := 0 ELSE len := LEN(t.par) END; wr.AsByte(len);
			i := 0; WHILE i < len DO ASSERT(t.par[i] IS Par); wr.Symbol(t.par[i]); INC(i) END
		ELSE HALT(126)
		END;
		wr.Set(t.flags); 
		wr.AsByte(t.what); wr.Int(t.pbfp); wr.Int(t.pvfp); wr.Int(t.size);
		IF (t.typeName # NIL) & (t.typeName.imno = 0) THEN wr.Symbol(t.typeName) ELSE wr.Symbol(NIL)
		END
	END Externalize;
	
	PROCEDURE (t: Type) Internalize (VAR rd: SymbolReader);
		VAR skip, len: INTEGER; s: Symbol; formals: POINTER TO ARRAY OF Par; 
			fields: POINTER TO ARRAY OF Field; proc: POINTER TO ARRAY OF TProc;
			mod, name: Name; type: Type;
	BEGIN
		WITH t: Basic DO
		|t: ArrayPtr DO rd.Symbol(s); 
			IF s = NIL THEN rd.Err(eElemTypeExpectedNilFound)
			ELSIF ~(s IS Array) THEN rd.Err(eElemTypeExpected)
			ELSE t.base := s(Array)
			END
		| t: RecPtr DO rd.Symbol(s);
			IF s = NIL THEN rd.Err(eRecPtrBaseTypeExpectedNilFound)
			ELSIF ~(s IS Record) THEN rd.Err(eRecPtrBaseTypeExpected)
			ELSE t.base := s(Record)
			END
		| t: Record DO rd.Symbol(s); 
			IF s # NIL THEN WITH s: Record DO t.base := s ELSE rd.Err(eRecBaseTypeExpected) END END;
			rd.FromShort(len);
			IF len < 0 THEN rd.Err(eNegativeFieldsLen)
			ELSIF len > 0 THEN NEW(fields, len); t.fields := fields;
				REPEAT DEC(len); rd.Symbol(s);
					IF s = NIL THEN rd.Err(eFieldExpectedNilFound)
					ELSE WITH s: Field DO fields[len] := s; Insert(t.tree, s) ELSE rd.Err(eFieldExpected) END
					END
				UNTIL len = 0
			END;
			rd.FromShort(len);
			IF len < 0 THEN rd.Err(eNegativeVerbsLen)
			ELSIF len > 0 THEN NEW(proc, len); t.proc := proc;
				REPEAT DEC(len); rd.Symbol(s);
					IF s = NIL THEN rd.Err(eVerbExpectedNilFound)
					ELSE WITH s: TProc DO proc[len] := s; AppendProc(t, s) ELSE rd.Err(eVerbExpected) END
					END
				UNTIL len = 0
			END;
			rd.FromByte(t.level); rd.FromByte(t.align)
		| t: Array DO rd.Int(t.len); rd.Int(t.lenoffs); rd.Symbol(s);
			IF s = NIL THEN rd.Err(eArrayElTypExpectedNilFound)
			ELSIF ~(s IS Type) THEN rd.Err(eArrayElTypExpected)
			ELSE t.elTyp := s(Type)
			END
		| t: Procedure DO rd.Symbol(s);
			IF s = NIL THEN rd.Err(eResTypExpectedNilFound)
			ELSIF ~(s IS Type) THEN rd.Err(eResTypExpected)
			ELSE t.res := s(Type)
			END;
			rd.FromByte(len);
			IF len < 0 THEN rd.Err(eNegativeParamLen)
			ELSIF len > 0 THEN NEW(formals, len); t.par := formals;
				REPEAT DEC(len); rd.Symbol(s);
					IF s = NIL THEN rd.Err(eParExpectedNilFound)
					ELSIF ~(s IS Par) THEN rd.Err(eParExpected)
					ELSE formals[len] := s(Par)
					END
				UNTIL len = 0
			END
		ELSE HALT(126)
		END;
		rd.Set(t.flags); rd.FromByte(t.what); rd.Int(t.pbfp); rd.Int(t.pvfp); rd.Int(t.size);
		rd.Symbol(s);

	PROCEDURE DoInternalize (OUT top: Segraph; VAR rd: SymbolReader);
		CONST errWrongDictLen = 1; errModuleExpected = 2;
		VAR ver, cap: INTEGER; s: Symbol;
	BEGIN 
		rd.errTop := 0;

			ELSE NEW(symDict, cap + 1)

	END DoInternalize;

	BEGIN
		i := 1; WHILE top.mod[i] # NIL DO INC(i) END; 
		IF name = "SYSTEM" THEN

		ELSIF name = "JAVASCRIPT" THEN
			mod := BuildJAVASCRIPT(alias); top.mod[i] := mod; Append(mod); done := TRUE
		ELSE
			rd.rider := HrM.lib.OldHrSym(name$, NIL); done := rd.rider # NIL;

			IF done THEN 
				rd.resetUsed := TRUE;	(* IMPORTed objects should not have used in .flags *)

						mod.imno := i;	(* correction necessary after Append! *)
						top.mod[i] := mod
					END
				ELSE done := FALSE
				END
			END
		END
	END Import;

	PROCEDURE DoDumpSymbols (VAR rd: SymbolReader; VAR log: TextMappers.Formatter);
		VAR tag, byte: BYTE; int: INTEGER; ch: CHAR; bool: BOOLEAN; set: SET; real: REAL; long: LONGINT;
			pstr: POINTER TO String;
	BEGIN ASSERT(rd.typed, 20);
		REPEAT ReadTag(rd, tag); 
			IF ~rd.rider.eof THEN
				rd.rider.SetPos(rd.rider.Pos() - SIZE(BYTE));
				CASE tag OF
				| inttg: rd.Int(int); log.WriteInt(int); log.WriteChar(" ")
				| chartg: rd.Char(ch); log.WriteChar(ch); log.WriteChar(" ")
				| settg: rd.Set(set); log.WriteSet(set); log.WriteChar(" ")
				| booltg: rd.Bool(bool); log.WriteBool(bool); log.WriteChar(" ")
				| realtg: rd.Real(real); log.WriteReal(real); log.WriteChar(" ")
				| stringtg: ReadAsStringPtr(rd, pstr); 
					IF pstr # NIL THEN log.WriteString(pstr) ELSE log.WriteString("''") END; log.WriteChar(" ")
				| bytetg: rd.FromByte(int); log.WriteInt(int); log.WriteChar(" ")
				| shorttg: rd.FromShort(int); log.WriteInt(int); log.WriteChar(" ")
				| longtg: rd.Long(long); log.WriteInt(long); log.WriteChar(" ")
				| symboltg: ReadTag(rd, byte); INC(logIndent); log.WriteLn; log.WriteString("SYMBOL"); log.WriteChar(" ")
				| symbolendtg: ReadTag(rd, byte); log.WriteLn; log.WriteString("SYMBOLEND"); DEC(logIndent); log.WriteLn
				ELSE ReadTag(rd, byte); log.WriteString("Unknown tag: "); log.WriteInt(int); log.WriteChar(" ")
				END;
				IF (rd.rider.Pos() >= globalErrPos) & (markerMaker # NIL) THEN markerMaker(log, globalErrNo);
					globalErrPos := MAX(INTEGER)
				END
			END
		UNTIL rd.rider.eof
	END DoDumpSymbols;
