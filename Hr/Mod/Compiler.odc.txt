	PROCEDURE (lib: Lib) OldHrSym* (mod: HrLib.Name; rd: Files.Reader): Files.Reader;
		VAR f: Files.File; res: Files.Reader; fname: Files.Name;
	BEGIN 
		Files.dir.GetFileName(mod$, herschelSymType, fname);
		f := Files.dir.Old(lib.symbols, fname, Files.shared); 
		IF f # NIL THEN res := f.NewReader(rd) END;
		RETURN res
	END OldHrSym;

	PROCEDURE (lib: Lib) NewHrSym* (IN mod: ARRAY OF CHAR): Files.File;
	BEGIN RETURN Files.dir.New(lib.symbols, Files.dontAsk)
	END NewHrSym;

		VAR _: INTEGER;

			WITH v: TextViews.View DO text := v; beg := 0; end := v.ThisModel().Length() ELSE END
		END
	END OldSourceView;

	PROCEDURE InsertMarks (t: TextModels.Model);
	BEGIN DevMarkers.SetDir(HrMarkers.dir); HrM.InsertMarks(t); DevMarkers.SetDir(DevMarkers.stdDir)
	END InsertMarks;
	
	PROCEDURE (VAR a: SafeAction) Do;
		CONST obj = 3; ref = 4; allref = 5; srcpos = 6; bigEnd = 15; pVarInd = 14;
		VAR sg: HrT.Segraph; vsize: INTEGER;
	BEGIN a.name := '';
		HrAlpha.Do(a.src, lib, HrSyn.dir, Log.text, {}, sg);
		IF HrM.noerr THEN
			a.name := sg.mod[0].name$;
			IF HrM.options * {HrM.interface, HrM.noCode } = {} THEN 
				IF traverser = NIL THEN NEW(traverser) END;
				HrV.Init({ ref, allref, srcpos }); HrV.Allocate(sg.mod[0], vsize); traverser.Module(sg); HrV.Close;
				IF HrM.noerr THEN HrM.RegisterObj END
			END
		END
	END Do;
	
	PROCEDURE CompileThisRd (src: TextModels.Reader; OUT name: HrT.Name);
		VAR a: SafeAction; 
	BEGIN
		a.src := src; Services.Try(a);
		IF a.trapped THEN HrM.Mark(1111, HrM.errpos); name := '' ELSE name := a.name$ END;
		IF ~HrM.noerr THEN InsertMarks(src.Base()) END
	END CompileThisRd;
	
	PROCEDURE GetSrc (OUT src: TextModels.Reader);
		VAR c: TextControllers.Controller; beg, _: INTEGER;
	BEGIN
		c := TextControllers.Focus();
		IF c # NIL THEN
			IF c.HasSelection() THEN c.GetSelection(beg, _) ELSE beg := 0 END;
			src := c.text.NewReader(NIL); src.SetPos(beg)
		END
	END GetSrc;
	
	PROCEDURE ShowDecoded*;
		VAR name: HrT.Name; res: TextModels.Model; src: TextModels.Reader; 
			fmt: POINTER TO TextMappers.Formatter;
	BEGIN
		GetSrc(src);
		IF src # NIL THEN
			CompileThisRd(src, name);
			IF HrM.noerr THEN
				res := TextModels.dir.New(); NEW(fmt); fmt.ConnectTo(res);
				IF HrM.options * {HrM.interface, HrM.noCode } = {} THEN
					HrOcfExplorer.Decode(lib.OldCode(name$, NIL), src, fmt);
				ELSE fmt.WriteString("Interface module ⇒ no code output"); fmt.WriteLn
				END;
				Views.OpenAux(TextViews.dir.New(res), name + " decoded")
			ELSE Log.String("Errors..."); Log.Ln
			END
		ELSE 
			Log.String("Nothing to compile; put caret into module text or select the word MODULE"); Log.Ln
		END
	END ShowDecoded;
		
	PROCEDURE OcfTo (ocf, out: Files.File; format: INTEGER; OUT res: INTEGER);
	BEGIN ASSERT(ocf # NIL, 20); ASSERT(out # NIL, 21);
		HrBinder.NewJob(HrBinder.amd64, out, HrBinder.executable, '', format);
		HrBinder.AddOcf(ocf, HrBinder.isEntry, res);
		IF res = 0 THEN HrBinder.CloseJob ELSE HALT(7); out.Close END
	END OcfTo;

	PROCEDURE DoCompile (VAR rd: TextModels.Reader);
		VAR name: HrT.Name;
	BEGIN 
		CompileThisRd(rd, name);
		IF HrM.noerr THEN 
			Log.ParamMsg("Herschel: ^0 compiled successfully, code and symbol files placed under ^1", name, demoLocation, ''); Log.Ln
		ELSE 
			Log.String("Herschel:"); Log.Int(HrM.errors); Log.String(" errors compiling "); Log.String(name); Log.Ln
		END
	END DoCompile;
	
	PROCEDURE Compile*;
		VAR rd: TextModels.Reader;
	BEGIN 
		GetSrc(rd);
		IF rd # NIL THEN DoCompile(rd) ELSE Log.String("No text in focus"); Log.Ln END
	END Compile;
	
	
	PROCEDURE Open;
	BEGIN
		Dialog.ShowStatus("#Dev:Compiling");
		StdLog.buf.Delete(0, StdLog.buf.Length())
	END Open;

	PROCEDURE Close;
	BEGIN
		StdLog.text.Append(StdLog.buf);
		IF HrM.noerr THEN Dialog.ShowStatus("#Dev:Ok")
		END;
		(*sourceR := NIL;*)
		Kernel.Cleanup
	END Close;

	PROCEDURE CompileList (beg, end: INTEGER; c: TextControllers.Controller);
		VAR (*v: Views.View;*) i: INTEGER; error, one: BOOLEAN; name: Files.Name; loc: Files.Locator;
			t: TextModels.Model; opts: SET; title, entry: ARRAY 64 OF CHAR;
			rd: TextModels.Reader; mname: HrT.Name; tbeg, tend: INTEGER; tv: TextViews.View;
	BEGIN
		s.SetPos(beg); s.Scan; one := FALSE;
		WHILE (s.start < end) & (s.type = TextMappers.string) & (s.len < LEN(mname)) DO
			s.Scan; one := TRUE;
			WHILE (s.start < end) & (s.type = TextMappers.char) &
				((s.char = "-") OR (s.char = "+") OR
				(s.char = "!") OR (s.char = "*") OR (s.char = "?") OR (s.char = "^") OR (s.char = "("))
			DO
				IF s.char = "(" THEN
					WHILE (s.start < end) & ((s.type # TextMappers.char) OR (s.char # ")")) DO s.Scan END
				END;
				s.Scan
			END
		END;
		IF one & (s.start >= end) THEN
			s.SetPos(beg); s.Scan; error := FALSE;
			WHILE (s.start < end) & (s.type = TextMappers.string) & ~error DO
				(*i := 0; WHILE i < LEN(name) DO name[i] := 0X; INC(i) END;*)
				
				Log.String(s.string); Log.Char(":");
				tv := NIL; lib.OldSourceView(s.string, tv, tbeg, tend, loc, name);
				IF tv # NIL THEN t := tv.ThisModel() ELSE t := NIL END;
				s.Scan; opts := defopt;
				WHILE (s.start < end) & (s.type = TextMappers.char) DO
					IF s.char = "-" THEN
						IF srcpos IN opts THEN EXCL(opts, srcpos)
						ELSIF allref IN opts THEN EXCL(opts, allref)
						ELSIF ref IN opts THEN EXCL(opts, ref)
						ELSE EXCL(opts, obj)
						END
					ELSIF s.char = "!" THEN
						IF assert IN opts THEN EXCL(opts, assert)
						ELSE EXCL(opts, checks)
						END
					ELSIF s.char = "+" THEN INCL(opts, allchecks)
					ELSIF s.char = "?" THEN INCL(opts, hint)
					ELSIF s.char = "@" THEN INCL(opts, errorTrap)
					ELSIF s.char = "$" THEN INCL(opts, oberon)
					ELSIF s.char = "(" THEN
						s.Scan;
						WHILE (s.start < end) & (s.type = TextMappers.string) DO
							title := s.string$; s.Scan;
							IF (s.start < end) & (s.type = TextMappers.char) & (s.char = ":") THEN
								s.Scan;
								IF (s.start < end) & (s.type = TextMappers.string) THEN
									entry := s.string$; s.Scan;
									IF t # NIL THEN DevSelectors.ChangeTo(t, title, entry) END
								END
							END;
							IF (s.start < end) & (s.type = TextMappers.char) & (s.char = ",") THEN s.Scan END
						END
					END;
					s.Scan
				END;
				IF t # NIL THEN
					rd := t.NewReader(rd); rd.SetPos(tbeg); DoCompile(rd)
				ELSE Log.String(" not found"); Log.Ln
				END
			END
		ELSE Dialog.ShowMsg("#Dev:NotOnlyFileNames")
		END;
		s.ConnectTo(NIL);
		IF error & (c # NIL) & c.HasSelection() & (s.start < end) THEN
			c.SetSelection(s.start, end)
		END;
		IF error & (tv # NIL) THEN
			Views.Open(tv, loc, name, NIL);
			DevMarkers.ShowFirstError(t, TextViews.any)
		END
	END CompileList;

	PROCEDURE CompileModuleList*;
		VAR c: TextControllers.Controller; beg, end: INTEGER;
	BEGIN
		Open;
		c := TextControllers.Focus();
		IF c # NIL THEN
			s.ConnectTo(c.text);
			IF c.HasSelection() THEN c.GetSelection(beg, end)
			ELSE beg := 0; end := c.text.Length()
			END;
			CompileList(beg, end, c)
		ELSE Dialog.ShowMsg("#Dev:NoTextViewFound")
		END;
		Close
	END CompileModuleList;

	PROCEDURE CompileThis*;
		VAR p: DevCommanders.Par; beg, end: INTEGER;
	BEGIN
		Open;
		p := DevCommanders.par;
		IF p # NIL THEN
			DevCommanders.par := NIL;
			s.ConnectTo(p.text); beg := p.beg; end := p.end;
			CompileList(beg, end, NIL)
		ELSE Dialog.ShowMsg("#Dev:NoTextViewFound")
		END;
		Close
	END CompileThis;

	PROCEDURE ModuleTo* (IN mod, format: ARRAY OF CHAR);
		VAR rd: Files.Reader; ocf, elf: Files.File; fmt, res: INTEGER; type: Files.Type;
			path: Dialog.String;
	BEGIN
		IF format = 'elf' THEN fmt := HrBinder.elf; type := ''
		ELSIF format = 'pe' THEN fmt := HrBinder.pe; type := 'exe'
		ELSE Log.String('Valid formats: elf, pe'); Log.Ln; fmt := -1
		END;
		rd := lib.OldCode(mod$, NIL);
		IF rd = NIL THEN
			Log.ParamMsg("Oops, couldn't find ^0Code/^1.ocf", demoLocation, mod, ''); Log.Ln
		END;
		IF (rd # NIL) & (fmt # -1) THEN ocf := rd.Base();
			elf := Files.dir.New(elfLoc, Files.dontAsk);
			OcfTo(ocf, elf, fmt, res);
			IF res = 0 THEN
				elf.Register(mod$, type, Files.dontAsk, res);
				
				IF res = 0 THEN
					Log.ParamMsg("ELF ^0^1 created", elfLocation, mod, '');
					Dialog.GetLocPath(elfLoc, path);
					IF path # "" THEN path := path + "/" END;
					LB.Chmod(path + LB.outputName, {0, 2, 3, 4, 5, 6, 7, 8}) (* rwxrwxr-x *)
				ELSE
					Log.String("Could not save ELF as "); Log.String(elfLocation);
					Log.String(mod); Log.Ln
				END
			ELSE Log.String("Smth went wrong enchanting the Ocf into an Elf, rse ="); Log.Int(res); Log.Ln
			END
		END
	END ModuleTo;

MAKING A DLL
1. Write your Component Pascal module and compile it with HrCompiler.Compile. This should produce the .ocf file for your module under Hr/Demo/Code.

2. Replace M with your module's name in the following example and run it:
"HrPeFmt.Install; 
HrBinder.OcfTo('Hr/Demo/Code/', 'M.ocf', 'Hr/Demo/Elf/', 'M.dll', 2, FALSE)"

This should produce the DLL for your module under Hr/Demo/Elf.
