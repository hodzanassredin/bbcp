MODULE ObxProbe24;

	IMPORT Console, Strings;

	VAR bad: INTEGER;

	PROCEDURE CHK (IN s: ARRAY OF CHAR; got, exp: LONGINT);
		VAR t: ARRAY 40 OF CHAR;
	BEGIN
		IF got # exp THEN
			INC(bad); Console.WriteStr("BAD "); Console.WriteStr(s);
			Strings.IntToString(got, t); Console.WriteStr(" got="); Console.WriteStr(t);
			Strings.IntToString(exp, t); Console.WriteStr(" exp="); Console.WriteStr(t);
			Console.WriteLn
		END
	END CHK;

	PROCEDURE Go*;
		VAR a, b, c: LONGINT;
	BEGIN
		bad := 0;
		a := 1234567890123; b := 10007;
		CHK("a+b", a + b, 1234567900130);
		CHK("a-b", a - b, 1234567880116);
		CHK("-a", -a, -1234567890123);
		CHK("a*2", a * 2, 2469135780246);
		CHK("a DIV b", a DIV b, 123370429);
		CHK("a MOD b", a MOD b, 7120);
		IF ~(a > b) THEN INC(bad); Console.WriteStr("BAD a>b"); Console.WriteLn END;
		c := -a;
		IF ~(c < 0) THEN INC(bad); Console.WriteStr("BAD c<0"); Console.WriteLn END;
		CHK("c DIV b", c DIV b, -123370430);
		CHK("c MOD b", c MOD b, 2887);
		IF bad = 0 THEN Console.WriteStr("Probe24 OK")
		ELSE Console.WriteStr("Probe24 FAILURES") END;
		Console.WriteLn
	END Go;

END ObxProbe24.
