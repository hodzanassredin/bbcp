	PROCEDURE AdAligned* (x: INTEGER): INTEGER;
	BEGIN RETURN (x + addrSize[version] - 1) DIV addrSize[version] * addrSize[version]
	END AdAligned;
	
	PROCEDURE PageAligned* (x: INTEGER): INTEGER;
	BEGIN RETURN (x + pageSize - 1) DIV pageSize * pageSize
	END PageAligned;
	
	
	
	PROCEDURE InsertExtSym (ext: Module; IN n: Name; type: INTEGER; OUT ind: INTEGER);
	(** Insert symbol n of type type from external module ext into external symbol table; the symbol's number is ind *)
		(* maybe, a check against duplicate names is required; in this case, ind won't be monotonously growing *)
		(* for now, simply append n without any checks *)
	BEGIN
		NEW(externals[extIndex].name, LEN(n$) + 1); externals[extIndex].name^ := n$; externals[extIndex].type := type;
		externals[extIndex].mod := ext;
		ext.used := TRUE;
		ind := extIndex; INC(extIndex)
	END InsertExtSym;
	
	PROCEDURE AppendExtReloc (virtAdr: LONGINT; symInd: INTEGER);
	(** Append an ELF relocation for virtual address virtAdr referring to external symbol symInd *)
	BEGIN
		IF relModulus = relocCap THEN INC(relPage); relModulus := 0 END;
		IF elfReloc[relPage] = NIL THEN NEW(elfReloc[relPage]) END;
		elfReloc[relPage, relModulus].virtAddr := virtAdr; elfReloc[relPage, relModulus].symInd := symInd;
		INC(relModulus) 
	END AppendExtReloc;
	
	PROCEDURE BindExternal (ext: Module; IN n: Name; type: INTEGER; code: POINTER TO ARRAY OF BYTE;proxyBase: LONGINT; offs: INTEGER; VAR res: INTEGER);
		VAR (*bs: BookmarkScanner; *)ind: INTEGER;
	BEGIN
		InsertExtSym(ext, n, type, ind);	(* changes global references to the currently inserted symbol *)
		IF ind >= 0 THEN (* go thru the chain of bookmarks and make each bookmark an entry in relocations *)
			(* external symbols don't actually have any chains, they are rerefenced in the proxies! *)
			AppendExtReloc(proxyBase + offs, ind)
			
		ELSE HALT(126)	(* possibly a duplicate symbol name in two different external libraries? *)
		END
	END BindExternal;
	
	PROCEDURE Bind (m: Module; VAR os: HrOcf.OcfScanner; VAR meta, desc, code, imp: POINTER TO ARRAY OF BYTE; 
		OUT res: INTEGER
	);
		VAR us: HrOcf.UseScanner; im: Module; 
	BEGIN
		os.ConnectUseScanner(us); res := OK;
		REPEAT us.Scan;
			CASE us.type OF
			| mEnd: (* end of use block *)
			| mNextImp: (* step over to next imported module *)
			| mProc:
				im := m.imports[us.imp];
				IF im.name[0] = '$' THEN ASSERT(us.chainOffset = 0); 
					BindExternal(im, us.name, mProc, code, m.meta.addr + AdAligned(m.ms), us.chainHead, res)
				ELSE HALT(126)
				END
			ELSE HALT(126)
			END
		UNTIL (us.type = mEnd) OR (res # OK) OR (m.imports[us.imp] = NIL)
	END Bind;
	
	PROCEDURE AddExtLib (IN n: Name; OUT m: Module);
	BEGIN NEW(m); m.name := n$; m.link := modules; modules := m
	END AddExtLib;
	
	PROCEDURE ThisMod (IN n: Name): Module;
		VAR m: Module;
	BEGIN m := modules; WHILE (m # NIL) & (m.name # n) DO m := m.link END; RETURN m
	END ThisMod;

	PROCEDURE Cmp (IN a, b: ARRAY OF CHAR): INTEGER;
		VAR i, cmp: INTEGER; c, d: CHAR;
	BEGIN
		i := 0; REPEAT c := a[i]; d := b[i]; INC(i) UNTIL (c # d) OR (c = 0X) OR (d = 0X); 
		IF c = d THEN cmp := equal ELSIF c < d THEN cmp := less ELSE cmp := greater END;
		RETURN cmp
	END Cmp;
	
	PROCEDURE InsExp (m: Module; IN es: HrOcf.ExpScanner);
		CONST mVar = 3; mProc = 4;	(* keep in sync with HrE *)
		VAR e, g: Export; cmp: INTEGER;
	BEGIN
		INC(nofExports);
		g := exports; e := exports; 
		WHILE e # NIL DO
			g := e; cmp := Cmp(es.name, e^.name);
			IF cmp = less THEN e := e.left ELSIF cmp = greater THEN e := e.right ELSE HALT(20) END
		END;
		NEW(e); e.name := es.name$; e.mod := m; e.nameOffs := es.nameOffs; 
		IF es.mode = mProc THEN e.addr := es.offset ELSIF es.mode = mVar THEN e.addr := -es.offset ELSE HALT(20) END; 
		IF g = NIL THEN exports := e ELSIF cmp = less THEN g.left := e ELSE g.right := e END
	END InsExp;
	
	PROCEDURE MergeExports (m: Module; VAR es: HrOcf.ExpScanner);
	BEGIN
		REPEAT es.Scan;
			IF ~es.eos THEN InsExp(m, es) END
		UNTIL es.eos
	END MergeExports;
	
	PROCEDURE DoAddOcf (ocf: Files.File; VAR res: INTEGER);
		VAR s: HrOcf.OcfScanner; _, i: INTEGER; n: Name; (*meta, desc, code,*) imps: POINTER TO ARRAY OF BYTE;
			m: Module; es: HrOcf.ExpScanner;
			c: ARRAY 3 OF LONGINT; 
	BEGIN
		s.ConnectTo(ocf.NewReader(NIL));	(* header and descriptor are in, module can be scanned *)
		IF s.res # HrOcf.OK THEN HALT(20)
		ELSE
			NEW(m); 
			m.ms := s.ms; m.ds := s.ds; m.cs:= s.cs; m.vs := s.vs; m.imps := s.imps; m.exts := s.exts; m.name := s.name$;
			(* add imported modules *)
			s.rider.SetPos(s.importsPos); 
			i := 0; WHILE (i < s.nofImports) & (res = OK) DO
				HrOcf.ReadName(s.rider, n, _); 	(* !!! potential utf8 errors ignored *)
				Log.String("Importing: "); Log.String(n); Log.Ln;
				IF ThisMod(n) # NIL THEN 	(* nothing *)
				ELSIF n[0] = '$' THEN AddExtLib(n, m.imports[i])
				ELSE HALT(126)
				END;
				INC(i)
			END;
			IF res = OK THEN
				
				formatter.AllocateMod(m);	(* allocate segments' virtual base addresses *)
				
				
				(* read in meta, desc, code blocks *)
				NEW(s.meta, m.ms); 
				s.rider.SetPos(s.metaPos); s.rider.ReadBytes(s.meta^, 0, m.ms);
				NEW(s.desc, m.ds); 
				s.rider.SetPos(s.descPos); s.rider.ReadBytes(s.desc^, 0, m.ds);
				NEW(s.code, m.cs);
				s.rider.SetPos(s.codePos); s.rider.ReadBytes(s.code^, 0, m.cs);
				IF m.imps > 0 THEN NEW(imps, m.imps * addrSize[version]) END;
				
				(* perform base relocation *)
				s.rider.SetPos(s.baseRelocPos); 
				HrOcf.Fixup(s, s.rider, NIL); s.ConnectToBlks;
				
				IF ~terminal THEN s.ConnectExpScanner(es); MergeExports(m, es) END;
				
				
				(* perform binding with already-prepared modules *)
				(*s.rider.SetPos(s.usePos);*)	(* after reading all the base fixups, the rider is exactly at the UseBlk *)
				Bind(m, s, s.meta, s.desc, s.code, imps, res);
				
				IF res = OK THEN	 (* write out segments into ELF *)
					formatter.WriteMod(m, s.meta, imps, s.code, s.desc);
					m.link := modules; modules := m
				END
			END
		END
		;ASSERT(res = OK);
	END DoAddOcf;
	
	PROCEDURE (format: Formatter) NewJob- (f: Files.File), NEW, ABSTRACT;

	PROCEDURE NewJob* (ver: INTEGER; f: Files.File; kind: BOOLEAN; IN libraryName: ARRAY OF CHAR; outFormat: INTEGER );
		CONST ET_REL = 1; ET_EXEC = 2; ET_DYN = 3; 
	BEGIN
		ASSERT(ver = amd64, 126); 
		IF outFormat = elf THEN 
			IF elfFmt # NIL THEN formatter := elfFmt ELSE HALT(23) END
		ELSIF outFormat = pe THEN
			IF peFmt # NIL THEN formatter := peFmt ELSE HALT(23) END
		ELSE HALT(24)
		END;
		ASSERT(f # NIL, 22);
		ASSERT(LEN(libraryName$) < LEN(libName) - 1, 23);
		libName := libraryName$;
		m64 := ver = amd64; 
		IF m64 THEN version := v64 ELSE version := v32; HALT(126) END;
		terminal := kind; (* FALSE for a library *)
		extIndex := 1;	(* start with 1: #0 is NULL symbol *)
		entryMod := NIL;
		modules := NIL;
		REPEAT elfReloc[relPage] := NIL; DEC(relPage) UNTIL relPage = -1; relPage := 0;
		relModulus := 0;
		nofExports := 0;
		exports := NIL;
		formatter.NewJob(f)
	END NewJob;
	
	PROCEDURE AddOcf* (ocf: Files.File; isEntry: BOOLEAN; OUT res: INTEGER);
	BEGIN
		ASSERT(formatter # NIL, 20);
		ASSERT(ocf # NIL, 22);
		IF isEntry THEN ASSERT(entryMod = NIL, 23) END;
		res := OK;
		DoAddOcf(ocf, res); 
		IF isEntry & (res = OK) THEN entryMod := modules END
	END AddOcf;
	
	PROCEDURE (format: Formatter) CloseJob-, NEW, ABSTRACT;
	
	PROCEDURE CloseJob*;
	BEGIN 
		ASSERT(formatter # NIL, 20);
		ASSERT(entryMod # NIL, 22);
		formatter.CloseJob;
		formatter := NIL;
		modules := NIL;
		exports := NIL;
		Kernel.Collect
	END CloseJob;
	
	PROCEDURE ResetJob*;
	BEGIN HALT(126)
	END ResetJob;

	PROCEDURE DoOcfTo (ocf, out: Files.File; format: INTEGER; kind: BOOLEAN; IN name: ARRAY OF CHAR; OUT res: INTEGER);
	BEGIN ASSERT(ocf # NIL, 20); ASSERT(out # NIL, 21);
		NewJob(amd64, out, kind, name, format);
		AddOcf(ocf, isEntry, res);
		IF res = 0 THEN CloseJob ELSE HALT(7); out.Close END
	END DoOcfTo;
	
	PROCEDURE OcfTo* (IN ocfLoc, ocfName, outLoc, outName: ARRAY OF CHAR; format: INTEGER; kind: BOOLEAN);
		VAR loc: Files.Locator; ocf, out: Files.File; _, res: INTEGER;
	BEGIN
		ocf := Files.dir.Old(Files.dir.This(ocfLoc), ocfName$, Files.shared); ASSERT(ocf # NIL, 20);
		out := Files.dir.New(Files.dir.This(outLoc), Files.dontAsk);
		DoOcfTo(ocf, out, format, kind, outName, res);
		IF res = 0 THEN out.Register(outName$, '', Files.dontAsk, _);
			IF kind = executable THEN Log.String("Executable written: ") 
			ELSE Log.String("Library written: ") 
			END;
			Log.String(outName); Log.Ln
		ELSE Log.String("Something went wrong:"); Log.Int(res); Log.Ln 
		END
	END OcfTo;

	PROCEDURE InitMod;
		VAR _: INTEGER;
	BEGIN
		addrSize[v32] := 4; addrSize[v64] := 8;
		relPage := 0; relModulus := 0;
		IF Dialog.platform = Dialog.linux THEN Dialog.Call("HrElfFmt.Install", '', _) 
		ELSE Dialog.Call("HrPeFmt.Install", '', _)
		END
	END InitMod;
